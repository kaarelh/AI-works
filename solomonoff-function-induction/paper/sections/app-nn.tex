\section{Proofs for Section~\ref{sec:nn} (neural-network Bayes)}\label{app:nn}

\begin{proposition}[Every prior nearly saturates the bound on its typical set (standard)]\label{prop:nn:typical}
\status{proved; known (standard)}\src{context Prop 4.3}
For every semimeasure $P$ on truth tables, every string $a$ and $j\ge0$: $P\{f:P(f)\le2^{-K(f\mid a)-j}\}\le2^{-j}$. Hence if $P_A$ is lower-semicomputed by $a$ and $\varpi$ is a probability measure, then with probability at least $1-2^{-j}$ over $\theta\sim\varpi$, $K(f_\theta\mid a)-c\le-\log P_A(f_\theta)<K(f_\theta\mid a)+j$.
\end{proposition}

\begin{proof}
The left side is at most $\sum_f2^{-K(f\mid a)-j}\le2^{-j}$ by Kraft's inequality; the lower bound is \cref{prop:nn:upper}.
\end{proof}

This is the standard fact that $-\log P\approx K$ on $P$-most objects \citep{li2008introduction} (cf.\ \cref{lem:ex:surprisal}). So $P_A$ looks like $2^{-K}$ on its own typical set, which can omit most simple functions.

\begin{proof}[Proof of \cref{prop:nn:uniform}]
Let $C$ be a circuit for $f$ of size $s=\csize(f)<S$. The circuits of size $S$ whose first $s$ gates equal $C$, whose gates $s+1,\dots,S-1$ are arbitrary, and whose last gate is $\mathrm{AND}(\mathrm{gate}_s,\mathrm{gate}_s)$ all compute $f$. Their fraction among all circuits of size $S$ is $\prod_{j=s+1}^{S-1}16(d+1+j)^2\big/\prod_{j=1}^{S}16(d+1+j)^2=1/(N_s\cdot16(d+1+S)^2)$. For the mistakes: a law of a random total function has order-independent cumulative log loss $-\log P_{{\rm unif},S}(f)$ on every ordering (\cref{prop:setting:order}), and the expected number of Gibbs mistakes is at most $\ln2$ times the log loss, as in \cref{thm:comp:circuitbayes}(b).
\end{proof}

\begin{remark}[What a bridge would give]\label{rem:nn:bridge}
\status{remark}\src{context \S4.3}
If $P_A(f)\ge2^{-a\ell(f)-b}$ for a description length $\ell$, then on every ordering NN Bayes makes at most $(\ln2)(a\ell(f)+b)$ Gibbs mistakes and at most $a\ell(f)+b$ posterior-majority mistakes (as in \cref{thm:ood:cumulative,rem:comp:majority}); if also $P_A(f)\le2^{-a'\ell(f)+b'}$, the log loss on the full domain is at least $a'\ell(f)-b'$. \Cref{prop:nn:uniform} is the lower half for uniform circuit priors. The converse, that few mistakes imply a small circuit, is stated for NN Bayes by the slides in their setting and in distribution (\cref{rem:nn:slides}), and holds for circuit Bayes on every ordering (\cref{thm:comp:circuitbayes}).
\end{remark}

\begin{proof}[Proof of \cref{thm:nn:walsh}]
(a) Write $\mathcal K(x,x')=h(x\odot x')$ with $h(z):=\Psi(\sum_jz_j/d)$ and $\odot$ the coordinatewise product, using $\langle x,x'\rangle=\sum_jx_jx'_j$. For fixed $x$, $x'=x\odot z$ with $z$ uniform, and $\chi_S(x\odot z)=\chi_S(x)\chi_S(z)$, so $(T_{\mathcal K}\chi_S)(x)=\Ex_z[h(z)\chi_S(z)]\,\chi_S(x)$. The eigenvalue $\Lambda_{|S|}:=\Ex_z[h(z)\chi_S(z)]$ depends only on $|S|$, because $h$ is invariant under permutations of the coordinates. For the NNGP and NTK recursions with i.i.d.\ Gaussian weights and biases, every layer's kernel depends on the inputs only through $\langle x,x\rangle$, $\langle x',x'\rangle$ and $\langle x,x'\rangle$, and the first two equal $d$ on the cube.
(b) The characters form an orthonormal basis of $L^2$ of the uniform measure. For each $S$, $\min_{g\in\mathcal V}\|\chi_S-g\|^2=1-\|P_{\mathcal V}\chi_S\|^2$ with $P_{\mathcal V}$ the orthogonal projection, and $\sum_{\text{all }T}\|P_{\mathcal V}\chi_T\|^2=\operatorname{tr}P_{\mathcal V}=\dim\mathcal V\le n$. So $\sum_{|S|=k}\|P_{\mathcal V}\chi_S\|^2\le n$.
\end{proof}

\begin{proof}[Proof of \cref{cor:nn:separation}]
$\chi_S$ is total and computable, so \cref{thm:ood:cumulative} gives $\sum_te_t\le(\ln2)(K(\chi_S)+c)$ on every sequence of distinct inputs; repeated inputs have $e_t=0$, so the bound holds for i.i.d.\ inputs too (\cref{prop:setting:order}). An enumerative code of $S$ given $d$ and $k$ gives the bound on $K(\chi_S)$. Let $y^*=\chi_S(x)\in\{\pm1\}$ and let $w\le e$ be the posterior mass of the wrong label at $x$; the posterior mean is $\hat g=y^*(1-e-w)$, so $|y^*-\hat g|=e+w\le2e$ and $(y^*-\hat g)^2\le4e^2\le4e$. For i.i.d.\ inputs the expected test error after $t-1$ samples equals the expected online error at step $t$, which gives the averaged bound. The kernel side is \cref{thm:nn:walsh}(b) with $\mathcal V=\mathrm{span}\{\mathcal K(x_i,\cdot)\}$. Since $\log\binom dk\le k\log d$, the SOI side needs $n=O(k\log d/\varepsilon)$ for the averaged error, while $\binom dk\ge(d/k)^k$. For the error at a single step, \cref{cor:learn:perstep}(a) bounds the expected error by $((K(\chi_S)+c)\ln2+\ln n+1)/n$, and the squared error by four times this.
\end{proof}

\begin{proof}[Proof of \cref{rem:nn:oddparity}]
(a) With $\arccos(-\rho)=\pi-\arccos\rho$ one gets $\Psi_1(\rho)-\Psi_1(-\rho)=\rho$ and $\rho\Psi_0(\rho)-(-\rho)\Psi_0(-\rho)=\rho$. So $\Psi_1(\rho)-\frac12\rho$ and $\Psi_1(\rho)+\rho\Psi_0(\rho)-\rho$ are even. The function $z\mapsto\Psi(\rho(z))-a\rho(z)$, $\rho(z)=\sum_jz_j/d$, is then invariant under $z\mapsto-z$, so its Walsh coefficients at odd $|S|$ vanish, while $a\rho(z)$ has Walsh coefficients only at $|S|=1$. Hence $\Lambda_{|S|}=0$ for odd $|S|\ge3$. Then $\langle\mathcal K(x_i,\cdot),\chi_S\rangle=(T_{\mathcal K}\chi_S)(x_i)=0$, so $\mathcal V\perp\chi_S$ and $\|\chi_S-\hat g\|^2=1+\|\hat g\|^2$.
(b) $\langle z,\chi_S\rangle:=\Ex_x[z(x)\chi_S(x)]$ is a linear functional of a centred Gaussian process, hence centred Gaussian, with variance $\Ex_{x,x'}\mathcal K(x,x')\chi_S(x)\chi_S(x')=\langle\chi_S,T_{\mathcal K}\chi_S\rangle=\Lambda_{|S|}=0$. So $\langle z,\chi_S\rangle=0$ almost surely. If $\mathrm{sign}\,z(x)=\chi_S(x)$ for all $x$, then $z(x)\chi_S(x)=|z(x)|$ and $\langle z,\chi_S\rangle=\Ex|z|>0$. So this event has prior probability $0$, and conditioning on an event of positive probability keeps it at $0$.
(c) From \cref{thm:nn:walsh}(a), $\mathcal K(x,x')=\sum_T\Lambda_{|T|}\chi_T(x)\chi_T(x')$, so $\sum_{x}\mathcal K(x_i,x)\chi_S(x)=2^d\Lambda_{|S|}\chi_S(x_i)=0$, and $\sum_{x}g(x)\chi_S(x)=0$ for every $g\in\mathcal V$; split the sum into training and unseen points. For kernel ridge regression $g=\sum_i\alpha_i\mathcal K(x_i,\cdot)$ with $\alpha=(G+\beta I)^{-1}y$ and $G$ the training Gram matrix, so $\sum_ig(x_i)y_i=y^\top G(G+\beta I)^{-1}y\ge0$: $G$ and $(G+\beta I)^{-1}$ are commuting positive semidefinite matrices. If some $g\in\mathcal V$ interpolates the training labels, its unseen sum is exactly $-n$.
\end{proof}
