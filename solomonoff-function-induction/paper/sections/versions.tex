\section{Kolmogorov and semimeasure versions}\label{sec:versions}

H\"anni's draft has a section headed ``Equivalence between Kolmogorov version and universal semimeasure version'', whose whole text is ``need coding theorem'' \citep{hanni2026notes}. He later asked that the paper ``include an equivalence between the kolmogorov complexity version and the universal semimeasure version, if these turn out to be equivalent (if they don't, then it should explain that)'' \citep{hanni2026request}. They are not equivalent: \cref{tab:versions:compare} summarises what is equal, what differs and by how much, and \cref{rem:versions:answer} answers in plain words. Proofs are in \cref{app:versions}.

\subsection{The variants}\label{sec:versions:defs}

\begin{definition}[Variants]\src{foundations Def 4.1}\label{def:versions:variants}
Let $(P_i)$ be an effective enumeration of all lsc conditional semimeasures (it exists by trimming each $P(\cdot\mid x)$ as in \cref{thm:setting:weakuniversal}).
\begin{itemize}
\item \emph{Program mixture} (deterministic programs):\\ $W(D):=\sum_{p\in\dom V}2^{-|p|}\,\ind[\varphi_{V(p)}\supseteq D]=\sum_e\mm(e)\,\ind[\varphi_e\supseteq D]$.
\item \emph{Kolmogorov version} (single shortest program): $\KF(D):=\min\{|p|:p\in\dom V,\ \varphi_{V(p)}\supseteq D\}=\min\{K(e):\varphi_e\supseteq D\}$.
\item \emph{Product mixture} (private randomness per input, on data sets):\\ $\Mprod(D):=\sum_i\mm(i)\prod_{(x,y)\in D}P_i(y\mid x)$.
\item \emph{Per-query mixture} (private randomness per query, on sequences): $\Mpq(y_{1:n}\Vert x_{1:n}):=\sum_i\mm(i)\prod_{t\le n}P_i(y_t\mid x_t)$, for every finite sequence of pairs, consistent or not.
\end{itemize}
\end{definition}

$W$ reads H\"anni's random program as a deterministic program; $\MF$ lets it read random bits. $W(\emptyset)$, $\Mprod(\emptyset)$, $\Mpq(\varepsilon\Vert\varepsilon)$ are below $1$, which changes losses by a constant only. $W$'s programs delimit themselves before reading the input, so $W$ is a mixture of fixed functions (\cref{rem:versions:Wdelim}).

\begin{remark}[Operator induction]\status{known; unverified; equality on distinct inputs proved}\src{foundations Def 4.1, claim F36; context \S6 item 1; history Prop 4.9}\label{rem:versions:operator}
Solomonoff's third kind of induction takes ``an unordered set of ordered pairs'' and predicts the second element of a new pair from the first \citep{solomonoff2008three} (abstract only seen). Secondary sources write its prediction as $\sum_ja_j\prod_{i\le n}O^j(A_i\mid Q_i)$, the product running over all question--answer pairs, repeats included \citep{ozkural2017ultimate} (search snippet only). On that description operator induction is $\Mpq$: equal to $\Mprod(D_n)$ on distinct inputs, different with repeats, where $\Mprod$ gives a conflicting label probability $0$ and $\Mpq$ allows fresh noise (\cref{prop:learn:perquery}). So operator induction and $\Mprod$ coincide only on distinct inputs.
\end{remark}

\begin{remark}[Where the randomness lives]\src{foundations Rem 4.11}\label{rem:versions:randomness}
Nowhere in $W$ and $\KF$ (fixed functions); privately per input in $\Mprod$; privately per query in $\Mpq$ (a repeated input is resampled); on one tape shared by all inputs in $\MF$, so one hypothesis may correlate labels across inputs, as $F(x)=\omega_{|x|+1}$ does. Every mixture also correlates labels through its posterior; the classes still differ by more than constants.
\end{remark}

\subsection{The chain and the gaps}\label{sec:versions:chain}

\begin{theorem}[The chain]\status{proved}\src{foundations Thm 4.2}\label{thm:versions:chain}
For every data set $D\ne\emptyset$,
\[
2^{-\KF(D)}\le W(D)\le2^{O(1)}\Mprod(D)\le2^{O(1)}\MF(D)\le2^{O(1)}\Mw(D)\le2^{K(\dom D)+O(1)}\,2^{-\KF(D)}.
\]
So the five quantities agree up to $K(\dom D)+O(1)$ bits.
\end{theorem}

The last step is the conditional coding theorem, $K(D\mid\dom D)\le-\log\Mw(D)+O(1)$ (\cref{app:versions:chain}). Let $\Omtail{\ell}:=\sum\{2^{-|p|}:p\in\dom V,\ |p|\ge\ell\}$, which tends to $0$.

\begin{proposition}[Deterministic programs cannot separate inputs cheaply]\status{proved}\src{foundations Prop 4.3 ((b) = context Prop 5.4(2))}\label{prop:versions:Wgap}
(a) \emph{Two points, binary labels} ($\{0,1\}\subseteq Y$). For every $\varepsilon>0$ and all large $d$ there are distinct $x_1,x_2\in\{0,1\}^d$ with $W(\{(x_1,0),(x_2,1)\})\le\varepsilon$, while $\Mprod(D)\ge2^{-O(1)}/4$ and $\MF(D)\ge2^{-O(1)}/4$ for every two-point $D$ with labels in $\{0,1\}$.
(b) \emph{One point, long labels} ($Y=\{0,1\}^*$). For every $d$ some $y\in\{0,1\}^d$ has $W(\{(0^d,y)\})\le2^{1-d}\,\Omtail{d-1}$, while $\Mprod(\{(0^d,y)\})\ge2^{-d-O(1)}$.
So $\Mprod/W$ and $\MF/W$ are unbounded.
\end{proposition}

(Pigeonhole: the fewer than $2^\ell$ programs shorter than $\ell$ give the $2^d$ inputs at most $3^{2^\ell}$ output patterns, so two inputs are never separated by them.) The two-point gap of $W$ against $\Mprod$ and $\MF$ lies between $\log(1/\Omtail{\ell_d})$, $\ell_d:=\lceil\log(d/\log3)\rceil-1$, and $\Kpad(2^{d+1})+O(1)$ (\cref{rem:versions:Wrate}), where $\Kpad(N):=\min_{L\ge N}K(L)$ is the \emph{padded complexity}, nondecreasing, unbounded and at most $K(N)$.

\begin{proposition}[Tape-length bound]\status{proved; finite analogue computed}\src{foundations Prop 4.4 (= its referee's Prop R1); checks c9(a), r5}\label{prop:versions:tape}
(a) Let $T$ be a function machine with law $\nu$, and for $L\ge0$ let $\nu^{(L)}(D):=\Unif\{\omega:$ every $(x,y)\in D$ has a triple $(x,w,y)\in T$ with $w$ a prefix of $\omega_{1:L}\}$. Then $\nu^{(L)}(D)\uparrow\nu(D)$ as $L\to\infty$, and for every $D\ne\emptyset$ and every $L$,
\[
W(D)\ \ge\ 2^{-K(T)-\Kpad(L)-c}\,\nu^{(L)}(D),
\]
with $c$ independent of $T$, $D$ and $L$. With $L_{1/2}(\nu,D):=\min\{L:\nu^{(L)}(D)\ge\nu(D)/2\}$ this gives $\log\big(\MF(D)/W(D)\big)\le\Kpad\big(L_{1/2}(\MF,D)\big)+O(1)$.
(b) \emph{Fair-coin labels.} For every $D$ with labels in $\{0,1\}$ and $\dom D\subseteq\{0,1\}^{\le d}$: $W(D)\ge2^{-|D|-\Kpad(2^{d+1})-c'}\ge2^{-|D|-K(\lceil\log(d+1)\rceil)-c''}$.
(c) $\Kpad(2^{2^s})\le K(s)+O(1)$, and $\Kpad(N)\le2\log(g(N)+2)+c_g$ for every computable, nondecreasing, unbounded $g:\mathbb N\to\mathbb N$.
\end{proposition}

The $2^{L'}$ programs ``simulate $T$ with stored tape prefix $w$'', $|w|=L'\ge L$, $K(L')=\Kpad(L)$, carry weight $2^{-K(T)-\Kpad(L)-O(1)}\nu^{(L')}(D)$. So $W$ imitates any shared-tape law, paying only for a self-delimiting bound on the number of tape bits used: a random program in H\"anni's sense can carry the bits it needs. (b) applies (a) to $F(x):=\omega_{j(x)+1}$.

\begin{proposition}[Shared randomness is not a mixture of private randomness]\status{proved; counting step computed}\src{foundations Prop 4.5; check r9(c)}\label{prop:versions:shared}
Let $\{0,1\}\subseteq Y$ and let $\nulen$ be the law of $F(x):=\omega_{|x|+1}$: one fair coin per input length, shared by all inputs of that length. $\nulen$ is a machine law of complexity $O(1)$, but $\sup_D\nulen(D)/\Mprod(D)=\infty$. Hence $\MF/\Mprod$ is unbounded.
\end{proposition}

(A product hypothesis labels $0^n$ and $1^n$ independently; a compactness argument shows that $\Mprod$ covers only a small part of the non-constant patterns of $k$ shared coins.)

\begin{remark}[Size of this gap]\status{proved}\src{foundations Rem 4.5a}\label{rem:versions:sharedsize}
The witnesses of \cref{prop:versions:shared} are not explicit, and their gap is small: on data with inputs of length at most $d$, $\Mprod(D)\ge2^{-O(1)}W(D)\ge2^{-\Kpad(d+1)-O(1)}\nulen(D)$ by \cref{prop:versions:tape}. A lower bound on the rate is open.
\end{remark}

\subsection{No coding theorem for function programs}\label{sec:versions:coding}

For strings, shortest description and algorithmic probability agree up to a constant (the coding theorem). For functions they do not.

\begin{theorem}[No coding theorem for function programs]\status{proved; arithmetic and covering numbers computed}\src{foundations Thm 4.7 (= its referee's Prop A); checks c8(a), c8(b), r11}\label{thm:versions:nocoding}
Let $\{0,1\}\subseteq Y$ and $d\ge2$. There are distinct $x_1,x_2\in\{0,1\}^d$ such that $D:=\{(x_1,0),(x_2,1)\}$ satisfies
\[
\text{(i) }\KF(D)\ge\log d-2;\qquad\text{(ii) }-\log W(D)\le2+K(\lceil\log(d+1)\rceil)+c;\qquad\text{(iii) }\LSFI(D)\le2+c,
\]
with $c$ independent of $d$. Hence $\KF(D)+\log W(D)\ge\log d-K(\lceil\log(d+1)\rceil)-c-4\to\infty$, and on these data sets $-\log W(D)\le\log\KF(D)+O(\log\log\KF(D))$.
\end{theorem}

(i) is the colouring argument with $\ell=\ell_d$, (ii) is \cref{prop:versions:tape}(b), (iii) the fair-coin product law. So a c.e.\ set of programs, $\{e:\varphi_e\supseteq D\}$, can have algorithmic probability $2^{-O(\log\log d)}$ while its simplest member has complexity $\log d-O(1)$ (related literature: \cref{app:versions:nocoding}).

\begin{corollary}[Two-point rate]\status{proved}\src{foundations Cor 4.8}\label{cor:versions:twopoint}
Let $\rho_K(d)$ be the largest value of $\KF(D)-\LSFI(D)$ over two-point data sets $D$ with binary labels and $d$-bit inputs. Then $\log d-O(1)\le\rho_K(d)\le\log d+K(\lceil\log(d+2)\rceil)+O(1)=\log d+O(\log\log d)$.
\end{corollary}

\begin{proposition}[Covering bound for the single shortest program]\status{proved; computed}\src{foundations Prop 4.9 (its referee's Prop A3, sharpened); checks c8(c,d), r11(d)}\label{prop:versions:covering}
For every $D$ with labels in $\{0,1\}$, $|D|=k$ and $\dom D\subseteq\{0,1\}^{\le d}$,
\[
\KF(D)\ \le\ k+\log\big(k(d+2)\big)+K\big(\lceil\log(d+2)\rceil,k\big)+O(1).
\]
Consequently, with $\delta_D:=k-K(D\mid\dom D)$ (the deficiency of randomness of the labels given the inputs), $\KF(D)\le\LSFI(D)+\delta_D+\log(k(d+2))+K(\lceil\log(d+2)\rceil,k)+O(1)$.
\end{proposition}

(About $2^kk(d+2)\ln2$ random truth tables cover all such data sets; a program indexes the first covering family.) So a logarithmic cost is real for the single shortest program, while $W$ pays no factor depending on $|D|$.

\begin{remark}[Witnesses of large gaps cannot be computable]\status{proved}\src{foundations Rem 4.10}\label{rem:versions:noncomputable}
If $k\mapsto D_k$ is computable, then $\KF(D_k)\le K(k)+O(1)$, so the five quantities of \cref{thm:versions:chain}, all between $2^{-\KF(D_k)-O(1)}$ and $1$, agree on $D_k$ up to $K(k)+O(1)$ bits; if $D_d$ has $d$-bit inputs and $d\mapsto D_d$ is computable, they agree up to $O(1)$. So the witnesses of \cref{prop:versions:Wgap,thm:versions:nocoding} and \cref{thm:setting:weakvsmachine}(iii) are non-constructive by necessity; \cref{thm:setting:weakvsmachine}(iv) is an explicit family with a linear gap on average.
\end{remark}

\paragraph{Open question.}\label{q:versions:explicit} Is there a computable family $k\mapsto D_k$ along which some gap of \cref{tab:versions:compare} tends to infinity (necessarily no faster than $K(k)$)?

\begin{remark}[What the gaps mean for prediction]\status{proved}\src{foundations \S4 ``What the differences mean for prediction''}\label{rem:versions:prediction}
(1) \emph{Gaps are cumulative losses:} by \cref{prop:setting:order}, a gap of $g$ bits between two of $W,\Mprod,\MF,\Mw$ on $D$ is a difference of $g$ bits in total log loss on $D$ in every order (semimeasure convention, up to the constants $\log W(\emptyset)$, $\log\Mprod(\emptyset)$); single steps can differ more or less.
(2) \emph{Random tape or not:} $W$ predicts as well as SFI up to $\Kpad(L_{1/2}(\MF,D))+O(1)$ bits in total, and up to $K(\lceil\log(d+1)\rceil)+O(1)$ bits on fair-coin labels however many are predicted. So whether H\"anni's random program has a random tape matters little, unless SFI's explanation of the data uses a number of tape bits with no simple bound; likewise private versus shared randomness, as $\Mprod$ lies between $W$ and $\MF$.
(3) \emph{The shortest program:} $\KF(D)$ is a description length (MDL with one hypothesis; the Occam learner of \cref{prop:learn:occam}), not a predictor's loss. It can exceed SFI's loss by $\log d-O(1)$ bits on two points and by about $\log(k(d+2))$ on $k$ incompressible binary labels; for deterministic targets it has SFI's learning rate (\cref{prop:learn:occam}).
(4) $\Mw$ can beat $\MF$ by linearly many bits on data no random function produces (blocks of $\nutri$), but has no posterior over functions.
(5) For a deterministic computable $f$ all five are at least $2^{-K(f)-O(1)}$ on its data, so the basic learning bounds (\cref{thm:learn:dominance}(b), \cref{cor:learn:mistakes}, \cref{thm:learn:batch}(a)) hold for all.
(6) Fresh label noise with repeated inputs separates $\Mpq$ from all set-based objects: finite against infinite loss (\cref{prop:learn:freshnoise,prop:learn:perquery}).
(7) On a single query SFI is the conditional universal distribution, $2^{-K(y\mid x)\pm O(1)}$; it commits to where the output ends, unlike sequence induction on an undelimited output, and with an end marker the two agree up to constants (\cref{prop:versions:structured}).
\end{remark}

\begin{remark}[Answer to H\"anni's request]\status{proved}\src{foundations \S4 ``Answer to the question in H\"anni's draft''; claims F16, F17}\label{rem:versions:answer}
There are three objects, and no two of them agree up to a constant.
\begin{itemize}
\item The \emph{single shortest program} $\KF$ falls short of $\MF$ by $\log d\pm O(\log\log d)$ bits at worst on two points with $d$-bit inputs (\cref{cor:versions:twopoint}): one program must be chosen, and short programs cannot separate every pair of inputs, while a random tape separates any two at constant cost.
\item The \emph{program mixture} $W$ falls short of $\MF$ unboundedly (\cref{prop:versions:Wgap}), but only by the cost of naming a padded tape length, since the $2^L$ programs storing $L$-bit tape prefixes together weigh $2^{-K(L)}$ (\cref{prop:versions:tape}).
\item There is \emph{no coding theorem for function programs}: $-\log W$ and $\KF$ differ unboundedly, not even within a constant factor (\cref{thm:versions:nocoding}); the same holds for history-aware SFI (\cref{rem:versions:hist}).
\end{itemize}
The coding theorem, which H\"anni's draft names as the tool, gives agreement of all five versions up to $K(\dom D)+O(1)$ bits (\cref{thm:versions:chain}), not up to a constant. For deterministic computable targets all have loss at most $K(f)+O(1)$ on the data of $f$.
\end{remark}

\paragraph{Open question.}\label{q:versions:open} Is $\log(\MF(D)/W(D))=O(K(d,|D|))$ for inputs of length at most $d$, or even $\Kpad$ of a computable function of $(d,|D|)$? Proved: $\min\{\Kpad(L_{1/2}(\MF,D)),K(\dom D)\}+O(1)$, and $\Kpad(2^{d+1})+O(1)$ against fair coins; $L_{1/2}$ is not computably bounded in $(d,|D|)$. Also open: the exact rate of the two-point gap of $W$ (between $\log(1/\Omtail{\ell_d})$ and $\Kpad(2^{d+1})$), a lower rate for $\MF/\Mprod$, and how close $\log(\Mw/\MF)$ comes to $K(\dom D)$.
