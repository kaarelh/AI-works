\section{Proofs for \texorpdfstring{\cref{sec:algstat}}{Section \ref*{sec:algstat}} (algorithmic statistics)}\label{app:algstat}

\subsection{Steeper codes over sets and products, and sets from products}\label{app:algstat:coarse}

\begin{proposition}[Steeper two-part codes over sets]\status{proved; a natural reading refuted}\src{steeper Prop 3.5}\label{prop:algstat:set}
Let $\lambda>1$ and let $A_\lambda$ minimise $\lambda K(A)+\log|A|$ over $A\ni z$. Then, for every $b\ge0$,
\[
K(A_\lambda)\le\sophset_b(z)+\frac{b+\kappa}{\lambda-1},
\]
and $A_\lambda$ is $\big(b+(\lambda-1)\sophset_b(z)\big)$-sufficient.
\end{proposition}

\begin{proof}
Let $A_b$ attain $\sophset_b(z)$. Then $\lambda K(A_\lambda)+\log|A_\lambda|\le\lambda K(A_b)+\log|A_b|\le(\lambda-1)\sophset_b(z)+K(z)+b$. Subtract the sufficiency line for the first claim; use $K(A_\lambda)\le\lambda K(A_\lambda)$ for the second. A minimiser exists because only finitely many $A$ have $K(A)\le L$.
\end{proof}

\begin{remark}[$\lambda$ near $1$]\label{rem:algstat:near1}
The bounds do not show that $\lambda$ slightly above $1$ selects an approximately minimal sufficient statistic: as $\lambda\to1$ the length bound becomes vacuous, and as $\lambda$ grows the sufficiency deficiency grows. The proposition is useful at $\lambda-1=\Theta(1)$ when the sophistication is small.
\end{remark}

\begin{proposition}[Steeper two-part codes over products]\status{proved}\src{steeper Prop 3.6}\label{prop:algstat:product}
Every $\lambda$-MAP over $\Hcl{\times}$ for $D$ has length at most $\sophx_b(D)+(b+\kappa)/(\lambda-1)$ and is $\big(b+(\lambda-1)\sophx_b(D)\big)$-product-sufficient.
\end{proposition}

\begin{proof}
As for \cref{prop:algstat:set}, with \cref{lem:setup:coding} as the sufficiency line. The same trade-off applies (\cref{rem:algstat:near1}).
\end{proof}

\begin{proposition}[Sets from products]\status{proved for computable hypotheses; that it is the standard set--distribution translation is believed (known; unverified)}\src{steeper Prop 3.7}\label{prop:algstat:coarse}
Let $\pi$ be product-$b$-sufficient for $D$, and let $\nu_\pi$ be a computable conditional measure with rational values. Put $r_D:=\lceil-\log\nu_\pi(D)\rceil$ (of order $n$) and $A:=\{y'\in Y^{\dom D}:\nu_\pi(\dom D\mapsto y')\ge2^{-r_D}\}$. Then $y_D\in A$, $K(A\mid\dom D)\le\ell(\pi)+2\log(r_D+1)+\kappa$, and $A$ is $(b+2\log(r_D+1)+\kappa)$-sufficient conditionally on $\dom D$: $K(A\mid\dom D)+\log|A|\le K(y_D\mid\dom D)+b+2\log(r_D+1)+\kappa$. Consequently $\sophset_{b'}(y_D\mid\dom D)\le\sophx_b(D)+2\log(r_D+1)+\kappa$ with $b'=b+2\log(r_D+1)+\kappa$, provided a computable rational product hypothesis attains $\sophx_b(D)$. Against the unconditional set sophistication of the string $y_D$, add $K(\dom D)$.
\end{proposition}

\begin{proof}
Since $Y$ is finite and $\nu_\pi$ is computable with rational values, the test $\nu_\pi(\dom D\mapsto y')\ge2^{-r_D}$ is decidable, so $A$ is computable from $(\pi,\dom D,r_D)$. By (F1), (F3) and (F5), $K(A\mid\dom D)\le\ell(\pi)+2\log(r_D+1)+\kappa$. It contains $y_D$ because $\nu_\pi(D)\ge2^{-r_D}$. Since $\sum_{y'}\nu_\pi(\dom D\mapsto y')\le1$, $|A|\le2^{r_D}$. Hence
\[
K(A\mid\dom D)+\log|A|\le\ell(\pi)+2\log(r_D+1)+\kappa+r_D\le\ell(\pi)-\log\nu_\pi(D)+1+2\log(r_D+1)+\kappa,
\]
and product-$b$-sufficiency of $\pi$ gives the claim. The consequence follows by taking $\pi$ to attain $\sophx_b(D)$. Against the unconditional set sophistication of the string $y_D$, describe $\dom D$ first, at cost $K(\dom D)$.
\end{proof}

\begin{question}[lower-semicomputable product models]\label{q:algstat:lsc}
Does \cref{prop:algstat:coarse} hold for lower-semicomputable product hypotheses, for which the set $A$ is only computably enumerable?
\end{question}

\subsection{Proof of the noisy-parity proposition}\label{app:algstat:parity}

We prove \cref{prop:algstat:parity}, in the notation of \cref{sec:algstat:parity}.

\emph{(i), injectivity and size.} If $f_{a'}\oplus\ind_{E'}=f_{a''}\oplus\ind_{E''}$ with $a'\ne a''$, then the nonzero parity $f_{a'}\oplus f_{a''}$, which has $N/2$ ones, equals $\ind_{E'}\oplus\ind_{E''}$, which has at most $2pN<N/2$ ones. So $a'=a''$ and then $E'=E''$. Hence $|A_0|=2^d\binom N{pN}=2^{d+L_E}$.

\emph{(i), complexity of $A_0$ and of $z$.} $A_0$ is computable from $d$ and $r$, so $K(A_0)\le2\log d+2\log r+\kappa$ by (F5). The table $z$ determines $a$ as the unique parity within distance $pN<N/4$ of $z$, and then $E=\{x:z(x)\ne f_a(x)\}$; $d$ is determined by $|z|=N$. So $K(z)\ge K(a,E)-\kappa$, and by symmetry of information (F4), $K(a,E)=K(a)+K(E\mid a,K(a))+O(1)\ge d+L_E-\delta_E-\kappa$. Therefore $K(A_0)+\log|A_0|\le K(z)+\delta_E+2\log d+2\log r+\kappa$.

\emph{(i), flatness.} For an integer $0\le v\le d$ let $A_v$ be the subset of $A_0$ whose parities $a'$ agree with $a$ on the first $d-v$ coordinates. Then $\log|A_v|=v+L_E$ and $K(A_v)\le d-v+2\log d+2\log(v+1)+2\log r+\kappa$, since $A_v$ is computable from $d$, $r$, $v$ and those $d-v$ bits. So every $A_v$ has two-part length $d+L_E+O(\log d)$, within $\delta_E+O(\log d)$ of the sufficiency line. With the sufficiency line itself as the lower bound, the structure function $h_z$ lies within $\delta_E+O(\log d)$ of the sufficiency line for every $\alpha$ from $O(\log d)$ up to $d+O(\log d)$.

\emph{(i), $\lambda$-optimal set models.} By \cref{prop:algstat:set} with $b=\delta_E+2\log d+2\log r+\kappa$, $K(A_\lambda)\le2\log d+2\log r+\kappa+(b+\kappa)/(\lambda-1)$. By (F3), $K(a)-K(a\mid A_\lambda)\le K(A_\lambda)+\kappa$, which is the claim.

\emph{(ii), upper bound.} Since $|E|=pN$ exactly, $-\log\nu_{\pi_g}(D)=NH(p)$, and $NH(p)\le L_E+\log(N+1)$ by the standard bounds on binomial coefficients \citep{cover2006elements}. Also $K(y_D\mid\dom D)\ge K(z\mid d)-\kappa\ge K(z)-2\log d-\kappa\ge d+L_E-\delta_E-2\log d-\kappa$ by (i). So
\[
\ell(\pi_g)-\log\nu_{\pi_g}(D)\le d+\kappa_g+L_E+\log(N+1)\le K(y_D\mid\dom D)+\delta_E+2\log d+\kappa_g+\log(N+1)+\kappa,
\]
and $\pi_g$ is product-$b$-sufficient for every $b$ in the range. Hence $\sophx_b(D)\le\ell(\pi_g)\le d+\kappa_g$.

\emph{(ii), lower bound.} The memoriser gives $K(z\mid d)\le d+2\log N+L_E+\kappa\le d+NH(p)+2\log N+\kappa$: describe $a$, then $|E|$, then the index of $E$ among the sets of that size. Let $\pi$ be product-$b$-sufficient. If $\lev(\pi)\ge2$, then $R(\pi)\ge G_1$ by \cref{lem:teacher:floor} (with $B=\infty$), so
\[
\ell(\pi)+NR(\pi)\ge NG_1>d+NH(p)+2\log N+\kappa+b\ge K(y_D\mid\dom D)+b,
\]
by the upper end of the range of $b$, which contradicts sufficiency. So $\lev(\pi)=1$, and (D1) of \cref{lem:teacher:twopart} with $s(1)=4$ gives $\ell(\pi)\ge d-4-\Lambda_1$.

\emph{(ii), the kink.} If $\ell(\pi)<d-4-\Lambda_1$, then $\lev(\pi)\ne1$ by (D1), so $R(\pi)\ge G_1$ and $-\log\nu_\pi(D)=NR(\pi)\ge NG_1$.

\emph{(iii)} follows from (i) and (ii), and from (i) and (iv) for every $p<1/4$.

\emph{(iv).} Let $\ell(\pi)<d-s(t)-\Lambda_1$. If $\lev(\pi)=t'\le t$, then (D1) and the monotonicity of $s$ give $\ell(\pi)\ge d-s(t')-\Lambda_1\ge d-s(t)-\Lambda_1$, a contradiction; so $\lev(\pi)>t$ and $u(\pi)\le\frac12+2^{-t-1}$. \Cref{lem:teacher:sharpfloor} with $m=d$, $B=\infty$ and the information budget $I(\pi)\le\ell(\pi)+\kappa<d+\kappa$ applies with a deviation term at most $\gamma_d$, because the deviation term is increasing in the budget. It gives
\[
\Ex_x\nu_\pi(y^*(x)\mid x)\le p+(1-2p)\Big(\frac12+2^{-t-1}\Big)+p\big(2\gamma_d+2^{1-g}\big)=\frac12\Big(1+(1-2p)2^{-t}+2p\big(2\gamma_d+2^{1-g}\big)\Big),
\]
and Jensen's inequality gives $R(\pi)\ge F_t$. For the sophistication: a product-$b$-sufficient $\pi$ has
\[
\ell(\pi)+NR(\pi)\le K(y_D\mid\dom D)+b\le K(z\mid d)+\kappa+b\le\ell(\pi_b)+2\kappa+b\le d+\lceil L_E\rceil+\kappa_b+2\kappa+b,
\]
so $NR(\pi)<NF_t$ in the stated range of $b$, and the kink forces $\ell(\pi)\ge d-s(t)-\Lambda_1$. The upper bound and its range of $b$ are those of (ii), whose proof uses only minimum-distance decoding, valid for $p<1/4$, and the good model, whose description of $p$ costs $K(p)$. For the limit: $j_d=d+\delta_E+O(\log d)+O(g)$ while $pN=p2^d$, so $\gamma_d\to0$ when $\delta_E=O(d)$; taking $g\to\infty$ slowly, $F_t\to1-\log(1+(1-2p)2^{-t})$, which tends to $1>H(p)$ as $t\to\infty$, so some fixed $t_p$ has $F_{t_p}>H(p)$ for all large $d$. \qed

\subsection{Numbers for the noisy parity at every flip rate below 1/4}\label{app:algstat:numbers}

\Cref{tab:algstat:numbers} evaluates \cref{prop:algstat:parity}(iv) for four dyadic flip rates of small complexity ($K(p)\le2\log(h'+1)+2\log(e'+1)+\kappa$ for $p=h'/2^{e'}$), with $\delta_E=0$, $t=2$ (so $s(t)=7.2$) and the least $g$ for which the range of $b$ is nonempty. A direct computation (code in the accompanying repository) produced the values, with every occurrence of $\kappa$ set to $0$ or to $30$ (so the constant in $j_d$ is $2\kappa$). The rows with $\kappa=0$ are illustrations, not results about a particular reference machine; the rows with $\kappa=30$ show how the bounds degrade. For this choice of $g$, at $d\ge30$, $F_2-H(p)$ lies between $0.016$ (for $p=15/64$) and $0.064$ (for $p=1/8$). For comparison, the floor of \cref{lem:teacher:floor} at large level, $1-\log(1+2p)$, exceeds $H(p)$ only for $p<0.1525$, and with it no admissible $t$ exists for $p\ge3/16$ at these sizes.
