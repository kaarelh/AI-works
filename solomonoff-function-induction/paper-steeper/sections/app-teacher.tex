\section{Proofs for \texorpdfstring{\cref{sec:teacher}}{Section \ref*{sec:teacher}} (the teacher's mistakes)}\label{app:teacher}

Throughout, the setting is \cref{def:teacher:instance}, and (F1)--(F5) are the standard facts of \cref{app:setup:facts}. For $z\in\{0,1\}^m$ and $w\in\{0,1\}^{d-m}$ we write $(z,w)$ for the input with prefix $z$; $w$ is uniform on $\{0,1\}^{d-m}$ in every expectation $\Ex_w$. Since $a\in\{0,1\}^d$, the length $d$ can be computed from $a$. In the proofs we write $\tau_B:=2^{1-B}$ and $\beta_B:=2^m\log(1+\tau_B)$, both $0$ if $B=\infty$, so that $G_j=1-\log(1+2^{-j}+2p+\tau_B)$ and $\Lambda_1=2\log d+2\log m+2\log B\cdot\ind[B<\infty]+\beta_B+\kappa$; and $\Gamma_g:=\delta_S+\Delta_\ell+\Lambda_3+1+2\log m+2\log s_0+g+2\log g+\kappa$, so that $j_g$ in \cref{thm:teacher:main}(e) is the least integer $j\ge0$ with $j-2\log(j+1)>\Gamma_g$.

\subsection{List decoding}\label{app:teacher:list}

\begin{proof}[Proof of \cref{lem:teacher:list}]
Write $h_y(x):=\nu_\pi(y\mid x)$, $s_h:=h_0-h_1\in[-1,1]$, $\chi_{a'}(x):=(-1)^{\langle a',x\rangle}$ and $\hat s(a'):=\Ex_x[s_h\chi_{a'}]$. Since $\nu_\pi(f_{a'}(x)\mid x)=\frac12(h_0+h_1)+\frac12\chi_{a'}(x)s_h(x)$, the agreement with $f_{a'}$ is
\[
u(\pi;a'):=\Ex_x\nu_\pi(f_{a'}(x)\mid x)=\tfrac12\big(\Ex_x[h_0+h_1]+\hat s(a')\big)\le\tfrac12\big(1+\hat s(a')\big).
\]
The characters are orthonormal under the uniform law on $\{0,1\}^d$, so Parseval's identity gives $\sum_{a'}\hat s(a')^2=\Ex_xs_h^2\le1$. Hence $A_t:=\{a':u(\pi;a')>\theta_t\}\subseteq\{a':\hat s(a')>2^{-t}\}$ has fewer than $2^{2t}$ elements. This is the list-size bound for the Hadamard code \citep{goldreich1989hard,kushilevitz1993learning}. The value $u(\pi;a')$ is a finite average of lower semicomputable values, so it is lower semicomputable uniformly in $(\pi,a',d)$, and $A_t$ is computably enumerable given $(\pi,d,t)$. If $\lev(\pi)=t$, then $a\in A_t$. So $a$ is described, given $\pi$, by $d$, by $t$, and by its index in the enumeration of $A_t$, which takes $2t$ bits given $t$. By (F5), $K(a\mid\pi)\le2t+2\log(t+1)+2\log d+\kappa=s(t)+2\log d+\kappa$.
\end{proof}

The argument uses the lower-semicomputable agreement $u$ rather than the correlation $\hat s$, which is not computable for semimeasures; this is what extends the computable-hypothesis version of the lemma in the earlier report \citep{claude2026whatfollows} to all product hypotheses. A direct computation (code in the accompanying repository) on $400$ random semimeasure hypotheses at $d=7,8$ confirms the identity for $u$, the list bound and the risk floor.

\subsection{The risk floors, the coding lemma and the two-part bounds}\label{app:teacher:lemmas}

\begin{proof}[Proof of \cref{lem:teacher:floor}]
Since $\min\{B,-\log v\}=-\log\max\{v,2^{-B}\}$ and $-\log$ is convex, Jensen's inequality gives
\[
R_B(\pi)\ge-\log\Ex_x\max\{\nu_\pi(y^*(x)\mid x),2^{-B}\}\ge-\log\big(\Ex_x\nu_\pi(y^*(x)\mid x)+2^{-B}\big).
\]
Off $E$, $y^*=f_a$; on $E$, $\nu_\pi(y^*\mid x)\le1$, and $\Pr[E]=p$. So $\Ex_x\nu_\pi(y^*(x)\mid x)\le u(\pi)+p$. If $\lev(\pi)=t\ge2$, then $u(\pi)\le\theta_{t-1}=\frac12+2^{-t}$, and $\frac12+2^{-t}+p+2^{-B}=\frac12\big(1+2^{-(t-1)}+2p+\tau_B\big)$, which gives $R_B\ge G_{t-1}$. If $\lev(\pi)=\infty$, then $u\le\frac12$ and $R_B\ge G_\infty$.
\end{proof}

\Cref{lem:teacher:floor} is tight as a function of $u$ alone: a hypothesis that gives the teacher's label probability $1$ on $E$ and is uniform elsewhere has $u=\frac{1-p}2$ and $\Ex_x\nu(y^*\mid x)=\frac12+\frac p2=u+p$. Such a hypothesis knows $S$. Removing the additive $p$ therefore needs the randomness of $S$ relative to the hypothesis, which the next lemma uses. It is the semimeasure version of a lemma of the earlier report, which handled computable hypotheses through their exact correlation; for semimeasures only lower-semicomputable quantities are available. The lemma is used for the kink of \cref{prop:algstat:parity} at every flip rate below $1/4$, not for \cref{thm:teacher:main} (see \cref{rem:limits:method}).

\begin{lemma}[Randomness of the error rule sharpens the floor]\status{proved; computed}\src{steeper Lemma 2.2c (T5 Lemma 2.5c for semimeasures); check c7(C)}\label{lem:teacher:sharpfloor}
For every program $\pi$, every integer $g\ge1$ and every $B\in[\log(1/p),\infty]$,
\[
R_B(\pi)\ \ge\ -\log\Big(p+(1-2p)\,u(\pi)+p\big(2\gamma(\pi)+2^{1-g}\big)+2^{-B}\Big),
\]
where $\gamma(\pi):=\sqrt{j\ln2/(2s_0)}$ and $j$ is the least integer with $j-2\log(j+1)>\delta_S+I(\pi)+1+2\log m+2\log s_0+g+2\log g+\kappa$. Always $I(\pi)\le\ell(\pi)+\kappa$, so the bound also holds with $\ell(\pi)$ in place of $I(\pi)$. The proof does not use $p=2^{-r}$; it holds for every rule size $s_0$, with $p:=s_0/2^m$.
\end{lemma}

\begin{proof}
For $z\in\{0,1\}^m$ let $A(z):=\Ex_w\nu_\pi(f_a(z,w)\mid(z,w))$ and $V(z):=\Ex_w\nu_\pi(f_a(z,w)\oplus1\mid(z,w))$. Then $A+V\le1$, $u=2^{-m}\sum_zA(z)$, and $\bar v:=2^{-m}\sum_zV(z)\le1-u$. Off $E$ the teacher's label is $f_a$ and on $E$ it is $f_a\oplus1$, so
\begin{equation}\label{eq:teacher:identity}
\Ex_x\nu_\pi(y^*(x)\mid x)=2^{-m}\Big[\sum_{z\notin S}A(z)+\sum_{z\in S}V(z)\Big]=u-p\,\mathrm{avg}_SA+p\,\mathrm{avg}_SV.
\end{equation}

\emph{Information budget.} By definition $K(S\mid a,\pi)=B_S-I(\pi)=L_S-\delta_S-I(\pi)$. By (F3), $B_S=K(S\mid a,K(a))\le K(\pi\mid a,K(a))+K(S\mid a,\pi)+\kappa$, so $I(\pi)\le K(\pi\mid a,K(a))+\kappa\le\ell(\pi)+\kappa$ by (F1).

\emph{Coding step.} Suppose $\mathcal B$ is a set of $s_0$-subsets of $\{0,1\}^m$ with $|\mathcal B|\le2^{L_S-j}$ that contains $S$ and is computably enumerable given $(\pi,a,m,s_0)$ and a parameter $\theta$ with $K(\theta)\le g+2\log g+\kappa$. Then $S$ is described by $j$, $m$, $s_0$, $\theta$ and its index among the first $2^{\lceil L_S\rceil-j}$ enumerated elements, so $K(S\mid a,\pi)\le\lceil L_S\rceil-j+2\log(j+1)+2\log m+2\log s_0+g+2\log g+\kappa$. With the budget this gives $j-2\log(j+1)\le\delta_S+I(\pi)+1+2\log m+2\log s_0+g+2\log g+\kappa$, which the choice of $j$ excludes. So no such $\mathcal B$ exists.

\emph{The $V$ side.} If $1-u+\gamma\ge1$, then $\mathrm{avg}_SV\le1\le1-u+\gamma$. Otherwise let $\theta\le1$ be the least multiple of $2^{-g}$ with $\theta\ge1-u+\gamma$; then $K(\theta)\le g+2\log g+\kappa$ and $\theta\ge\bar v+\gamma$. If $\mathrm{avg}_SV>\theta$, the set $\mathcal B:=\{S':|S'|=s_0,\ \mathrm{avg}_{S'}V>\theta\}$ contains $S$ and is computably enumerable, because $V$ is lower semicomputable. For $S'$ uniform among the $s_0$-subsets, Hoeffding's inequality for sampling without replacement \citep{hoeffding1963probability} gives $\Pr[\mathrm{avg}_{S'}V-\bar v\ge\gamma]\le e^{-2s_0\gamma^2}=2^{-j}$, so $|\mathcal B|\le2^{L_S-j}$, which the coding step excludes. Hence $\mathrm{avg}_SV\le1-u+\gamma+2^{-g}$.

\emph{The $A$ side.} $A$ is only lower semicomputable, so a set of the form ``$\mathrm{avg}_{S'}A$ is small'' need not be enumerable; we use a finite stage. Let $\nu_s\uparrow\nu_\pi$ be the stage-$s$ lower approximations, $A_s(z):=\Ex_w\nu_s(f_a(z,w)\mid(z,w))$ and $u_s:=2^{-m}\sum_zA_s(z)\uparrow u$. If $u\le2^{-g}$, then $\mathrm{avg}_SA\ge0>u-2^{-g}-\gamma$. Otherwise let $\theta'$ be the largest multiple of $2^{-g}$ strictly below $u$, so $\theta'\ge u-2^{-g}$ and $K(\theta')\le g+2\log g+\kappa$, and let $s^*$ be the first stage with $u_{s^*}>\theta'$. Then $A_{s^*}$ is computable from $(\pi,a,m,\theta')$. If $\mathrm{avg}_SA_{s^*}<u_{s^*}-\gamma$, the set $\mathcal B:=\{S':|S'|=s_0,\ \mathrm{avg}_{S'}A_{s^*}<u_{s^*}-\gamma\}$ is decidable from these data and contains $S$; the lower-tail form of the same inequality gives $|\mathcal B|\le2^{L_S-j}$, which the coding step excludes. Hence $\mathrm{avg}_SA\ge\mathrm{avg}_SA_{s^*}\ge u_{s^*}-\gamma>u-2^{-g}-\gamma$.

\emph{Combination.} By \eqref{eq:teacher:identity}, $\Ex_x\nu_\pi(y^*\mid x)\le u-p(u-2^{-g}-\gamma)+p(1-u+\gamma+2^{-g})=p+(1-2p)u+p(2\gamma+2^{1-g})$. Conclude with Jensen's inequality as in \cref{lem:teacher:floor}.
\end{proof}

A direct computation (code in the accompanying repository) on $400$ random semimeasure hypotheses at $d=8$, $m\in\{6,8\}$, a third of them correlated with $S$, confirms \eqref{eq:teacher:identity} and the form $\Ex\nu(y^*)\le p+(1-2p)u+p(\mathrm{dev}_A+\mathrm{dev}_V)$ with the actual deviations.

\begin{proof}[Proof of \cref{lem:teacher:coding}]
For $z\in\{0,1\}^m$ and $b\in\{0,1\}$ let
\[
q_z(b):=2^{-\Ex_w\min\{B,\,-\log\nu_\pi(f_a(z,w)\oplus b\mid(z,w))\}}.
\]
By Jensen's inequality for the convex function $t\mapsto2^{-t}$, $q_z(b)\le\Ex_w\max\{\nu_\pi(f_a(z,w)\oplus b\mid(z,w)),2^{-B}\}$, so $\sum_bq_z(b)\le1+2\cdot2^{-B}=1+\tau_B$. Each $q_z(b)$ is lower semicomputable from $(\pi,a,m,B)$, because the exponent is an average of upper semicomputable values. Define, on subsets $S'\subseteq\{0,1\}^m$,
\[
Q(S'):=\prod_{z\in\{0,1\}^m}\frac{q_z(\ind_{S'}(z))}{1+\tau_B},
\]
normalised factor by factor. Its total mass is $\prod_z\sum_bq_z(b)/(1+\tau_B)\le1$, so $Q$ is a lower-semicomputable semimeasure given $(\pi,a,m,B)$. Since $y^*(z,w)=f_a(z,w)\oplus\ind_S(z)$,
\[
-\log Q(S)=\sum_z\Ex_w\min\{B,-\log\nu_\pi(y^*(z,w)\mid(z,w))\}+2^m\log(1+\tau_B)=2^mR_B(\pi)+\beta_B.
\]
By (F2), $K(S\mid\pi,a,m,B)\le2^mR_B(\pi)+\beta_B+\kappa$, and by (F3) and (F5) the descriptions of $m$ and of the integer $B$ cost at most $2\log m+2\log B+\kappa$ more. For $B=\infty$ the terms in $B$ vanish.
\end{proof}

The normalisation must be per factor: with a single division by $1+\tau_B$, $Q$ is in general not a semimeasure (for a deterministic hypothesis at the smallest admissible $B$ its total mass is $1.119$).

\begin{proof}[Proof of \cref{lem:teacher:twopart}]
(D1). By (F3), (F1) and \cref{lem:teacher:list}, $d\le K(a)\le K(\pi)+K(a\mid\pi)+\kappa\le\ell(\pi)+s(t)+2\log d+\kappa$. All terms of $\Lambda_1$ other than $2\log d+\kappa$ are nonnegative.

(D2). By symmetry of information (F4), $K(a,S)=K(a)+K(S\mid a,K(a))+O(1)\ge d+B_S-\kappa$. By (F3), $K(a,S)\le K(\pi)+K(a\mid\pi)+K(S\mid a,\pi)+\kappa$. Insert (F1), \cref{lem:teacher:list} and \cref{lem:teacher:coding}: $d+B_S\le\ell(\pi)+s(t)+2\log d+2^mR_B(\pi)+\beta_B+2\log m+2\log B\cdot\ind[B<\infty]+\kappa$.
\end{proof}

\subsection{Proof of the theorem}\label{app:teacher:main}

\begin{proof}[Proof of \cref{thm:teacher:main}]
\emph{(o).} If $\lev(\pi)=1$, then (D1) and (D2) with $s(1)=4$ give $\ell(\pi)\ge d-4-\Lambda_1+\max\{0,B_S-2^mR_B(\pi)\}$; multiply by $c$ and add $R_B(\pi)$. If $\lev(\pi)=t\in[2,\infty)$, then (D1), $\ell\ge0$ and \cref{lem:teacher:floor} give
\[
J^B_c(\pi)\ge c\,(d-s(t)-\Lambda_1)^++G_{t-1}\ge\varphi(t):=c\,(d-4t-\Lambda_1)^++G_{t-1},
\]
using $s(t)\le4t$. Extend $\varphi$ to real $t\ge2$. If $t^*\ge2$: on $[2,t^*]$ the first term is linear and $G_{t-1}$ is concave, so $\varphi\ge\min\{\varphi(2),\varphi(t^*)\}=\min\{c(d-8-\Lambda_1)+G_1,\,G_{t^*-1}\}$; for $t>t^*$, $\varphi(t)\ge G_{t-1}\ge G_{t^*-1}$ because $G$ is increasing. If $t^*<2$: $\varphi(t)\ge G_{t-1}\ge G_1$. If $\lev(\pi)=\infty$: $J^B_c\ge R_B\ge G_\infty\ge G_j$ for every $j$, so $J^B_c\ge\Phi_2(c)$ in both cases. ($G_j$ is increasing and concave: with $v=2^{-j}$, $\mathrm dG/\mathrm dj=v/(1+v+2p+\tau_B)>0$ and $\mathrm d^2G/\mathrm dj^2=-\ln2\cdot v(1+2p+\tau_B)/(1+v+2p+\tau_B)^2<0$.)

\emph{(i).} When $c2^m>1$, $\psi_c$ has slope $1-c2^m<0$ on $[0,B_S/2^m]$ and slope $1$ after it, so $\min_{R\ge0}\psi_c(R)=\psi_c(B_S/2^m)=B_S/2^m$, and every program of level $1$ has $J^B_c\ge\Phi_1(c)$. With (o), $\Phi^B(c)\ge\min\{\Phi_1(c),\Phi_2(c)\}$, and $\Phi\ge\Phi^B$ because $R\ge R_B$. Since $c\le c_{\rm hi}$, $J_g(c)\le\Phi_2(c)$, so
\[
J_c(\pi_g)-\Phi(c)\le J_g(c)-\min\{\Phi_1,\Phi_2\}=\max\{J_g-\Phi_1,J_g-\Phi_2\}\le\max\{J_g-\Phi_1,0\}=\Delta(c).
\]
Finally $H(p)-B_S/2^m=(2^mH(p)-L_S+\delta_S)/2^m\le(\log(2^m+1)+\delta_S)/2^m$, by the lower bound on $L_S$.

\emph{(ii).} $J^B_c(\pi)\le J_g(c)+\epsilon<\Phi_2(c)$, so $\lev(\pi)=1$ by (o). Every $\pi$ with $J_c(\pi)\le\Phi(c)+\epsilon$ qualifies, since $J^B_c\le J_c$ and $\Phi(c)\le J_c(\pi_g)\le J_g(c)$.

(a) By \cref{lem:teacher:floor}, $-\log(u+p+2^{-B})\le R_B\le J^B_c\le J_g+\epsilon$; and $\lev(\pi)=1$ gives $u>\frac34$.

(b) By (o), $c(d-4-\Lambda_1)+\psi_c(R_B)\le J_g+\epsilon$, that is, $\psi_c(R_B)\le c(4+\Lambda_1+\kappa_g)+H(p)+\epsilon$. If $R_B<B_S/2^m$, then $\psi_c(R_B)=cB_S-(c2^m-1)R_B$. Using the identity $cB_S-H(p)=(c2^m-1)H(p)-c(2^mH(p)-B_S)$,
\[
(c2^m-1)R_B\ge(c2^m-1)H(p)-c\big(2^mH(p)-B_S+4+\Lambda_1+\kappa_g\big)-\epsilon=(c2^m-1)H(p)-c\Lambda_2-\epsilon.
\]
If $R_B\ge B_S/2^m$ and $2^mH(p)-B_S>0$, then $R_B\ge H(p)-(2^mH(p)-B_S)/2^m\ge H(p)-c\Lambda_2/(c2^m-1)$, because $\Lambda_2\ge2^mH(p)-B_S$ and $c/(c2^m-1)>2^{-m}$; if $2^mH(p)-B_S\le0$ the claim is immediate.

(c) $c\,\ell=J^B_c-R_B\le c(d+\kappa_g)+H(p)+\epsilon-R_B$; apply (b) and divide by $c$.

(d) By (F3), $B_S\le K(\pi\mid a,K(a))+K(S\mid a,\pi)+\kappa$. By (F4) and (F3), $K(\pi\mid a,K(a))=K(\pi,a)-K(a)+O(1)\le K(\pi)+K(a\mid\pi)-K(a)+O(1)$. By (F1), \cref{lem:teacher:list} with $t=1$, $K(a)\ge d$ and (c), this is at most $\ell(\pi)-d+4+2\log d+\kappa\le\Delta_\ell+\Lambda_3$.

(e) Let $v(z):=\Ex_w\nu_\pi(f_a(z,w)\oplus1\mid(z,w))\in[0,1]$. Given $x\in E$, $z$ is uniform on $S$ and $y^*=f_a\oplus1$, so $e(\pi)=\mathrm{avg}_Sv$. Since $\nu_\pi$ is a semimeasure, $\bar v:=2^{-m}\sum_zv(z)\le1-u(\pi)$. If $1-u+\gamma_g\ge1$ there is nothing to prove. Otherwise let $\theta\le1$ be the least multiple of $2^{-g}$ with $\theta\ge1-u(\pi)+\gamma_g$; then $K(\theta)\le g+2\log g+\kappa$ and $\theta\ge\bar v+\gamma_g$. Suppose $e(\pi)>\theta$. The set $\mathcal B_\theta:=\{S':|S'|=s_0,\ \mathrm{avg}_{S'}v>\theta\}$ is computably enumerable given $(\pi,a,m,s_0,\theta)$, because $v$ is lower semicomputable, and it contains $S$. By Hoeffding's inequality for sampling without replacement, $|\mathcal B_\theta|\le\binom{2^m}{s_0}e^{-2s_0\gamma_g^2}=2^{L_S-j_g}$. Describing $S$ by $j_g$ and its index among the first $2^{\lceil L_S\rceil-j_g}$ enumerated elements gives $K(S\mid a,\pi)\le\lceil L_S\rceil-j_g+2\log(j_g+1)+2\log m+2\log s_0+g+2\log g+\kappa$. By (d), $K(S\mid a,\pi)\ge L_S-\delta_S-\Delta_\ell-\Lambda_3$. Together, $j_g-2\log(j_g+1)\le\Gamma_g$, contradicting the choice of $j_g$. So $e(\pi)\le\theta\le1-u(\pi)+\gamma_g+2^{-g}$.

\emph{(iii).} $\lev(\pi)=1$ follows from (o) as in (ii). For $c2^m<1$ and $R\ge0$, $\psi_c(R)\ge cB_S+(1-c2^m)\min\{R,B_S/2^m\}$: for $R\le B_S/2^m$ this is an equality, and for $R\ge B_S/2^m$, $\psi_c(R)=R\ge B_S/2^m=cB_S+(1-c2^m)B_S/2^m$. Compare with $J^B_c\le J_b(c)+\epsilon$ and (o), and use $L_S=B_S+\delta_S$:
\[
(1-c2^m)\min\{R_B,B_S/2^m\}\le c\big(\lceil L_S\rceil+\kappa_b-B_S+4+\Lambda_1\big)+\epsilon=\varrho.
\]
If $\varrho<(1-c2^m)B_S/2^m$, the minimum cannot be $B_S/2^m$, so $R_B\le\varrho/(1-c2^m)$. For the error label, $p\,\Ex[\min\{B,-\log\nu_\pi(y^*\mid x)\}\mid x\in E]\le R_B$, and by Jensen's inequality
\[
\Ex[\nu_\pi(y^*\mid x)\mid E]\ge\Ex[\max\{\nu_\pi(y^*\mid x),2^{-B}\}\mid E]-2^{-B}\ge2^{-R_B/p}-2^{-B}.
\]

\emph{(iv).} By (D1) with $t=1$ and $R(\pi)\ge H(p)$, $J_c(\pi)\ge c(d-4-\Lambda_1)+H(p)>c(d-4-\Lambda_1)+c(\lceil L_S\rceil+\kappa_b+4+\Lambda_1)=c(d+\lceil L_S\rceil+\kappa_b)\ge J_c(\pi_b)$.
\end{proof}

\subsection{The full truth table}\label{app:teacher:fulltable}

\begin{proof}[Proof of \cref{cor:teacher:fulltable}]
With $m=d$ the error set is $E=S$ and $2^m=N$. On the whole truth table the empirical loss equals $N$ times the population risk under uniform inputs, so $J_\lambda(\pi;D)=\lambda\ell(\pi)+NR(\pi)=N\,J_{\lambda/N}(\pi)$, and a $\lambda$-MAP minimises $J_{\lambda/N}$ exactly.

(i) Put $c=\lambda/N$. Then $c2^m=\lambda>1$, and the hypothesis $J_g(c)<\Phi_2(c)$ holds for every $c<c_{\rm hi}$, by the form of $\Phi_2$. A $\lambda$-MAP has $J_c(\pi)=\Phi(c)$, so \cref{thm:teacher:main}(ii) applies with $\epsilon=0$, $B=\infty$. Substituting $c2^m-1=\lambda-1$ and $c/(c2^m-1)=\lambda/(N(\lambda-1))$ gives the stated forms of (a)--(d).

(ii) At $\lambda=1$ the objective is $\ell+NR$. First, $\ell(\pi_g)+NR(\pi_g)=\ell(\pi_g)+NH(p)\le d+\kappa_g+L_E+\log(N+1)$, because $NH(p)\le L_E+\log(N+1)$. Next, $\ell(\pi_b)+NR(\pi_b)\le d+\lceil L_E\rceil+\kappa_b$. Every program of level $1$ has $\ell+NR\ge\ell+NR_B\ge d+B_E-4-\Lambda_1=d+L_E-\delta_E-4-\Lambda_1$, by (D2). Every program of level at least $2$ has $\ell+NR\ge NR\ge NG_1$ by \cref{lem:teacher:floor}, and by the hypotheses $G_1>H(p)$ and $N(G_1-H(p))>d+\kappa_g+\log(N+1)$ this exceeds $\ell(\pi_g)+NR(\pi_g)$, since $L_E\le NH(p)$; so the optimum is attained at level $1$. Here $d=\log N$, and $\kappa_g$, $\kappa_b$ and $\Lambda_1$ are $O(\log N)$, so the optimum and the two values lie within $O(\log N)+\delta_E$ of each other.
\end{proof}

\subsection{Finite samples}\label{app:teacher:sampled}

\begin{lemma}[Finite sample]\status{proved}\src{steeper Lemma 2.6 (adapts T5 Lemma 1.2)}\label{lem:teacher:finite}
Let $m\le d$ and $B<\infty$. Draw $n\le2^d$ inputs uniformly without replacement and label them by $y^*$; let $\hat J_c(\pi):=c\,\ell(\pi)+\frac1n\sum_{i\le n}-\log\nu_\pi(y^*(x_i)\mid x_i)$. Let $\pi_{\rm ref}$ be a fixed program whose per-input losses lie in $[0,A]$. With probability at least $1-\delta$, every minimiser $\hat\pi$ of $\hat J_c$ satisfies
\[
J^B_c(\hat\pi)\le J_c(\pi_{\rm ref})+\epsilon_1+B\sqrt{\frac{\hat L\ln2+\ln(2/\delta)}{2n}},\qquad \epsilon_1:=A\sqrt{\frac{\ln(2/\delta)}{2n}},\quad\hat L:=\frac{J_c(\pi_{\rm ref})+\epsilon_1}{c}.
\]
\end{lemma}

\begin{proof}
Each sampled input is marginally uniform. By Hoeffding's inequality for sampling without replacement \citep{hoeffding1963probability}, with probability at least $1-\delta/2$, $\hat J_c(\pi_{\rm ref})\le J_c(\pi_{\rm ref})+\epsilon_1$. For each program $\pi$ the truncated losses lie in $[0,B]$; apply the same inequality with confidence $(\delta/2)2^{-\ell(\pi)}$ and take a union bound, which costs at most $\delta/2$ by Kraft's inequality. With probability at least $1-\delta/2$, simultaneously for all $\pi$,
\[
\hat J_c(\pi)\ge c\,\ell(\pi)+\frac1n\sum_{i\le n}\min\{B,-\log\nu_\pi(y^*(x_i)\mid x_i)\}\ge J^B_c(\pi)-B\sqrt{\frac{\ell(\pi)\ln2+\ln(2/\delta)}{2n}}.
\]
On both events, a minimiser satisfies $c\,\ell(\hat\pi)\le\hat J_c(\hat\pi)\le\hat J_c(\pi_{\rm ref})\le J_c(\pi_{\rm ref})+\epsilon_1$, so $\ell(\hat\pi)\le\hat L$, and the two displays combine to the claim.
\end{proof}

The bound is one-sided and uses truncation, so no noise floor on the hypotheses is needed; this adapts the empirical-selection lemma of the earlier report. The value of $c$ is fixed for the given $n$, so the lemma also applies when $\lambda=\lambda_n$ follows any schedule; \cref{prop:limits:plain}(b) uses this.

\begin{proof}[Proof of \cref{cor:teacher:sampled}]
A $\lambda$-MAP with $\lambda=cn$ minimises $J_\lambda/n=\hat J_c$. Apply \cref{lem:teacher:finite} with $\pi_{\rm ref}=\pi_g$, whose per-input losses $-\log(1-p)$ and $-\log p$ lie in $[0,\log(1/p)]$, so $A=\log(1/p)$. Then $J^B_c(\hat\pi)\le J_c(\pi_g)+\epsilon_n\le J_g(c)+\epsilon_n$, which is the hypothesis of \cref{thm:teacher:main}(ii) with $\epsilon=\epsilon_n$.
\end{proof}

\subsection{Concentration of the power posterior}\label{app:teacher:concentration}

The power posterior with $\lambda=cn$, not only its maximiser, concentrates on near-optimal programs. Let $\hat J_c(\pi):=c\,\ell(\pi)+\frac1n\sum_{i\le n}-\log\nu_\pi(y_i\mid x_i)$ (\cref{rem:setup:persample}) and $\hat J_g:=\hat J_c(\pi_g)$.

\begin{proposition}[The power posterior concentrates on near-minimisers]\status{proved; computed}\src{steeper Prop 2.8; checks c5(e), c7(F); referee q1(F)}\label{prop:teacher:concentration}
Let $\lambda=cn$ with $cn\ge1$, and let $D$ be any data set of size $n$. For every $\epsilon>0$,
\[
\frac{\Pi_\lambda\big(\{\pi:\hat J_c(\pi)\ge\hat J_g+\epsilon\}\,\big|\,D\big)}{\Pi_\lambda(\pi_g\mid D)}\ \le\ 2^{-n\epsilon+(\hat J_g+\epsilon)/c}.
\]
Consequently, for data sampled as in \cref{cor:teacher:sampled}, with probability at least $1-\delta$ all but this fraction of the posterior mass lies on programs that satisfy \cref{thm:teacher:main}(ii) with $\epsilon':=\epsilon+\epsilon_1+B\sqrt{(\hat L'\ln2+\ln(2/\delta))/(2n)}$ in place of $\epsilon$, where $\hat L':=(J_g(c)+\epsilon_1+\epsilon)/c$, provided $J_g(c)+\epsilon'<\Phi_2(c)$.
\end{proposition}

\begin{proof}
For every $\pi$, $2^{-\lambda\ell(\pi)}\nu_\pi(D)=2^{-n\hat J_c(\pi)}$. Write $\hat R(\pi):=\frac1n\sum_i-\log\nu_\pi(y_i\mid x_i)\ge0$ for the empirical risk. Then $n\hat J_c(\pi)-\ell(\pi)=(\lambda-1)\ell(\pi)+n\hat R(\pi)\ge(1-1/\lambda)\big(\lambda\ell(\pi)+n\hat R(\pi)\big)=(1-1/\lambda)\,n\hat J_c(\pi)$. So, on the set $\{\pi:\hat J_c(\pi)\ge\hat J_g+\epsilon\}$,
\[
2^{-n\hat J_c(\pi)}=2^{-\ell(\pi)}\,2^{-(n\hat J_c(\pi)-\ell(\pi))}\le2^{-\ell(\pi)}\,2^{-(1-1/\lambda)\,n(\hat J_g+\epsilon)},
\]
and by Kraft's inequality the sum over the set is at most $2^{-(1-1/\lambda)n(\hat J_g+\epsilon)}$. Divide by $2^{-n\hat J_g}$, the unnormalised weight of $\pi_g$, and use $n/\lambda=1/c$: the ratio is at most $2^{-n\epsilon+(\hat J_g+\epsilon)/c}$.

For the consequence, work on the event of \cref{lem:teacher:finite} with $\pi_{\rm ref}=\pi_g$ and $A=\log(1/p)$, on which $\hat J_g\le J_g(c)+\epsilon_1$ and the uniform deviation bound holds for every program. A program with $\hat J_c(\pi)<\hat J_g+\epsilon$ has $\ell(\pi)\le\hat J_c(\pi)/c<\hat L'$, hence
\[
J^B_c(\pi)\le\hat J_c(\pi)+B\sqrt{\frac{\hat L'\ln2+\ln(2/\delta)}{2n}}\le J_g(c)+\epsilon',
\]
and \cref{thm:teacher:main}(ii) applies with $\epsilon'$ when $J_g(c)+\epsilon'<\Phi_2(c)$. The bad set has posterior mass at most the ratio times $\Pi_\lambda(\pi_g\mid D)\le1$.
\end{proof}

Near-minimisers can be $\epsilon/c$ bits longer than the minimiser, which is why the length budget is $\hat L'$ and not the $\hat L$ of \cref{lem:teacher:finite}. For instance, with $\hat J_g$ replaced by its population value $J_g=0.0232$, $c=10^{-4}$, $\epsilon=10^{-3}$, $n=10^6$ and the machine constant set to $0$ (an illustration), $-n\epsilon+(J_g+\epsilon)/c=-1000+241.7=-758.3$, so the ratio is at most $2^{-758}$. At $c=10^{-4}$, $\epsilon=10^{-3}$, $n=10^{13}$, $B=30$, $\delta=0.01$ and $\kappa=0$, a direct computation (code in the accompanying repository) gives $\hat L'=241.7$ bits and $\epsilon'=1.0933\cdot10^{-3}$, and the condition $J_g+\epsilon'<\Phi_2$ holds. Two further consequences: the steeper predictive, averaged over $E$, gives the erroneous label probability at most $\sup e(\pi)+2^{-n\epsilon+(\hat J_g+\epsilon)/c}$, the supremum over the programs just described; averaged over the unseen error inputs $E\setminus\dom D$ instead, the bound is multiplied by $|E|/(|E|-n)$, which is negligible at $d=100$.
