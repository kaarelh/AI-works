\section{Kinks: the model--randomness split on one data set}\label{sec:kinks}

Besides a steeper penalty, one can look for the split between model and noise by plotting the entropy left given the hypothesis against the hypothesis length and looking for a change of slope. With shared randomness, as in the sequence case, one expects none where a random parameter of the model ends, because each of its bits pays for about one bit of data (\cref{rem:algstat:reading}).\footnote{The suggestion to look for this change of slope in function induction, and the reason why none is expected with shared randomness, first appeared in the author's note \citep{hanni2026grokking}.} We make it precise as a \emph{kink}: every complexity weight $\lambda$ between the two slopes of a kink selects the model there, up to explicit windows, and stable selection happens only at kinks. The geometry is common to both reasons for a steeper penalty: right of the kink lies sampled noise (\cref{sec:recovery}) or a teacher's error rule (\cref{sec:teacher}). Proofs are in \cref{app:kinks}.

\subsection{The product structure function}\label{sec:kinks:structure}\label{sec:algstat:product}

\begin{definition}[Structure functions and sufficiency]\src{steeper \S3.5, product versions; split/kink \S1; split/teacher \S4}\label{def:algstat:product}
A \emph{point family} is a countable set $\mathcal P$ of programs $\pi$ with integer lengths $\ell(\pi)\ge0$, finitely many of each length, and fits $L(\pi)\in[0,\infty]$, not all infinite; for example, a data-set class on one observation $o$ (\cref{def:setup:class,def:setup:data}) with $L(\pi):=L(\pi;o)$. Its \emph{structure function} is $\rho(\alpha):=\min\{L(\pi):\ell(\pi)\le\alpha\}$ ($\min\emptyset:=\infty$), and $\breve\rho$ is the greatest convex minorant of $\rho$. On a data set $D$ we write $\rho_D$, and $\rho^\times_D$ for $\Hcl{\times}$, the \emph{product structure function}; in population form, $\rho_{\rm pop}(\alpha):=\inf\{R(\pi):\ell(\pi)\le\alpha\}$, with $R$ as in \cref{rem:setup:persample}. In a data-set class, $\pi$ is \emph{$b$-sufficient} for $D$ if $\ell(\pi)+L(\pi;D)\le K(y_D\mid\dom D)+b$; in $\Hcl{\times}$ we say \emph{product-$b$-sufficient}, and $\sophx_b(D):=\min\{\ell(\pi):\pi\in\Hcl{\times}\text{ is product-$b$-sufficient for }D\}$ is the \emph{product sophistication}.
\end{definition}

\paragraph{The hull.} A \emph{$\lambda$-MAP} minimises $J_\lambda:=\lambda\ell+L$ ($\lambda>0$). Let $C$ be the hull of \cref{lem:setup:hull} for the points $(\ell(\pi),L(\pi))$, with vertices $v_i=(\alpha_i,\rho_i)$ in order of length and edge slopes $-\sigma_i$, so that $\sigma_0=\infty>\sigma_1>\sigma_2>\dots>0$ and $\sigma_{i+1}=0$ at a last vertex $v_i$. By \cref{lem:setup:hull}(v), for $\lambda$ in the \emph{$\lambda$-interval} $(\sigma_{i+1},\sigma_i)$ of $v_i$ the $\lambda$-MAPs are the programs at $v_i$. The lower boundary of $C$ is the graph of $\breve\rho$ (\cref{app:kinks:hull}), so the $\lambda$-MAPs lie where the slope of $\breve\rho$ crosses $-\lambda$.

\subsection{Kinks, the kink lemma and the split}\label{sec:kinks:lemma}

\begin{definition}[Kinks, right lines and secants]\src{split/kink Def 3.1; split/teacher Def 1.4, Remark 1.3; split/verify-teacher \S3}\label{def:kink:kinks}
Let $\mathcal P$ be a countable set of programs with lengths $\ell(\pi)\ge0$ and two fits $0\le L'(\pi)\le L(\pi)\le\infty$ ($L'=L$ on a point family). A point $(\alpha_{\rm r},L_{\rm r})\in\mathbb R^2$ is \emph{bounded} if some program $\pi_{\rm r}$ has $\ell(\pi_{\rm r})\le\alpha_{\rm r}$ and $L(\pi_{\rm r})\le L_{\rm r}$. A bounded point is a \emph{left kink of steepness $\sigma\in(0,\infty]$ and window $w\ge0$} if every $\pi$ with $\ell(\pi)<\alpha_{\rm r}-w$ has $L(\pi)\ge L_{\rm r}+\sigma(\alpha_{\rm r}-\ell(\pi))$, and a \emph{right kink of flatness $\tau\ge0$ and window $w_+\ge0$} if every $\pi$ with $\ell(\pi)>\alpha_{\rm r}+w_+$ has $L(\pi)\ge L_{\rm r}-\tau(\ell(\pi)-\alpha_{\rm r})$; both together form a \emph{$(\sigma,\tau)$-kink}. A set $\mathcal M\subseteq\mathcal P$ has a \emph{right line} $(\tau,g)$, with $\tau\ge0$ and $g\in\mathbb R$, and \emph{length floor} $\ell_{\mathcal M}\le\alpha_{\rm r}$ if every $\pi\in\mathcal M$ has $\ell(\pi)\ge\ell_{\mathcal M}$ and $L'(\pi)\ge L_{\rm r}-g-\tau(\ell(\pi)-\alpha_{\rm r})$. The \emph{right secant} on $\mathcal M$ is $s_+:=\max\{0,\sup\{(L_{\rm r}-L'(\pi))/(\ell(\pi)-\alpha_{\rm r}):\pi\in\mathcal M,\ \ell(\pi)>\alpha_{\rm r}\}\}$, and the \emph{left steepness} is $s_-:=\inf\{(L(\pi)-L_{\rm r})/(\alpha_{\rm r}-\ell(\pi)):\ell(\pi)<\alpha_{\rm r}\}$ ($\inf\emptyset:=\infty$).
\end{definition}

For $\mathcal M=\mathcal P$ and $L'=L$, a bounded point is a zero-window $(\sigma,\tau)$-kink iff $s_-\ge\sigma$ and $s_+\le\tau$; a right line $(\tau,g)$, $g\ge0$, gives right kinks of flatness $\tau'>\tau$ and window $g/(\tau'-\tau)$ (\cref{lem:kink:secant}).

For $c>0$ put $J_c:=c\,\ell+L$, $J'_c:=c\,\ell+L'$, $\Phi(c):=\inf_\pi J_c(\pi)$, the reference value $J_{\rm r}(c):=c\alpha_{\rm r}+L_{\rm r}$, and, for a partition $\mathcal P=\mathcal M\sqcup\mathcal N$, the \emph{left margin} $\eta_{\mathcal N}(c):=\inf_{\pi\in\mathcal N}J'_c(\pi)-J_{\rm r}(c)\in(-\infty,\infty]$. In the \emph{data reading} $L=L'=L(\cdot;o)$ and $c=\lambda$, in total bits; in the \emph{population reading} $L=R$, $L'\le R$ (for example the truncated risk $R_B$ of \cref{def:teacher:instance}) and $c=\lambda/n$, per sample.

\begin{lemma}[The kink lemma]\status{proved}\src{split/teacher Lemma 1.1 (final form), Remark 1.2; split/verify-teacher \S2}\label{lem:kink:main}
Let $(\alpha_{\rm r},L_{\rm r})$ be bounded, $\mathcal P=\mathcal M\sqcup\mathcal N$, $c>0$ and $\epsilon\ge0$. In (i) and (ii) let $\mathcal M$ have a right line $(\tau,g)$ with length floor $\ell_{\mathcal M}$, let $c>\tau$ and put $\Delta(c):=g+(c-\tau)(\alpha_{\rm r}-\ell_{\mathcal M})$.
\begin{enumerate}[label=(\roman*),itemsep=0pt,topsep=2pt]
\item $\inf_\pi J'_c(\pi)\ge J_{\rm r}(c)+\min\{\eta_{\mathcal N}(c),-\Delta(c)\}$. If a program of $\mathcal M$ bounds the point, then $g\ge0$ and $\Delta(c)\ge0$. If $\eta_{\mathcal N}(c)\ge0$ and $\Delta(c)\ge0$, then $J_{\rm r}(c)-\Delta(c)\le\Phi(c)\le J_{\rm r}(c)$.
\item If every $\pi\in\mathcal N$ has $J'_c(\pi)>J_{\rm r}(c)+\epsilon$ (for instance, if $\eta_{\mathcal N}(c)>\epsilon$), then every $\pi$ with $J'_c(\pi)\le J_{\rm r}(c)+\epsilon$, in particular every $\pi$ with $J_c(\pi)\le\Phi(c)+\epsilon$, lies in $\mathcal M$ and has $\ell(\pi)\le\alpha_{\rm r}+(g+\epsilon)/(c-\tau)$ and $L'(\pi)\ge L_{\rm r}-(cg+\tau\epsilon)/(c-\tau)$.
\item $\eta_{\mathcal N}$ is concave on $(0,\infty)$.
\item \emph{No right line needed:} if every $\pi\in\mathcal M$ has $\ell(\pi)\ge\ell_{\mathcal M}$, every $\pi\in\mathcal M$ with $J'_c(\pi)\le J_{\rm r}(c)+\epsilon$ has $L'(\pi)\le L_{\rm r}+\epsilon+c(\alpha_{\rm r}-\ell(\pi))\le L_{\rm r}+\epsilon+c(\alpha_{\rm r}-\ell_{\mathcal M})$.
\item \emph{Wedge criterion:} if $\sigma>0$, $\vartheta\ge0$ and every $\pi\in\mathcal N$ has $L'(\pi)\ge L_{\rm r}+\max\{\sigma(\alpha_{\rm r}-\ell(\pi)),\vartheta\}$, then $\eta_{\mathcal N}(c)\ge\vartheta(1-c/\sigma)\ge0$ for $c\in(0,\sigma]$.
\end{enumerate}
\end{lemma}

\emph{Proof idea.} On $\mathcal M$ the line gives $J'_c\ge J_{\rm r}(c)-\Delta(c)$, as $c>\tau$. A program at most $\epsilon$ worse than the bounding one avoids $\mathcal N$ by the margin, and the line bounds its length (\cref{app:kinks:lemma}).

\begin{corollary}[Kinks locate the $\lambda$-MAPs]\status{proved}\src{split/kink Thm 3.2, Remark 3.2$'$; split/verify-kink \S6}\label{cor:kink:locate}
Let $\mathcal P$ be a point family and $\lambda>0$.
(a) If $\mathcal P$ has a left kink $(\sigma,w)$ at $(\alpha_-,L_-)$ and $\lambda<\sigma$, every $\lambda$-MAP $\hat\pi$ has $\ell(\hat\pi)\ge\alpha_--w$ and $L(\hat\pi)\le L_-+\lambda(\alpha_--\ell(\hat\pi))\le L_-+\lambda w$; if also $\lambda\ge1$, then $\ell(\hat\pi)+L(\hat\pi)\le\alpha_-+L_-+(\lambda-1)w$.
(b) If $\mathcal P$ has a right kink $(\tau,w_+)$ at $(\alpha_+,L_+)$ and $\lambda>\tau$, every $\lambda$-MAP has $\ell(\hat\pi)\le\alpha_++w_+$.
(c) If $\mathcal P$ has a zero-window $(\sigma,\tau)$-kink at $(\alpha_{\rm r},L_{\rm r})$ with $\tau<\sigma$, then $\rho(\alpha_{\rm r})$ is attained at length exactly $\alpha_{\rm r}$, $(\alpha_{\rm r},\rho(\alpha_{\rm r}))$ is a vertex $v_i$ with $(\tau,\sigma)\subseteq(\sigma_{i+1},\sigma_i)$, and for $\lambda\in(\tau,\sigma)$ the $\lambda$-MAPs are exactly the programs at $v_i$. Conversely, every vertex $v_i$ is a zero-window $(\sigma_i,\sigma_{i+1})$-kink, and $\sigma_{i+1}<\sigma_i$.
\end{corollary}

\emph{Proof idea.} A $\lambda$-MAP is no worse than the bounding program; beyond the left window one loses at least $\sigma-\lambda>0$ per bit, beyond the right one gains at most $\tau<\lambda$ per bit. In (c) the $\lambda$-MAP point is fixed on $(\tau,\sigma)$, hence a vertex. No strict inequality can be dropped (\cref{app:kinks:data}).

\begin{corollary}[The split on one data set]\status{proved}\src{split/kink Cor 3.3}\label{cor:kink:split}
Let $D$ be a data set and $\mathcal C$ a hypothesis class of data-set hypotheses, with the constant $\kappa$ of the sufficiency line (\cref{lem:setup:coding}); let $\pi_s\in\mathcal C$ be $b$-sufficient for $D$, and $(\alpha_{\rm r},L_{\rm r}):=(\ell(\pi_s),L(\pi_s;D))$.
(a) Every $\pi$ has $L(\pi;D)\ge L_{\rm r}-(b+\kappa)-(\ell(\pi)-\alpha_{\rm r})$: a right line $(1,b+\kappa)$ on all programs. So for every $\tau>1$ there is a right kink of flatness $\tau$ and window $(b+\kappa)/(\tau-1)$ at $(\alpha_{\rm r},L_{\rm r})$, and for every $\lambda>1$ every $\lambda$-MAP $\hat\pi$ has $\ell(\hat\pi)\le\alpha_{\rm r}+(b+\kappa)/(\lambda-1)$ and $L(\hat\pi;D)\ge L_{\rm r}-\lambda(b+\kappa)/(\lambda-1)$.
(b) If moreover $D$ has a left kink $(\sigma,w)$ at $(\alpha_{\rm r},L_{\rm r})$, then for every $\lambda\in(1,\sigma)$ every $\lambda$-MAP $\hat\pi$ is $(b+(\lambda-1)w)$-sufficient and satisfies
\[
\alpha_{\rm r}-w\le\ell(\hat\pi)\le\alpha_{\rm r}+\frac{b+\kappa}{\lambda-1},\qquad L_{\rm r}-\frac{\lambda(b+\kappa)}{\lambda-1}\le L(\hat\pi;D)\le L_{\rm r}+\lambda w.
\]
\end{corollary}

\emph{Proof idea.} The right line is the coding bound rearranged around $\pi_s$, and (a) is \cref{lem:kink:main}(ii) with $\mathcal N=\emptyset$, $\tau=1$ and $c=\lambda$; (b) adds \cref{cor:kink:locate}(a) (\cref{app:kinks:data}).

At a program attaining $\sophx_b(D)$, (a) gives every $\lambda$-MAP over $\Hcl{\times}$ length at most $\sophx_b(D)+(b+\kappa)/(\lambda-1)$ (\cref{prop:algstat:product}), a bound vacuous as $\lambda\to1$ (\cref{rem:algstat:near1}). The content of (b) is a steepness $\sigma>1$, which the coding bound alone does not give; per-query samples have one with high probability (\cref{thm:recovery:kink}, where the Ville line of \cref{thm:recovery:kink}(iii) replaces the coding bound), and so does the full table of a noisy random parity under the conditions of \cref{prop:parity:kink}.

\begin{theorem}[Stable MAP lengths occur only at kinks]\status{proved}\src{split/kink Thm 3.4}\label{thm:kink:converse}
Let $\mathcal P$ be a point family, $0\le\tau<\sigma\le\infty$ and $\alpha'\le\alpha''$. Every $\lambda$-MAP with $\lambda\in(\tau,\sigma)$ has length in $[\alpha',\alpha'']$ if and only if $\mathcal P$ has a zero-window left kink of steepness $\sigma$ at a point $(\alpha_-,\rho(\alpha_-))$ with $\alpha_-\ge\alpha'$ and a zero-window right kink of flatness $\tau$ at a point $(\alpha_+,\rho(\alpha_+))$ with $\alpha_+\le\alpha''$. With the indices $i,j$ such that $\sigma_{i+1}<\sigma\le\sigma_i$ and $\sigma_{j+1}\le\tau<\sigma_j$, one can take $\alpha_-=\alpha_i$, $\alpha_+=\alpha_j$, and the $\lambda$-MAP lengths over $(\tau,\sigma)$ have minimum $\alpha_i$ and maximum $\alpha_j$. The index $j$ exists unless $\tau=0$ and there are infinitely many vertices; then both conditions fail for every finite $\alpha''$.
\end{theorem}

\emph{Proof idea.} One direction is \cref{cor:kink:locate}(a),(b). Conversely, the $\lambda$-MAPs sit at $v_i$ for $\lambda$ just below $\sigma$ and at $v_j$ just above $\tau$, and by \cref{cor:kink:locate}(c) these are zero-window kinks of steepness $\sigma_i\ge\sigma$ and flatness $\sigma_{j+1}\le\tau$ (\cref{app:kinks:data}).

\begin{proposition}[A universal member flattens every kink far beyond its length]\status{proved}\src{split/kink Prop 3.5; split/verify-kink \S5}\label{prop:kink:universal}
Let a point family contain $\pi_U$, with $\ell_U:=\ell(\pi_U)$ and $L(\pi_U)\le L(\pi)+\ell(\pi)+\kappa_0$ for every $\pi$, which is \eqref{eq:absorb:dom} on one observation.
(i) Every left kink $(\sigma,w)$ at $(\alpha_{\rm r},L_{\rm r})$ with $\alpha_{\rm r}-w>\ell_U$ has $\sigma\le1+(\ell_U+\kappa_0)/(\alpha_{\rm r}-\ell_U)$; so for $\epsilon>0$ every such kink steeper than $1+\epsilon$ has $\alpha_{\rm r}\le\ell_U+(\ell_U+\kappa_0)/\epsilon$.
(ii) For every $\lambda>1$ every $\lambda$-MAP has length at most $\ell_U+(\ell_U+\kappa_0)/(\lambda-1)$. For $\epsilon>0$ this is at most $\ell_U+(\ell_U+\kappa_0)/\epsilon$ when $\lambda\ge1+\epsilon$, and, if $\ell_U+\kappa_0>0$, only then.
\end{proposition}

\emph{Proof idea.} (i) The hypothesis puts $L(\pi_U)$ at most $\alpha_{\rm r}+\kappa_0$ above $L_{\rm r}$, the kink at least $\sigma(\alpha_{\rm r}-\ell_U)$ above. (ii) Insert it into $J_\lambda(\hat\pi)\le J_\lambda(\pi_U)$ (\cref{app:kinks:data}).

\begin{remark}[Reading]\status{remark}\src{split/kink \S6, Remark 3.6}\label{rem:kink:reading}
The $\lambda$-MAP takes exactly the hull edges with $\sigma_i>\lambda$: the model keeps the blocks of program bits that explain more than $\lambda$ bits of data per bit, on average over an edge, not bit by bit. Several vertices with $\sigma_i>1$ give different models for different $\lambda$ (\cref{sec:limits}). Under (Dom), which holds in $\Hcl{F}$ and fails in $\Hcl{\times}$ (\cref{cor:absorb:nouniversal}), \cref{prop:kink:universal}(ii) is \cref{thm:absorb:absorption}(a).
\end{remark}
