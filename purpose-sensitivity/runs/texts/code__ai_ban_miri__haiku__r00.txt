import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"


def _normalize_email(raw):
    s = raw
    if "<" in s and ">" in s:
        start = s.index("<")
        end = s.find(">", start)
        if end != -1:
            s = s[start + 1:end]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw.lower()
    cuts = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    z = raw.strip()
    if re.fullmatch(r"[0-9]{4,5}-[0-9]{4}", z):
        z = z.split("-")[0]
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    if re.fullmatch(r"[0-9]{5}", z):
        return z
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(raw.strip(), fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in {"y", "yes", "true", "1"}


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _merge(group):
    """group: cleaned records for one person, in input order."""

    def pick(key):
        cands = [rec for rec in group if rec[key] != ""]
        if not cands:
            return ""
        # Stable sorts: recency first, then manual-source preference.
        # Ties keep input order, so the earlier row wins.
        cands.sort(key=lambda rec: rec["updated"], reverse=True)
        cands.sort(key=lambda rec: not rec["manual"])
        return cands[0][key]

    tags = set().union(*(rec["tags"] for rec in group))
    return {
        "first_name": pick("first_name"),
        "last_name": pick("last_name"),
        "zip": pick("zip"),
        "email": pick("email"),
        "phone": pick("phone"),
        "source": pick("source"),
        "tags": ";".join(sorted(tags)),
        "opted_out": "Y" if any(rec["opted_out"] for rec in group) else "N",
        "updated": max(rec["updated"] for rec in group).strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    # Clean each row; drop rows with no valid email or phone.
    recs = []
    for row in rows:
        email = _normalize_email(row["email"])
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(row["phone"])
        if not email and not phone:
            continue
        source = row["source"].strip().lower()
        recs.append({
            "first_name": _normalize_name(row["first_name"]),
            "last_name": _normalize_name(row["last_name"]),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row["zip"]),
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row["tags"]),
            "opted_out": _is_opted_out(row["opted_out"]),
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over records; matching is transitive.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    first_by_email = {}
    first_by_phone_last = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in first_by_email:
                union(i, first_by_email[rec["email"]])
            else:
                first_by_email[rec["email"]] = i
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].lower())
            if key in first_by_phone_last:
                union(i, first_by_phone_last[key])
            else:
                first_by_phone_last[key] = i

    # Group records by root. Groups are ordered by earliest input row,
    # because records are iterated in input order.
    groups = {}
    order = []
    for i in range(len(recs)):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[i])

    return [_merge(groups[root]) for root in order]

Notes on judgment calls, since the rules didn't cover these:

- **ZIP+4 with a dropped leading zero:** I treat `2139-4307` as ZIP+4 (4–5 digit prefix), so it becomes `02139`. The rules' example only shows a 5-digit prefix, so tell me if you want that strictly 5 digits.
- **Display names:** Angle-bracket extraction only applies when both `<` and `>` are present.
- **Bad `updated` values:** An unparseable value raises `ValueError` rather than being silently ranked. The rules say `updated` is always present, so a failure would indicate bad data worth looking at before the send.
- **Source field:** The `source` output uses the same selection as the other fields (non-empty, manual preferred, then most recent, then earliest row).
