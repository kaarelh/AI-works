```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, filter, and merge duplicate CRM contacts per the written rules."""

    def s(v):
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^<>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
                e = local + "@" + domain
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"ext|x|#", p, re.IGNORECASE)
        if m:
            p = p[:m.start()]
        d = re.sub(r"[^0-9]", "", p)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_dt(raw):
        t = s(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(t, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable: treated as oldest

    def parse_tags(raw):
        return [t.strip().lower() for t in re.split(r"[;,]", s(raw)) if t.strip()]

    # ---------- build normalized records, drop rows without valid contact ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        email = email if email_valid(email) else ""
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip")),
            "source": source,
            "rank": 0 if source == "manual" else 1,
            "dt": parse_dt(row.get("updated")),
            "tags": parse_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower() in ("y", "yes", "true", "1"),
        })

    # ---------- union-find ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # ---------- build output ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]  # already in input order
        if not cands:
            return ""
        cands.sort(key=lambda r: r["dt"], reverse=True)  # stable: ties keep input order
        cands.sort(key=lambda r: r["rank"])              # manual first
        return cands[0][field]

    out = []
    for root in order:
        members = groups[root]
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["dt"] for i in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })
    return out
```

Notes on a few judgment calls:

- **Opt-outs:** a person is flagged `"Y"` if any merged row opted out, so when you send, filter on `opted_out == "N"`. Rows dropped by rule 9 (no valid email or phone) never appear, so they can't be emailed anyway.
- **Rows with a phone but no valid email:** these are still returned with `email == ""`, so skip those for an email send.
- **ZIP+4:** I treat anything like `digits-dddd` as ZIP+4 and then apply the 4-digit fix, so `2139-4307` becomes `02139`.
- **Unparseable `updated`:** treated as the oldest possible date rather than raising an error. Your spec says this shouldn't occur.
- **Name matching (rule 10):** compares the normalized last name lowercased.
