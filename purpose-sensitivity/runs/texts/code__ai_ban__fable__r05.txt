```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one dict per person."""

    OUT_KEYS = ("first_name", "last_name", "email", "phone", "zip",
                "tags", "opted_out", "updated", "source")

    # ---------- helpers ----------
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        e = (m.group(1) if m else raw).strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        s = raw.lower()
        cut = len(s)
        for marker in ("x", "ext", "#"):
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return raw.strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---------- normalize rows (and drop unmatchable ones) ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(get(row, "phone"))
        if not phone_valid(phone):
            phone = ""
        if not email and not phone:
            continue  # rule 9
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "tags": split_tags(get(row, "tags")),
            "opted_out": is_opted_out(get(row, "opted_out")),
            "dt": parse_updated(get(row, "updated")),
            "source": get(row, "source").strip().lower(),
        })

    # ---------- union-find for transitive matching ----------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so groups stay ordered
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email, by_phone_last = {}, {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}      # root -> list of recs, in input order
    order = []       # roots in order of first appearance (rule 12)
    for i, r in enumerate(recs):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(r)

    # ---------- build one output dict per person ----------
    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["source"] == "manual", r["dt"], -r["idx"]))
        return best[field]

    out = []
    for root in order:
        g = groups[root]
        all_tags = set()
        for r in g:
            all_tags |= r["tags"]
        person = {
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "zip": pick(g, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["dt"] for r in g).strftime("%Y-%m-%d"),
            "source": pick(g, "source"),
        }
        out.append({k: person[k] for k in OUT_KEYS})
    return out
```

A few notes on how I interpreted the rules:

- **Phone extensions**: the cut happens at the earliest occurrence of `x`, `ext`, or `#` (case-insensitive), so `"617-555-1234 Ext. 5"` and `"6175551234x5"` both become `6175551234`. Output phones are the bare 10 digits.
- **Field selection** (rule 11): ties on `manual` and `updated` go to the earlier input row. For `source`, this means any group containing a `manual` row outputs `"manual"`.
- **Phone+last-name matching** requires a non-empty last name; rows with a valid phone but blank last name only match via email.
- **`updated`**: the function raises a `ValueError` naming the bad value if a row's date isn't in one of the three formats, rather than silently guessing—better to catch it before the send than to misorder a group.
- Missing keys or `None` values are treated as `""`, so a slightly ragged export won't crash it.
