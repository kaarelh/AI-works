import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _field(row, key):
    """Return a field as a string, treating missing/None as ''."""
    value = row.get(key, "")
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = raw
    # Rule 1: "Ann Lee <ann@x.org>" -> "ann@x.org"
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    # Rule 2
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, else ''."""
    m = re.search(r"ext|x|#", raw, flags=re.IGNORECASE)
    if m:
        raw = raw[: m.start()]
    digits = re.sub(r"[^0-9]", "", raw)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    # Rule 4
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    # Rule 5
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    # Rule 6
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    # Rule 8
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email or phone (rule 9).
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source_raw = _field(row, "source").strip()
        records.append({
            "idx": idx,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": source_raw.lower(),
            "is_manual": source_raw.casefold() == "manual",
            "tags": _parse_tags(_field(row, "tags")),
            "opted_out": _field(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(_field(row, "updated")),
        })

    # ---- Union-find for transitive matching (rule 10).
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the root as the smaller index (earlier in input).
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_with_key = {}
    for i, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_with_key:
                union(i, first_with_key[key])
            else:
                first_with_key[key] = i

    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(records[i])

    # ---- Build one output dict per person (rule 11).
    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda r: (
                not r["is_manual"],                 # manual first
                -(r["updated"] - datetime.min),     # most recent first
                r["idx"],                           # earlier in input first
            ),
        )
        return best[field]

    output = []
    # Rule 12: order by each person's earliest row in the input.
    for members in sorted(groups.values(), key=lambda m: min(r["idx"] for r in m)):
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        latest = max(r["updated"] for r in members)
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
            "source": pick(members, "source"),
        })
    return output

A few notes before tomorrow's send:

- **Filter opt-outs before sending.** The function flags opt-outs but doesn't remove anyone. Send only to rows where `opted_out == "N"` and `email != ""`. One opted-out row marks the whole merged person as opted out.
- **Some opt-outs can be lost by design.** Rule 9 drops rows that have neither a valid email nor a valid phone *before* matching. If an opt-out was recorded on such a row (for example, a malformed email), that opt-out won't carry over. Run a quick check on the dropped opted-out rows before sending.
- **Interpretation choices I made where the spec was open:**
  - Phones are output as plain 10 digits (e.g. `6175551234`).
  - "ZIP+4" means 4–5 digits, a hyphen, then 4 digits, so `2139-4307` also becomes `02139`.
  - "Ext" is detected at its earliest position, so `555-1234 ext 5` cuts at "ext". Any letter "x" counts as a marker.
  - Unparseable `updated` values raise a `ValueError` with the bad value rather than being silently skipped.
  - Digit checks use ASCII `[0-9]` only, so stray Unicode digits don't slip through.
- **Merged people with different emails get one email.** Rows joined by phone and last name may have different addresses, and only the chosen address appears in the output. That avoids double-sends, but it means the other address won't receive this edition.
