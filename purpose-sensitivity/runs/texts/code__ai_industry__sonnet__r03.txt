```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    DIGITS = "0123456789"

    def s(value):
        return "" if value is None else str(value)

    # ---------- normalizers ----------
    def norm_email(raw):
        v = s(raw)
        m = re.search(r"<([^<>]*)>", v)
        if m:
            v = m.group(1)
        v = v.strip().lower()
        if v.count("@") == 1:
            local, domain = v.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
                v = local + "@" + domain
        # validity
        if v.count("@") != 1:
            return ""
        local, domain = v.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return v

    def norm_phone(raw):
        v = s(raw)
        m = re.search(r"ext|x|#", v, re.IGNORECASE)
        if m:
            v = v[:m.start()]
        d = "".join(c for c in v if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        v = s(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(v, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated value: %r" % (raw,))

    def is_opted_out(raw):
        return s(raw).strip().lower() in ("y", "yes", "true", "1")

    def parse_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize rows, drop unusable ones ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": s(row.get("source")).strip().lower(),
            "tags": parse_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
        })

    n = len(recs)

    # ---------- union-find ----------
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("e", r["email"]))
        last = r["last_name"].lower()
        if r["phone"] and last:
            keys.append(("p", r["phone"], last))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda r: (
                0 if r["source"] == "manual" else 1,
                -(r["updated"] - datetime.min),
                r["idx"],
            ),
        )
        return best[field]

    ordered = sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m))

    result = []
    for members in ordered:
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": "%04d-%02d-%02d" % (latest.year, latest.month, latest.day),
        })
    return result
```

Notes on interpretation choices:

- **Opt-outs:** The function doesn't filter anyone out. It sets `opted_out` to "Y" if any row in the merged group opted out. Before sending, exclude every output row where `opted_out == "Y"`.
- **ZIP+4:** Only the strict form `5digits-4digits` is trimmed at the hyphen. Something like "1234-5" becomes "".
- **Dropped rows:** Rows with no valid email or phone are discarded before matching (rule 9). If one of them had opted out, that flag is not carried anywhere, but it has no contact info you could email anyway.
- **Gmail rules:** They only apply when the address has exactly one "@". Addresses that become empty before the "@" (like `+x@gmail.com`) are invalid.
- **Phone-only matches:** These require a non-empty last name, compared case-insensitively after normalization.
- **Unparseable `updated`:** A value that doesn't match one of the three formats raises `ValueError` rather than being silently guessed, so bad data surfaces before the send.
