```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DIGITS = "0123456789"
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")


def _normalize_email(raw):
    s = (raw or "").strip()
    lt = s.find("<")
    if lt != -1:
        gt = s.find(">", lt + 1)
        if gt != -1:
            s = s[lt + 1:gt]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = (raw or "").lower()
    markers = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if markers:
        s = s[:min(markers)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(group, field):
    # Stable sorts: first order by recency (newest first), then move
    # "manual" rows to the front. Ties keep input order.
    cands = [r for r in group if r[field]]
    if not cands:
        return ""
    cands.sort(key=lambda r: r["updated"], reverse=True)
    cands.sort(key=lambda r: r["source"] != "manual")
    return cands[0][field]


def _merge(group):
    tags = sorted({t for r in group for t in r["tags"]})
    return {
        "first_name": _pick(group, "first_name"),
        "last_name": _pick(group, "last_name"),
        "zip": _pick(group, "zip"),
        "email": _pick(group, "email"),
        "phone": _pick(group, "phone"),
        "source": _pick(group, "source"),
        "tags": ";".join(tags),
        "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
        "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    # Parse and normalize every row.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        phone = _normalize_phone(row.get("phone"))
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "email": email if _valid_email(email) else "",
            "phone": phone if _valid_phone(phone) else "",
            "zip": _normalize_zip(row.get("zip")),
            "source": (row.get("source") or "").strip().lower(),
            "tags": [
                t.strip().lower()
                for t in re.split(r"[;,]", row.get("tags") or "")
                if t.strip()
            ],
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # Rule 9: drop rows with neither a valid email nor a valid phone.
    valid = [r for r in recs if r["email"] or r["phone"]]

    # Rule 10: union-find so matching is transitive.
    parent = {r["idx"]: r["idx"] for r in valid}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for r in valid:
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], r["idx"])
            else:
                first_by_email[r["email"]] = r["idx"]
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], r["idx"])
            else:
                first_by_phone_last[key] = r["idx"]

    # Rule 12: groups in order of their earliest row.
    groups = {}
    for r in valid:
        groups.setdefault(find(r["idx"]), []).append(r)

    return [_merge(group) for group in groups.values()]
```

Notes on the implementation:

- **Opt-outs are flagged, not removed.** Per rule 11, `dedupe_contacts` returns opted-out people with `opted_out="Y"`. Filter on `opted_out == "N"` before the send.
- **Opt-outs only propagate through matches.** If someone opts out with an email or phone that doesn't match their other records, the function won't connect them, and their other rows will still show `N`. Before tomorrow's send, I'd check the final recipient list against a raw suppression list of every opted-out email and phone from the original export.
- **Interpretations:** ZIP+4 is treated as 4–5 digits, a hyphen, and 4 digits. An `updated` value in an unrecognized format raises a `ValueError` rather than being silently mis-sorted, so a bad export fails loudly.
