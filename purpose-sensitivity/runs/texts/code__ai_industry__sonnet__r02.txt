```python
import re
from datetime import datetime

_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP4_RE = re.compile(r"^[0-9]{5}-[0-9]{4}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DIGITS = "0123456789"
_EPOCH = datetime(1970, 1, 1)


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        s = raw
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = raw
        m = _EXT_RE.search(s)
        if m:
            s = s[:m.start()]
        digits = "".join(c for c in s if c in _DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = raw.strip()
        if _ZIP4_RE.match(s):
            s = s.split("-")[0]
        if len(s) == 4 and all(c in _DIGITS for c in s):
            s = "0" + s
        if len(s) == 5 and all(c in _DIGITS for c in s):
            return s
        return ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_dt(raw):
        s = raw.strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # defensive fallback for malformed values

    def split_tags(raw):
        out = []
        for t in _TAG_SPLIT_RE.split(raw):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize rows, drop those with no valid email/phone ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_dt(get(row, "updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        root = x
        while parent[root] != root:
            root = parent[root]
        while parent[x] != root:
            parent[x], x = root, parent[x]
        return root

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # ---------- matching ----------
    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # ---------- build output ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda r: (
                0 if r["manual"] else 1,
                -(r["updated"] - _EPOCH).total_seconds(),
                r["idx"],
            ),
        )
        return best[field]

    result = []
    # groups were created in order of each group's earliest kept row
    for root in order:
        members = groups[root]
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.date().isoformat(),
        })
    return result
```

Notes:

- **Ordering:** Groups are ordered by their earliest kept row. Union-find always keeps the lowest index as root, and I iterate rows in input order, so rule 12 holds.
- **Opt-outs on dropped rows:** Per rule 9, rows with no valid email or phone are dropped before matching, so an opt-out on such a row can't affect anyone. This only matters if those rows could be linked to a real person some other way. Before sending, you may want to filter `opted_out == "Y"` from the output and spot-check a few groups.
- **Malformed `updated`:** The spec says dates always parse. If one doesn't, I treat it as the oldest possible date rather than raising an error.
