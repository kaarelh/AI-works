```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_UPDATED_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _norm_phone(raw):
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _norm_zip(raw):
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _all_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_digits(s):
        return s
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _UPDATED_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in ("y", "yes", "true", "1")


def _norm_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # Rules 1-9: normalize each row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        email = email if _valid_email(email) else ""
        phone = _norm_phone(row.get("phone", ""))
        phone = phone if _valid_phone(phone) else ""
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _norm_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row["updated"]),
        })

    # Rule 10: group rows into people (transitive via union-find).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    first_by_email = {}
    first_by_phone_name = {}
    for j, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(j, first_by_email[r["email"]])
            else:
                first_by_email[r["email"]] = j
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in first_by_phone_name:
                union(j, first_by_phone_name[key])
            else:
                first_by_phone_name[key] = j

    groups = {}
    for j in range(len(recs)):
        groups.setdefault(find(j), []).append(j)
    # Each group's indices are ascending, so g[0] is the group's earliest row.
    ordered = sorted(groups.values(), key=lambda g: g[0])

    # Rule 11: choose field values and build one output dict per person.
    def pick(group, field):
        candidates = [recs[j] for j in group if recs[j][field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[field]

    result = []
    for group in ordered:
        members = [recs[j] for j in group]
        tags = set()
        for m in members:
            tags |= m["tags"]
        result.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
        })
    return result
```

Notes before you use this:

- **Opted-out contacts are not removed.** Per rule 11, they come back with `opted_out == "Y"`, and a person is flagged "Y" if any of their rows opted out. Filter with `opted_out == "N"` before the send list is built.
- **ZIP+4 handling:** I matched the ZIP+4 pattern with a 4- or 5-digit prefix, so an Excel-mangled `2139-4307` becomes `02139`. If you want only 5-digit prefixes, change `{4,5}` to `{5}`.
- **Bad `updated` values raise `ValueError`** rather than being silently dropped or defaulted, so a malformed export fails loudly. Only rows that survive rule 9 are parsed.
- **I haven't executed this.** This chat has no code execution, so please run it on a small sample before the export. Useful checks: a Gmail address with dots and a `+` alias matching a plain one, a phone with `ext.` matching the same number without the extension, a last-name mismatch that should not merge, and a chain A→B→C where A and C share no direct key.
