import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_DIGITS = "0123456789"


def _norm_email(raw):
    """Return the normalized email, or "" if it is not valid."""
    s = (raw or "").strip()
    lt = s.find("<")
    if lt != -1:
        gt = s.find(">", lt + 1)
        if gt != -1:
            s = s[lt + 1:gt]
    s = s.strip().lower()
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if domain in _GMAIL_DOMAINS:
        local = local.replace(".", "").split("+", 1)[0]
        domain = "gmail.com"
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return f"{local}@{domain}"


def _norm_phone(raw):
    """Return the normalized 10-digit phone, or "" if it is not valid."""
    s = raw or ""
    m = re.search(r"x|ext|#", s, flags=re.IGNORECASE)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Return a 5-digit ZIP, or "" if it cannot be normalized."""
    z = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{5})-([0-9]{4})", z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    if re.fullmatch(r"[0-9]{5}", z):
        return z
    return ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return (raw or "").strip().lower() in {"y", "yes", "true", "1"}


def _tag_set(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # Steps 1-5 and 9: normalize fields and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email"))
        phone = _norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _tag_set(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Step 10: union-find over rows that match on email, or on phone + last name.
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in phone_seen:
                union(i, phone_seen[key])
            else:
                phone_seen[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)
    # Members are appended in input order, so g[0] is each group's earliest row.
    ordered = sorted(groups.values(), key=lambda g: g[0])

    def pick(members, field):
        # Prefer manual rows, then the latest "updated", then the earlier input row.
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    # Step 11: build one output record per person.
    out = []
    for members in ordered:
        group = [recs[i] for i in members]
        tags = set().union(*(r["tags"] for r in group))
        latest = max(r["updated"] for r in group)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return out

Notes before tomorrow's send:

- **Filter opted-out people before sending.** The function flags them with `opted_out == "Y"`, but it doesn't remove them. Send only to rows where `opted_out == "N"`.
- **Opt-outs only carry across matches.** If someone opted out using one email, and their other record has a different email and a phone number that doesn't match on last name, the function won't link them. Check your final list against your raw opt-out list by email and phone before sending.
- **Bad `updated` values raise `ValueError`.** I chose this over silently guessing a date. If it fails, fix the bad row in the export and rerun.
