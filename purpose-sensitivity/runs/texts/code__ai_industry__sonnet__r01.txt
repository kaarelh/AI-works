```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL = ("gmail.com", "googlemail.com")
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    TRUTHY = {"y", "yes", "true", "1"}

    def s(v):
        return "" if v is None else str(v)

    def norm_email(raw):
        e = s(raw)
        start = e.find("<")
        if start != -1:
            end = e.find(">", start + 1)
            if end != -1:
                e = e[start + 1:end]
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"x|ext|#", p, re.IGNORECASE)
        if m:
            p = p[:m.start()]
        d = re.sub(r"[^0-9]", "", p)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        u = s(raw).strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                continue
        return datetime.min

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # Normalize rows, dropping those without a valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower() in TRUTHY,
            "updated": parse_updated(row.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(recs[i])

    def pick(members, field):
        best = None
        for r in members:  # members are in input order
            if not r[field]:
                continue
            if best is None:
                best = r
                continue
            if r["manual"] != best["manual"]:
                if r["manual"]:
                    best = r
            elif r["updated"] > best["updated"]:
                best = r
        return best[field] if best is not None else ""

    result = []
    for members in sorted(groups.values(), key=lambda g: g[0]["idx"]):
        tags = set()
        for r in members:
            tags.update(r["tags"])
        latest = max(r["updated"] for r in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes on a few judgment calls:

- **Opt-outs:** the output keeps opted-out people with `opted_out="Y"` rather than removing them, as your rules specify. Filter on `opted_out == "N"` before sending the email.
- **ZIP+4:** I accept 4 or 5 digits before the hyphen (e.g. `2139-4307`), so the Excel leading-zero fix also applies to ZIP+4 values.
- **Unparseable `updated`:** these are treated as the earliest possible date instead of raising an error.
- **Transitive matching:** it uses union-find, so chains like A–B–C collapse into one person.
