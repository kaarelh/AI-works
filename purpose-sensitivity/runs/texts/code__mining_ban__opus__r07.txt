```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _s(value):
    """Treat None as an empty string so a stray null doesn't crash the run."""
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = _s(raw)
    # Rule 1: "Ann Lee <ann@x.org>" -> "ann@x.org"
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    # Rule 2
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, else ''."""
    s = _s(raw).lower()
    # Cut at the earliest extension marker ("ext", "x", or "#").
    s = re.split(r"ext|x|#", s, maxsplit=1)[0]
    digits = re.sub(r"[^0-9]", "", s)  # ASCII digits only
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    # Rule 4
    s = _s(raw).strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    # Rule 5
    s = _s(raw).strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw, index):
    # Rule 6
    s = _s(raw).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Row {index}: unrecognized 'updated' value {raw!r}")


def _is_opted_out(raw):
    # Rule 7
    return _s(raw).strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    # Rules 8 and 11 (tags)
    return {t.strip().lower() for t in re.split(r"[;,]", _s(raw)) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row, dropping ones with no valid contact (rule 9) ----
    recs = []
    for index, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = _s(row.get("source")).strip().lower()
        recs.append({
            "index": index,
            "email": email,
            "phone": phone,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "source": source,
            "manual": source == "manual",
            "dt": _parse_updated(row.get("updated"), index),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "tags": _split_tags(row.get("tags")),
        })

    # ---- Union-find over records (rule 10) ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the smaller (earlier) index as root; not required, just tidy.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}  # match key -> first record position with that key
    for pos, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(pos, first_seen[key])
            else:
                first_seen[key] = pos

    groups = {}
    for pos, r in enumerate(recs):
        groups.setdefault(find(pos), []).append(r)

    # ---- Field selection (rule 11) ----
    def pick(members, field):
        # members are in input order; replace only on a strictly better rank,
        # so ties go to the earlier row.
        best = None
        for r in members:
            if not r[field]:
                continue
            rank = (r["manual"], r["dt"])
            if best is None or rank > (best["manual"], best["dt"]):
                best = r
        return best[field] if best is not None else ""

    output = []
    # Rule 12: order people by their earliest row in the input.
    for members in sorted(groups.values(), key=lambda m: m[0]["index"]):
        latest = max(r["dt"] for r in members)
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })
    return output
```

**Things to check before tomorrow's send:**

1. **This function doesn't remove opted-out people.** It only marks them with `opted_out: "Y"`. Before you load the list into your email tool, filter with `[p for p in result if p["opted_out"] == "N" and p["email"]]`. Doing this filter after deduping is the right order, because an opt-out on any duplicate row suppresses the whole person.
2. **An opt-out row with no valid email and no valid phone gets dropped** by rule 9, so it can't suppress anyone. If your CRM keeps opt-outs as a separate list, also check it against the final emails.
3. **Phone + last name matching can merge different people.** For example, Jane and John Smith sharing a household landline become one person. That person gets only one email address, so the other one won't get the alert. If either of them opted out, both are suppressed. Suppressing both is the safe direction, but if you see many of these merges you may want to review them.
4. **A bad `updated` value raises an error** that names the row, instead of guessing. I chose a loud failure over a silent mis-ranking. If you'd rather skip or default those rows, tell me and I'll change it.
5. Small behaviors to know about. The gmail rule only applies when the address has exactly one `@`. A ZIP+4 whose 5-digit part lost its leading zero (`2139-4307`) is also repaired to `02139`. Names like `mcdonald` become `Mcdonald` because `str.title()` works that way.
