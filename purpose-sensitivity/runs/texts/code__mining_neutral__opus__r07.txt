```python
import re
from datetime import datetime

_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_EXT_MARKER = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DATETIME_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_EPOCH = datetime(1, 1, 1)


def _field(row, key):
    """Return a field as a string, treating missing/None as ""."""
    value = row.get(key, "")
    return "" if value is None else str(value)


def _normalize_email(raw):
    """Rule 1: extract from angle brackets, strip, lowercase, gmail canonicalization."""
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    """Rule 2."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, otherwise ""."""
    s = raw
    m = _EXT_MARKER.search(s)
    if m:
        s = s[: m.start()]
    digits = re.sub(r"[^0-9]", "", s)  # ASCII digits only
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    s = raw.strip()
    m = _ZIP_PLUS4.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw, index):
    """Rule 6. A bare date means midnight."""
    s = raw.strip()
    for fmt in _DATETIME_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Row {index}: unrecognized 'updated' value {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    """Rule 8 (plus the stripping/lowercasing from rule 11)."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email or phone (rule 9) ----
    records = []
    for index, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        records.append({
            "index": index,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _parse_tags(_field(row, "tags")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
            "updated": _parse_updated(_field(row, "updated"), index),
        })

    # ---- Union-find for transitive matching (rule 10) ----
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the lower position (earlier input row).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}  # match key -> record position
    for pos, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(pos, first_seen[key])
            else:
                first_seen[key] = pos

    groups = {}
    for pos in range(len(records)):
        groups.setdefault(find(pos), []).append(records[pos])

    # ---- Build one output dict per person (rule 11) ----
    def preference(rec):
        # manual first, then most recent updated, then earlier input row
        seconds = (rec["updated"] - _EPOCH).total_seconds()
        return (0 if rec["is_manual"] else 1, -seconds, rec["index"])

    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        return min(candidates, key=preference)[field]

    people = []
    for members in groups.values():
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        latest = max(r["updated"] for r in members)
        people.append((
            min(r["index"] for r in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
                "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
                "source": pick(members, "source"),
            },
        ))

    # ---- Order by earliest input row (rule 12) ----
    people.sort(key=lambda p: p[0])
    return [person for _, person in people]
```

**Notes and judgment calls:**

- **This function marks opt-outs. It does not remove them.** As specified, people who opted out still appear in the output with `opted_out: "Y"`. Before sending, filter with `[p for p in result if p["opted_out"] == "N" and p["email"]]`. The `p["email"]` check matters because some people will only have a phone number.
- **Opt-outs can be missed.** An opt-out is only caught if it's on a row that matches the person's other rows. If someone opted out under a different email and has no matching phone and last name, they'll look like a separate person and still get the email. You may want to also check against a standalone suppression list.
- **Ambiguous points where I picked an interpretation:**
  - ZIP+4 with a missing leading zero (`2139-4307`) is treated as ZIP+4, then padded to `02139`.
  - Gmail normalization only runs when there's exactly one `@`. Addresses with more than one `@` are invalid either way.
  - Only ASCII digits 0–9 count in phones and ZIPs.
  - Last names are compared with `casefold()`, after the rule 5 normalization.
- **Bad dates raise errors.** An unrecognized `updated` value raises a `ValueError` that includes the row number, so bad data doesn't pass silently. Whitespace around dates is tolerated.
- **Tie-breaking is deterministic.** Recency comparisons use time elapsed since a fixed reference date, not `.timestamp()`, so time zones and daylight saving can't affect the results.
