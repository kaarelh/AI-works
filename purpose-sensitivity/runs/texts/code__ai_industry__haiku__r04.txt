```python
import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ASCII_DIGITS = "0123456789"


def _normalize_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local:
        return ""
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in _ASCII_DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    z = raw.strip()
    m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    if re.fullmatch(r"[0-9]{5}", z):
        return z
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _normalize_tags(raw):
    return sorted({t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()})


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def dedupe_contacts(rows):
    # Normalize every row and drop rows with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _valid_email(_normalize_email(row.get("email", "")))
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = row.get("source", "").strip().lower()
        recs.append({
            "idx": idx,
            "first": _normalize_name(row.get("first_name", "")),
            "last": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _normalize_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
            "dt": _parse_updated(row.get("updated", "")),
        })

    # Union-find so matching is transitive.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_first = {}
    phone_first = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_first:
                union(email_first[r["email"]], i)
            else:
                email_first[r["email"]] = i
        last_key = r["last"].lower()
        if r["phone"] and last_key:
            key = (r["phone"], last_key)
            if key in phone_first:
                union(phone_first[key], i)
            else:
                phone_first[key] = i

    # Group by root. Dict insertion order follows each group's earliest row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        pool = [r for r in cands if r["manual"]] or cands
        best_dt = max(r["dt"] for r in pool)
        pool = [r for r in pool if r["dt"] == best_dt]
        return min(pool, key=lambda r: r["idx"])[field]

    people = []
    for members in groups.values():
        group_recs = [recs[i] for i in members]
        all_tags = sorted({t for r in group_recs for t in r["tags"]})
        people.append({
            "first_name": pick(members, "first"),
            "last_name": pick(members, "last"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(all_tags),
            "opted_out": "Y" if any(r["opted_out"] for r in group_recs) else "N",
            "updated": max(r["dt"] for r in group_recs).strftime("%Y-%m-%d"),
        })
    return people
```

Notes for the send:

- **Opt-outs stay in the output.** Your spec returns opted-out people with `opted_out="Y"`, so filter them out before the send, for example: `recipients = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N"]`. Because a person is flagged if any of their rows opted out, the filter also excludes everyone linked to an opted-out row through matching.
- **Rows with unparseable `updated` values raise `ValueError`.** I chose this over guessing, so a bad date stops the run instead of silently changing the winner. Only rows that survive the email/phone check are parsed, so junk in dropped rows won't stop the run.
- **ZIP+4 matching is literal.** Only 5-digit, hyphen, 4-digit values are treated as ZIP+4. An Excel-mangled value like `2139-4307` becomes `""`. If you want those kept, change the regex in `_normalize_zip` to `[0-9]{4,5}-[0-9]{4}`.
