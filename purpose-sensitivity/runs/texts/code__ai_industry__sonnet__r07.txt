```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable: treat as oldest

    def parse_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---- normalize rows, dropping those with no valid email or phone ----
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(get(row, "updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # ---- matching ----
    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---- build output, ordered by earliest row in input ----
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    result = []
    for root in sorted(groups, key=lambda k: min(groups[k])):
        members = groups[root]
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Opt-outs on dropped rows:** Per rule 9, rows with no valid email or phone are dropped before matching, so an opt-out on such a row can't carry over to anyone. That shouldn't matter for an email send, since those rows have no usable address anyway.
- **Final send filter:** The function returns opted-out people with `opted_out == "Y"` rather than removing them. Before sending, filter to `opted_out == "N"` and a non-empty `email`. Phone-only contacts will have an empty email.
- **Unparseable `updated` values:** These are treated as the oldest possible date instead of raising an error.
- **ZIP+4 with a lost leading zero:** A value like `2139-4307` becomes `02139`. The spec didn't cover this case, so I chose to handle it.
