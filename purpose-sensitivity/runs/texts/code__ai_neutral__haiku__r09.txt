import re
from datetime import datetime

_ASCII_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")


def _is_ascii_digits(s):
    return bool(s) and all(c in _ASCII_DIGITS for c in s)


def _clean_email(raw):
    """Return the normalized, valid email, or None."""
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    if s.count("@") != 1:
        return None
    local, domain = s.split("@")

    if domain in _GMAIL_DOMAINS:
        local = local.replace(".", "").split("+", 1)[0]
        domain = "gmail.com"

    if not local:
        return None
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return None
    return f"{local}@{domain}"


def _clean_phone(raw):
    """Return the normalized, valid 10-digit phone, or None."""
    s = raw.lower()
    cut_points = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cut_points:
        s = s[:min(cut_points)]

    digits = "".join(c for c in s if c in _ASCII_DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]

    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return None


def _clean_zip(raw):
    """Return a normalized 5-digit ZIP, or ""."""
    s = raw.strip()
    if re.fullmatch(r"[0-9]{4,5}-[0-9]{4}", s):
        s = s.split("-")[0]
    if len(s) == 4 and _is_ascii_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_ascii_digits(s):
        return s
    return ""


def _clean_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return datetime.min


def _is_opted_out(raw):
    return raw.strip().lower() in {"y", "yes", "true", "1"}


def _clean_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # Clean every row and drop rows with neither a valid email nor a valid phone.
    recs = []
    for row in rows:
        email = _clean_email(row.get("email", "")) or ""
        phone = _clean_phone(row.get("phone", "")) or ""
        if not email and not phone:
            continue
        recs.append({
            "first_name": _clean_name(row.get("first_name", "")),
            "last_name": _clean_name(row.get("last_name", "")),
            "zip": _clean_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _clean_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Union-find over row positions; recs keeps input order.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email = {}
    by_phone_last = {}
    for j, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(j, by_email[r["email"]])
            else:
                by_email[r["email"]] = j
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(j, by_phone_last[key])
            else:
                by_phone_last[key] = j

    # Groups keyed by root; dict order follows each group's earliest row.
    groups = {}
    for j in range(len(recs)):
        groups.setdefault(find(j), []).append(j)

    def pick(members, field):
        # Preference: source "manual", then latest updated, then earliest input row.
        # Python's sort is stable, so members (already in input order) break ties.
        cands = [recs[j] for j in members if recs[j][field]]
        if not cands:
            return ""
        cands.sort(key=lambda r: r["updated"], reverse=True)
        cands.sort(key=lambda r: r["source"] != "manual")
        return cands[0][field]

    result = []
    for members in groups.values():
        group = [recs[j] for j in members]
        all_tags = set().union(*(r["tags"] for r in group))
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return result

Notes on interpretation:

- **ZIP+4:** I treat "looks like ZIP+4" as a 4–5 digit prefix, a hyphen, and a 4-digit suffix. That way "2139-4307" becomes "02139" as the Excel-leading-zero rule intends.
- **Empty or unparseable `updated`:** Your rules say `updated` is always filled, so I didn't add error handling beyond a fallback to `datetime.min`. Such a row sorts last, and a group whose rows all have it outputs "0001-01-01".
- **Extension markers:** The cut happens at the earliest occurrence of "x", "ext", or "#", so "ext" is cut at its start.
- **Opt-outs:** Rows dropped in step 9 don't count toward `opted_out`, since they're removed before matching.
