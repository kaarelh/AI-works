import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL = {"gmail.com", "googlemail.com"}
    DIGITS = "0123456789"

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL:
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
            s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local or "." not in domain:
            return ""
        if domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"x|ext|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        d = "".join(c for c in s if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = (raw or "").strip()
        if re.fullmatch(r"[0-9]+-[0-9]+", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_dt(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min

    def norm_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw or ""):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize rows, dropping those with no valid email and no valid phone
    recs = []
    for idx, r in enumerate(rows):
        email = norm_email(r.get("email"))
        phone = norm_phone(r.get("phone"))
        if not email and not phone:
            continue
        source = (r.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(r.get("first_name")),
            "last_name": norm_name(r.get("last_name")),
            "zip": norm_zip(r.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": norm_tags(r.get("tags")),
            "opted_out": (r.get("opted_out") or "").strip().lower()
                         in ("y", "yes", "true", "1"),
            "updated": parse_dt(r.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            k = rec["email"]
            if k in email_seen:
                union(i, email_seen[k])
            else:
                email_seen[k] = i
        last = rec["last_name"].lower()
        if rec["phone"] and last:
            k = (rec["phone"], last)
            if k in phone_last_seen:
                union(i, phone_last_seen[k])
            else:
                phone_last_seen[k] = i

    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        # members are in input order; stable sorts apply the priorities
        cands.sort(key=lambda c: c["updated"], reverse=True)
        cands.sort(key=lambda c: 0 if c["source"] == "manual" else 1)
        return cands[0][field]

    result = []
    for root in order:
        members = groups[root]  # already in input order
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({

            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),

            "phone": pick(members, "phone"),
            "source": pick(members, "source"),

            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result

Notes on a few interpretation choices:

- **Opt-outs:** If any row in a merged person is opted out, the whole person is `"Y"`. Filter on `opted_out == "N"` before sending tomorrow's alert.
- **Matching:** I only match on the email and phone+last name rules, with transitive merging via union-find. Rows without a valid email or phone are dropped first, as in rule 9.
- **Unparseable `updated` values:** These are treated as the earliest possible date rather than raising an error. The spec says that shouldn't happen, but it keeps one bad cell from breaking the export.
- **Gmail addresses:** If the local part is empty after stripping dots and `+` tags (e.g. `+foo@gmail.com`), the email is invalid.
- **Email validity:** Beyond your rules, there is no check for internal whitespace or other syntax problems, so a malformed address that passes rule 2 would still count as valid.

It's worth spot-checking the output on a small sample of known duplicates before the send.
