import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_OPT_OUT_VALUES = ("y", "yes", "true", "1")
_EXT_MARKERS = ("x", "ext", "#")


def _normalize_email(raw):
    s = raw.strip()
    if "<" in s and ">" in s:
        lt = s.index("<")
        gt = s.rindex(">")
        if lt < gt:
            s = s[lt + 1:gt]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    if not local:
        return False
    return "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in _EXT_MARKERS:
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(d):
    return len(d) == 10 and d[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
        s = s[:5]
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # Normalize each row and drop rows with no valid email or phone.
    prepared = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""

        phone = _normalize_phone(row.get("phone", ""))
        if not _is_valid_phone(phone):
            phone = ""

        if not email and not phone:
            continue

        last_name = _normalize_name(row.get("last_name", ""))
        source = row.get("source", "").strip().lower()

        prepared.append({
            "index": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": last_name,
            "last_key": last_name.casefold(),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Union-find to group rows transitively.
    parent = list(range(len(prepared)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_index = {}
    phone_last_index = {}
    for idx, p in enumerate(prepared):
        if p["email"]:
            if p["email"] in email_index:
                union(idx, email_index[p["email"]])
            else:
                email_index[p["email"]] = idx
        if p["phone"] and p["last_key"]:
            key = (p["phone"], p["last_key"])
            if key in phone_last_index:
                union(idx, phone_last_index[key])
            else:
                phone_last_index[key] = idx

    # Collect groups; iterating in input order keeps groups ordered by earliest row.
    groups = {}
    order = []
    for idx in range(len(prepared)):
        root = find(idx)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(prepared[idx])

    def pick(ranked, field):
        for m in ranked:
            if m[field]:
                return m[field]
        return ""

    result = []
    for root in order:
        members = groups[root]

        # Preference: manual source first, then most recent updated, then earliest row.
        ranked = sorted(members, key=lambda m: m["index"])
        ranked = sorted(ranked, key=lambda m: m["updated"], reverse=True)
        ranked = sorted(ranked, key=lambda m: not m["manual"])

        all_tags = set()
        for m in members:
            all_tags |= m["tags"]

        result.append({
            "first_name": pick(ranked, "first_name"),
            "last_name": pick(ranked, "last_name"),
            "zip": pick(ranked, "zip"),
            "email": pick(ranked, "email"),
            "phone": pick(ranked, "phone"),
            "source": pick(ranked, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
        })

    return result
