import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one dict per person (see rules 1-12)."""

    DIGITS = "0123456789"

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        s = raw
        lt = s.find("<")
        if lt != -1:
            gt = s.find(">", lt + 1)
            if gt != -1:
                s = s[lt + 1:gt]
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        if not local:
            return False
        return "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        d = "".join(c for c in raw if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_date(raw):
        s = raw.strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized date format: %r" % raw)

    def is_opted_out(raw):
        return raw.strip().lower() in ("y", "yes", "true", "1")

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize rows, drop those with no valid contact ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(get(row, "tags")),
            "opted_out": is_opted_out(get(row, "opted_out")),
            "updated": parse_date(get(row, "updated")),
        })

    n = len(recs)
    if n == 0:
        return []

    # ---------- union-find ----------
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            k = r["email"]
            if k in email_seen:
                union(i, email_seen[k])
            else:
                email_seen[k] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            k = (r["phone"], last)
            if k in phone_last_seen:
                union(i, phone_last_seen[k])
            else:
                phone_last_seen[k] = i

    # ---------- group, preserving order of earliest row ----------
    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[i])

    # ---------- build output ----------
    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    result = []
    for root in order:
        g = groups[root]
        tags = set()
        for r in g:
            tags |= r["tags"]
        result.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "zip": pick(g, "zip"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "source": pick(g, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["updated"] for r in g).strftime("%Y-%m-%d"),
        })
    return result

Some notes on judgment calls where the rules were slightly open:

- **Opt-outs:** Rows dropped under rule 9 (no valid email or phone) are removed before matching, so their opt-out flag and tags don't carry over. Such a row can't be matched to anyone anyway.
- **Opt-out filtering:** The function outputs `opted_out` but doesn't remove opted-out people. Before sending, filter with `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`.
- **ZIP+4:** I treat any digits followed by a hyphen and 4 digits as ZIP+4, so `2139-4307` also becomes `02139`.
- **Email quirks:** Angle-bracket extraction uses the first `<...>` pair. Gmail handling only applies when there is exactly one `@`. An address like `+promo@gmail.com` ends up with an empty local part and is invalid.
- **Extension markers:** Because "ext" contains "x", the pattern `ext|x|#` cuts at the first occurrence of any marker.
- **Dates:** An unparseable `updated` value raises `ValueError` rather than being silently guessed, so bad data surfaces before the send.
