import re
from datetime import datetime

_EMAIL_BRACKET_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"[0-9]{4,5}-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DIGITS = "0123456789"


def _is_ascii_digits(s):
    return s != "" and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    """Return (normalized_email, is_valid)."""
    s = raw or ""
    m = _EMAIL_BRACKET_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    local, sep, domain = s.partition("@")
    if domain in ("gmail.com", "googlemail.com"):
        local = local.split("+", 1)[0].replace(".", "")
        domain = "gmail.com"
    s = local + sep + domain

    valid = (
        s.count("@") == 1
        and local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )
    return s, valid


def _normalize_phone(raw):
    """Return (normalized_digits, is_valid)."""
    s = (raw or "").lower()
    markers = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if markers:
        s = s[:min(markers)]

    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]

    valid = len(digits) == 10 and digits[0] not in "01"
    return digits, valid


def _normalize_zip(raw):
    s = (raw or "").strip()
    if _ZIP_PLUS4_RE.fullmatch(s):
        s = s.split("-", 1)[0]
    if len(s) == 4 and _is_ascii_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_ascii_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _normalize_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def _ranked(group):
    """Return group records best-first: manual before others, then most
    recent updated, then earliest input position. Relies on stable sorts."""
    ranked = sorted(group, key=lambda r: r["idx"])
    ranked.sort(key=lambda r: r["updated"], reverse=True)
    ranked.sort(key=lambda r: r["source"] != "manual")
    return ranked


def dedupe_contacts(rows):
    # Normalize rows and drop any without a valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        email, email_ok = _normalize_email(row.get("email"))
        phone, phone_ok = _normalize_phone(row.get("phone"))
        if not (email_ok or phone_ok):
            continue
        records.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": (row.get("source") or "").strip().lower(),
            "tags": _normalize_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over matching rows (transitive matching).
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(records):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], i)
            else:
                first_by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    # Collect groups; each group's first member is its earliest row.
    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(i)
    ordered_groups = sorted(groups.values(), key=lambda g: g[0])

    # Build one output dict per person.
    output = []
    for members in ordered_groups:
        grp = [records[i] for i in members]
        ranked = _ranked(grp)

        def pick(field):
            for r in ranked:
                if r[field]:
                    return r[field]
            return ""

        all_tags = set().union(*(r["tags"] for r in grp))
        output.append({
            "first_name": pick("first_name"),
            "last_name": pick("last_name"),
            "zip": pick("zip"),
            "email": pick("email"),
            "phone": pick("phone"),
            "source": pick("source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in grp) else "N",
            "updated": max(r["updated"] for r in grp).strftime("%Y-%m-%d"),
        })

    return output

Notes on decisions the rules didn't spell out:

- **ZIP+4:** I matched 4 or 5 digits before the hyphen, so an Excel-mangled "2139-4307" is recognized and padded to "02139".
- **Bad `updated` values:** the function raises a `ValueError` rather than guessing. Dropped rows are never parsed, so a bad date on a row with no valid contact won't stop the run.
- **Missing keys:** string fields default to `""` if a key is absent. `updated` is required.
