```python
import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_MIN_DT = datetime.min
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_PHONE_MARKER_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_DIGITS = "0123456789"
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _clean_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    # Validity check
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _clean_phone(raw):
    s = raw or ""
    m = _PHONE_MARKER_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _clean_zip(raw):
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _clean_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _clean_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return _MIN_DT  # unparseable: treat as oldest


def dedupe_contacts(rows):
    # Normalize each row; drop rows with neither a valid email nor a valid phone.
    prepared = []
    for idx, row in enumerate(rows):
        email = _clean_email(row.get("email"))
        phone = _clean_phone(row.get("phone"))
        if not email and not phone:
            continue
        prepared.append({
            "first_name": _clean_name(row.get("first_name")),
            "last_name": _clean_name(row.get("last_name")),
            "zip": _clean_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": (row.get("source") or "").strip().lower(),
            "tags": _clean_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Union-find over prepared row positions.
    parent = list(range(len(prepared)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    # Bucket rows by match keys. Rows in the same bucket are the same person;
    # union-find makes the relation transitive.
    by_email = {}
    by_phone_last = {}
    for i, p in enumerate(prepared):
        if p["email"]:
            by_email.setdefault(p["email"], []).append(i)
        if p["phone"] and p["last_name"]:
            key = (p["phone"], p["last_name"].lower())
            by_phone_last.setdefault(key, []).append(i)
    for bucket in list(by_email.values()) + list(by_phone_last.values()):
        for j in bucket[1:]:
            union(bucket[0], j)

    # Collect groups; each group's members are in ascending input order.
    groups = {}
    for i in range(len(prepared)):
        groups.setdefault(find(i), []).append(i)
    ordered_groups = sorted(groups.values(), key=lambda g: g[0])

    def pick(ranked, field):
        for i in ranked:
            if prepared[i][field]:
                return prepared[i][field]
        return ""

    result = []
    for members in ordered_groups:
        # Rank: manual sources first, then most recent updated, then earlier input.
        # Stable sorts applied in reverse priority order.
        ranked = sorted(members)
        ranked = sorted(ranked, key=lambda i: prepared[i]["updated"], reverse=True)
        ranked = sorted(ranked, key=lambda i: prepared[i]["source"] != "manual")

        tags = set()
        for i in members:
            tags |= prepared[i]["tags"]

        result.append({
            "first_name": pick(ranked, "first_name"),
            "last_name": pick(ranked, "last_name"),
            "zip": pick(ranked, "zip"),
            "email": pick(ranked, "email"),
            "phone": pick(ranked, "phone"),
            "source": pick(ranked, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(prepared[i]["opted_out"] for i in members) else "N",
            "updated": max(prepared[i]["updated"] for i in members).strftime("%Y-%m-%d"),
        })

    return result
```

Notes for the send:

1. **Opted-out people are still in the output.** Per rule 7, they come back with `opted_out == "Y"`. Filter them out before building the send list, e.g. `recipients = [c for c in dedupe_contacts(rows) if c["opted_out"] == "N"]`. Because a person who opted out on any row is marked "Y", this also excludes people who appear under an old, non-opted-out record.
2. **Unparseable `updated` values** are treated as the oldest possible date, so they lose ties to any valid date. Your spec says `updated` is always one of the three formats, so this only matters if the export has bad data.
3. **Gmail addresses with more than one "@"** are left unchanged and then rejected by the validity check, as the rules imply.
4. **ZIP+4 with a dropped leading zero** (e.g. `2139-4307`) is accepted and becomes `02139`.
