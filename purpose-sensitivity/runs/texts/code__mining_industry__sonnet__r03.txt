```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    DIGITS = "0123456789"
    GMAIL = {"gmail.com", "googlemail.com"}
    OPT_YES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    EPOCH = datetime(1, 1, 1)

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        d = "".join(c for c in raw if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = raw.strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_date(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized updated format: %r" % raw)

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # Normalize and drop rows with no valid email and no valid phone
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        dt = parse_date(get(row, "updated"))
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": split_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in OPT_YES,
            "dt": dt,
            "secs": (dt - EPOCH).total_seconds(),
        })

    n = len(recs)
    parent = list(range(n))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root (earliest row)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # Groups ordered by earliest row in input (recs are in input order)
    order.sort(key=lambda root: min(groups[root]))

    def pick(members, field):
        best = None
        best_key = None
        for i in members:
            r = recs[i]
            if not r[field]:
                continue
            key = (0 if r["source"] == "manual" else 1, -r["secs"], r["idx"])
            if best_key is None or key < best_key:
                best_key = key
                best = r[field]
        return best if best is not None else ""

    result = []
    for root in order:
        members = groups[root]
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["dt"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes on a few choices:

- **Opt-outs:** if any row in a merged person has opted out, the whole person is marked `Y`. Filter on `opted_out == "N"` before sending tomorrow's email.
- **Invalid values:** an invalid email or phone is output as `""`. It is also ignored when matching, so two rows never merge on a malformed value.
- **Bad dates:** an `updated` value that isn't one of the three formats raises `ValueError`. Rows dropped under rule 9 aren't parsed, so a bad date there won't raise.
- **Extension markers:** the search is `ext|x|#`, so the leftmost marker wins. For example, `555-123-4567 ext. 89` is cut at "ext".
- **Source and `manual`:** the preference for `manual` uses each row's own source. The output `source` field is picked from rows with a non-empty source, using the same ranking as the other fields.
