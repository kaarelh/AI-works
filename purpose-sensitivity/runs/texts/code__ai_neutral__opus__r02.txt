import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_FIVE_DIGITS_RE = re.compile(r"[0-9]{5}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_FIELDS = ("first_name", "last_name", "email", "phone", "zip",
           "tags", "opted_out", "updated", "source")


def _get(row, key):
    val = row.get(key, "")
    return "" if val is None else str(val)


def _norm_email(raw):
    # Rule 1: display name -> part inside angle brackets
    m = _ANGLE_RE.search(raw)
    if m:
        raw = m.group(1)
    email = raw.strip().lower()
    if email.count("@") == 1:
        local, domain = email.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            email = local + "@gmail.com"
    return email


def _email_is_valid(email):
    # Rule 2
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _norm_phone(raw):
    # Rule 3: returns 10 valid digits, or "" if invalid
    low = raw.lower()
    cut = len(low)
    for marker in ("x", "ext", "#"):
        i = low.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in raw[:cut] if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    # Rule 4
    z = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if len(z) == 4 and z.isascii() and z.isdigit():
        z = "0" + z
    return z if _FIVE_DIGITS_RE.fullmatch(z) else ""


def _norm_name(raw):
    # Rule 5
    name = raw.strip()
    if name.isupper() or name.islower():
        return name.title()
    return name


def _parse_updated(raw):
    # Rule 6
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    # Rule 7
    return raw.strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    # Rule 8 (+ normalization from rule 11)
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email or phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(_get(row, "email"))
        if not _email_is_valid(email):
            email = ""
        phone = _norm_phone(_get(row, "phone"))
        if not email and not phone:
            continue
        source = _get(row, "source").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _norm_name(_get(row, "first_name")),
            "last_name": _norm_name(_get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(_get(row, "zip")),
            "source": source,
            "tags": _split_tags(_get(row, "tags")),
            "opted_out": _is_opted_out(_get(row, "opted_out")),
            "updated": _parse_updated(_get(row, "updated")),
            "manual": source == "manual",
        })

    # ---- Union-find over records (rule 10, transitive)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(first_seen[key], i)
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Build one output dict per person (rule 11)
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates,
                   key=lambda r: (r["manual"], r["updated"], -r["index"]))
        return best[field]

    people = []
    for group in groups.values():
        latest = max(r["updated"] for r in group)
        tags = set()
        for r in group:
            tags |= r["tags"]
        out = {
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
            "source": pick(group, "source"),
        }
        earliest = min(r["index"] for r in group)
        people.append((earliest, {k: out[k] for k in _FIELDS}))

    # ---- Order by earliest input row (rule 12)
    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

1. **The function keeps opted-out people.** Rule 11 says to output every person, so people who opted out are still in the result, marked `opted_out == "Y"`. To keep them from getting the email, filter them out at send time:

   The `p["email"]` check also skips people who only have a phone number.

2. **Rule 9 can lose opt-outs.** A row with no valid email or phone is dropped before matching, so its opt-out flag is dropped too. Suppose someone opted out on a row where their email was mistyped, and a separate row has their correct email. That opt-out is lost and they will get the email. If you keep a separate suppression list, apply it to the final list as well.

3. **Gmail addresses are output in normalized form.** For example, `Ann.Lee+news@googlemail.com` comes out as `annlee@gmail.com`. Gmail delivers both to the same inbox, but your email platform may treat the normalized address as a new contact. If the platform tracks unsubscribes by exact address, this could slip past its suppression. You can either check the output against the platform's own suppression list or keep the original address alongside the normalized one.

4. **Bad dates stop the run.** A malformed `updated` value raises a `ValueError` that shows the bad value, rather than dropping the row quietly. Dropping it could also drop an opt-out.

5. **Shared emails merge people.** Two people who share an email address (for example, a household or office account) are merged into one, as rule 10 says.
