```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """
    Merge duplicate CRM contacts.

    Input:  list of dicts with string fields first_name, last_name, email,
            phone, zip, tags, opted_out, updated, source.
    Output: list of dicts (same keys), one per person, ordered by the
            position of each person's earliest row in the input.

    Assumptions beyond the written rules:
      * A missing/None field is treated as "".
      * An `updated` value that is not one of the three documented formats
        raises ValueError (naming the row) rather than being silently guessed.
      * For ZIPs, anything before the first hyphen is kept (so "2139-4307"
        also becomes "02139").
    """

    # ------------------------------------------------------------------ #
    # Field normalizers
    # ------------------------------------------------------------------ #
    angle_re = re.compile(r"<([^<>]*)>")
    ext_re = re.compile(r"ext|x|#", re.IGNORECASE)
    zip5_re = re.compile(r"^[0-9]{5}$")
    zip4_re = re.compile(r"^[0-9]{4}$")
    date_formats = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")

    def text(row, key):
        v = row.get(key)
        return v if isinstance(v, str) else ("" if v is None else str(v))

    def norm_email(raw):
        s = raw
        m = angle_re.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        s = raw
        m = ext_re.search(s)
        if m:
            s = s[: m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = raw.strip()
        if "-" in s:
            s = s.split("-", 1)[0].strip()
        if zip4_re.match(s):
            s = "0" + s
        return s if zip5_re.match(s) else ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw, idx):
        s = raw.strip()
        for fmt in date_formats:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(
            f"Row {idx}: cannot parse 'updated' value {raw!r} "
            "(expected YYYY-MM-DD, MM/DD/YYYY, or YYYY-MM-DDTHH:MM:SS)"
        )

    def is_opted_out(raw):
        return raw.strip().lower() in ("y", "yes", "true", "1")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ------------------------------------------------------------------ #
    # Normalize every row; drop rows with no valid email and no valid phone
    # ------------------------------------------------------------------ #
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(text(row, "email"))
        phone = norm_phone(text(row, "phone"))
        recs.append(
            {
                "idx": idx,
                "first_name": norm_name(text(row, "first_name")),
                "last_name": norm_name(text(row, "last_name")),
                "email": email if email_valid(email) else "",
                "phone": phone if phone_valid(phone) else "",
                "zip": norm_zip(text(row, "zip")),
                "tags": norm_tags(text(row, "tags")),
                "opted_out": is_opted_out(text(row, "opted_out")),
                "updated_dt": parse_updated(text(row, "updated"), idx),
                "source": text(row, "source").strip().lower(),
            }
        )
    recs = [r for r in recs if r["email"] or r["phone"]]

    # ------------------------------------------------------------------ #
    # Group rows into people (union-find, so matching is transitive)
    # ------------------------------------------------------------------ #
    parent = list(range(len(recs)))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so roots track earliest rows
            parent[max(ra, rb)] = min(ra, rb)

    key_owner = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(key_owner[k], i)
            else:
                key_owner[k] = i

    groups = {}  # root -> members, insertion order == earliest-row order
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ------------------------------------------------------------------ #
    # Build one output record per person
    # ------------------------------------------------------------------ #
    def pick(members, field):
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        best = max(
            cands,
            key=lambda m: (m["source"] == "manual", m["updated_dt"], -m["idx"]),
        )
        return best[field]

    out = []
    for members in groups.values():
        all_tags = set()
        for m in members:
            all_tags |= m["tags"]
        out.append(
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
                "updated": max(m["updated_dt"] for m in members).strftime("%Y-%m-%d"),
                "source": pick(members, "source"),
            }
        )
    return out
```

A few notes on how I resolved spots the rules left open, so you can adjust if your data person disagrees:

- **ZIP+4:** I keep whatever is before the first hyphen, so both `02139-4307` and the Excel-mangled `2139-4307` come out as `02139`. Anything not exactly five digits after that becomes `""`.
- **Unparseable `updated`:** the function raises `ValueError` naming the row rather than guessing. For a send like tomorrow's, I'd rather you see a bad row than have it silently sorted to the wrong date.
- **Extension markers:** the first occurrence of `x`, `ext`, or `#` (case-insensitive) cuts the phone, per the rule as written. Be aware this means a value like `Fax: 202-555-0100` is treated as having no phone; if your export has that pattern, tell me and I'll tighten it.
- **Output order:** people appear in order of their earliest input row, and all tie-breaks for field selection follow the "manual" → most recent `updated` → earlier row sequence, including for `source` itself (so a group with any "manual" row reports `source: "manual"`).

Opted-out people are still in the output (flagged `"Y"`) rather than removed, since the spec asks for the flag; just filter on `opted_out == "N"` before loading into your email tool.
