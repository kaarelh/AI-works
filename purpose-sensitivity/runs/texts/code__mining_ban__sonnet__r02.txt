import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, filter, and merge duplicate contact rows into one dict per person."""

    # ---------- normalizers ----------
    def s(value):
        return "" if value is None else str(value)

    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^<>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"x|ext|#", p, re.IGNORECASE)
        if m:
            p = p[:m.start()]
        digits = re.sub(r"[^0-9]", "", p)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        u = s(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated value: %r" % (raw,))

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize rows, drop those with no valid email/phone ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower()
                         in ("y", "yes", "true", "1"),
            "updated": parse_updated(row.get("updated")),
        })

    # ---------- union-find grouping ----------
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def pick(members, field):
        best = None
        for i in members:  # members are in input order
            r = recs[i]
            if not r[field]:
                continue
            if best is None:
                best = r
                continue
            if r["manual"] != best["manual"]:
                if r["manual"]:
                    best = r
            elif r["updated"] > best["updated"]:
                best = r
        return best[field] if best else ""

    result = []
    for root in sorted(groups, key=lambda g: min(groups[g])):
        members = sorted(groups[root])
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
        })
    return result

Notes on how I read a few of the rules:

- **Opt-outs:** If any row in a merged group is opted out, the whole person is marked `Y`. Filter on `opted_out == "N"` before sending tomorrow's alert.
- **Gmail normalization:** It happens before validation, so an address like `+news@gmail.com` (empty local part after stripping) is invalid.
- **Extension markers:** Any "x" in the phone field cuts the string there, as the rules specify, so stray letters like "x" in text will truncate the number.
- **Merge order:** Rows are merged transitively, and people come out in order of their earliest surviving row. Rows dropped by rule 9 never take part in matching.
- **Bad dates:** An `updated` value in an unexpected format raises `ValueError` rather than being silently skipped. I'd rather it fail loudly than risk a wrong merge.
