import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DIGITS = "0123456789"
_EPOCH = datetime.min


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    """Return the normalized valid email, or "" if invalid."""
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if domain in _GMAIL_DOMAINS:
        local = local.replace(".", "").split("+", 1)[0]
        domain = "gmail.com"
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return f"{local}@{domain}"


def _norm_phone(raw):
    """Return the normalized 10-digit phone, or "" if invalid."""
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Return the normalized 5-digit ZIP, or "" if invalid."""
    s = raw.strip()
    if re.fullmatch(r"[0-9]{4,5}-[0-9]{4}", s):
        s = s.split("-", 1)[0]
    if len(s) == 4 and _is_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_digits(s):
        return s
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _best(group, field):
    """Pick the value of `field` from the best row in the group that has it.

    Preference: source is "manual", then most recent `updated`, then earliest input row.
    """
    cands = [r for r in group if r[field]]
    if not cands:
        return ""
    best = min(
        cands,
        key=lambda r: (
            r["source"] != "manual",
            -(r["dt"] - _EPOCH).total_seconds(),
            r["idx"],
        ),
    )
    return best[field]


def dedupe_contacts(rows):
    # Normalize rows and drop those with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
            "dt": _parse_updated(row.get("updated", "")),
        })

    # Union-find over record positions (positions follow input order).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_seen = {}
    for k, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(email_seen[r["email"]], k)
            else:
                email_seen[r["email"]] = k
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in phone_seen:
                union(phone_seen[key], k)
            else:
                phone_seen[key] = k

    # Group records; groups are ordered by their earliest input row.
    groups = {}
    order = []
    for k in range(len(recs)):
        root = find(k)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[k])

    out = []
    for root in order:
        g = groups[root]
        tags = set().union(*(r["tags"] for r in g))
        out.append({
            "first_name": _best(g, "first_name"),
            "last_name": _best(g, "last_name"),
            "zip": _best(g, "zip"),
            "email": _best(g, "email"),
            "phone": _best(g, "phone"),
            "source": _best(g, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["dt"] for r in g).strftime("%Y-%m-%d"),
        })
    return out

Notes for the send:

- **Filter opted-out people before sending.** The function returns everyone, including people flagged `opted_out == "Y"`. Build the send list with `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N"]`. Because a person is one group, one opt-out row flags the whole person, so an opt-out can't be bypassed by a duplicate row.
- **Assumptions I made where the rules were ambiguous:**
  - ZIP+4 is recognized as 4–5 digits, a hyphen, and 4 digits, so `2139-4307` becomes `02139`. Other hyphenated values become `""`.
  - A display-name email uses the first `<...>` pair in the field.
  - An unparseable `updated` raises `ValueError` rather than being silently skipped, so a bad date fails loudly before the send.
- **Limits of the matching rules:** two records with different emails and different last names are kept as separate people even if they share a phone number. If the export has duplicates like that, they'll still get the email twice, and that needs a rule change, not a code fix.
