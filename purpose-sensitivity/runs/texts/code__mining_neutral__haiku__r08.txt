import re
from datetime import datetime

_DIGITS = "0123456789"
_BRACKETED_EMAIL = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4 = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EPOCH = datetime(1970, 1, 1)
_TRUE_VALUES = {"y", "yes", "true", "1"}


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw
    m = _BRACKETED_EMAIL.search(raw)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        pos = s.find(marker)
        if pos != -1 and pos < cut:
            cut = pos
    digits = [c for c in s[:cut] if c in _DIGITS]
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return "".join(digits)
    return ""


def _normalize_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4.match(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _all_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _recency(dt):
    # Lower sorts first: most recent first; missing dates sort last.
    if dt is None:
        return float("inf")
    return -(dt - _EPOCH).total_seconds()


def _merge(members):
    def pick(field):
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda m: (0 if m["manual"] else 1, _recency(m["updated"]), m["idx"]),
        )
        return best[field]

    dates = [m["updated"] for m in members if m["updated"] is not None]
    latest = max(dates).strftime("%Y-%m-%d") if dates else ""
    tags = sorted({t for m in members for t in m["tags"]})

    return {
        "first_name": pick("first_name"),
        "last_name": pick("last_name"),
        "zip": pick("zip"),
        "email": pick("email"),
        "phone": pick("phone"),
        "source": pick("source"),
        "tags": ";".join(tags),
        "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
        "updated": latest,
    }


def dedupe_contacts(rows):
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email") or "")
        phone = _normalize_phone(row.get("phone") or "")
        last = _normalize_name(row.get("last_name") or "")
        source = (row.get("source") or "").strip().lower()
        tags = []
        for t in re.split(r"[;,]", row.get("tags") or ""):
            t = t.strip().lower()
            if t:
                tags.append(t)
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name") or ""),
            "last_name": last,
            "last_key": last.casefold(),
            "zip": _normalize_zip(row.get("zip") or ""),
            "email": email if _is_valid_email(email) else "",
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": (row.get("opted_out") or "").strip().lower() in _TRUE_VALUES,
            "updated": _parse_updated(row.get("updated") or ""),
        })

    # Rule 9: drop rows with neither a valid email nor a valid phone.
    kept = [r for r in recs if r["email"] or r["phone"]]

    # Rule 10: union-find, so matches are transitive.
    parent = {}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_owner = {}
    phone_owner = {}
    for r in kept:
        i = r["idx"]
        parent[i] = i
        if r["email"]:
            if r["email"] in email_owner:
                union(email_owner[r["email"]], i)
            else:
                email_owner[r["email"]] = i
        if r["phone"] and r["last_key"]:
            key = (r["phone"], r["last_key"])
            if key in phone_owner:
                union(phone_owner[key], i)
            else:
                phone_owner[key] = i

    # Rule 12: groups are created in order of their earliest row.
    groups = {}
    for r in kept:
        groups.setdefault(find(r["idx"]), []).append(r)

    return [_merge(members) for members in groups.values()]

Notes on decisions I made:

- **ZIP+4:** I treat a value as ZIP+4 when it matches 4 or 5 digits, a hyphen, and 4 digits. This covers the Excel case where the prefix lost its leading zero (e.g., `2139-4307` becomes `02139`).
- **Opted-out rows:** Per rule 11, the output includes a `opted_out` flag rather than dropping those people. Filter for `opted_out == "Y"` before sending. A person who has any opted-out row in the group is flagged "Y," so the merged record will be excluded even if other duplicates haven't opted out. That's the conservative result you want.
- **Unparseable `updated` values:** Your rules say every `updated` value is one of the three formats, so I didn't add error handling beyond treating an unparseable value as missing. Missing dates sort last when choosing the preferred row.
