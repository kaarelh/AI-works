import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _is_ascii_digits(s):
    return s != "" and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw
    start = s.find("<")
    if start != -1:
        end = s.find(">", start + 1)
        if end != -1:
            s = s[start + 1:end]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    lower = raw.lower()
    cuts = [i for i in (lower.find(m) for m in ("x", "ext", "#")) if i != -1]
    if cuts:
        lower = lower[:min(cuts)]
    digits = "".join(c for c in lower if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    z = raw.strip()
    parts = z.split("-")
    if len(parts) == 2 and len(parts[1]) == 4 and _is_ascii_digits(parts[1]):
        z = parts[0]
    if len(z) == 4 and _is_ascii_digits(z):
        z = "0" + z
    if len(z) == 5 and _is_ascii_digits(z):
        return z
    return ""


def _normalize_name(raw):
    n = raw.strip()
    if n.isupper() or n.islower():
        n = n.title()
    return n


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in {"y", "yes", "true", "1"}


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(group, field):
    """Choose one row's value for a field: manual first, then most recent
    `updated`, then earliest input row. Normalized empties are falsy, so
    truthiness doubles as the "non-empty (and valid)" check."""
    cands = [r for r in group if r[field]]
    if not cands:
        return ""
    best = min(
        cands,
        key=lambda r: (
            not r["manual"],
            -(r["updated"] - datetime.min).total_seconds(),
            r["idx"],
        ),
    )
    return best[field]


def dedupe_contacts(rows):
    # Normalize every row and drop those with no valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row["email"])
        if not _valid_email(email):
            email = ""
        phone = _normalize_phone(row["phone"])
        if not _valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue
        source = row["source"].strip().lower()
        records.append({
            "idx": idx,
            "first_name": _normalize_name(row["first_name"]),
            "last_name": _normalize_name(row["last_name"]),
            "zip": _normalize_zip(row["zip"]),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row["tags"]),
            "opted_out": _is_opted_out(row["opted_out"]),
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find: link rows sharing a valid email, or a valid phone plus
    # the same non-empty last name (case-insensitive). Transitive by design.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the smaller index as root so a root is always a group's earliest row.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_name = {}
    for i, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in first_by_email:
                union(first_by_email[rec["email"]], i)
            else:
                first_by_email[rec["email"]] = i
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].casefold())
            if key in first_by_phone_name:
                union(first_by_phone_name[key], i)
            else:
                first_by_phone_name[key] = i

    # Group rows. Iterating in index order means groups come out ordered
    # by each group's earliest input row.
    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(records[i])

    result = []
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        result.append({
            "first_name": _pick(group, "first_name"),
            "last_name": _pick(group, "last_name"),
            "zip": _pick(group, "zip"),
            "email": _pick(group, "email"),
            "phone": _pick(group, "phone"),
            "source": _pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return result
