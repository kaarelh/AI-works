```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    ASCII_DIGITS = "0123456789"

    def s(v):
        return "" if v is None else str(v)

    def norm_email(raw):
        v = s(raw)
        m = re.search(r"<([^<>]*)>", v)
        if m:
            v = m.group(1)
        v = v.strip().lower()
        if v.count("@") == 1:
            local, domain = v.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                v = local + "@" + domain
        # validity
        if v.count("@") != 1:
            return ""
        local, domain = v.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return v

    def norm_phone(raw):
        v = s(raw)
        m = re.search(r"ext|x|#", v, re.IGNORECASE)
        if m:
            v = v[:m.start()]
        d = "".join(c for c in v if c in ASCII_DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        v = s(raw).strip()
        if re.fullmatch(r"[0-9]+-[0-9]{4}", v):
            v = v.split("-", 1)[0]
        if re.fullmatch(r"[0-9]{4}", v):
            v = "0" + v
        if re.fullmatch(r"[0-9]{5}", v):
            return v
        return ""

    def norm_name(raw):
        v = s(raw).strip()
        if v.isupper() or v.islower():
            v = v.title()
        return v

    def parse_updated(raw):
        v = s(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(v, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated value: %r" % (raw,))

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize rows and drop those with neither valid email nor valid phone
    recs = []
    for i, r in enumerate(rows):
        email = norm_email(r.get("email"))
        phone = norm_phone(r.get("phone"))
        if not email and not phone:
            continue
        source = s(r.get("source")).strip().lower()
        recs.append({
            "idx": i,
            "first_name": norm_name(r.get("first_name")),
            "last_name": norm_name(r.get("last_name")),
            "zip": norm_zip(r.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": parse_tags(r.get("tags")),
            "opted_out": s(r.get("opted_out")).strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_updated(r.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for k, rec in enumerate(recs):
        if rec["email"]:
            key = rec["email"]
            if key in email_seen:
                union(k, email_seen[key])
            else:
                email_seen[key] = k
        last = rec["last_name"].lower()
        if rec["phone"] and last:
            key = (rec["phone"], last)
            if key in phone_last_seen:
                union(k, phone_last_seen[key])
            else:
                phone_last_seen[key] = k

    groups = {}
    for k in range(n):
        groups.setdefault(find(k), []).append(k)

    # Order people by their earliest row in the input
    ordered = sorted(groups.values(), key=lambda g: min(recs[k]["idx"] for k in g))

    def pick(members, field):
        cands = [recs[k] for k in members if recs[k][field]]
        if not cands:
            return ""
        cands.sort(key=lambda r: r["idx"])                       # earlier row wins last
        cands.sort(key=lambda r: r["updated"], reverse=True)     # then most recent
        cands.sort(key=lambda r: 0 if r["source"] == "manual" else 1)  # manual first
        return cands[0][field]

    result = []
    for members in ordered:
        tags = set()
        for k in members:
            tags |= recs[k]["tags"]
        latest = max(recs[k]["updated"] for k in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[k]["opted_out"] for k in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Opt-outs:** the function doesn't remove opted-out people, it only flags them with `opted_out == "Y"` (if any row in the group opted out). Before sending, filter with something like `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`. People with only a valid phone have an empty `email` and can't receive the email.
- **Bad dates:** an `updated` value in an unrecognized format raises `ValueError` rather than being silently guessed, so you'll find out about bad data instead of mis-ordering records.
- **ZIP+4:** I treat any `digits-4digits` value as ZIP+4, so `2139-4307` becomes `02139`. Anything that doesn't end up as exactly 5 digits becomes `""`.
- **Gmail:** an address like `+news@gmail.com` normalizes to an empty local part and is treated as invalid.
