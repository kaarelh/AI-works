import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"^[0-9]{4,5}-[0-9]{4}$")


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    s = (raw or "").lower()
    cut = len(s)
    for marker in ("x", "ext", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    z = (raw or "").strip()
    if _ZIP_PLUS4_RE.match(z):
        z = z.split("-", 1)[0]
    if len(z) == 4 and _is_digits(z):
        z = "0" + z
    if len(z) == 5 and _is_digits(z):
        return z
    return ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _tags(raw):
    return [t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()]


def _pick(members, recs, field):
    # Candidates: group rows where this normalized field is non-empty.
    # Members are already in input order, so stable sorts keep input order as the final tiebreak.
    cands = [k for k in members if recs[k][field]]
    if not cands:
        return ""
    cands.sort(key=lambda k: recs[k]["updated"], reverse=True)  # most recent first
    cands.sort(key=lambda k: not recs[k]["manual"])             # manual rows first
    return recs[cands[0]][field]


def dedupe_contacts(rows):
    # Steps 1-9: normalize each row and drop rows with no valid email or phone.
    recs = []
    for row in rows:
        email_raw = _norm_email(row.get("email", ""))
        phone_digits = _norm_phone(row.get("phone", ""))
        email_ok = _valid_email(email_raw)
        phone_ok = _valid_phone(phone_digits)
        if not (email_ok or phone_ok):
            continue

        source = (row.get("source") or "").strip().lower()
        recs.append({
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email_raw if email_ok else "",
            "phone": phone_digits if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": _tags(row.get("tags", "")),
            "opted_out": _opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Step 10: union-find over rows that share a valid email, or a valid phone plus last name.
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for k, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], k)
            else:
                first_by_email[r["email"]] = k
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], k)
            else:
                first_by_phone_last[key] = k

    groups = {}
    for k in range(n):
        groups.setdefault(find(k), []).append(k)

    # Step 11-12: one output dict per person, ordered by earliest input row.
    out = []
    for root in dict.fromkeys(find(k) for k in range(n)):
        members = groups[root]
        out.append({
            "first_name": _pick(members, recs, "first_name"),
            "last_name": _pick(members, recs, "last_name"),
            "zip": _pick(members, recs, "zip"),
            "email": _pick(members, recs, "email"),
            "phone": _pick(members, recs, "phone"),
            "source": _pick(members, recs, "source"),
            "tags": ";".join(sorted({t for k in members for t in recs[k]["tags"]})),

            "updated": max(recs[k]["updated"] for k in members).strftime("%Y-%m-%d"),
        })
    return out

Notes on the implementation:

- **Opt-outs are flagged, not removed.** Per rule 11, a person with any opted-out row gets `opted_out == "Y"`, and they still appear in the output. Filter on `opted_out == "N"` before the send so no one who opted out gets the special edition.
- **ZIP+4 pattern:** I read "looks like ZIP+4" as a 4–5 digit prefix, a hyphen, and 4 digits. That handles both `02139-4307` and `2139-4307`. If the data person meant only exactly five digits before the hyphen, change `{4,5}` to `{5}`.
- **Unparseable `updated` raises `ValueError`.** The spec says `updated` is always one of the three formats, so I'd rather fail loudly than silently sort a bad date to the bottom.
- **Matching is linear, not pairwise.** Rows are linked through dictionaries keyed by email and by (phone, last name), so large CRM exports don't need an O(n²) comparison.
