import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"


def dedupe_contacts(rows):
    # ---------- normalizers ----------
    def s(v):
        return "" if v is None else str(v)

    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^<>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"ext|x|#", p, re.IGNORECASE)
        if m:
            p = p[:m.start()]
        d = "".join(c for c in p if c in _DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        u = s(raw).strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable: treated as oldest

    def is_opted_out(raw):
        return s(raw).strip().lower() in {"y", "yes", "true", "1"}

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize rows, drop rows with no valid contact ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        src = s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": src,
            "manual": src == "manual",
            "tags": parse_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
        })

    n = len(recs)

    # ---------- union-find ----------
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(n):  # ascending, so dict order = earliest row order
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    result = []
    for members in groups.values():
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.date().isoformat(),
        })
    return result

- **Opt-outs:** `opted_out` is "Y" if any row in the group opted out, so filter the send list to `opted_out == "N"`. Also skip anyone whose output `email` is empty (phone-only people), since they have no address to email.
- **Merging:** The phone + last name rule can merge distinct people who share a phone, such as family members. Under rule 11, one person's opt-out then suppresses the whole merged record. That errs toward not over-emailing, but it may drop a few real supporters.
- **Bad dates:** An unparseable `updated` value is treated as the oldest possible date instead of raising an error. If you'd rather it fail loudly, replace the `return datetime.min` line with a `raise`.
