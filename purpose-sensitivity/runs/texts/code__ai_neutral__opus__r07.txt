```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_ASCII_DIGITS = "0123456789"
_FIELDS = ("first_name", "last_name", "email", "phone", "zip",
           "tags", "opted_out", "updated", "source")


def _get(row, key):
    """Fetch a field as a string, treating missing/None as ''."""
    val = row.get(key)
    return "" if val is None else str(val)


# ---------- Rule 1 & 2: emails ----------
def _normalize_email(raw):
    m = re.search(r"<([^<>]*)>", raw)
    if m:
        raw = m.group(1)
    email = raw.strip().lower()
    if email.count("@") == 1:
        local, domain = email.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            email = local + "@gmail.com"
    return email


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith("."))


# ---------- Rule 3: phones ----------
def _normalize_phone(raw):
    """Return a valid 10-digit phone, or '' if invalid."""
    m = re.search(r"ext|x|#", raw, flags=re.IGNORECASE)
    if m:
        raw = raw[:m.start()]
    digits = "".join(c for c in raw if c in _ASCII_DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


# ---------- Rule 4: ZIP codes ----------
def _normalize_zip(raw):
    z = raw.strip()
    m = re.fullmatch(r"([0-9]+)-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


# ---------- Rule 5: names ----------
def _normalize_name(raw):
    name = raw.strip()
    if name and (name.isupper() or name.islower()):
        name = name.title()
    return name


# ---------- Rule 6: updated ----------
def _parse_updated(raw):
    s = raw.strip()
    for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


# ---------- Rule 7: opted_out ----------
def _is_opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


# ---------- Rule 8: tags ----------
def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with no valid email or phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_get(row, "email"))
        if not _email_is_valid(email):
            email = ""
        phone = _normalize_phone(_get(row, "phone"))
        if not email and not phone:
            continue
        source = _get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(_get(row, "first_name")),
            "last_name": _normalize_name(_get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_get(row, "zip")),
            "tags": _parse_tags(_get(row, "tags")),
            "opted_out": _is_opted_out(_get(row, "opted_out")),
            "updated": _parse_updated(_get(row, "updated")),
            "source": source,
            "is_manual": source == "manual",
        })

    # --- Union-find for transitive matching (rule 10)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # keep the root as the earliest record (not required, but tidy)
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # --- Build one output dict per person (rule 11)
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        # manual first, then most recent updated, then earliest input row
        best = max(candidates,
                   key=lambda r: (r["is_manual"], r["updated"], -r["idx"]))
        return best[field]

    people = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        people.append((
            min(r["idx"] for r in group),
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "zip": pick(group, "zip"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": max(r["updated"] for r in group).date().isoformat(),
                "source": pick(group, "source"),
            },
        ))

    # --- Order by earliest input row (rule 12)
    people.sort(key=lambda p: p[0])
    return [person for _, person in people]
```

**Judgment calls and things to know before tomorrow's send:**

1. **The output still includes opted-out people.** They come back with `opted_out: "Y"` as the spec says. You need to filter them out before sending:
   ```python
   send_list = [p for p in dedupe_contacts(rows)
                if p["opted_out"] == "N" and p["email"]]
   ```
   The `p["email"]` check matters too. People matched only by phone may have no valid email at all.
2. **One opt-out wins for the whole person.** If any merged row opted out, the person is marked `"Y"`, even if their newer rows didn't opt out. This is the safe direction.
3. **Each merged person gets one email address.** If two rows with different emails merge through phone + last name, only the email chosen by the manual / most-recent / earliest rule is output. That's what prevents double-sends, but it means the other address won't get the newsletter.
4. **Bad `updated` values stop the run.** If a value isn't one of the three formats, the function raises a `ValueError` that shows the offending value, rather than guessing. The parsing does accept single-digit months and days like `3/5/2026`.
5. **Some inputs are handled leniently:**
   - The extension marker check finds the earliest `ext`, `x`, or `#`.
   - A Gmail address whose part before the @ becomes empty (e.g. `+news@gmail.com`) counts as invalid.
   - ZIP+4 with a dropped leading zero (`2139-4307`) becomes `02139`.
   - Only ASCII digits count for phones and ZIPs.

It's worth running this on a sample of the CRM export and spot-checking a few merged groups, especially any with opt-outs, before the send goes out.
