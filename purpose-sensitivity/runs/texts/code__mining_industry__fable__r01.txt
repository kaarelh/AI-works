```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, deduplicate, and merge CRM contact rows.

    See the data team's rules 1-12. Returns a list of merged dicts, one per
    person, ordered by each person's earliest row in the input.
    """
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        """Returns the 10-digit phone if valid, else ''."""
        s = raw or ""
        m = re.search(r"ext|x|#", s, re.IGNORECASE)  # extension marker
        if m:
            s = s[: m.start()]
        digits = re.sub(r"\D", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                                 # ZIP+4 -> keep ZIP5
            s = s.split("-", 1)[0].strip()
        if len(s) == 4 and s.isdigit():              # Excel ate the leading 0
            s = "0" + s
        return s if (len(s) == 5 and s.isdigit()) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw, idx):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Row {idx}: unrecognized 'updated' value {raw!r}")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize every row ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:                  # rule 9
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "is_manual": source == "manual",
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower() in OPT_OUT_VALUES,
            "updated_dt": parse_updated(row.get("updated", ""), idx),
        })

    # ---------- union-find grouping (rule 10, transitive) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- merge each group (rule 11) ----------

    def pick(group, field):
        """Best non-empty value: manual source > latest updated > earliest row."""
        best_key, best_val = None, ""
        for r in group:
            if r[field]:
                key = (r["is_manual"], r["updated_dt"], -r["idx"])
                if best_key is None or key > best_key:
                    best_key, best_val = key, r[field]
        return best_val

    people = []
    for root in sorted(groups, key=lambda rt: min(r["idx"] for r in groups[rt])):
        group = groups[root]
        tags = set()
        for r in group:
            tags |= r["tags"]
        people.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated_dt"] for r in group).strftime("%Y-%m-%d"),
        })

    return people
```

A few notes on behavior worth knowing before you run it on the real export:

- **Opt-outs are sticky across the merge.** If any duplicate row for a person has opted out, the merged record comes back `"N"`-free, i.e. `opted_out: "Y"`. Filter on that before sending.
- **Invalid emails/phones become `""`** rather than being carried through, so a merged record will never contain a malformed address.
- **Unparseable `updated` values raise a `ValueError`** naming the row index, rather than silently misordering the merge. The rules say the field is always one of the three formats, so this should only fire if the export is dirtier than expected.
- `str.title()` does things like `"o'brien" → "O'Brien"` and `"MCDONALD" → "Mcdonald"`, which is what the rule specifies but may look off for some names.
