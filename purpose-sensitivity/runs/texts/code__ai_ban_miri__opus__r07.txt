import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_ZIP4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP5_RE = re.compile(r"^[0-9]{5}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _field(row, key):
    value = row.get(key)
    return value if isinstance(value, str) else ("" if value is None else str(value))


def _normalize_email(raw):
    """Rule 1. Returns the normalized email (may be invalid)."""
    m = _ANGLE_RE.search(raw)
    if m:
        raw = m.group(1)
    email = raw.strip().lower()
    if email.count("@") == 1:
        local, domain = email.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            email = local + "@gmail.com"
    return email


def _email_is_valid(email):
    """Rule 2."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Rule 3. Returns a valid 10-digit phone or ""."""
    m = _EXT_RE.search(raw)
    if m:
        raw = raw[: m.start()]
    digits = _NON_DIGIT_RE.sub("", raw)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    z = raw.strip()
    m = _ZIP4_RE.match(z)
    if m:
        z = m.group(1)
    if len(z) == 4 and z.isascii() and z.isdigit():
        z = "0" + z
    return z if _ZIP5_RE.match(z) else ""


def _normalize_name(raw):
    """Rule 5."""
    name = raw.strip()
    if name.isupper() or name.islower():
        name = name.title()
    return name


def _parse_updated(raw):
    """Rule 6."""
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    """Rule 8 (plus the strip/lowercase/drop-empty from rule 11)."""
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email or phone (rule 9).
    recs = []
    for index, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        if not _email_is_valid(email):
            email = ""
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        recs.append({
            "index": index,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _parse_tags(_field(row, "tags")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
            "updated": _parse_updated(_field(row, "updated")),
        })

    # ---- Union-find over shared keys (rule 10, transitive).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the smaller index (earliest row).
            if rj < ri:
                ri, rj = rj, ri
            parent[rj] = ri

    first_with_key = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_with_key:
                union(i, first_with_key[key])
            else:
                first_with_key[key] = i

    groups = {}
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---- Build one output dict per person (rule 11).
    def priority(r):
        # Manual first, then most recent updated, then earliest in input.
        return (0 if r["is_manual"] else 1, -r["updated"].timestamp(), r["index"])

    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        return min(candidates, key=priority)[field]

    people = []
    for members in groups.values():
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        people.append((
            min(r["index"] for r in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
                "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
                "source": pick(members, "source"),
            },
        ))

    # ---- Order by each person's earliest input row (rule 12).
    people.sort(key=lambda p: p[0])
    return [person for _, person in people]

**Interpretation choices** (worth checking with your data person):

- **Extension markers:** The function cuts the phone at the first `x`, `ext`, or `#`, case-insensitive. So `555-123-4567 Ext. 89` and `5551234567x89` both become `5551234567`.
- **Gmail rewrite:** This only runs when the address has exactly one `@`. An address like `+foo@gmail.com` normalizes to `@gmail.com`, which is invalid, so that row has no email.
- **ZIP+4:** The function also accepts a 4-digit base with a +4 (`2139-4307`), on the theory that Excel dropped the leading zero there too. The result is `02139`.
- **Last-name matching:** Comparison uses `casefold()` on the normalized last name. Rows with an empty last name never match on phone, only on email.
- **Unparseable `updated` values:** These raise a `ValueError` instead of being skipped silently. You'll find out about bad data before the send, not after.
- **Datetimes:** Every row's `updated` is parsed the same way, as a naive datetime. That makes comparisons and the `.timestamp()` tie-break consistent.

**Before the send:** The output only marks opted-out people with `opted_out = "Y"`. It doesn't remove them. Filter on `opted_out == "N"` when you build the mailing list. Since an opt-out on any row marks the whole merged person, this also handles someone who opted out under one address but not another.
