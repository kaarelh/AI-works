```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    _MARKER_RE = re.compile(r"ext|x|#", re.IGNORECASE)
    _ANGLE_RE = re.compile(r"<([^<>]*)>")
    _ZIP4_RE = re.compile(r"^[0-9]{5}-[0-9]{4}$")
    _DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    _TRUE_VALUES = {"y", "yes", "true", "1"}

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                domain = "gmail.com"
                s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        if not local:
            return False
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return False
        return True

    def norm_phone(raw):
        m = _MARKER_RE.search(raw)
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        if _ZIP4_RE.match(z):
            z = z.split("-", 1)[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        return datetime.min  # unparseable: treat as oldest

    # ---- normalize rows, dropping those with no valid email/phone ----
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        tags = set()
        for t in re.split(r"[;,]", get(row, "tags")):
            t = t.strip().lower()
            if t:
                tags.add(t)
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": get(row, "source").strip().lower(),
            "tags": tags,
            "opted_out": get(row, "opted_out").strip().lower() in _TRUE_VALUES,
            "updated": parse_updated(get(row, "updated")),
        })

    # ---- union-find grouping ----
    n = len(recs)
    parent = list(range(n))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # ---- build output ----
    def pick(members, field):
        best = None
        for i in members:  # members are in input order
            r = recs[i]
            if not r[field]:
                continue
            if best is None:
                best = r
                continue
            r_manual = r["source"] == "manual"
            b_manual = best["source"] == "manual"
            if r_manual != b_manual:
                if r_manual:
                    best = r
            elif r["updated"] > best["updated"]:
                best = r
        return best[field] if best else ""

    result = []
    for root in order:
        members = groups[root]
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.date().isoformat(),
            "source": pick(members, "source"),
        })
    return result
```

Notes on a few judgment calls:

- **Opt-outs:** Because opt-out is group-wide, if any duplicate row opted out, the merged person is `"Y"`. Before sending, filter on `opted_out == "N"`.
- **Order of rules:** Rows with no valid email and no valid phone are dropped before matching, so they can't contribute tags, opt-outs, or dates to a group. A dropped row that carries an opt-out therefore won't suppress anyone. If you'd rather be conservative, run a separate opt-out check on the raw rows.
- **Unparseable `updated` values:** The spec says these can't occur, but any that do are treated as the oldest possible date rather than raising an error.
- **ZIP+4:** Only the strict `#####-####` pattern is treated as ZIP+4. Anything else that isn't 5 digits (after the 4-digit fix) becomes `""`.
- **Last-name matching:** Phone-plus-last-name matching uses the normalized last name lowercased, and rows with an empty last name never match on phone alone.
