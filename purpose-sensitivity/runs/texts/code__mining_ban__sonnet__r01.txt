import re
from datetime import datetime


def dedupe_contacts(rows):
    ASCII_DIGITS = "0123456789"

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, _, domain = s.partition("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, _, domain = e.partition("@")
        return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"x|ext|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        digits = "".join(c for c in s if c in ASCII_DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s and (s.isupper() or s.islower()):
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min

    def parse_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw or ""):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # Normalize rows and drop those with no valid email and no valid phone.
    recs = []
    for r in rows:
        email = norm_email(r.get("email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(r.get("phone"))
        if not email and not phone:
            continue
        source = (r.get("source") or "").strip().lower()
        recs.append({
            "first_name": norm_name(r.get("first_name")),
            "last_name": norm_name(r.get("last_name")),
            "zip": norm_zip(r.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(r.get("tags")),
            "opted_out": (r.get("opted_out") or "").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_updated(r.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in email_seen:
                union(i, email_seen[rec["email"]])
            else:
                email_seen[rec["email"]] = i
        last = rec["last_name"].lower()
        if rec["phone"] and last:
            key = (rec["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    def pick(indices, field):
        best = None
        for i in indices:  # indices are in input order
            rec = recs[i]
            if not rec[field]:

            elif rec["updated"] > best["updated"]:

        return best[field] if best else ""

Notes on a few judgment calls where the rules were silent:

- **Gmail rules** apply only when the address has exactly one `@`. Anything with multiple `@` signs stays invalid.
- **Unparseable `updated` values** are treated as the earliest possible datetime, so they never win a "most recent" comparison. Per the spec this shouldn't occur.
- **Last-name matching** for the phone rule uses the normalized (stripped, title-cased where applicable) last name, compared with `.lower()`.
- **Grouping** uses union-find, so chains like A–B (email) and B–C (phone + last name) merge into one person.
- **Opt-outs** are group-wide: if any row for a person is opted out, the output is `"Y"`. Your send step should skip every row with `opted_out == "Y"`.
