from datetime import datetime

_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_TRUE_VALUES = {"y", "yes", "true", "1"}


def _is_digits(s):
    return s != "" and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw.strip()
    start = s.find("<")
    if start != -1:
        end = s.find(">", start + 1)
        if end != -1:
            s = s[start + 1:end]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (local != ""
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith("."))


def _normalize_phone(raw):
    """Return the 10-digit phone, or "" if it is not valid."""
    s = raw.lower()
    cuts = [i for i in (s.find(m) for m in ("x", "ext", "#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    z = raw.strip()
    head, sep, tail = z.partition("-")
    # ZIP+4 such as "02139-4307": keep the part before the hyphen.
    if sep and _is_digits(head) and len(tail) == 4 and _is_digits(tail):
        z = head
    # Excel dropped the leading zero.
    if len(z) == 4 and _is_digits(z):
        z = "0" + z
    if len(z) == 5 and _is_digits(z):
        return z
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    parts = raw.replace(",", ";").split(";")
    return {p.strip().lower() for p in parts if p.strip()}


def _pick(group, field):
    """Pick a field value: manual rows first, then most recent, then earliest input row."""
    candidates = [r for r in group if r[field]]
    if not candidates:
        return ""
    best = max(candidates, key=lambda r: (r["manual"], r["updated"], -r["index"]))
    return best[field]


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one dict per person.

    Each output dict has: first_name, last_name, zip, email, phone, source,
    tags (";"-joined), opted_out ("Y"/"N"), updated ("YYYY-MM-DD").
    Opted-out people are still returned with opted_out == "Y"; filter them
    out before sending.
    """
    # Step 1: normalize rows and drop those with no valid email or phone.
    records = []
    for i, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = row.get("source", "").strip().lower()
        records.append({
            "index": i,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _TRUE_VALUES,
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Step 2: group rows into people with union-find (handles transitivity).
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_seen = {}
    for pos, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in email_seen:
                union(pos, email_seen[rec["email"]])
            else:
                email_seen[rec["email"]] = pos
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].lower())
            if key in phone_seen:
                union(pos, phone_seen[key])
            else:
                phone_seen[key] = pos

    groups = {}
    for pos, rec in enumerate(records):
        groups.setdefault(find(pos), []).append(rec)

    # Step 3: build one output dict per person, ordered by earliest input row.
    # Each group's list is already in input order, so group[0] is the earliest.
    ordered = sorted(groups.values(), key=lambda g: g[0]["index"])

    output = []
    for group in ordered:
        all_tags = set().union(*(r["tags"] for r in group))
        output.append({
            "first_name": _pick(group, "first_name"),
            "last_name": _pick(group, "last_name"),
            "zip": _pick(group, "zip"),
            "email": _pick(group, "email"),
            "phone": _pick(group, "phone"),
            "source": _pick(group, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return output
