```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    def field(row, key):
        v = row.get(key)
        return "" if v is None else str(v)

    # ---- rule 1 & 2: email ----
    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    # ---- rule 3: phone ----
    def norm_phone(raw):
        base = re.split(r"ext|x|#", raw, maxsplit=1, flags=re.IGNORECASE)[0]
        digits = re.sub(r"[^0-9]", "", base)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return re.fullmatch(r"[2-9][0-9]{9}", p) is not None

    # ---- rule 4: zip ----
    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    # ---- rule 5: names ----
    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    # ---- rule 6: updated ----
    def parse_updated(raw):
        v = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(v, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable: treat as oldest possible

    # ---- rule 7 & 8: opted_out, tags ----
    def parse_opted_out(raw):
        return raw.strip().lower() in OPT_OUT_VALUES

    def parse_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---- normalize every row; rule 9: drop rows with no valid contact ----
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(field(row, "email"))
        phone = norm_phone(field(row, "phone"))
        ev = email_valid(email)
        pv = phone_valid(phone)
        if not ev and not pv:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(field(row, "first_name")),
            "last_name": norm_name(field(row, "last_name")),
            "email": email if ev else "",
            "phone": phone if pv else "",
            "zip": norm_zip(field(row, "zip")),
            "tags": parse_tags(field(row, "tags")),
            "opted_out": parse_opted_out(field(row, "opted_out")),
            "updated": parse_updated(field(row, "updated")),
            "source": field(row, "source").strip().lower(),
        })

    # ---- rule 10: transitive matching via union-find ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # ---- rule 11: build one output dict per person ----
    def pick(members, name):
        cands = [recs[i] for i in members if recs[i][name]]
        if not cands:
            return ""
        best = max(
            cands,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[name]

    out = []
    # rule 12: order by earliest row in the input
    for root in sorted(groups, key=lambda g: recs[g]["idx"]):
        members = groups[root]
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return out
```

A few notes on how I resolved details the rules leave open:

- **Gmail rewrite** only applies when the address has exactly one `@` (otherwise it's invalid anyway and is left as-is).
- **Phone extension markers**: the split looks for the earliest of `ext`, `x`, or `#`; since the digits are extracted afterward, a bare `x` inside `ext` gives the same result either way.
- **ZIP**: anything with a hyphen is treated as ZIP+4 and truncated at the first hyphen; a non-numeric remainder becomes `""`.
- **Names**: `str.isupper()`/`islower()` require at least one cased character, so names with no letters (or already mixed case like "McDonald") are left untouched.
- **Field selection** uses the key `(source == "manual", updated, -input_index)`, so manual wins first, then the newest `updated`, then the earliest row — exactly the priority in rule 11. Because the `source` field is chosen the same way, a group containing any manual row will output `"manual"`.
- **Unparseable `updated` values** (which the spec says shouldn't occur) are treated as `datetime.min` rather than crashing the run. If you'd prefer a hard failure so bad data gets caught before send, change the `return datetime.min` line to `raise ValueError(f"Bad updated value: {raw!r}")`.

Given this is going out to the whole subscriber list tomorrow, it's worth running the output through a quick sanity check (e.g. count of `opted_out == "Y"`, and spot-checking a few known duplicate pairs) before you load it into the mailer.
