import re
from datetime import datetime

_DIGITS = frozenset("0123456789")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPTED_OUT_VALUES = frozenset({"y", "yes", "true", "1"})


def _get(row, key):
    value = row.get(key)
    return "" if value is None else str(value)


def _is_ascii_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(addr):
    if addr.count("@") != 1:
        return False
    local, domain = addr.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw
    m = re.search(r"x|ext|#", s, re.IGNORECASE)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and _is_ascii_digits(digits) and digits[0] not in "01"


def _norm_zip(raw):
    z = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if len(z) == 4 and _is_ascii_digits(z):
        z = "0" + z
    if len(z) == 5 and _is_ascii_digits(z):
        return z
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _norm_tags(raw):
    parts = re.split(r"[;,]", raw)
    return {p.strip().lower() for p in parts if p.strip()}


def _is_opted_out(raw):
    return raw.strip().lower() in _OPTED_OUT_VALUES


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    return None


def _pick_value(recs, members, field, ok_flag=None):
    """Choose a field value from a group's rows.

    Preference: source == "manual", then most recent `updated`,
    then earliest row in the input. Only rows where the field is
    non-empty (or valid, if ok_flag is given) qualify.
    """
    if ok_flag:
        cands = [p for p in members if recs[p][ok_flag]]
    else:
        cands = [p for p in members if recs[p][field]]
    if not cands:
        return ""
    # Stable sorts: members are already in input order, so ties keep input order.
    cands.sort(key=lambda p: recs[p]["updated"] or datetime.min, reverse=True)
    cands.sort(key=lambda p: not recs[p]["manual"])
    return recs[cands[0]][field]


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one output dict per person.

    See the data rules for normalization, matching, and selection details.
    Note: opted-out people are included in the output with opted_out == "Y";
    filter them out before sending.
    """
    # Step 1: normalize every row and drop rows with no valid email or phone.
    recs = []
    for row in rows:
        email = _norm_email(_get(row, "email"))
        phone = _norm_phone(_get(row, "phone"))
        email_ok = _valid_email(email)
        phone_ok = _valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        source = _get(row, "source").strip().lower()
        recs.append({
            "first": _norm_name(_get(row, "first_name")),
            "last": _norm_name(_get(row, "last_name")),
            "zip": _norm_zip(_get(row, "zip")),
            "email": email,
            "email_ok": email_ok,
            "phone": phone,
            "phone_ok": phone_ok,
            "source": source,
            "manual": source == "manual",
            "tags": _norm_tags(_get(row, "tags")),
            "opted_out": _is_opted_out(_get(row, "opted_out")),
            "updated": _parse_updated(_get(row, "updated")),
        })

    # Step 2: union-find. Rows that share a valid email, or a valid phone
    # plus the same non-empty last name, are joined. Shared-key indexing
    # keeps this linear, and union-find gives the transitive closure.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    first_by_email = {}
    first_by_phone_last = {}
    for pos, r in enumerate(recs):
        if r["email_ok"]:
            key = r["email"]
            if key in first_by_email:
                union(pos, first_by_email[key])
            else:
                first_by_email[key] = pos
        if r["phone_ok"] and r["last"]:
            key = (r["phone"], r["last"].lower())
            if key in first_by_phone_last:
                union(pos, first_by_phone_last[key])
            else:
                first_by_phone_last[key] = pos

    # Step 3: group rows. Dict insertion order is each group's earliest row.
    groups = {}
    for pos in range(len(recs)):
        groups.setdefault(find(pos), []).append(pos)

    # Step 4: build one output dict per person.
    result = []
    for members in groups.values():
        all_tags = set().union(*(recs[p]["tags"] for p in members))
        dates = [recs[p]["updated"] for p in members if recs[p]["updated"]]
        result.append({
            "first_name": _pick_value(recs, members, "first"),
            "last_name": _pick_value(recs, members, "last"),
            "zip": _pick_value(recs, members, "zip"),
            "email": _pick_value(recs, members, "email", ok_flag="email_ok"),
            "phone": _pick_value(recs, members, "phone", ok_flag="phone_ok"),
            "source": _pick_value(recs, members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(recs[p]["opted_out"] for p in members) else "N",
            "updated": max(dates).strftime("%Y-%m-%d") if dates else "",
        })
    return result

**Notes on interpretation:**

- **Opt-outs are not removed.** As specified, the function marks people with `opted_out == "Y"` and keeps them in the output. Filter those out before the send. Because duplicates are merged, one opted-out row marks the whole person as opted out, which is the conservative choice.
- **ZIP+4:** I accept a 4- or 5-digit prefix before the hyphen, so `2139-4307` becomes `02139`. This matches the leading-zero rule.
- **Unparseable `updated` values** are treated as older than any valid date. If a group has no valid dates, its `updated` is `""`.
- **Matching** uses shared keys with union-find, so it runs in linear time on large exports.
