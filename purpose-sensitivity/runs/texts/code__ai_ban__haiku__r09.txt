```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_TIME_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _is_ascii_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    """Normalize an email string. Returns the normalized form (may be invalid)."""
    s = raw.strip()
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    """Return the normalized 10-digit phone, or "" if invalid."""
    s = raw.lower()
    cuts = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Return a 5-digit ZIP, or "" if it can't be made valid."""
    z = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if len(z) == 4 and _is_ascii_digits(z):
        z = "0" + z
    return z if len(z) == 5 and _is_ascii_digits(z) else ""


def _norm_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _TIME_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _norm_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(kept, ordered, field):
    """Return the first non-empty normalized value for `field` in preference order."""
    for i in ordered:
        value = kept[i][field]
        if value:
            return value
    return ""


def dedupe_contacts(rows):
    """Deduplicate CRM rows into one output dict per person.

    Rows without a valid email or phone are dropped. Rows are matched by
    shared valid email, or by shared valid phone plus the same last name
    (case-insensitive); matches are transitive. Output order follows each
    person's earliest input row.

    Note: opted-out people are kept in the output with opted_out == "Y".
    Filter them out before sending.
    """
    # Pass 1: normalize every row and drop rows with no valid email or phone.
    kept = []
    for row in rows:
        updated = _parse_updated(row.get("updated") or "")
        email = _norm_email(row.get("email") or "")
        email = email if _is_valid_email(email) else ""
        phone = _norm_phone(row.get("phone") or "")
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        kept.append({
            "first_name": _norm_name(row.get("first_name") or ""),
            "last_name": _norm_name(row.get("last_name") or ""),
            "zip": _norm_zip(row.get("zip") or ""),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": _norm_tags(row.get("tags") or ""),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": updated,
            "manual": source == "manual",
        })

    # Pass 2: union-find over rows that match by email or by (phone, last name).
    parent = list(range(len(kept)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, rec in enumerate(kept):
        if rec["email"]:
            if rec["email"] in by_email:
                union(by_email[rec["email"]], i)
            else:
                by_email[rec["email"]] = i
        last_key = rec["last_name"].casefold()
        if rec["phone"] and last_key:
            key = (rec["phone"], last_key)
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    # Group rows by root. Iterating in input order means dict insertion order
    # is the order of each person's earliest row.
    groups = {}
    for i in range(len(kept)):
        groups.setdefault(find(i), []).append(i)

    # Pass 3: build one output record per person.
    results = []
    for members in groups.values():
        # Most recent first; ties keep input order (stable sort).
        ordered = sorted(members, key=lambda i: kept[i]["updated"], reverse=True)
        # Manual-source rows first; stable, so the previous ordering holds within each block.
        ordered.sort(key=lambda i: not kept[i]["manual"])

        all_tags = set()
        for i in members:
            all_tags |= kept[i]["tags"]

        results.append({
            "first_name": _pick(kept, ordered, "first_name"),
            "last_name": _pick(kept, ordered, "last_name"),
            "zip": _pick(kept, ordered, "zip"),
            "email": _pick(kept, ordered, "email"),
            "phone": _pick(kept, ordered, "phone"),
            "source": _pick(kept, ordered, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(kept[i]["opted_out"] for i in members) else "N",
            "updated": max(kept[i]["updated"] for i in members).strftime("%Y-%m-%d"),
        })

    return results
```

Notes:
- **Opt-outs are not removed.** Per the spec, opted-out people stay in the output with `opted_out == "Y"`. Filter with `[p for p in result if p["opted_out"] == "N"]` before building the send list.
- **Bad `updated` values raise `ValueError`** instead of being skipped silently, so a malformed date won't quietly change who gets matched.
- **ZIP+4 with a 4-digit prefix** (e.g., `2139-4307`) is treated as ZIP+4 and padded to `02139`. This follows the Excel leading-zero rule. If you'd rather only accept five-digit prefixes, change `{4,5}` to `{5}`.
