import re
from datetime import datetime

_EPOCH = datetime(1970, 1, 1)
_DIGITS = "0123456789"


def dedupe_contacts(rows):
    # ---------- normalizers ----------
    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"x|ext|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        d = "".join(c for c in s if c in _DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated value: %r" % (raw,))

    def is_opted_out(raw):
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def parse_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw or ""):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize and filter rows ----------
    recs = []
    for idx, r in enumerate(rows):
        email = norm_email(r.get("email"))
        phone = norm_phone(r.get("phone"))
        if not email and not phone:
            continue
        dt = parse_updated(r.get("updated"))
        recs.append({
            "idx": idx,
            "first_name": norm_name(r.get("first_name")),
            "last_name": norm_name(r.get("last_name")),
            "zip": norm_zip(r.get("zip")),
            "email": email,
            "phone": phone,
            "source": (r.get("source") or "").strip().lower(),
            "tags": parse_tags(r.get("tags")),
            "opted_out": is_opted_out(r.get("opted_out")),
            "updated": dt,
            "ts": (dt - _EPOCH).total_seconds(),
        })

    # ---------- union-find ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root (not required, just tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            k = rec["email"]
            if k in by_email:
                union(i, by_email[k])
            else:
                by_email[k] = i
        if rec["phone"] and rec["last_name"]:
            k = (rec["phone"], rec["last_name"].lower())
            if k in by_phone_last:
                union(i, by_phone_last[k])
            else:
                by_phone_last[k] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output ----------
    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda r: (0 if r["source"] == "manual" else 1, -r["ts"], r["idx"]),
        )
        return best[field]

    ordered = sorted(groups.values(), key=lambda g: min(r["idx"] for r in g))

    result = []
    for g in ordered:
        tags = set()
        for r in g:
            tags.update(r["tags"])
        latest = max(r["updated"] for r in g)
        result.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "zip": pick(g, "zip"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "source": pick(g, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result

Notes on choices I made where the rules left room:

- **Gmail handling** only applies when the address has exactly one `@`. If the local part ends up empty after removing the `+` suffix (e.g. `+news@gmail.com`), the email is invalid.
- **Invalid emails and phones** are treated as empty, so they never count for matching or for picking output values.
- **Name matching** for rule 10 uses the normalized last name (after rule 5), lowercased.
- **The "manual" preference** also applies to the `source` field itself, so a group with a manual row outputs `"manual"`.
- **Opt-outs** only consider rows that survive rule 9. A row dropped for having no valid email or phone can't mark anyone as opted out. If you want opt-outs from dropped rows honored, tell me, since that's a safety consideration for tomorrow's send.
- **Unparseable `updated` values** raise a `ValueError` rather than being silently guessed. Better to catch a bad row before sending than to mis-merge.
