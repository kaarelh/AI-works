import re
from datetime import datetime

_EMAIL_IN_BRACKETS = re.compile(r"<([^<>]*)>")
_EXT_MARKER = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_FIVE_DIGITS = re.compile(r"[0-9]{5}")
_FOUR_DIGITS = re.compile(r"[0-9]{4}")
_TAG_SPLIT = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the data team's rules.

    Returns one dict per person, in order of each person's earliest row.
    Opted-out people are still included (opted_out == "Y"); filter them
    out before sending.
    """

    def field(row, name):
        value = row.get(name)
        return value if isinstance(value, str) else ("" if value is None else str(value))

    # ---- Normalizers -------------------------------------------------

    def norm_email(raw):
        s = raw
        m = _EMAIL_IN_BRACKETS.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_is_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return (
            local != ""
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        m = _EXT_MARKER.search(raw)
        s = raw[: m.start()] if m else raw
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_is_valid(digits):
        return len(digits) == 10 and digits[0] not in "01"

    def norm_zip(raw):
        s = raw.strip()
        m = _ZIP_PLUS4.fullmatch(s)
        if m:
            s = m.group(1)
        if _FOUR_DIGITS.fullmatch(s):
            s = "0" + s
        return s if _FIVE_DIGITS.fullmatch(s) else ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return raw.strip().lower() in _OPT_OUT_VALUES

    def parse_tags(raw):
        return {t.strip().lower() for t in _TAG_SPLIT.split(raw) if t.strip()}

    # ---- Normalize rows; drop rows with no valid email or phone (rule 9)

    recs = []  # each rec keeps its original input index
    for idx, row in enumerate(rows):
        email = norm_email(field(row, "email"))
        phone = norm_phone(field(row, "phone"))
        email_ok = email_is_valid(email)
        phone_ok = phone_is_valid(phone)
        if not (email_ok or phone_ok):
            continue
        source = field(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(field(row, "first_name")),
            "last_name": norm_name(field(row, "last_name")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "zip": norm_zip(field(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(field(row, "tags")),
            "opted_out": is_opted_out(field(row, "opted_out")),
            "updated": parse_updated(field(row, "updated")),
        })

    # ---- Union-find grouping (rule 10) --------------------------------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], i)
            else:
                first_by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # ---- Build output (rules 11, 12) ----------------------------------

    def preference(i):
        r = recs[i]
        # manual first, then most recent updated, then earliest input row
        return (0 if r["manual"] else 1, datetime.max - r["updated"], r["idx"])

    def pick(members, name):
        candidates = [i for i in members if recs[i][name]]
        if not candidates:
            return ""
        return recs[min(candidates, key=preference)][name]

    ordered = sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m))

    out = []
    for members in ordered:
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
            "source": pick(members, "source"),
        })
    return out

A few notes before tomorrow's send:

- **Filter out opt-outs before sending.** The function follows the spec and keeps opted-out people in the output with `opted_out: "Y"`. Your send list should be something like `[c for c in dedupe_contacts(rows) if c["opted_out"] == "N" and c["email"]]`.
- **Opt-outs only carry over when rows actually match.** If someone opted out under one email and also appears under a different email, with no shared phone and last name, they'll still look like two people and the second one will get the email. If you have a separate suppression list, apply it as a final check.
- **Bad dates fail loudly.** If an `updated` value doesn't match one of the three formats, the function raises a `ValueError` naming the bad value. I chose that over guessing silently.
- **Judgment calls on unclear points:**
  - A ZIP+4 whose first part lost its leading zero (e.g. `2139-4307`) is treated as ZIP+4 and padded to `02139`.
  - Gmail normalization only runs when the address has exactly one `@`. Addresses with more than one are invalid anyway.
  - Only ASCII digits 0–9 count for phones and ZIPs.
