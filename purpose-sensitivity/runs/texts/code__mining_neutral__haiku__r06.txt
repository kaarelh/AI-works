```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPTED_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ZIP_PLUS4 = re.compile(r"([0-9]+)-([0-9]{4})")


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = (raw or "").strip()
    if "<" in s:
        start = s.index("<") + 1
        end = s.find(">", start)
        if end != -1:
            s = s[start:end]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _normalize_phone(raw):
    s = (raw or "").lower()
    cut = len(s)
    for marker in ("x", "ext", "#"):
        idx = s.find(marker)
        if idx != -1:
            cut = min(cut, idx)
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _all_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(members, field):
    """Pick the field value from the best row: manual first, then most recent, then earliest."""
    candidates = [m for m in members if m[field]]
    if not candidates:
        return ""
    best = max(candidates, key=lambda m: (m["manual"], m["updated"], -m["idx"]))
    return best[field]


def dedupe_contacts(rows):
    # Rules 1-9: normalize each row and drop rows with no valid email or phone.
    recs = []
    for row in rows:
        email = _normalize_email(row.get("email"))
        if not _is_valid_email(email):
            email = ""

        digits = _normalize_phone(row.get("phone"))
        phone = digits if _is_valid_phone(digits) else ""

        if not email and not phone:
            continue

        source = (row.get("source") or "").strip().lower()
        last = _normalize_name(row.get("last_name"))
        tags = {t.strip().lower() for t in re.split(r"[;,]", row.get("tags") or "")} - {""}

        recs.append({
            "idx": len(recs),  # position among kept rows; preserves input order
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": last,
            "last_key": last.casefold(),
            "zip": _normalize_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPTED_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # Rule 10: union-find, so matching is transitive.
    n = len(recs)
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], i)
            else:
                first_by_email[r["email"]] = i
        if r["phone"] and r["last_key"]:
            key = (r["phone"], r["last_key"])
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    # Group rows by root. Iterating in input order means groups come out
    # ordered by each group's earliest row (rule 12).
    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # Rule 11: build one output dict per person.
    people = []
    for member_idx in groups.values():
        members = [recs[i] for i in member_idx]

        tags = set()
        for m in members:
            tags |= m["tags"]

        people.append({
            "first_name": _pick(members, "first_name"),
            "last_name": _pick(members, "last_name"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
        })

    return people
```

Notes on decisions you may want to check:

- **Opt-outs:** The function returns opted-out people with `opted_out="Y"` and does not remove them. Filter with `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N"]` before building the send list. Because a group is flagged if any of its rows opted out, this also suppresses people who opted out under a different address or phone.
- **ZIP+4:** I treat "looks like ZIP+4" as digits, a hyphen, then exactly four digits. That lets a Excel-mangled prefix like `2139-4307` become `02139`. If the data person meant strictly five digits before the hyphen, change the regex to `([0-9]{5})-([0-9]{4})`.
- **Unparseable `updated`:** The function raises `ValueError` rather than guessing. A silent fallback could make the wrong row win the "most recent" tiebreak, and I'd rather the send job fail loudly than pick the wrong data quietly.
- **Extension markers:** "x" matches anywhere in the phone string, as the rules say. That means a stray letter x in a phone field will truncate the number, which is probably what you want but is worth a spot-check.
