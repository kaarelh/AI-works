```python
import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to CPR's data rules.

    Returns a list of dicts with keys: first_name, last_name, email, phone,
    zip, tags, opted_out, updated, source. One dict per person, ordered by
    each person's earliest row in the input.
    """

    # ---------- normalizers ----------

    def field(row, key):
        val = row.get(key)
        return "" if val is None else str(val)

    def norm_email(raw):
        s = raw
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_is_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        """Return the 10-digit phone if valid, else ''."""
        s = re.split(r"(?i)ext|x|#", raw, maxsplit=1)[0]   # drop extension
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = raw.strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)     # ZIP+4
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):                 # Excel ate the 0
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = raw.strip()
        if s and (s.isupper() or s.islower()):
            return s.title()
        return s

    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def parse_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---------- normalize rows, drop unreachable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(field(row, "email"))
        if not email_is_valid(email):
            email = ""
        phone = norm_phone(field(row, "phone"))
        if not email and not phone:
            continue
        source = field(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "email": email,
            "phone": phone,
            "first_name": norm_name(field(row, "first_name")),
            "last_name": norm_name(field(row, "last_name")),
            "zip": norm_zip(field(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "updated": parse_updated(field(row, "updated")),
            "tags": parse_tags(field(row, "tags")),
            "opted_out": field(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
        })

    # ---------- union-find matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # keep the earlier row as root (not required, but tidy)
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output (rules 11-12) ----------

    def priority(r):
        # higher is better: manual first, then most recent, then earliest row
        return (r["is_manual"], r["updated"], -r["idx"])

    def pick(group, key):
        candidates = [r for r in group if r[key]]
        if not candidates:
            return ""
        return max(candidates, key=priority)[key]

    people = []
    for group in groups.values():
        latest = max(r["updated"] for r in group)
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        people.append((min(r["idx"] for r in group), {
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
            "source": pick(group, "source"),
        }))

    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]
```

How some ambiguous parts of the spec are handled:

- **Opted-out people are still in the output.** They are marked `opted_out: "Y"`, as the spec says. Before you send, filter them out, and also drop anyone without an email:
  ```python
  send_list = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]
  ```
- **Rule 9 can lose an opt-out.** A row with an invalid email and no valid phone (e.g. a typo like `ann@xorg`) is dropped before matching. If that row was the only place the person opted out, the opt-out is lost. Tomorrow's send is safe, since it can only go to valid addresses. Still, it may be worth checking your raw export for opted-out rows that rule 9 drops. Your suppression list should probably keep them.
- **Email validity follows the rule exactly.** It doesn't reject spaces or other odd characters inside an address. Your email platform will likely bounce those anyway.
- **Extension markers:** the first occurrence of `ext`, `x`, or `#` cuts off the rest of the phone, case-insensitive.
- **ZIP+4:** it's recognized as 4–5 digits, a hyphen, then 4 digits. That way `2139-4307` also becomes `02139`.
- **Bad dates:** an `updated` value in none of the three formats raises a `ValueError` rather than being silently guessed.
- **Ties:** when two rows qualify for a field, the order is manual source first, then most recent `updated`, then the earlier row in the input.
