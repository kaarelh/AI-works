import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ASCII_DIGITS = "0123456789"


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows per the data team's rules (1-12)."""

    # ---------- field normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_is_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        s = raw or ""
        s = re.split(r"ext|x|#", s, maxsplit=1, flags=re.IGNORECASE)[0]
        digits = "".join(c for c in s if c in _ASCII_DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_is_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)   # ZIP+4 (also tolerates 4-digit base)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s and (s.isupper() or s.islower()):
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return (raw or "").strip().lower() in _OPT_OUT_VALUES

    def split_tags(raw):
        return [t.strip().lower() for t in re.split(r"[;,]", raw or "")]

    # ---------- normalize & filter (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        email_ok = email_is_valid(email)
        phone_ok = phone_is_valid(phone)
        if not email_ok and not phone_ok:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "zip": norm_zip(row.get("zip")),
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
        })

    # ---------- union-find matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- merge each group (rule 11) ----------

    def pick(members, field):
        candidates = [m for m in members if m[field]]
        if not candidates:
            return ""
        best = max(candidates,
                   key=lambda m: (m["manual"], m["updated"], -m["idx"]))
        return best[field]

    people = []
    for members in groups.values():
        tags = sorted({t for m in members for t in m["tags"] if t})
        people.append((min(m["idx"] for m in members), {
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        }))

    # ---------- order by earliest row (rule 12) ----------

    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

A few things to check before tomorrow's send:

1. **The function flags opt-outs but does not remove them.** Rule 11 says to output `opted_out: "Y"`, so opted-out people are still in the result. When you build the send list, filter it explicitly:
   ```python
   send_list = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]
   ```
   The `p["email"]` check also matters, because people kept only for a valid phone will have `email == ""`.

2. **Opt-outs only carry over through matches.** If someone opted out under one email and appears elsewhere under a different email with no matching phone and last name, the two records won't merge, and the second one will show `"N"`. Also, an opted-out row with neither a valid email nor a valid phone is dropped by rule 9, so it can't suppress anyone. If you keep a separate unsubscribe or suppression list, apply it as well.

3. **Matching on phone plus last name can merge household members.** Two people who share a landline and a last name become one person, and only one email is output for them. If either one opted out, both are suppressed. This follows the spec, but some family members may not get the alert.

4. **Choices I made where the spec was ambiguous:**
   - ZIP+4 with a 4-digit base (`2139-4307`) is also accepted and padded to `02139`.
   - Only ASCII digits count for phones and ZIPs.
   - An `updated` value that doesn't parse raises a `ValueError` instead of being silently skipped, so bad data surfaces before the send.
   - The gmail rules only apply when the address has exactly one `@`.

I'd recommend running it on a small sample with known duplicates and opt-outs and spot-checking the output before the full send.
