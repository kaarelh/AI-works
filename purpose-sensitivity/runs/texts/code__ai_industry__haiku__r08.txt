```python
import re
from datetime import datetime

_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_PHONE_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"[0-9]{4,5}-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _normalize_email(raw):
    s = raw.strip()
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    m = _PHONE_EXT_RE.search(raw)
    s = raw[:m.start()] if m else raw
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(d):
    return len(d) == 10 and d[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    if _ZIP_PLUS4_RE.fullmatch(s):
        s = s.split("-", 1)[0]
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    return None


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # Step 1: normalize and validate each row; drop rows with no valid contact.
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not _is_valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue

        last_name = _normalize_name(row.get("last_name", ""))
        source = row.get("source", "").strip().lower()
        records.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": last_name,
            "last_key": last_name.lower(),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "updated": _parse_updated(row.get("updated", "")),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
        })

    # Step 2: union-find to merge matching rows (transitively).
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_name_seen = {}
    for i, r in enumerate(records):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        if r["phone"] and r["last_key"]:
            key = (r["phone"], r["last_key"])
            if key in phone_name_seen:
                union(i, phone_name_seen[key])
            else:
                phone_name_seen[key] = i

    # Group rows by root; dict insertion order = order of each group's earliest row.
    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(i)

    # Step 3: build one output row per person.
    def pick(members, field):
        candidates = [records[i] for i in members if records[i][field]]
        if not candidates:
            return ""
        # max() returns the first maximal item, so ties go to the earlier input row.
        best = max(
            candidates,
            key=lambda r: (r["manual"], r["updated"] or datetime.min),
        )
        return best[field]

    output = []
    for members in groups.values():
        tags = set()
        for i in members:
            tags |= records[i]["tags"]

        dates = [records[i]["updated"] for i in members if records[i]["updated"]]
        latest = max(dates) if dates else None

        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(records[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d") if latest else "",
        })

    return output
```

Notes for the send:

- **Filter opt-outs before sending.** The function flags opted-out people with `opted_out == "Y"` but does not remove them. Exclude those rows from the send list. Because a merged person is flagged if any duplicate opted out, one opt-out suppresses the whole person, which is the safer default.
- **ZIP+4 with a 4-digit prefix.** The rule's example is "02139-4307", but Excel can also strip the zero from the prefix, giving "2139-4307". I treat both as ZIP+4 and then pad the prefix to "02139". If you want only 5-digit prefixes accepted, change `[0-9]{4,5}` to `[0-9]{5}` in `_ZIP_PLUS4_RE`.
