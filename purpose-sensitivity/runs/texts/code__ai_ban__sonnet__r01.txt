```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    epoch = datetime(1970, 1, 1)

    def s(v):
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        raw = s(raw)
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        raw = s(raw)
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        raw = s(raw).strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(raw, fmt)
            except ValueError:
                pass
        return None

    def ts(dt):
        return float("-inf") if dt is None else (dt - epoch).total_seconds()

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize rows, drop those with no valid email/phone ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        upd = parse_updated(row.get("updated"))
        source = s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower() in ("y", "yes", "true", "1"),
            "updated": upd,
            "ts": ts(upd),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # ---------- matching ----------
    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # ---------- build output ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = min(cands, key=lambda r: (0 if r["manual"] else 1, -r["ts"], r["idx"]))
        return best[field]

    result = []
    for root in order:
        members = groups[root]
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        dates = [recs[i]["updated"] for i in members if recs[i]["updated"] is not None]
        latest = max(dates).strftime("%Y-%m-%d") if dates else ""
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest,
        })
    return result
```

Notes on a few judgment calls:

- **Opt-outs:** `opted_out` is "Y" if any row in a merged group has opted out. Before the send, filter out every person with `opted_out == "Y"`, since the function outputs them rather than removing them.
- **Order of output:** people are ordered by their earliest surviving row. Rows dropped under rule 9 don't count.
- **Email extraction:** the display-name step takes the first `<...>` pair it finds. An address with no angle brackets is used whole.
- **ZIP+4:** I accept any digit string before a hyphen followed by exactly 4 digits, so an Excel-damaged `2139-4307` also becomes `02139`.
- **Unparseable `updated`:** the spec says this shouldn't happen. If it does, the row sorts as oldest and doesn't affect the group's latest date. If no row in a group has a parseable date, `updated` is `""`.
- **Invalid values:** an invalid email or phone is never used as a match key or output value, even if the row is kept because its other contact field is valid.
