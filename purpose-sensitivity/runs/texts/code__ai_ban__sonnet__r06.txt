from datetime import datetime


def dedupe_contacts(rows):
    GMAIL = {"gmail.com", "googlemail.com"}
    OPT_YES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    def s(v):
        return "" if v is None else str(v)

    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL:
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"ext|x|#", p, flags=re.IGNORECASE)
        if m:
            p = p[:m.start()]
        d = re.sub(r"[^0-9]", "", p)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        m = re.fullmatch(r"([0-9]{5})-([0-9]{4})", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

        v = s(raw).strip()
        for f in DATE_FORMATS:
            try:
                return datetime.strptime(v, f)
            except ValueError:
                pass
        return datetime.min

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)

    # Normalize and filter
    recs = []
    for idx, r in enumerate(rows):

            "opted_out": s(r.get("opted_out")).strip().lower() in OPT_YES,
            "updated": parse_date(r.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}

    for i, rec in enumerate(recs):

- **Dropped rows:** per rule 9, rows with no valid email or phone are removed before matching, so they don't affect ordering, tags, or opt-out status. If a dropped row was marked opted out and shares a person with a kept row, that opt-out will not be seen. Check this before sending, since it matters for compliance.

- **ZIP+4:** only the strict `ddddd-dddd` form is stripped to five digits. Something like `2139-4307` becomes `""`.
- **Source selection:** the output `source` is chosen from rows with a non-empty source, using the same manual-first, newest-first, earliest-row ordering as the other fields.
- **Unparseable `updated`:** if a date doesn't match any of the three formats, it is treated as the oldest possible date rather than raising an error.
