import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_MIN_DT = datetime.min


def _normalize_email(raw):
    s = (raw or "").strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = (raw or "").lower()
    markers = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if markers:
        s = s[: min(markers)]
    digits = "".join(c for c in s if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = (raw or "").strip()
    if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
        s = s.split("-")[0]
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return _MIN_DT


def _normalize_tags(raw):
    parts = re.split(r"[;,]", raw or "")
    return {p.strip().lower() for p in parts if p.strip()}


def dedupe_contacts(rows):
    # 1. Normalize every row; drop rows with neither a valid email nor phone.
    recs = []
    for row in rows:
        email = _normalize_email(row.get("email"))
        phone = _normalize_phone(row.get("phone"))
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not email_ok and not phone_ok:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "first": _normalize_name(row.get("first_name")),
            "last": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": _normalize_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # 2. Union-find to group rows into people (transitive matching).
    n = len(recs)
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in first_by_email:
                union(first_by_email[rec["email"]], i)
            else:
                first_by_email[rec["email"]] = i
        if rec["phone"] and rec["last"]:
            key = (rec["phone"], rec["last"].lower())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    # Group indices; dict insertion order is each group's earliest row.
    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    def pick(group, field):
        # Candidates: rows where the field is non-empty (email/phone already valid-only).
        cands = [i for i in group if recs[i][field]]
        if not cands:
            return ""
        # Most recent first; sort is stable, so ties keep input order.
        ordered = sorted(cands, key=lambda i: recs[i]["updated"], reverse=True)
        manual_first = [i for i in ordered if recs[i]["manual"]]
        return recs[(manual_first or ordered)[0]][field]

    # 3. Build one output record per person.
    result = []
    for group in groups.values():
        tags = set()
        for i in group:
            tags |= recs[i]["tags"]
        result.append({
            "first_name": pick(group, "first"),
            "last_name": pick(group, "last"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in group) else "N",
            "updated": max(recs[i]["updated"] for i in group).strftime("%Y-%m-%d"),
        })
    return result
