import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    # ---------- per-field normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return bool(local) and "." in domain \
            and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        s = (raw or "").lower()
        cut = len(s)
        for marker in ("ext", "x", "#"):          # ignore everything from the earliest marker on
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"\D", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = (raw or "").strip()
        if "-" in z:                               # ZIP+4 -> keep part before hyphen
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", z):              # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"\d{5}", z) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min                        # unparseable: treat as oldest possible

    def parse_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    def opted(raw):
        return (raw or "").strip().lower() in OPT_OUT_VALUES

    # ---------- normalize rows, drop unmatchable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        ev, pv = email_valid(email), phone_valid(phone)
        if not (ev or pv):
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email if ev else "",
            "phone": phone if pv else "",
            "zip": norm_zip(row.get("zip", "")),
            "source": (row.get("source") or "").strip().lower(),
            "tags": parse_tags(row.get("tags", "")),
            "opted_out": opted(row.get("opted_out", "")),
            "updated": parse_updated(row.get("updated", "")),
        })

    # ---------- group into people with union-find (rule 10, transitive) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)   # root is always the earliest row

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in seen:
                union(seen[k], i)
            else:
                seen[k] = i

    groups = {}                                   # insertion order = order of earliest row (rule 12)
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---------- build one output dict per person (rule 11) ----------

    def pick(members, field):
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        # prefer source=="manual", then latest updated, then earliest input row
        best = max(cands, key=lambda m: (m["source"] == "manual", m["updated"], -m["idx"]))
        return best[field]

    out = []
    for members in groups.values():
        tags = set().union(*(m["tags"] for m in members))
        latest = max(m["updated"] for m in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": latest.date().isoformat(),
        })
    return out

A few judgment calls worth knowing about:

- **Phone extension markers:** I cut at the *earliest* occurrence of `ext`, `x`, or `#`, so `555-123-4567 ext 12` and `555-123-4567x12` both work, and `ext` isn't mangled by its own `x`.
- **ZIP+4:** anything before a hyphen is kept (then the 4-digit/5-digit checks run), so `2139-4307` also becomes `02139`.
- **Gmail with an invalid address** (e.g. no `@`): the gmail rewriting only runs when there's exactly one `@`, so such rows just fail validation as expected.
- **Unparseable `updated`:** the spec says it's always one of the three formats; just in case, I treat anything else as the oldest possible date rather than crashing the whole export.
- **Names:** `str.title()` gives `Mcdonald` for `MCDONALD` and `O'Brien` for `O'BRIEN`, per the rule as written.
