import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_TAG_SPLIT_RE = re.compile(r"[;,]")
_ZIP4_RE = re.compile(r"^([0-9]{5})-[0-9]{4}$")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _s(value):
    return "" if value is None else str(value)


def _normalize_email(raw):
    text = _s(raw)
    m = _ANGLE_RE.search(text)
    if m:
        text = m.group(1)
    text = text.strip().lower()
    if text.count("@") == 1:
        local, domain = text.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        text = local + "@" + domain
    return text


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    if not local:
        return False
    return "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    text = _s(raw)
    m = _EXT_RE.search(text)
    if m:
        text = text[:m.start()]
    digits = re.sub(r"[^0-9]", "", text)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    text = _s(raw).strip()
    m = _ZIP4_RE.match(text)
    if m:
        text = m.group(1)
    if re.fullmatch(r"[0-9]{4}", text):
        text = "0" + text
    if re.fullmatch(r"[0-9]{5}", text):
        return text
    return ""


def _normalize_name(raw):
    text = _s(raw).strip()
    if text.isupper() or text.islower():
        text = text.title()
    return text


def _parse_updated(raw):
    text = _s(raw).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(text, fmt)
        except ValueError:
            continue
    return datetime.min  # unparseable / missing: treated as oldest


def _parse_tags(raw):
    tags = set()
    for part in _TAG_SPLIT_RE.split(_s(raw)):
        part = part.strip().lower()
        if part:
            tags.add(part)
    return tags


def dedupe_contacts(rows):
    # ---- Normalize rows, dropping those with no valid email or phone ----
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        email = email if _email_is_valid(email) else ""
        phone = _normalize_phone(row.get("phone"))
        phone = phone if _phone_is_valid(phone) else ""
        if not email and not phone:
            continue
        source = _s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _s(row.get("opted_out")).strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # ---- Union-find over kept rows ----
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root (not required, but tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            key = r["email"]
            if key in email_seen:
                union(i, email_seen[key])
            else:
                email_seen[key] = i
        last = r["last_name"].casefold()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)  # members in input order

    # ---- Build output ----
    def pick(members, field):
        best = None
        for i in members:
            r = recs[i]
            if not r[field]:
                continue
            if best is None:
                best = r
            elif r["manual"] and not best["manual"]:
                best = r
            elif r["manual"] == best["manual"] and r["updated"] > best["updated"]:
                best = r
        return best[field] if best is not None else ""

    ordered_groups = sorted(groups.values(), key=lambda m: m[0])
    result = []
    for members in ordered_groups:
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.date().isoformat(),
        })
    return result

Notes on a few judgment calls:

- **Opt-outs:** because `opted_out` is "Y" if any row in a merged group opted out, you should filter on `opted_out == "N"` before sending. Rows with no valid email are also in the output, so filter on `email != ""` as well.
- **Gmail normalization** only applies when the address has exactly one `@`. An address like `+tag@gmail.com` normalizes to an empty local part and is treated as invalid.
- **Source preference:** `source` is picked only from rows with a non-empty source, as with the other fields. A "manual" row wins even if it's older.
- **Unparseable `updated` values** are treated as the oldest possible date rather than raising an error.
- **Last-name matching** uses the normalized name compared case-insensitively. A phone-only match with an empty last name does not merge rows.
