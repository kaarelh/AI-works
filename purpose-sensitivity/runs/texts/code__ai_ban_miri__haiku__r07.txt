import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_TRUE_VALUES = {"y", "yes", "true", "1"}


def _s(value):
    return "" if value is None else str(value)


def _norm_email(raw):
    m = _ANGLE_RE.search(raw)
    email = (m.group(1) if m else raw).strip().lower()
    if email.count("@") == 1:
        local, domain = email.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            email = f"{local}@gmail.com"
    return email


def _valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    m = _EXT_RE.search(raw)
    if m:
        raw = raw[:m.start()]
    digits = "".join(c for c in raw if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    z = raw.strip()
    if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
        z = z[:5]
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    if re.fullmatch(r"[0-9]{5}", z):
        return z
    return ""


def _norm_name(raw):
    name = raw.strip()
    if name.isupper() or name.islower():
        name = name.title()
    return name


def _parse_updated(raw):
    value = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(value, fmt)
        except ValueError:
            pass
    return None  # unparseable or blank


def _opted_out(raw):
    return raw.strip().lower() in _TRUE_VALUES


def _norm_tags(raw):
    tags = set()
    for part in re.split(r"[;,]", raw):
        tag = part.strip().lower()
        if tag:
            tags.add(tag)
    return tags


def _pick(recs, members, field):
    """Choose the group's value for one field.

    Candidates are members whose normalized field is non-empty. Preference:
    source 'manual', then most recent updated (unparseable counts as oldest),
    then earliest input position.
    """
    candidates = [recs[i] for i in members if recs[i][field]]
    if not candidates:
        return ""
    best = min(
        candidates,
        key=lambda r: (
            r["source"] != "manual",
            -((r["updated"] if r["updated"] is not None else datetime.min) - datetime.min),
            r["index"],
        ),
    )
    return best[field]


def dedupe_contacts(rows):
    # Normalize every row; drop rows with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(_s(row.get("email")))
        phone = _norm_phone(_s(row.get("phone")))
        rec = {
            "index": idx,
            "first_name": _norm_name(_s(row.get("first_name"))),
            "last_name": _norm_name(_s(row.get("last_name"))),
            "email": email if _valid_email(email) else "",
            "phone": phone if _valid_phone(phone) else "",
            "zip": _norm_zip(_s(row.get("zip"))),
            "source": _s(row.get("source")).strip().lower(),
            "tags": _norm_tags(_s(row.get("tags"))),
            "opted_out": _opted_out(_s(row.get("opted_out"))),
            "updated": _parse_updated(_s(row.get("updated"))),
        }
        if not rec["email"] and not rec["phone"]:
            continue
        recs.append(rec)

    # Union-find: rows sharing a valid email, or a valid phone plus the same
    # non-empty last name, are one person. Union-find gives the transitive closure.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_first = {}
    phone_first = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_first:
                union(email_first[r["email"]], i)
            else:
                email_first[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in phone_first:
                union(phone_first[key], i)
            else:
                phone_first[key] = i

    # Group members by root. Iterating in input order means each group's first
    # member is its earliest row, so insertion order gives the output order.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    result = []
    for members in groups.values():
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        stamps = [recs[i]["updated"] for i in members if recs[i]["updated"] is not None]
        result.append({
            "first_name": _pick(recs, members, "first_name"),
            "last_name": _pick(recs, members, "last_name"),
            "zip": _pick(recs, members, "zip"),
            "email": _pick(recs, members, "email"),
            "phone": _pick(recs, members, "phone"),
            "source": _pick(recs, members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(stamps).strftime("%Y-%m-%d") if stamps else "",
        })
    return result

Notes for the send:

- **The function keeps opted-out people and flags them `opted_out == "Y"`.** Your spec doesn't say to drop them, so filter on `opted_out == "N"` before the send. A person who opted out on any duplicate row comes back flagged, so the merge can't reintroduce them.
- **Recipients should also have a non-empty `email`.** Phone-only people will be in the output, so exclude them from an email send.
- **Choices where the spec was open:**
  - ZIP+4 is recognized only in the strict `NNNNN-NNNN` form. A value like `1234-5678` becomes `""`.
  - A blank or unparseable `updated` counts as the oldest. If no row in a group has a parseable date, `updated` is `""`.
  - Among candidates with equal source and date, the earliest input row wins.
