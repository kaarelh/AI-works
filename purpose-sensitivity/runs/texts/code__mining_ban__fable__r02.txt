```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows per the data team's rules.

    Takes a list of dicts (first_name, last_name, email, phone, zip, tags,
    opted_out, updated, source) and returns one merged dict per person,
    ordered by each person's earliest row in the input.
    """

    # ---------- field normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        s = (raw or "").lower()
        cut = len(s)
        for marker in ("x", "ext", "#"):           # ignore everything from the earliest marker on
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                               # ZIP+4 -> keep the part before the hyphen
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):           # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize rows, drop unusable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        e_ok = email_valid(email)
        p_ok = phone_valid(phone)
        if not e_ok and not p_ok:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email if e_ok else "",
            "phone": phone if p_ok else "",
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "updated_dt": parse_updated(row.get("updated", "")),
            "source": source,
            "manual": source == "manual",
        })

    # ---------- group rows into people with union-find (rule 10) ----------

    parent = list(range(len(recs)))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so roots are stable/earliest
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen_email = {}
    seen_phone = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone:
                union(i, seen_phone[key])
            else:
                seen_phone[key] = i

    groups = {}                                    # root -> rows, in input order
    for i, r in enumerate(recs):                   # dict insertion order = earliest row (rule 12)
        groups.setdefault(find(i), []).append(r)

    # ---------- merge each group (rule 11) ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated_dt"], -r["idx"]))
        return best[field]

    out = []
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated_dt"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return out
```

A few notes on choices where the rules left a little room:

- **Matching key for phones** is the valid 10-digit phone plus the *normalized* last name lowercased, so `SMITH`, `smith`, and `Smith` all match each other.
- **Field selection** uses the normalized row data, so a row whose email was invalid is skipped for the `email` field but can still supply name/ZIP/source if it was kept (because it had a valid phone).
- **`source`** is picked with the same manual > newest > earliest preference, so if any row in a group is `manual`, the merged record's source will be `manual`.
- **Unparseable `updated`** values raise a `ValueError` naming the bad value, rather than being silently guessed; since the spec says `updated` is never empty and always one of three formats, I'd rather you see the bad row before tomorrow's send than have it quietly sorted wrong.
- `str.title()` is used exactly as specified, so `O'BRIEN` → `O'Brien` but `MCDONALD` → `Mcdonald`. If that bothers anyone, it's a one-line change in `norm_name`.

Since your opted-out supporters are merged into their person's single record with `opted_out = "Y"`, you can filter on that column before the send and be confident nobody who opted out on any duplicate row slips through.
