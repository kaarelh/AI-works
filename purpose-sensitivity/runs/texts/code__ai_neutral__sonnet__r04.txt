```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_TRUE_VALUES = {"y", "yes", "true", "1"}


def dedupe_contacts(rows):
    def s(v):
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^<>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"x|ext|#", p, re.IGNORECASE)
        if m:
            p = p[: m.start()]
        digits = re.sub(r"[^0-9]", "", p)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = s(raw).strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-", 1)[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if not re.fullmatch(r"[0-9]{5}", z):
            return ""
        return z

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        t = s(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(t, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated value: %r" % (raw,))

    def is_opted_out(raw):
        return s(raw).strip().lower() in _TRUE_VALUES

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize rows, drop those with no valid contact ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        e_ok = email_valid(email)
        p_ok = phone_valid(phone)
        if not (e_ok or p_ok):
            continue
        source = s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email if e_ok else "",
            "phone": phone if p_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
        })

    n = len(recs)

    # ---------- union-find ----------
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def pick(members, field):
        best = None
        best_key = None
        for i in members:  # members are in input order
            r = recs[i]
            if not r[field]:
                continue
            key = (r["manual"], r["updated"])
            if best is None or key > best_key:  # strict: ties keep earlier row
                best, best_key = r, key
        return best[field] if best is not None else ""

    result = []
    for members in sorted(groups.values(), key=lambda m: m[0]):
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Sending:** the function keeps opted-out people and flags them with `opted_out == "Y"`. Filter those out before the send, e.g. `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`. People with only a phone have an empty `email` and can't receive the newsletter.
- **ZIP+4:** I read "looks like ZIP+4" strictly as 5 digits, a hyphen, then 4 digits. A value like `2139-4307` (leading zero lost) becomes `""` under this reading. If you want those rescued, relax the regex to `[0-9]{4,5}-[0-9]{4}`.
- **Opt-out scope:** if any row in a merged group has opted out, the whole person is marked `Y`, per rule 11.
- **Dropped rows:** rows dropped under rule 9 never join a group, so an opt-out on such a row has no effect. It has no valid email or phone, so it couldn't be matched to anyone anyway.
- **Dates:** an `updated` value in an unexpected format raises `ValueError` rather than being silently guessed.
