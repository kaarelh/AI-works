```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"[0-9]{4,5}-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _clean_email(raw):
    """Return the normalized email, or "" if it is not valid."""
    s = (raw or "").strip()
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = f"{local}@gmail.com"

    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _clean_phone(raw):
    """Return the normalized 10-digit phone, or "" if it is not valid."""
    s = (raw or "").lower()
    cut = len(s)
    for marker in ("x", "ext", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _clean_zip(raw):
    z = (raw or "").strip()
    if _ZIP_PLUS4_RE.fullmatch(z):
        z = z.split("-", 1)[0]
    if len(z) == 4 and _is_digits(z):
        z = "0" + z
    if len(z) == 5 and _is_digits(z):
        return z
    return ""


def _clean_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return datetime.min  # Not expected per the spec; sorts oldest.


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    return [t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()]


def _pick(group_recs, field):
    """Choose the best row for a field: manual first, then latest updated,
    then earliest input position. Returns "" if no row has a value."""
    cands = [r for r in group_recs if r[field]]
    if not cands:
        return ""
    best = max(cands, key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]))
    return best[field]


def dedupe_contacts(rows):
    # Normalize each row; drop rows with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _clean_email(row.get("email"))
        phone = _clean_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "email": email,
            "phone": phone,
            "first_name": _clean_name(row.get("first_name")),
            "last_name": _clean_name(row.get("last_name")),
            "zip": _clean_zip(row.get("zip")),
            "source": (row.get("source") or "").strip().lower(),
            "updated": _parse_updated(row.get("updated")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "tags": _split_tags(row.get("tags")),
        })

    # Union-find; rows sharing any match key are joined (transitively).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}

    def link(key, pos):
        if key in first_seen:
            union(first_seen[key], pos)
        else:
            first_seen[key] = pos

    for pos, r in enumerate(recs):
        if r["email"]:
            link(("email", r["email"]), pos)
        if r["phone"] and r["last_name"]:
            link(("phone", r["phone"], r["last_name"].lower()), pos)

    groups = {}
    for pos in range(len(recs)):
        groups.setdefault(find(pos), []).append(pos)

    # Output people in order of their earliest input row.
    ordered = sorted(groups.values(), key=lambda g: g[0])

    people = []
    for group in ordered:
        group_recs = [recs[p] for p in group]
        tags = sorted({t for r in group_recs for t in r["tags"]})
        latest = max(r["updated"] for r in group_recs)
        people.append({
            "first_name": _pick(group_recs, "first_name"),
            "last_name": _pick(group_recs, "last_name"),
            "zip": _pick(group_recs, "zip"),
            "email": _pick(group_recs, "email"),
            "phone": _pick(group_recs, "phone"),
            "source": _pick(group_recs, "source"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(r["opted_out"] for r in group_recs) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })

    return people
```

Notes:

- **Opt-outs:** Any row in a duplicate group that has opted out marks the whole person `opted_out = "Y"`. Filter on that field before the send, and don't rely on the email tool's own unsubscribe list alone.
- **Interpretations I made:** Rule 4's ZIP+4 check accepts 4 or 5 digits before the hyphen, so an Excel-mangled `2139-4307` becomes `02139`. Rule 5 treats names with no cased letters (such as "123") as unchanged.
