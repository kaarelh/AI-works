import re
from datetime import datetime


def dedupe_contacts(rows):
    DIGITS = "0123456789"
    EPOCH = datetime(1970, 1, 1)

    def get(row, key):
        v = row.get(key)
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        lt = s.find("<")
        if lt != -1:
            gt = s.find(">", lt + 1)
            if gt != -1:
                s = s[lt + 1:gt]
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
            s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        d = "".join(ch for ch in raw if ch in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = raw.strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_dt(raw):
        s = raw.strip()
        if "T" in s:
            return datetime.strptime(s, "%Y-%m-%dT%H:%M:%S")
        if "/" in s:
            return datetime.strptime(s, "%m/%d/%Y")
        return datetime.strptime(s, "%Y-%m-%d")

    # Normalize rows; drop those with neither valid email nor valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        dt = parse_dt(get(row, "updated"))
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": source,
            "dt": dt,
            "secs": (dt - EPOCH).total_seconds(),
            "tags": [t.strip().lower() for t in re.split(r"[;,]", get(row, "tags"))],
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
        })

    # Union-find over kept records.
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("e", r["email"]))
        last = r["last_name"].lower()
        if r["phone"] and last:
            keys.append(("p", r["phone"], last))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda r: (0 if r["source"] == "manual" else 1, -r["secs"], r["idx"]),
        )
        return best[field]

    ordered = sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m))

    out = []
    for members in ordered:
        tags = set()
        for i in members:
            for t in recs[i]["tags"]:
                if t:
                    tags.add(t)
        latest = max(recs[i]["dt"] for i in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return out

Notes on a few choices:

- **Opt-outs and dropped rows:** Per rule 9, rows with no valid email or phone are discarded before matching, so an opt-out on such a row can't propagate to a person. That can only matter if the same person appears under a different, valid row. Before sending, you may want to filter your send list against the raw export's opted-out emails as well.
- **Transitive matching:** This uses union-find, so chains like A–B–C collapse into one person.
- **Selection ordering:** For each field, the preference is `manual` source first, then the latest `updated` (full datetime), then earliest input position.
- **Unparseable dates:** The function raises `ValueError` rather than guessing. If your export may contain malformed dates, wrap `parse_dt` with whatever fallback you prefer.
- **Output `updated`:** It is the latest across all kept rows in the group, formatted as `YYYY-MM-DD`.
