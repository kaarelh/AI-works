```python
import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_LIKE_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"


def _normalize_email(raw):
    s = raw
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _email_is_valid(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    m = _EXT_RE.search(raw)
    s = raw[:m.start()] if m else raw
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    s = raw.strip()
    m = _ZIP4_LIKE_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and all(c in _DIGITS for c in s):
        s = "0" + s
    if len(s) == 5 and all(c in _DIGITS for c in s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in {"y", "yes", "true", "1"}


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(members, field):
    """Choose the value of `field` from the group's rows where it is non-empty.

    Preference: manual source, then most recent `updated`, then earliest
    input row. `members` is in input order and max() returns the first
    maximal element, so ties go to the earlier row.
    """
    cands = [r for r in members if r[field]]
    if not cands:
        return ""
    best = max(cands, key=lambda r: (r["manual"], r["updated"]))
    return best[field]


def dedupe_contacts(rows):
    # Normalize every row and drop rows with no valid email or phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _email_is_valid(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source_raw = row.get("source", "")
        source = source_raw.strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over positions in `recs` (which are in input order).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # Rule 10: same valid email, or same valid phone + same non-empty last name.
    first_by_email = {}
    first_by_phone_last = {}
    for pos, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in first_by_email:
                union(first_by_email[rec["email"]], pos)
            else:
                first_by_email[rec["email"]] = pos
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].casefold())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], pos)
            else:
                first_by_phone_last[key] = pos

    # Group rows by root. Iterating in input order means groups are
    # encountered in order of their earliest row (rule 12).
    groups = {}
    order = []
    for pos in range(len(recs)):
        root = find(pos)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[pos])

    results = []
    for root in order:
        members = groups[root]
        tags = set()
        for r in members:
            tags |= r["tags"]
        results.append({
            "first_name": _pick(members, "first_name"),
            "last_name": _pick(members, "last_name"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        })
    return results
```

Notes on the choices I made where the rules were open to interpretation:

- **ZIP+4 with a 4-digit prefix:** I treat "2139-4307" as ZIP+4-like, since that's the Excel-mangled form the rule is meant to repair. It becomes "02139". If the data person wants a strict 5-4 match only, change `{4,5}` to `{5}` in `_ZIP4_LIKE_RE`.
- **Extension markers:** the regex `ext|x|#` takes the earliest marker, so "ext" is handled as a whole and not as a stray "x".
- **Bad `updated` values:** the function raises `ValueError` and names the value rather than silently sorting a bad row. It's better to fail loudly than to mis-rank a contact.
- **Rows dropped by rule 9:** they don't count toward a group's `opted_out` flag. Since they have no valid contact info, they can't receive the email anyway.

Before the send, filter the output to `opted_out == "N"` and `email != ""`. Phone-only contacts are in the output too, and they won't get the email.
