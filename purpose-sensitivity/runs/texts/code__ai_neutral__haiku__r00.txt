import re
from datetime import datetime

_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _normalize_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    m = re.search(r"ext|x|#", raw, flags=re.IGNORECASE)
    s = raw[:m.start()] if m else raw
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return None


def _normalize_zip(raw):
    z = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if len(z) == 4 and all(c in _DIGITS for c in z):
        z = "0" + z
    if len(z) == 5 and all(c in _DIGITS for c in z):
        return z
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in {"y", "yes", "true", "1"}


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw)} - {""}


def dedupe_contacts(rows):
    # Normalize each row and drop rows with neither a valid email nor phone.
    kept = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        email = email if _is_valid_email(email) else None
        phone = _normalize_phone(row.get("phone", ""))
        if email is None and phone is None:
            continue
        source = row.get("source", "").strip().lower()
        kept.append({
            "idx": idx,
            "email": email,
            "phone": phone,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over kept rows so matching is transitive.
    parent = list(range(len(kept)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for j, rec in enumerate(kept):
        if rec["email"] is not None:
            if rec["email"] in by_email:
                union(j, by_email[rec["email"]])
            else:
                by_email[rec["email"]] = j
        if rec["phone"] is not None and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].casefold())
            if key in by_phone_last:
                union(j, by_phone_last[key])
            else:
                by_phone_last[key] = j

    # Group rows; dict insertion order gives groups ordered by earliest row.
    groups = {}
    for j in range(len(kept)):
        groups.setdefault(find(j), []).append(kept[j])

    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    people = []
    for members in groups.values():
        tags = set().union(*(r["tags"] for r in members))
        latest = max(r["updated"] for r in members)
        people.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return people
