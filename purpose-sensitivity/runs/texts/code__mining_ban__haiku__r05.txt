import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_BRACKET_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = (raw or "").strip()
    m = _BRACKET_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[:m.start()]
    d = "".join(c for c in s if c in _DIGITS)
    if len(d) == 11 and d[0] == "1":
        d = d[1:]
    return d


def _is_valid_phone(d):
    return len(d) == 10 and d[0] not in "01"


def _normalize_zip(raw):
    z = (raw or "").strip()
    m = _ZIP_PLUS4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if len(z) == 4 and _all_digits(z):
        z = "0" + z
    return z if len(z) == 5 and _all_digits(z) else ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    parts = re.split(r"[;,]", raw or "")
    return {p.strip().lower() for p in parts if p.strip()}


def _pick(group, field):
    """Choose the value for `field` from rows where it is non-empty.
    Preference: manual source, then most recent updated, then earliest input row."""
    cands = [r for r in group if r[field]]
    if not cands:
        return ""
    best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["index"]))
    return best[field]


def dedupe_contacts(rows):
    # Normalize rows and drop any without a valid email or phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone"))
        if not _is_valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue

        source = (row.get("source") or "").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _parse_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Union-find so matches are transitive (rule 10).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], i)
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    # Collect groups; members are already in input order.
    groups = {}
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)
    ordered_groups = sorted(groups.values(), key=lambda g: g[0]["index"])

    # Build one output dict per person.
    output = []
    for g in ordered_groups:
        tags = set()
        for r in g:
            tags |= r["tags"]
        output.append({
            "first_name": _pick(g, "first_name"),
            "last_name": _pick(g, "last_name"),
            "zip": _pick(g, "zip"),
            "email": _pick(g, "email"),
            "phone": _pick(g, "phone"),
            "source": _pick(g, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["updated"] for r in g).strftime("%Y-%m-%d"),
        })
    return output

Notes for your data person:

1. **Opt-outs:** `opted_out` is "Y" if *any* row for a person has opted out. Filter on `opted_out == "Y"` before sending the alert, since a person with one opted-out duplicate will still appear in the output.
2. **ZIP+4 ambiguity:** I treated any 4–5 digit prefix followed by a hyphen and 4 digits as ZIP+4, so "2139-4307" becomes "02139". If you want only exact 5-digit prefixes to count, change `{4,5}` to `{5}` in `_ZIP_PLUS4_RE`.
3. **Households:** Per rule 10, two people with the same phone and last name are merged. A shared household landline with a spouse or relative sharing a last name will collapse into one record, so that second person won't receive the alert.
4. **Bad dates:** An unparseable `updated` value raises `ValueError`, but only for rows that survive the step-9 drop. I'd rather fail loudly than silently misorder the "most recent" pick.
5. **Tags** are deduplicated, since "union" implies a set.
