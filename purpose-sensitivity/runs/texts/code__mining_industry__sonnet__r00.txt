```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL = {"gmail.com", "googlemail.com"}
    DIGITS = "0123456789"

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"x|ext|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        d = "".join(c for c in s if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
            s = s.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized updated value: %r" % raw)

    def is_opted_out(raw):
        return (raw or "").strip().lower() in {"y", "yes", "true", "1"}

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw or ""):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize rows and drop those with neither a valid email nor valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
        })

    # Union-find over the kept records.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        last = r["last_name"].casefold()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    def pick(group, field):
        best = None
        for r in group:  # group is in input order
            if not r[field]:
                continue
            if best is None:
                best = r
                continue
            if r["manual"] != best["manual"]:
                if r["manual"]:
                    best = r
            elif r["updated"] > best["updated"]:
                best = r
        return best[field] if best else ""

    result = []
    for group in sorted(groups.values(), key=lambda g: g[0]["idx"]):
        tags = set()
        for r in group:
            tags |= r["tags"]
        result.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return result
```

Notes on choices:

- **Opt-outs:** `opted_out` is "Y" for a person if any of their rows opted out, so filter on `opted_out == "N"` before sending. Rows dropped under rule 9 (no valid email or phone) don't contribute to this flag, per the spec's "before matching."
- **Phone output:** the normalized phone is the bare 10 digits, e.g. `6175551234`.
- **Bad dates:** an unparseable `updated` value raises `ValueError` rather than being silently guessed at. Since this is going out tomorrow, that's better than quietly mis-ordering records, but you may want to run it on the export first to see whether any rows trip it.
- **Group order:** rows within each group are processed in input order, so ties on source and date go to the earlier row.

Before you send, I'd spot-check a few known duplicates (for example, a Gmail address with dots or a `+tag`, and a pair linked only by phone and last name) and confirm that anyone you know has opted out lands in the "Y" bucket.
