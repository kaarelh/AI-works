```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"


def _normalize_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw.strip().lower()
    cut = len(s)
    for marker in ("x", "ext", "#"):
        i = s.find(marker)
        if i != -1:
            cut = min(cut, i)
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)  # ZIP+4, incl. Excel-stripped prefix
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _first(ordered, field):
    """Return the first non-empty normalized value for field in preference order."""
    for r in ordered:
        if r[field]:
            return r[field]
    return ""


def dedupe_contacts(rows):
    # Normalize each row; keep only rows with a valid email or phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        email = email if _email_is_valid(email) else ""
        phone = _normalize_phone(row.get("phone", ""))
        phone = phone if _phone_is_valid(phone) else ""
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip(),
            "tags": row.get("tags", ""),
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Union-find over rows: matches are transitive (rule 10).
    parent = {r["idx"]: r["idx"] for r in recs}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email = {}
    by_phone_last = {}
    for r in recs:
        i = r["idx"]
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    # Group rows. Dict insertion order follows each group's earliest row.
    groups = {}
    for r in recs:
        groups.setdefault(find(r["idx"]), []).append(r)

    out = []
    for group in groups.values():
        # Preference: manual source first, then most recent updated, then earliest row.
        # Stable sorts applied from least to most significant key.
        ordered = sorted(group, key=lambda r: r["idx"])
        ordered.sort(key=lambda r: r["updated"], reverse=True)
        ordered.sort(key=lambda r: r["source"].lower() != "manual")

        tags = set()
        for r in group:
            for t in re.split(r"[;,]", r["tags"]):
                t = t.strip().lower()
                if t:
                    tags.add(t)

        out.append({
            "first_name": _first(ordered, "first_name"),
            "last_name": _first(ordered, "last_name"),
            "zip": _first(ordered, "zip"),
            "email": _first(ordered, "email"),
            "phone": _first(ordered, "phone"),
            "source": _first(ordered, "source").lower(),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })

    return out
```

Notes for the send:

- **Filter before sending.** Use `opted_out == "N"` and a non-empty `email`. Opt-out is applied at the group level, so if any duplicate of a person opted out, the merged record is marked `"Y"` even if their other rows did not say so.
- **Phone-only contacts** come out with `email == ""` and can't receive this email.
- **Judgment calls:** the ZIP+4 rule accepts a 4- or 5-digit prefix before the hyphen so Excel-stripped ZIPs still work. An unparseable `updated` raises `ValueError` instead of being silently dropped, so a bad row stops the run rather than changing the send list.
