import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"


def _str(row, key):
    value = row.get(key)
    return "" if value is None else str(value)


def _parse_updated(value):
    text = value.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(text, fmt)
        except ValueError:
            pass
    return datetime.min  # unparseable: sorts as oldest


def _normalize_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = s.find(marker)
        if i != -1:
            cut = min(cut, i)
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(phone):
    return len(phone) == 10 and phone[0] not in "01"


def _normalize_zip(raw):
    z = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)  # ZIP+4, incl. Excel-mangled
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):  # Excel ate the leading zero
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _is_opted_out(raw):
    return raw.strip().lower() in ("y", "yes", "true", "1")


def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(group, kept, field):
    """Return the best row in the group where `field` is non-empty, or None."""
    cands = [j for j in group if kept[j][field]]
    if not cands:
        return None
    cands.sort()  # earliest input row first (final tie-breaker)
    cands.sort(key=lambda j: kept[j]["updated"], reverse=True)  # most recent; stable
    cands.sort(key=lambda j: not kept[j]["manual"])  # manual rows first; stable
    return kept[cands[0]]


def _value(group, kept, field):
    best = _pick(group, kept, field)
    return best[field] if best else ""


def dedupe_contacts(rows):
    # Normalize each row; drop rows with neither a valid email nor a valid phone.
    kept = []
    for row in rows:
        email = _normalize_email(_str(row, "email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(_str(row, "phone"))
        if not _is_valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue

        first = _normalize_name(_str(row, "first_name"))
        last = _normalize_name(_str(row, "last_name"))
        source = _str(row, "source").strip().lower()
        kept.append({
            "first_name": first,
            "last_name": last,
            "zip": _normalize_zip(_str(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _split_tags(_str(row, "tags")),
            "opted_out": _is_opted_out(_str(row, "opted_out")),
            "updated": _parse_updated(_str(row, "updated")),
            "last_key": last.casefold(),
        })

    # Union-find over rows, so matching is transitive.
    parent = list(range(len(kept)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    first_by_email = {}
    first_by_phone_last = {}
    for j, r in enumerate(kept):
        if r["email"]:
            if r["email"] in first_by_email:
                union(j, first_by_email[r["email"]])
            else:
                first_by_email[r["email"]] = j
        if r["phone"] and r["last_key"]:
            key = (r["phone"], r["last_key"])
            if key in first_by_phone_last:
                union(j, first_by_phone_last[key])
            else:
                first_by_phone_last[key] = j

    groups = {}
    for j in range(len(kept)):
        groups.setdefault(find(j), []).append(j)
    # Each group's indices are ascending, so g[0] is its earliest row.
    ordered_groups = sorted(groups.values(), key=lambda g: g[0])

    # Build one output record per person.
    people = []
    for group in ordered_groups:
        tags = set()
        for j in group:
            tags |= kept[j]["tags"]
        latest = max(kept[j]["updated"] for j in group)
        people.append({
            "first_name": _value(group, kept, "first_name"),
            "last_name": _value(group, kept, "last_name"),
            "zip": _value(group, kept, "zip"),
            "email": _value(group, kept, "email"),
            "phone": _value(group, kept, "phone"),
            "source": _value(group, kept, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(kept[j]["opted_out"] for j in group) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return people
