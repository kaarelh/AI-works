```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_EPOCH = datetime(1, 1, 1)
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = s.find(marker)
        if i != -1:
            cut = min(cut, i)
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(d):
    return len(d) == 10 and d[0] not in "01"


def _normalize_zip(raw):
    z = raw.strip()
    if re.fullmatch(r"[0-9]{4,5}-[0-9]{4}", z):
        z = z.split("-")[0]
    if len(z) == 4 and _all_digits(z):
        z = "0" + z
    if len(z) == 5 and _all_digits(z):
        return z
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _is_opted_out(raw):
    return raw.strip().lower() in ("y", "yes", "true", "1")


def _tag_set(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _ts(d):
    # Seconds since the epoch-like origin; unparseable dates rank oldest.
    return (d - _EPOCH).total_seconds() if d is not None else float("-inf")


def dedupe_contacts(rows):
    # Normalize every row and drop those without a valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email") or "")
        phone = _normalize_phone(row.get("phone") or "")
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name") or ""),
            "last_name": _normalize_name(row.get("last_name") or ""),
            "zip": _normalize_zip(row.get("zip") or ""),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": (row.get("source") or "").strip().lower(),
            "tags": _tag_set(row.get("tags") or ""),
            "opted_out": _is_opted_out(row.get("opted_out") or ""),
            "updated": _parse_updated(row.get("updated") or ""),
        })

    # Union-find so that matches are transitive.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in seen:
                a, b = find(i), find(seen[k])
                if a != b:
                    parent[max(a, b)] = min(a, b)
            else:
                seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    def choose(members, field):
        # Prefer manual rows, then most recent updated, then earlier input row.
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda r: (r["source"] != "manual", -_ts(r["updated"]), r["idx"]),
        )
        return best[field]

    out = []
    for members in sorted(groups.values(), key=min):
        members = sorted(members)
        updates = [recs[i]["updated"] for i in members if recs[i]["updated"] is not None]
        all_tags = set().union(*(recs[i]["tags"] for i in members))
        out.append({
            "first_name": choose(members, "first_name"),
            "last_name": choose(members, "last_name"),
            "zip": choose(members, "zip"),
            "email": choose(members, "email"),
            "phone": choose(members, "phone"),
            "source": choose(members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(updates).strftime("%Y-%m-%d") if updates else "",
        })
    return out
```

Notes on the implementation:

- **Opted-out contacts are still returned**, with `opted_out="Y"`, as the spec requires. The send step for tomorrow's email must filter on `opted_out == "N"`, or those people will receive it.
- **ZIP+4 handling:** I treat anything shaped like `NNNN-NNNN` or `NNNNN-NNNN` as ZIP+4, so an Excel-mangled `2139-4307` becomes `02139`. If you want only exact 5-digit prefixes to count, change the regex to `[0-9]{5}-[0-9]{4}`.
- **Unparseable `updated` values** are ranked as the oldest. If no row in a group has a parseable date, `updated` is `""`. The spec doesn't cover this case.
