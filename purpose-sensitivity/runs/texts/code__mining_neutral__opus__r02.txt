import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _field(row, key):
    """Fetch a field as a string, treating missing/None as ''."""
    value = row.get(key)
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = re.split(r"ext|x|#", raw, maxsplit=1, flags=re.IGNORECASE)[0]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(phone):
    return len(phone) == 10 and phone[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw, index):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Row {index}: unrecognized 'updated' value {raw!r}")


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize rows; drop rows with neither a valid email nor a valid phone.
    records = []
    for index, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        phone = _normalize_phone(_field(row, "phone"))
        email = email if _email_is_valid(email) else ""
        phone = phone if _phone_is_valid(phone) else ""
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        records.append({
            "index": index,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(_field(row, "tags")),
            "opted_out": _field(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(_field(row, "updated"), index),
        })

    # --- Union-find for transitive matching.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the smaller (earlier) index as root.
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_by_email = {}
    first_by_phone_last = {}
    for i, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in first_by_email:
                union(i, first_by_email[rec["email"]])
            else:
                first_by_email[rec["email"]] = i
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].casefold())
            if key in first_by_phone_last:
                union(i, first_by_phone_last[key])
            else:
                first_by_phone_last[key] = i

    # --- Group (dict insertion order = order of each person's earliest row).
    groups = {}
    for i, rec in enumerate(records):
        groups.setdefault(find(i), []).append(rec)

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates, key=lambda r: (r["manual"], r["updated"], -r["index"]))
        return best[field]

    # --- Build output.
    output = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return output

**Before tomorrow's send:**

- **Filter the output.** The function keeps opted-out people (marked `opted_out: "Y"`) and phone-only people (`email: ""`), as the spec says. Your mailing list should be only the rows where `opted_out == "N"` and `email != ""`.
- **Opt-outs only carry over within a matched group.** If someone opted out under one address and is subscribed under a completely different one, with no shared phone and last name, they are two separate people and the second address will still get the email.
- **Bad dates stop the run.** If an `updated` value isn't in one of the three formats, the function raises `ValueError` and names the row. I chose that over guessing so that bad data can't quietly change which record is picked.

**Places where I had to interpret the spec:**

- **ZIP+4:** I treat `NNNNN-NNNN` and `NNNN-NNNN` (Excel dropped the leading zero) as ZIP+4. Anything else containing a hyphen becomes `""`.
- **Extension markers:** Text is cut at the first `x`, `ext`, or `#`, regardless of case.
- **Unclosed angle brackets:** An email like `Ann <ann@x.org` (missing `>`) passes the spec's validity rule as `ann <ann@x.org`. That follows the spec literally. If you want to reject addresses containing spaces or `<`, add that check to `_email_is_valid`.
- **`str.title()` quirks:** It produces results like `Mcdonald` from `MCDONALD`, which is what the spec asks for.
- **Digits only:** Phone and ZIP checks accept only ASCII digits 0–9, so Unicode digit characters are rejected.
