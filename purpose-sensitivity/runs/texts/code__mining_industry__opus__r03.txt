import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_FIVE_DIGITS_RE = re.compile(r"^[0-9]{5}$")
_FOUR_DIGITS_RE = re.compile(r"^[0-9]{4}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _norm_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
            s = local + "@" + domain
    return s


def _email_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Return the valid 10-digit phone, or '' if invalid."""
    s = raw or ""
    lower = s.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = lower.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP4_RE.match(s)
    if m:
        s = m.group(1)
    if _FOUR_DIGITS_RE.match(s):
        s = "0" + s
    return s if _FIVE_DIGITS_RE.match(s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "") if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with neither valid email nor valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _email_valid(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip", "")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # --- Union-find over records.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the lower index as root (purely cosmetic; order is set later).
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}  # match key -> record position
    for pos, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(pos, first_seen[k])
            else:
                first_seen[k] = pos

    # --- Group records (records are already in input order).
    groups = {}
    for pos, r in enumerate(recs):
        groups.setdefault(find(pos), []).append(r)

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda r: (not r["is_manual"], -r["updated"].timestamp(), r["idx"]),
        )
        return best[field]

    # --- Build output, ordered by each person's earliest input row.
    ordered = sorted(groups.values(), key=lambda g: min(r["idx"] for r in g))
    out = []
    for group in ordered:
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return out

A few notes on how I read the rules where they left room for interpretation:

- **Opt-outs aren't removed.** The function flags them (`opted_out: "Y"`) as specified, but it doesn't filter them out. Before sending, make sure your send step does something like `[c for c in result if c["opted_out"] == "N"]`. Because opt-out is "any row in the group," someone who opted out under one address and not under another is correctly suppressed everywhere.
- **Each email goes to one address per person.** Every person gets one `email` value. If someone's group merged two different addresses, only the preferred one gets the email.
- **Extension markers:** the function cuts at the earliest occurrence of "ext", "x", or "#". So `713-555-0100 x12` and `713-555-0100 EXT. 12` both become `7135550100`.
- **ZIP+4 with a dropped zero** (e.g. `2139-4307`) is also handled. The pattern accepts 4 or 5 digits before the hyphen, then pads to `02139`.
- **Digits are ASCII only.** Unicode digit characters (e.g. `²`) don't count toward ZIPs or phones.
- **Name casing** uses `isupper()`/`islower()`. These ignore non-letters, so `O'BRIEN` becomes `O'Brien` and `McDonald` is left alone. Note that `str.title()` has its own quirks, e.g. `o'neil` becomes `O'Neil` but `mcdonald` becomes `Mcdonald`.
- **Phone matching** only happens when both rows have a valid phone and the same non-empty last name, compared with `casefold()`. A phone-only row with no last name stays on its own.
- **A malformed `updated` value** raises a `ValueError` that names the bad value. Per the spec that field is always present, so failing loudly seemed better than guessing.
