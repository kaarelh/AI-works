```python
import re
from datetime import datetime

_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_ZIP4_DIGITS_RE = re.compile(r"[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _norm_email(raw):
    s = (raw or "").strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[: m.start()]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = (raw or "").strip()
    if _ZIP4_RE.fullmatch(s):
        s = s[:5]  # ZIP+4: keep the part before the hyphen
    if _ZIP4_DIGITS_RE.fullmatch(s):
        s = "0" + s  # Excel dropped the leading zero
    if _ZIP5_RE.fullmatch(s):
        return s
    return ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    tags = set()
    for t in re.split(r"[;,]", raw or ""):
        t = t.strip().lower()
        if t:
            tags.add(t)
    return tags


def _pick_row(members, field):
    """Return the preferred member row whose normalized `field` is non-empty.

    Preference: source == 'manual', then latest `updated`, then earliest input row.
    `members` is already in input order, and sorted() is stable.
    """
    cands = [r for r in members if r[field]]
    if not cands:
        return None
    cands.sort(key=lambda r: r["updated"], reverse=True)   # most recent first
    cands.sort(key=lambda r: r["source"] != "manual")      # manual first (stable)
    return cands[0]


def dedupe_contacts(rows):
    # Step 1: normalize every row and drop rows with no valid email or phone.
    recs = []
    for i, row in enumerate(rows):
        def g(key):
            return row.get(key) or ""

        email = _norm_email(g("email"))
        email = email if _valid_email(email) else ""
        phone = _norm_phone(g("phone"))
        if not email and not phone:
            continue

        first = _norm_name(g("first_name"))
        last = _norm_name(g("last_name"))
        recs.append({
            "idx": i,
            "first_name": first,
            "last_name": last,
            "last_key": last.lower(),
            "zip": _norm_zip(g("zip")),
            "email": email,
            "phone": phone,
            "source": g("source").strip().lower(),
            "tags": _parse_tags(g("tags")),
            "opted_out": g("opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # Step 2: group matches with union-find (handles transitive matches).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    email_first = {}
    phone_first = {}
    for j, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_first:
                union(j, email_first[r["email"]])
            else:
                email_first[r["email"]] = j
        if r["phone"] and r["last_key"]:
            key = (r["phone"], r["last_key"])
            if key in phone_first:
                union(j, phone_first[key])
            else:
                phone_first[key] = j

    # Step 3: collect groups. Insertion order = order of each group's earliest row.
    groups = {}
    for j in range(len(recs)):
        groups.setdefault(find(j), []).append(recs[j])

    # Step 4: build one output record per person.
    out = []
    for members in groups.values():
        first = _pick_row(members, "first_name")
        last = _pick_row(members, "last_name")
        zip_row = _pick_row(members, "zip")
        email = _pick_row(members, "email")
        phone = _pick_row(members, "phone")
        source = _pick_row(members, "source")

        tags = set()
        for r in members:
            tags |= r["tags"]

        out.append({
            "first_name": first["first_name"] if first else "",
            "last_name": last["last_name"] if last else "",
            "zip": zip_row["zip"] if zip_row else "",
            "email": email["email"] if email else "",
            "phone": phone["phone"] if phone else "",
            "source": source["source"] if source else "",
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        })

    return out
```

Notes for your send list:

- **The function does not remove opted-out people.** It marks them `opted_out == "Y"`, so filter them out before sending. A person is marked "Y" if any duplicate row opted out, so the filter is conservative. For the email send, also require a non-empty email:

- **Suppression lists from other systems:** if opt-outs are kept anywhere else, such as an unsubscribe tool or a suppression file, check those against `send_list` too.
- **Unparseable `updated` values raise `ValueError`** rather than being guessed, so bad data stops the run instead of silently picking the wrong record.
- **ZIP+4 handling** only treats the strict `12345-6789` form as ZIP+4. A value like `2139-4307` becomes `""`.
