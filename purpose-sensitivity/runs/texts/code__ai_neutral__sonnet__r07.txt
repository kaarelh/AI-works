```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    _EPOCH = datetime(1970, 1, 1)
    _DIGITS = "0123456789"

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        if not local:
            return False
        return "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        m = re.search(r"x|ext|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        digits = "".join(c for c in raw if c in _DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized updated value: %r" % raw)

    def is_opted_out(raw):
        return raw.strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # Normalize rows and drop those with neither valid email nor valid phone
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": get(row, "source").strip().lower(),
            "tags": split_tags(get(row, "tags")),
            "opted_out": is_opted_out(get(row, "opted_out")),
            "updated": parse_updated(get(row, "updated")),
        })

    # Union-find over record positions
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            key = r["email"]
            if key in email_seen:
                union(i, email_seen[key])
            else:
                email_seen[key] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # Order people by earliest row (recs are already in input order)
    ordered = sorted(groups.values(), key=lambda members: min(members))

    def pick(members, field):
        best = None
        best_key = None
        for i in members:
            r = recs[i]
            if not r[field]:
                continue
            key = (
                0 if r["source"] == "manual" else 1,
                -(r["updated"] - _EPOCH).total_seconds(),
                r["idx"],
            )
            if best_key is None or key < best_key:
                best_key = key
                best = r[field]
        return best if best is not None else ""

    result = []
    for members in ordered:
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

A few implementation notes on how I read the rules:

- **Extension markers:** the earliest occurrence of `x`, `ext`, or `#` is used as the cut point, so `555-123-4567 ext 9` and `555-123-4567x9` both become `5551234567`.
- **ZIP+4:** only the strict `ddddd-dddd` form is converted. Something like `2139-4307` becomes `""`.
- **Gmail with an empty local part:** an address like `+news@gmail.com` normalizes to `@gmail.com`, which fails validity, so it's treated as having no email.
- **Dropped rows:** these are removed before matching, so they can't link groups or affect output ordering.
- **Opted-out contacts:** the function doesn't remove them. It sets `opted_out` to `"Y"` if any row in the group opted out. For tomorrow's send, filter on `opted_out == "N"` after calling it.
- **Bad dates:** an unparseable `updated` value raises `ValueError` rather than silently guessing, so you'll find out about bad data before the send.
