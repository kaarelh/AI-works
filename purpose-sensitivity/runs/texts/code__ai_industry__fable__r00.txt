It follows the twelve rules as written; the one thing I added is that a malformed `updated` value raises a `ValueError` naming the bad value (the spec says it's always one of three formats, and silently guessing a date could mis-pick which row's email wins, so failing loudly seemed safer).

```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    ANGLE_RE = re.compile(r"<([^>]*)>")
    EXT_RE = re.compile(r"[x#]|ext", re.IGNORECASE)
    FIVE_DIGITS = re.compile(r"[0-9]{5}")
    FOUR_DIGITS = re.compile(r"[0-9]{4}")

    # ---------- field normalizers ----------

    def get(row, key):
        val = row.get(key, "")
        return "" if val is None else str(val)

    def norm_email(raw):
        m = ANGLE_RE.search(raw)
        e = (m.group(1) if m else raw).strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        m = EXT_RE.search(raw)
        s = raw[: m.start()] if m else raw
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if FOUR_DIGITS.fullmatch(z):
            z = "0" + z
        return z if FIVE_DIGITS.fullmatch(z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def norm_opted_out(raw):
        return raw.strip().lower() in OPT_OUT_VALUES

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---------- normalize rows, dropping unreachable ones (rule 9) ----------

    records = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        email = email if email_valid(email) else ""
        phone = phone if phone_valid(phone) else ""
        if not email and not phone:
            continue
        records.append(
            {
                "idx": idx,
                "first_name": norm_name(get(row, "first_name")),
                "last_name": norm_name(get(row, "last_name")),
                "email": email,
                "phone": phone,
                "zip": norm_zip(get(row, "zip")),
                "tags": norm_tags(get(row, "tags")),
                "opted_out": norm_opted_out(get(row, "opted_out")),
                "updated_dt": parse_updated(get(row, "updated")),
                "source": get(row, "source").strip().lower(),
            }
        )

    # ---------- group rows into people with union-find (rule 10) ----------

    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so groups track their earliest row
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    key_owner = {}
    for i, r in enumerate(records):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(i, key_owner[k])
            else:
                key_owner[k] = i

    groups = {}  # root -> list of records, in input order
    order = []   # roots in order of their earliest row
    for i, r in enumerate(records):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(r)

    # ---------- build one output dict per person (rules 11, 12) ----------

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["source"] == "manual", r["updated_dt"], -r["idx"]),
        )
        return best[field]

    output = []
    for root in order:
        group = groups[root]
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        output.append(
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "zip": pick(group, "zip"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "source": pick(group, "source"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": max(r["updated_dt"] for r in group).strftime("%Y-%m-%d"),
            }
        )
    return output
```

A few notes on judgment calls where the rules left a little room:

- **Email/phone stored as "" when invalid.** Since rule 11 only lets valid values win, an invalid email or phone never appears in the output, and rule 10 only matches on valid ones. Storing them as "" keeps that simple.
- **Gmail rule only applies when there's exactly one `@`.** If there's zero or two-plus, we can't tell what the domain is; the address is left alone and then fails validation anyway.
- **ZIP+4:** anything before the first hyphen is kept, so `"2139-4307"` → `"02139"` and `"abc-123"` → `""`.
- **Phone extension marker:** the regex finds the leftmost of `x`, `ext`, or `#`, so `"555-123-4567 ext. 12"` and `"5551234567x12"` both truncate correctly. One side effect to be aware of: a phone written as `"+1 (555) 123-4567"` is fine, but one containing the letter `x` anywhere (rare, but e.g. `"tel: 555-1234 x"`) truncates at the `x`, which is what the rule asks for.
- **Union-find root choice:** I always keep the smaller index as the root, which isn't required for correctness (ordering is computed separately in the `order` list), just a tidy invariant.
- **Tie-breaking in `pick`:** `max` with the key `(is_manual, updated_dt, -idx)` gives exactly "manual first, then latest `updated`, then earliest input row."
