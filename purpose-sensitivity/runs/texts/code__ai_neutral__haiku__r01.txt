```python
import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_BRACKETED = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP5 = re.compile(r"[0-9]{5}")
_DIGITS = "0123456789"


def _str(value):
    return (value or "").strip()


def _parse_updated(raw):
    s = _str(raw)
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _norm_email(raw):
    s = _str(raw)
    m = _BRACKETED.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    s = _str(raw).lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    s = _str(raw)
    m = _ZIP_PLUS4.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and all(c in _DIGITS for c in s):
        s = "0" + s
    if _ZIP5.fullmatch(s):
        return s
    return ""


def _norm_name(raw):
    s = _str(raw)
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _is_opted_out(raw):
    return _str(raw).lower() in {"y", "yes", "true", "1"}


def _parse_tags(raw):
    parts = re.split(r"[;,]", raw or "")
    return {p.strip().lower() for p in parts if p.strip()}


def _pick(group, field):
    """Pick the field value per rule 11: manual source first, then most recent updated,
    then earliest input row. `group` is in input order; only rows with a non-empty
    normalized value for `field` are considered."""
    cands = [r for r in group if r[field]]
    # Stable sorts: first by recency (descending), then by manual-ness.
    cands.sort(key=lambda r: r["ts"] or datetime.min, reverse=True)
    cands.sort(key=lambda r: not r["manual"])
    return cands[0][field] if cands else ""


def _latest(group):
    dated = [r["ts"] for r in group if r["ts"] is not None]
    return max(dated).strftime("%Y-%m-%d") if dated else ""


def dedupe_contacts(rows):
    # Normalize rows and drop any with neither a valid email nor a valid phone.
    recs = []
    for row in rows:
        email = _norm_email(row.get("email"))
        phone = _norm_phone(row.get("phone"))
        email = email if _valid_email(email) else ""
        phone = phone if _valid_phone(phone) else ""
        if not email and not phone:
            continue
        source = _str(row.get("source")).lower()
        recs.append({
            "first": _norm_name(row.get("first_name")),
            "last": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "ts": _parse_updated(row.get("updated")),
        })

    # Union-find over record positions (which preserve input order).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_seen = {}
    for pos, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(pos, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = pos
        last_key = r["last"].lower()
        if r["phone"] and last_key:
            key = (r["phone"], last_key)
            if key in phone_seen:
                union(pos, phone_seen[key])
            else:
                phone_seen[key] = pos

    # Group positions by root; dict insertion order = earliest input row per person.
    groups = {}
    for pos in range(len(recs)):
        groups.setdefault(find(pos), []).append(pos)

    output = []
    for members in groups.values():
        grp = [recs[p] for p in members]
        all_tags = set()
        for r in grp:
            all_tags |= r["tags"]
        output.append({
            "first_name": _pick(grp, "first"),
            "last_name": _pick(grp, "last"),
            "zip": _pick(grp, "zip"),
            "email": _pick(grp, "email"),
            "phone": _pick(grp, "phone"),
            "source": _pick(grp, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in grp) else "N",
            "updated": _latest(grp),
        })
    return output
```

Notes:

- **Opt-outs are flagged, not removed.** Per rule 11, `opted_out` is just a field on the output. Before the send, filter out every record where `opted_out == "Y"`, and every record where `email` is empty (phone-only contacts can't get an email).
- **ZIP+4 handling:** I read "looks like ZIP+4" as 4–5 digits, a hyphen, then 4 digits. That way an Excel-mangled `2139-4307` still becomes `02139`. If the data person wants a strict 5-digit prefix, change `[0-9]{4,5}` to `[0-9]{5}` in `_ZIP_PLUS4`.
- **Unparseable `updated` values** are ignored when picking the latest date and sort last in recency ordering. The spec doesn't cover this case.
- **Matching** uses union-find with hash lookups, so it runs in roughly linear time and handles large exports fine.
