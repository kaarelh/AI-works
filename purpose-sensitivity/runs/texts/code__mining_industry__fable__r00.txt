```python
import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)   # leftmost match wins
_TAG_SPLIT_RE = re.compile(r"[;,]")


def _s(row, key):
    """Field as a string ('' for missing/None)."""
    v = row.get(key, "")
    return "" if v is None else str(v)


def _normalize_email(raw):
    """Rules 1-2. Returns the normalized email, or '' if invalid."""
    m = _ANGLE_RE.search(raw)
    if m:
        raw = m.group(1)
    email = raw.strip().lower()
    parts = email.split("@")
    if len(parts) != 2:
        return ""
    local, domain = parts
    if domain in _GMAIL_DOMAINS:
        local = local.split("+", 1)[0].replace(".", "")
        domain = "gmail.com"
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return local + "@" + domain


def _normalize_phone(raw):
    """Rule 3. Returns 10 digits, or '' if invalid."""
    raw = _EXT_RE.split(raw, maxsplit=1)[0]
    digits = re.sub(r"[^0-9]", "", raw)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) != 10 or digits[0] in "01":
        return ""
    return digits


def _normalize_zip(raw):
    """Rule 4."""
    z = raw.strip()
    if "-" in z:
        z = z.split("-", 1)[0].strip()
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    """Rule 5."""
    n = raw.strip()
    if n.isupper() or n.islower():
        n = n.title()
    return n


def _parse_updated(raw):
    """Rule 6."""
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(group, field):
    """Rule 11 selection: non-empty value, prefer manual, then latest updated,
    then earliest input row."""
    candidates = [r for r in group if r[field]]
    if not candidates:
        return ""
    best = max(candidates, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
    return best[field]


def dedupe_contacts(rows):
    # --- normalize every row; drop rows with no valid email and no valid phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_s(row, "email"))
        phone = _normalize_phone(_s(row, "phone"))
        if not email and not phone:
            continue
        source = _s(row, "source").strip().lower()
        tags = [t.strip().lower() for t in _TAG_SPLIT_RE.split(_s(row, "tags"))]
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(_s(row, "first_name")),
            "last_name": _normalize_name(_s(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_s(row, "zip")),
            "tags": [t for t in tags if t],
            "opted_out": _s(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(_s(row, "updated")),
            "source": source,
            "manual": source == "manual",
        })

    # --- union-find for transitive matching (rule 10)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    key_owner = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(key_owner[k], i)
            else:
                key_owner[k] = i

    # --- group, preserving order of each group's earliest row (rule 12)
    groups = {}
    order = []
    for i, r in enumerate(recs):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(r)

    # --- build one output dict per person (rule 11)
    out = []
    for root in order:
        g = groups[root]
        out.append({
            "first_name": _pick(g, "first_name"),
            "last_name": _pick(g, "last_name"),
            "zip": _pick(g, "zip"),
            "email": _pick(g, "email"),
            "phone": _pick(g, "phone"),
            "source": _pick(g, "source"),
            "tags": ";".join(sorted({t for r in g for t in r["tags"]})),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["updated"] for r in g).strftime("%Y-%m-%d"),
        })
    return out
```

A few notes on judgment calls so you can check them against what your data person intended:

- **Extension markers:** the regex splits at the leftmost `x`, `ext`, or `#`. Since `ext` starts before its own `x`, both forms are handled; a phone with no marker is untouched.
- **ZIP+4:** anything with a hyphen is split at the first hyphen, so `2139-4307` (Excel ate the zero *and* it's ZIP+4) still becomes `02139`. A 9-digit ZIP with no hyphen becomes `""`.
- **`updated`:** the spec says it's always one of the three formats, so an unrecognized value raises a `ValueError` naming the bad value rather than silently misordering rows. If you'd rather it never crash the morning send, change the `raise` to `return datetime.min`.
- **Output phone** is the bare 10 digits (e.g. `5551234567`); format it for display elsewhere if needed.
- Missing or `None` fields are treated as `""`, which is slightly more forgiving than the spec requires.

One operational suggestion: since an opted-out row anywhere in a group marks the whole person `"Y"`, make sure your send step filters on `opted_out == "N"` after this function runs rather than on the raw export.
