```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    ANGLE_RE = re.compile(r"<([^>]*)>")
    EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
    TAG_SPLIT_RE = re.compile(r"[;,]")
    DIGITS5_RE = re.compile(r"[0-9]{5}")
    DIGITS4_RE = re.compile(r"[0-9]{4}")

    # ---------- field normalizers ----------

    def norm_email(raw):
        s = raw
        m = ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if domain in GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if (not local or "." not in domain
                or domain.startswith(".") or domain.endswith(".")):
            return ""
        return local + "@" + domain

    def norm_phone(raw):
        s = raw
        m = EXT_RE.search(s)
        if m:
            s = s[:m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) != 10 or digits[0] in "01":
            return ""
        return digits

    def norm_zip(raw):
        s = raw.strip()
        if "-" in s:
            s = s.split("-", 1)[0].strip()
        if DIGITS4_RE.fullmatch(s):
            s = "0" + s
        return s if DIGITS5_RE.fullmatch(s) else ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable -> treated as oldest possible

    def norm_tags(raw):
        return {t.strip().lower() for t in TAG_SPLIT_RE.split(raw) if t.strip()}

    # ---------- normalize rows, drop unusable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        def g(key):
            return str(row.get(key, "") or "")

        email = norm_email(g("email"))
        phone = norm_phone(g("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(g("first_name")),
            "last_name": norm_name(g("last_name")),
            "email": email,          # "" unless valid
            "phone": phone,          # "" unless valid
            "zip": norm_zip(g("zip")),
            "tags": norm_tags(g("tags")),
            "opted_out": g("opted_out").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(g("updated")),
            "source": g("source").strip().lower(),
        })

    # ---------- transitive matching via union-find (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in seen:
                union(seen[k], i)
            else:
                seen[k] = i

    # Group in input order; the first time a root appears is at the group's
    # earliest row, so dict insertion order gives us rule 12 for free.
    groups = {}
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---------- build one output dict per person (rule 11) ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        # Prefer: source == "manual", then latest updated, then earliest in input.
        best = max(cands, key=lambda r: (r["source"] == "manual",
                                         r["updated"],
                                         -r["idx"]))
        return best[field]

    out = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        latest = max(r["updated"] for r in group)
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return out
```

Notes on two small judgment calls:

- **ZIP+4 with a lost leading zero** (e.g. `"2139-4307"`): I take the part before the hyphen first, then pad to 5, so this comes out as `"02139"`. That seemed to be the intent of combining rules 4a and 4b.
- **Unparseable `updated`**: the spec says it's always one of the three formats, so this shouldn't happen, but rather than crash the whole send I treat a bad value as the oldest possible date (so that row never "wins" a tiebreak and never sets the group's `updated`). If you'd rather it raise so bad rows get caught before the send, change `return datetime.min` to `raise ValueError(f"bad updated value: {raw!r}")`.

One thing I'd suggest before tomorrow: run the output through a quick check that no email appears in more than one output row and that every row with `opted_out == "Y"` is excluded from the actual send list, since this function keeps opted-out people in the output (with the flag set) rather than removing them, which is what you want for suppression-list auditing.
