import re
from datetime import datetime

_EPOCH = datetime.min  # sentinel for unparseable `updated` values (sorts oldest)
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"


def _norm_email(raw):
    """Rule 1: extract angle-bracket address, strip, lowercase, canonicalize gmail."""
    m = re.search(r"<([^<>]*)>", raw)
    addr = (m.group(1) if m else raw).strip().lower()
    if addr.count("@") == 1:
        local, domain = addr.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            addr = local + "@gmail.com"
    return addr


def _valid_email(addr):
    """Rule 2."""
    if addr.count("@") != 1:
        return False
    local, domain = addr.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Rule 3 (normalization only; validity checked separately)."""
    m = re.search(r"ext|x|#", raw, re.IGNORECASE)  # leftmost marker wins
    body = raw[:m.start()] if m else raw
    digits = "".join(c for c in body if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    """Rule 3."""
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    """Rule 4."""
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)  # ZIP+4, incl. Excel-mangled prefix
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    """Rule 5."""
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6. Unparseable values are treated as oldest rather than crashing."""
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return _EPOCH


def _is_opted_out(raw):
    """Rule 7."""
    return raw.strip().lower() in {"y", "yes", "true", "1"}


def _parse_tags(raw):
    """Rule 8. Returns a set of stripped, lowercased, non-empty tags."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # Normalize every row, keeping the original input index.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        phone = _norm_phone(row.get("phone", ""))
        source = row.get("source", "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            # Invalid values are stored as "" so "non-empty" means "qualifies".
            "email": email if _valid_email(email) else "",
            "phone": phone if _valid_phone(phone) else "",
            "source": source,
            "manual": source == "manual",
            "updated": _parse_updated(row.get("updated", "")),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
        })

    # Rule 9: drop rows with neither a valid email nor a valid phone.
    kept = [r for r in recs if r["email"] or r["phone"]]

    # Rule 10: union-find over kept rows (transitive matching).
    parent = list(range(len(kept)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email = {}
    by_phone_last = {}
    for pos, r in enumerate(kept):
        if r["email"]:
            by_email.setdefault(r["email"], []).append(pos)
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            by_phone_last.setdefault(key, []).append(pos)

    for bucket in list(by_email.values()) + list(by_phone_last.values()):
        for pos in bucket[1:]:
            union(bucket[0], pos)

    # Group rows; `order` lists roots by earliest member (rule 12).
    groups = {}
    order = []
    for pos, r in enumerate(kept):
        root = find(pos)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(r)

    def best(members, field):
        """Rule 11: manual first, then most recent updated, then earliest input row."""
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        winner = min(
            cands,
            key=lambda m: (
                0 if m["manual"] else 1,
                -(m["updated"] - _EPOCH),
                m["idx"],
            ),
        )
        return winner[field]

    result = []
    for root in order:
        members = groups[root]
        result.append({
            "first_name": best(members, "first_name"),
            "last_name": best(members, "last_name"),
            "zip": best(members, "zip"),
            "email": best(members, "email"),
            "phone": best(members, "phone"),
            "source": best(members, "source"),
            "tags": ";".join(sorted(set().union(*(m["tags"] for m in members)))),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
        })

    return result

A few interpretation choices to check:

- **ZIP+4 detection:** I treat a value as ZIP+4 if it matches 4–5 digits, a hyphen, and 4 digits. That way "2139-4307" becomes "02139" as well as "02139-4307" becoming "02139"; the spec's example only shows the 5-digit form.
- **Unparseable `updated` values:** These sort as oldest instead of raising an error. If any `updated` value is malformed, it will show up as "0001-01-01" in the output if it's the group's latest value, so it's worth checking the export before sending.
- **Opt-outs:** Opt-out status is merged across each group, so if any duplicate row for a person has opted out, the output is marked "Y" and should be excluded from the send.
