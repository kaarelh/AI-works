```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize and merge duplicate CRM contact rows (see data rules 1-12)."""

    ANGLE_RE = re.compile(r"<([^<>]*)>")
    EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    def text(value):
        return "" if value is None else str(value)

    # ---- field normalizers -------------------------------------------------

    def norm_email(raw):
        """Rules 1-2. Returns the normalized email, or "" if it is not valid."""
        s = text(raw)
        m = ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if domain in GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return local + "@" + domain

    def norm_phone(raw):
        """Rule 3. Returns 10 digits, or "" if not valid."""
        s = text(raw)
        m = EXT_RE.search(s)
        if m:
            s = s[: m.start()]
        digits = re.sub(r"\D", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        """Rule 4."""
        s = text(raw).strip()
        if "-" in s:
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", s):
            s = "0" + s
        return s if re.fullmatch(r"\d{5}", s) else ""

    def norm_name(raw):
        """Rule 5."""
        s = text(raw).strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw):
        """Rule 6."""
        s = text(raw).strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        """Rule 7."""
        return text(raw).strip().lower() in OPT_OUT_VALUES

    def norm_tags(raw):
        """Rule 8 (+ the stripping/lowercasing from rule 11)."""
        return {t.strip().lower() for t in re.split(r"[;,]", text(raw)) if t.strip()}

    def norm_source(raw):
        return text(raw).strip().lower()

    # ---- normalize every row, dropping unreachable ones (rule 9) -----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = norm_source(row.get("source"))
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip")),
            "tags": norm_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated_dt": parse_updated(row.get("updated")),
            "source": source,
            "is_manual": source == "manual",
        })

    # ---- group matching rows with union-find (rule 10, transitive) ---------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # keep the smaller (earlier) index as root
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}  # root -> list of recs, in input order
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---- build one output dict per person (rules 11-12) --------------------

    def pick(group, field):
        """Non-empty values only; prefer manual, then newest, then earliest row."""
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates, key=lambda r: (r["is_manual"], r["updated_dt"], -r["idx"]))
        return best[field]

    output = []
    # groups were inserted in order of each group's earliest rec, so this is rule 12
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        latest = max(r["updated_dt"] for r in group)
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })

    return output
```

A few judgment calls worth knowing about, since you're sending this tomorrow:

- **`updated` that doesn't match any of the three formats raises `ValueError`** (naming the bad value) rather than silently guessing, since a wrong date could change which row's email/phone wins. If your export might contain stray formats, tell me and I'll add a fallback.
- **ZIP+4** is handled as "anything before the first hyphen," so an Excel-mangled `2139-4307` also comes out as `02139`.
- **Phone extensions** are cut at the first `x`, `ext`, or `#` anywhere in the string, exactly as specified — so a value like `"Fax: 617-555-0123"` loses everything from the `x` on and becomes invalid. That's per the rules, but it's worth a quick scan of your export for that pattern.
- Invalid emails and phones become `""` internally, which is why a row with neither gets dropped and why an invalid email can never be the one selected for output.
- Requires Python 3.7+ (relies on dict insertion order for rule 12).
