import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, deduplicate, and merge CRM contact rows (see rules 1-12)."""

    def text(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    # --- Rule 1 & 2: email ---------------------------------------------------
    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        e = (m.group(1) if m else raw).strip().lower()
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return local + "@" + domain

    # --- Rule 3: phone -------------------------------------------------------
    def norm_phone(raw):
        low = raw.lower()
        cuts = [i for i in (low.find("x"), low.find("ext"), low.find("#")) if i != -1]
        if cuts:
            low = low[:min(cuts)]
        digits = re.sub(r"\D", "", low)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    # --- Rule 4: zip ---------------------------------------------------------
    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"\d{5}", z) else ""

    # --- Rule 5: names -------------------------------------------------------
    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    # --- Rule 6: updated -----------------------------------------------------
    def parse_updated(raw):
        v = raw.strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(v, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    # --- Rule 7: opted_out ---------------------------------------------------
    def is_opted_out(raw):
        return raw.strip().lower() in ("y", "yes", "true", "1")

    # --- Rule 8: tags --------------------------------------------------------
    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # --- Normalize every row, drop unmatchable ones (Rule 9) -----------------
    recs = []
    for idx, row in enumerate(rows):
        rec = {
            "idx": idx,
            "first_name": norm_name(text(row, "first_name")),
            "last_name": norm_name(text(row, "last_name")),
            "email": norm_email(text(row, "email")),
            "phone": norm_phone(text(row, "phone")),
            "zip": norm_zip(text(row, "zip")),
            "tags": norm_tags(text(row, "tags")),
            "opted_out": is_opted_out(text(row, "opted_out")),
            "updated": parse_updated(text(row, "updated")),
            "source": text(row, "source").strip().lower(),
        }
        if rec["email"] or rec["phone"]:
            recs.append(rec)

    # --- Rule 10: union-find on email / (phone, last name) -------------------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            parent[max(ri, rj)] = min(ri, rj)

    by_email, by_phone_last = {}, {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in by_email:
                union(i, by_email[rec["email"]])
            else:
                by_email[rec["email"]] = i
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # --- Rule 11: merge each group -------------------------------------------
    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]))
        return best[field]

    out = []
    for members in groups.values():
        members.sort(key=lambda r: r["idx"])
        tags = set()
        for r in members:
            tags |= r["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })

    # --- Rule 12: order by earliest input row --------------------------------
    out.sort(key=lambda d: 0)  # groups already keyed by lowest idx; keep stable
    return sorted(out, key=lambda d: next(

A few notes on judgment calls, so you can sanity-check them against your data person's intent:

- **Email**: the output only ever contains *valid* normalized emails; an invalid email is treated as empty for both matching and output (so a row with only an invalid email and no valid phone is dropped per rule 9).
- **Phone extensions**: I cut at the earliest of `x`, `ext`, `#`. Since "ext" contains an "x", either way the digits after it are dropped. Normalized phone is the bare 10 digits.
- **ZIP**: any hyphen is treated as ZIP+4, so `2139-4307` → `02139`. Nine digits without a hyphen is left as-is and therefore becomes `""`.
- **Dates**: an unparseable `updated` raises a `ValueError` rather than silently guessing, since the spec says it's always present and in one of three formats. If you'd rather skip or default those rows, say so and I'll adjust.
- **Tie-breaking for picked fields**: manual source wins, then latest `updated` (compared as full datetimes), then earliest input row. The `source` output is `"manual"` whenever any row in the group is manual.
- **Names** use `str.isupper()`/`str.islower()`, so `O'BRIEN` → `O'Brien` and `mcdonald` → `Mcdonald` (that's what `title()` does; mixed-case names like `McDonald` are left alone).
