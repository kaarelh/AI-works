import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_ZIP4_RE = re.compile(r"[0-9]{4}")
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_OPT_OUT_VALUES = ("y", "yes", "true", "1")


def _norm_email(raw):
    """Rule 1: normalize an email address."""
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_is_valid(email):
    """Rule 2: exactly one @, non-empty local part, dotted domain."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Rule 3: return the 10-digit phone if valid, else ""."""
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = _NON_DIGIT_RE.sub("", s)  # ASCII digits only
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Rule 4: return a 5-digit ZIP or ""."""
    s = (raw or "").strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if _ZIP4_RE.fullmatch(s):
        s = "0" + s
    return s if _ZIP5_RE.fullmatch(s) else ""


def _norm_name(raw):
    """Rule 5: strip; title-case names that are all upper or all lower."""
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6: parse one of the three allowed formats."""
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    """Rule 8 (plus the normalization from rule 11)."""
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _email_is_valid(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        updated = _parse_updated(row.get("updated", ""))
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip", "")),
            "source": source,
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": updated,
            "is_manual": source == "manual",
            # Integer timestamp for sorting (seconds resolution is all the formats allow).
            "ts": updated.toordinal() * 86400
                  + updated.hour * 3600 + updated.minute * 60 + updated.second,
        })

    # ---- Union-find for transitive matching (rule 10).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the smaller position (i.e., earlier in input).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        last = r["last_name"].casefold()
        if r["phone"] and last:
            keys.append(("phone_last", r["phone"], last))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    # ---- Collect groups; recs are in input order, so each group's member list is too.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    def pick(members, field):
        candidates = [i for i in members if recs[i][field]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda i: (0 if recs[i]["is_manual"] else 1, -recs[i]["ts"], recs[i]["index"]),
        )
        return recs[best][field]

    # ---- Build output, ordered by each person's earliest input row (rule 12).
    output = []
    for members in sorted(groups.values(), key=lambda m: recs[m[0]]["index"]):
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })
    return output

**Things to know before tomorrow's send:**

1. **This function flags opt-outs but doesn't remove them.** Following rule 11, opted-out people are still in the output with `opted_out: "Y"`. Filter them out before sending:
   ```python
   send_list = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]
   ```
   The `p["email"]` check also skips phone-only contacts, who have nothing to email.

2. **Opt-outs can be missed (rule 9).** A row with an opt-out but no valid email or phone is dropped before matching. If someone opted out using a malformed address, their flag won't reach their other rows. Please check how many opted-out rows get dropped this way before sending.

3. **Households may lose an email (rule 10).** Two people with the same landline and last name, such as spouses, become one "person" with one email. The other spouse won't get the alert, and if either opted out, both are suppressed. That's the safe direction for opt-outs, but it reduces reach. If you'd rather keep them separate, matching on phone + last name + first name would fix it. Ask your data person.

4. **Implementation choices where the spec was ambiguous:**
   - Digits are matched as ASCII 0–9 only.
   - ZIP+4 also accepts a 4-digit base (e.g. `2139-4307` → `02139`).
   - An unparseable `updated` value raises a `ValueError` instead of being skipped silently.
   - The extension cut happens at the first `x`, `ext`, or `#`.
