import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]+)-[0-9]{4}")
_DIGITS = "0123456789"


def _normalize_email(raw):
    s = raw
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _normalize_phone(raw):
    s = raw.lower()
    markers = [s.index(m) for m in ("x", "ext", "#") if m in s]
    if markers:
        s = s[:min(markers)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    z = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(group, field):
    """Pick a field value: manual rows first, then most recent updated,
    then earliest input row. Only rows with a non-empty value qualify."""
    cands = [r for r in group if r[field]]
    # Stable sorts, applied from least to most significant key.
    # Input order (earliest row first) is preserved for ties.
    cands.sort(
        key=lambda r: r["updated"] if r["updated"] is not None else datetime.min,
        reverse=True,
    )
    cands.sort(key=lambda r: not r["is_manual"])
    return cands[0][field] if cands else ""


def dedupe_contacts(rows):
    # Normalize each row and drop rows with no valid email or phone (rule 9).
    recs = []
    for row in rows:
        email = _normalize_email(row.get("email") or "")
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(row.get("phone") or "")
        phone = phone if _is_valid_phone(phone) else ""
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "first_name": _normalize_name(row.get("first_name") or ""),
            "last_name": _normalize_name(row.get("last_name") or ""),
            "zip": _normalize_zip(row.get("zip") or ""),
            "email": email,
            "phone": phone,
            "source": source,
            "is_manual": source == "manual",
            "tags": _parse_tags(row.get("tags") or ""),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated") or ""),
        })

    # Union-find: rows sharing a valid email, or a valid phone plus the
    # same non-empty last name, are the same person (transitively).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_name = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_name:
                union(i, by_phone_name[key])
            else:
                by_phone_name[key] = i

    # Group rows; groups are ordered by their earliest input row.
    groups = {}
    order = []
    for i in range(len(recs)):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[i])

    out = []
    for root in order:
        group = groups[root]
        tags = set()
        for r in group:
            tags |= r["tags"]
        dates = [r["updated"] for r in group if r["updated"] is not None]
        out.append({
            "first_name": _pick(group, "first_name"),
            "last_name": _pick(group, "last_name"),
            "zip": _pick(group, "zip"),
            "email": _pick(group, "email"),
            "phone": _pick(group, "phone"),
            "source": _pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(dates).strftime("%Y-%m-%d") if dates else "",
        })
    return out
