import re
from datetime import datetime

def dedupe_contacts(rows):
    """
    Deduplicate CRM contact rows according to CADM's data rules.

    Takes a list of dicts with string fields first_name, last_name, email,
    phone, zip, tags, opted_out, updated, source. Returns a list of merged
    person dicts, ordered by each person's earliest row in the input.

    NOTE: Opted-out people are still returned, with opted_out == "Y".
    Filter them out before sending.
    """
    DIGITS = "0123456789"
    UPDATED_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}

    def field(row, key):
        val = row.get(key)
        return val if isinstance(val, str) else ("" if val is None else str(val))

    # Rules 1 and 2: emails
    def norm_email(raw):
        m = re.search(r"<([^>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    # Rule 3: phones
    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, flags=re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        d = "".join(c for c in raw if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        return d

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    # Rule 4: ZIP codes
    def norm_zip(raw):
        z = raw.strip()
        if re.fullmatch(r"[0-9]+-[0-9]{4}", z):
            z = z.split("-", 1)[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    # Rule 5: names
    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    # Rule 6: updated
    def parse_updated(raw, idx):
        s = raw.strip()
        for fmt in UPDATED_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Row {idx}: unrecognized 'updated' value {raw!r}")

    # Normalize every row, applying rule 9
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(field(row, "email"))
        e_ok = email_valid(email)
        phone = norm_phone(field(row, "phone"))
        p_ok = phone_valid(phone)
        if not e_ok and not p_ok:
            continue  # Rule 9: drop before matching
        source = field(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(field(row, "first_name")),
            "last_name": norm_name(field(row, "last_name")),
            "email": email if e_ok else "",
            "phone": phone if p_ok else "",
            "zip": norm_zip(field(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "updated": parse_updated(field(row, "updated"), idx),
            "opted_out": field(row, "opted_out").strip().lower() in OPT_OUT_VALUES,
            "tags": {t.strip().lower()
                     for t in re.split(r"[;,]", field(row, "tags"))
                     if t.strip()},
        })

    # Rule 10: transitive matching with union-find
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the earlier record as root (not required, but tidy)
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_with_key = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_with_key:
                union(i, first_with_key[k])
            else:
                first_with_key[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # Rule 11: field selection
    def pick(group, key):
        cands = [r for r in group if r[key]]
        if not cands:
            return ""
        # Stable sorts, least significant criterion first:
        cands.sort(key=lambda r: r["idx"])                        # earlier row
        cands.sort(key=lambda r: r["updated"], reverse=True)      # most recent
        cands.sort(key=lambda r: 0 if r["manual"] else 1)         # manual first
        return cands[0][key]

    # Rule 12: order by earliest row
    ordered = sorted(groups.values(), key=lambda g: min(r["idx"] for r in g))

    out = []
    for group in ordered:
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return out

Notes on the send and a few judgment calls:

- **Opt-outs:** The function marks opted-out people with `opted_out == "Y"` but still returns them, as the spec requires. Before sending, filter with `[p for p in result if p["opted_out"] == "N" and p["email"]]`. People with only a phone number have `email == ""`, so the second condition removes them too.
- **One email per person:** If someone's rows have two different valid emails and were merged by phone and last name, only the preferred email is output. They get the alert once, at that address. This is what you want for avoiding double sends.
- **Gmail addresses** come out canonicalized. For example, `Ann.Lee+cadm@googlemail.com` becomes `annlee@gmail.com`, which delivers to the same inbox.
- **Bad `updated` values:** If a row's `updated` isn't in one of the three formats, the function raises a `ValueError` naming the row index. It doesn't guess. Better to find out tonight than to mis-merge silently.
- **Extension marker:** Since `"x"` counts as a marker, anything after an `x` anywhere in the phone field is cut off. That matches the rule as written.
- **ZIP+4:** The hyphen form is recognized for any digit run before `-NNNN`. A mangled `2139-4307` becomes `02139`, and anything that doesn't end up as 5 digits becomes `""`.
- **Digit handling** is ASCII-only, so stray Unicode digits like superscripts can't sneak into phones or ZIPs.
