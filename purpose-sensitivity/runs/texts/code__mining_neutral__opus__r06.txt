import re
from datetime import datetime

_FIELDS = ("first_name", "last_name", "email", "phone", "zip",
           "tags", "opted_out", "updated", "source")

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP5_RE = re.compile(r"^[0-9]{5}$")
_ZIP4_RE = re.compile(r"^[0-9]{4}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _norm_email(raw):
    """Rules 1 and 2. Returns the normalized email if valid, else ''."""
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
            s = local + "@" + domain

    # Validity check (rule 2)
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local:
        return ""
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _norm_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, else ''."""
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Rule 4."""
    s = (raw or "").strip()
    m = _ZIP_PLUS4_RE.match(s)
    if m:
        s = m.group(1)
    if _ZIP4_RE.match(s):
        s = "0" + s
    return s if _ZIP5_RE.match(s) else ""


def _norm_name(raw):
    """Rule 5."""
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw, index):
    """Rule 6. Raises ValueError on an unrecognized format."""
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Row {index}: unrecognized 'updated' value {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _norm_tags(raw):
    """Rule 8 (plus the normalization from rule 11)."""
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "")
            if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email or phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email"))
        phone = _norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _norm_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated"), idx),
        })

    # ---- Union-find for transitive matching (rule 10)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the lower position (earlier input row).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_by_key = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_by_key:
                union(i, first_by_key[k])
            else:
                first_by_key[k] = i

    # ---- Group rows; recs is already in input order, so groups are created
    # ---- in order of each person's earliest row (rule 12).
    groups = {}
    order = []
    for i in range(len(recs)):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[i])

    # ---- Build output (rule 11)
    def preference(r):
        # manual first, then most recent updated, then earliest in input
        return (0 if r["is_manual"] else 1,
                -(r["updated"] - datetime.min),
                r["index"])

    out = []
    for root in order:
        members = sorted(groups[root], key=preference)
        person = {}
        for field in ("first_name", "last_name", "email", "phone", "zip", "source"):
            person[field] = next((r[field] for r in members if r[field]), "")

        all_tags = set()
        for r in members:
            all_tags |= r["tags"]

        result = {
            "first_name": person["first_name"],
            "last_name": person["last_name"],
            "email": person["email"],
            "phone": person["phone"],
            "zip": person["zip"],
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
            "source": person["source"],
        }
        out.append(result)

    return out

**How it handles the rules:**

- **Matching (rule 10):** Rows are grouped with union-find, so chains like A↔B by email and B↔C by phone + last name end up as one person. Phone matches only happen when the last name is non-empty, compared with `casefold()`.
- **Picking values (rule 11):** Each group is sorted once by (manual first, newest `updated`, earliest input row). Then each field takes the first non-empty normalized value. Invalid emails and phones are already normalized to `""`, so only valid ones can be picked.
- **Bad dates:** An `updated` value that doesn't match any of the three formats raises a `ValueError` with the row number. I chose this over guessing so a data problem can't silently reorder who wins a merge.
