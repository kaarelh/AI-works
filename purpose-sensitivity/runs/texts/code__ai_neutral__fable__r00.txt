```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize CRM contact rows, merge duplicates, and return one dict per person."""

    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    # ---------- normalization helpers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        s = (raw or "").lower()
        cut = len(s)
        for marker in ("ext", "x", "#"):      # ignore everything from the first marker on
            i = s.find(marker)
            if i != -1:
                cut = min(cut, i)
        digits = re.sub(r"[^0-9]", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                          # ZIP+4 -> keep the part before the hyphen
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):      # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw, idx):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Row {idx}: unrecognized 'updated' value {raw!r}")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize every row, drop rows with no valid email or phone ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        ev, pv = email_valid(email), phone_valid(phone)
        if not ev and not pv:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email if ev else "",
            "phone": phone if pv else "",
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", ""), idx),
            "source": (row.get("source", "") or "").strip().lower(),
        })

    # ---------- group rows into people (union-find, so matching is transitive) ----------

    parent = list(range(len(recs)))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}  # root -> members in input order; dict order = order of earliest row
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---------- build one output dict per person ----------

    def pick(members, field):
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda m: (
                m["source"] != "manual",          # manual first
                datetime.max - m["updated"],      # then most recent updated
                m["idx"],                         # then earliest in input
            ),
        )
        return best[field]

    out = []
    for members in groups.values():
        tags = set()
        for m in members:
            tags |= m["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })

    return out
```

A few implementation notes so you can sanity-check against your data:

- **Phones** are output as the bare 10-digit string (e.g. `2025551234`), which is the normalized value per rule 3. Extension markers are cut at the first occurrence of `ext`, `x`, or `#`.
- **Emails/phones that are invalid** are treated as empty for both matching and output, so an invalid email never links two rows and never appears in the result.
- **Phone + last name matching** only fires when the normalized last name is non-empty, as specified; a phone alone does not merge rows.
- **`updated`** values that don't match one of the three formats raise a `ValueError` naming the row index rather than silently sorting wrong. If your export occasionally has blanks or other formats, tell me what you'd like them treated as and I'll adjust.
- Output field order matches the input field order (first_name, last_name, email, phone, zip, tags, opted_out, updated, source).
