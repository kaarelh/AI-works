import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    # ---------- field normalizers ----------
    def get(row, key):
        v = row.get(key)
        return "" if v is None else str(v)

    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        parts = e.split("@")
        if len(parts) == 2 and parts[1] in GMAIL_DOMAINS:
            local = parts[0].split("+", 1)[0].replace(".", "")
            e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        # "x", "ext" and "#" all start an extension; cut at the first x/X/# seen
        p = re.split(r"[xX#]", raw, maxsplit=1)[0]
        digits = re.sub(r"\D", "", p)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:                                # ZIP+4 -> keep ZIP5
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", z):               # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"\d{5}", z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min                         # unparseable -> treated as oldest

    # ---------- normalize rows, drop unreachable ones (rule 9) ----------
    records = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        ev, pv = email_valid(email), phone_valid(phone)
        if not ev and not pv:
            continue
        tags = [t.strip().lower() for t in re.split(r"[;,]", get(row, "tags"))]
        records.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email if ev else "",
            "phone": phone if pv else "",
            "zip": norm_zip(get(row, "zip")),
            "tags": {t for t in tags if t},
            "opted_out": get(row, "opted_out").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(get(row, "updated")),
            "source": get(row, "source").strip().lower(),
        })

    # ---------- group matching rows (rule 10, transitive via union-find) ----------
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen = {}
    for i, r in enumerate(records):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in seen:
                union(seen[k], i)
            else:
                seen[k] = i

    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(i)
    # members are in input order, so g[0] is each person's earliest row (rule 12)
    ordered = sorted(groups.values(), key=lambda g: g[0])

    # ---------- build one output row per person (rule 11) ----------
    def pick(members, field):
        cands = [records[i] for i in members if records[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["source"] == "manual",
                                         r["updated"],
                                         -r["idx"]))
        return best[field]

    out = []
    for members in ordered:
        recs = [records[i] for i in members]
        tags = set().union(*(r["tags"] for r in recs))
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in recs) else "N",
            "updated": max(r["updated"] for r in recs).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return out

A few judgment calls worth knowing before you run it on tomorrow's list:

- **Extension markers:** cutting at the first `x`/`X`/`#` covers `x`, `ext`, `Ext.`, etc. It would also truncate at the `x` in a stray word like "fax" if that ever appears in the phone field.
- **Unparseable `updated` values:** the spec says they're always one of the three formats, so I don't raise; if one slips through it's treated as the oldest possible date rather than crashing the whole run.
- **Opt-outs:** since `opted_out` is `"Y"` if *any* row in the group opted out, make sure your send step filters on that column — this function dedupes but doesn't remove opted-out people from the output.
- Missing/`None` fields are treated as `""` so a slightly ragged export won't blow up.
