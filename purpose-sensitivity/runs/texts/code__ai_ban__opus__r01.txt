import re
from datetime import datetime

_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _normalize_email(raw):
    """Rule 1: extract from display name, strip, lowercase, canonicalize Gmail."""
    s = raw or ""
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_is_valid(email):
    """Rule 2."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, else ""."""
    s = raw or ""
    lower = s.lower()
    cut_points = [i for i in (lower.find("x"), lower.find("ext"), lower.find("#")) if i != -1]
    if cut_points:
        s = s[:min(cut_points)]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]+)-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6. Raises ValueError on an unrecognized format rather than guessing."""
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    """Rules 8 and 11 (tags): split on ; or , then strip, lowercase, drop empties."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row, then drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        email = email if _email_is_valid(email) else ""
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "is_manual": source == "manual",
            "updated": _parse_updated(row.get("updated", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "tags": _split_tags(row.get("tags", "")),
        })

    # --- Union-find for transitive matching (rule 10).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the smaller index so roots stay stable.
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(i, first_by_email[r["email"]])
            else:
                first_by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in first_by_phone_last:
                union(i, first_by_phone_last[key])
            else:
                first_by_phone_last[key] = i

    # --- Collect groups, preserving order of earliest row (rule 12).
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])
    ordered = sorted(groups.values(), key=lambda g: min(r["idx"] for r in g))

    # --- Build one output record per person (rule 11).
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        # Prefer manual, then most recent updated, then earliest in input.
        best = min(
            candidates,
            key=lambda r: (not r["is_manual"], -r["updated"].timestamp(), r["idx"]),
        )
        return best[field]

    out = []
    for group in ordered:
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return out

Because opt-out is "any row in the group," one opt-out row suppresses the whole merged person. That is the safe behavior.
- **Gmail addresses come out canonicalized.** For example, `Jane.Doe+cas@googlemail.com` becomes `janedoe@gmail.com`. Gmail delivers both forms to the same inbox, so sending still works. The address will just look different from what's in the CRM.
- **Rows dropped by rule 9 can't contribute an opt-out.** If someone's only opt-out record has no valid email or phone, it can't be linked to anyone, so it won't suppress anything. Spot-check your opt-out list against the output if you can.
- **Bad dates raise an error.** If an `updated` value isn't in one of the three formats, the function raises `ValueError` and names the bad value. I chose that over silently skipping or misordering rows, so you'll see the problem immediately.
- **Interpretation calls I made:**
  - For ZIP+4, I take the digits before the hyphen even if there are only 4, so `2139-4307` becomes `02139`.
  - Phone-plus-last-name matching uses the title-cased last name, compared with `casefold()`.
