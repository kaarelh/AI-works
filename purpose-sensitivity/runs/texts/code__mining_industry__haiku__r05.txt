```python
import re
from datetime import datetime

_EMAIL_IN_BRACKETS = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_TAG_SPLIT = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"


def _norm_email(raw):
    s = raw.strip()
    m = _EMAIL_IN_BRACKETS.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") != 1:
        return s  # Invalid either way; the validity check rejects it.
    local, domain = s.split("@")
    if domain in _GMAIL_DOMAINS:
        local = local.replace(".", "").split("+", 1)[0]
        domain = "gmail.com"
    return f"{local}@{domain}"


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i >= 0]
    if cuts:
        s = s[: min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None  # Unparseable: treated as the oldest possible value.


def _parse_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT.split(raw) if t.strip()}


def _pick_key(rec):
    # Manual rows first, then most recent updated, then earliest input row.
    if rec["updated"] is None:
        recency = float("inf")
    else:
        recency = -(rec["updated"] - datetime.min).total_seconds()
    return (0 if rec["manual"] else 1, recency, rec["idx"])


def _best(members, field):
    cands = [m for m in members if m[field]]
    return min(cands, key=_pick_key)[field] if cands else ""


def _merge_group(members):
    tags = set()
    for m in members:
        tags.update(m["tags"])
    dates = [m["updated"] for m in members if m["updated"] is not None]
    latest = max(dates).strftime("%Y-%m-%d") if dates else ""
    return {
        "first_name": _best(members, "first_name"),
        "last_name": _best(members, "last_name"),
        "zip": _best(members, "zip"),
        "email": _best(members, "email"),
        "phone": _best(members, "phone"),
        "source": _best(members, "source"),
        "tags": ";".join(sorted(tags)),
        "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
        "updated": latest,
    }


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email") or "")
        phone = _norm_phone(row.get("phone") or "")
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue

        last_name = _norm_name(row.get("last_name") or "")
        source = (row.get("source") or "").strip().lower()
        records.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name") or ""),
            "last_name": last_name,
            "zip": _norm_zip(row.get("zip") or ""),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "updated": _parse_updated(row.get("updated") or ""),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "tags": _parse_tags(row.get("tags") or ""),
            "match_last": last_name.lower(),
        })

    # Step 2: union-find over matching rules (transitive by construction).
    n = len(records)
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            parent[max(ri, rj)] = min(ri, rj)

    by_email = {}
    by_phone_last = {}
    for i, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in by_email:
                union(by_email[rec["email"]], i)
            else:
                by_email[rec["email"]] = i
        if rec["phone"] and rec["match_last"]:
            key = (rec["phone"], rec["match_last"])
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    # Step 3: group rows. Dict insertion order follows each group's earliest row.
    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(records[i])

    return [_merge_group(members) for members in groups.values()]
```

Notes on the implementation:

- **Opt-outs:** `opted_out` is "Y" if *any* row for a person has opted out, so filter on `opted_out == "N"` before the send. Opted-out people are still returned so you can see them, but they should not go into the send list.
- **ZIP+4 from Excel:** The ZIP+4 pattern accepts a 4- or 5-digit prefix, so a mangled value like "2139-4307" becomes "02139".
- **Dates:** An `updated` value that doesn't match one of the three formats is treated as oldest. If every row in a group has an unparseable date, the output `updated` is "".
- **Testing:** I couldn't run this in the chat environment. Before the send, run it on a sample export and spot-check a few merged groups, especially any with gmail dot/plus variants and phone-only matches.
