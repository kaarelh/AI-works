import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_ZIP_PLUS4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP5_RE = re.compile(r"^[0-9]{5}$")
_ZIP4_RE = re.compile(r"^[0-9]{4}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _field(row, key):
    value = row.get(key)
    return "" if value is None else str(value)


def _normalize_email(raw):
    """Rule 1: normalize. Returns the normalized string (may be invalid)."""
    s = raw
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    """Rule 2."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    if not local:
        return False
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return False
    return True


def _normalize_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, else ""."""
    s = raw
    m = _EXT_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = _NON_DIGIT_RE.sub("", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    s = raw.strip()
    m = _ZIP_PLUS4_RE.match(s)
    if m:
        s = m.group(1)
    if _ZIP4_RE.match(s):
        s = "0" + s
    return s if _ZIP5_RE.match(s) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):

    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:

def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        try:
            updated = _parse_updated(_field(row, "updated"))
        except ValueError as e:
            raise ValueError(f"Row {idx}: {e}") from None
        source = _field(row, "source").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),

            "manual": source == "manual",
            "updated": updated,
            "tags": _parse_tags(_field(row, "tags")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
        })

    # --- Union-find over the kept rows (rule 10, transitive).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the earlier row as root (not required, but tidy).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(i, first_by_email[r["email"]])
            else:
                first_by_email[r["email"]] = i

        if r["phone"] and last_key:
            key = (r["phone"], last_key)
            if key in first_by_phone_last:
                union(i, first_by_phone_last[key])
            else:
                first_by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # --- Build one output dict per person (rule 11).
    def pick(members, field):
        candidates = [i for i in members if recs[i][field]]
        if not candidates:
            return ""
        # Prefer manual, then latest updated, then earliest input position.
        best = max(
            candidates,
            key=lambda i: (recs[i]["manual"], recs[i]["updated"], -recs[i]["index"]),
        )
        return recs[best][field]

    people = []
    for members in groups.values():
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        people.append((
            min(recs[i]["index"] for i in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),

                "tags": ";".join(sorted(tags)),
                "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
                "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
            },
        ))

    # --- Order by earliest input row (rule 12).
    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

A few things to check before tomorrow's send:

1. **Filter after deduping.** This function returns everyone, including opted-out people. For the send list, keep only rows where `opted_out == "N"` and `email != ""`. Some people will only have a phone number.

2. **Decisions I made where the spec was unclear:**
   - **ZIP+4 with a dropped zero:** I treat `"2139-4307"` as ZIP+4, keep `2139`, and pad it to `02139`. If you want only a 5-digit prefix to count as ZIP+4, change `{4,5}` to `{5}` in `_ZIP_PLUS4_RE`.
   - **Gmail rule:** It only runs when the address has exactly one `@`. Addresses with more than one `@` are invalid anyway.
   - **Bad `updated` values:** The function stops with an error that gives the row number. I chose this over silently guessing a date. If you'd rather skip or default those rows, let me know.
   - **Extension markers:** Any `x` counts as a marker, as the spec says. So a phone typed with a stray letter, like `555x...`, gets cut off at that point.

3. **Shared phone numbers can merge different people.** Two people with the same phone and last name, such as spouses on a home landline, become one person. That has two effects:
   - Only one of their email addresses is kept, so the other person won't get the alert.
   - If either one opted out, the merged record is marked opted out, so neither gets the alert.

   This is what rule 10 specifies, and it errs on the side of respecting opt-outs. But it's worth knowing that some missing recipients come from this.

4. **Emails are output in normalized form.** For example, `Jane.Doe+cadm@gmail.com` becomes `janedoe@gmail.com`.
