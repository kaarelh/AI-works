```python
import re
from datetime import datetime

_UPDATED_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_OPTED_OUT_VALUES = {"y", "yes", "true", "1"}


def dedupe_contacts(rows):
    """Normalize, merge duplicate people, and return one dict per person.

    Input rows are not modified. Rows with neither a valid email nor a valid
    phone are dropped before matching (rule 9).
    """

    def field(row, key):
        v = row.get(key)
        return "" if v is None else str(v)

    # ---- rules 1 & 2: email -------------------------------------------------
    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        e = (m.group(1) if m else raw).strip().lower()
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""  # invalid -> treated as empty
        return local + "@" + domain

    # ---- rule 3: phone ------------------------------------------------------
    def norm_phone(raw):
        p = raw.lower()
        m = re.search(r"ext|x|#", p)  # leftmost extension marker wins
        if m:
            p = p[: m.start()]
        digits = re.sub(r"[^0-9]", "", p)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""  # invalid -> treated as empty

    # ---- rule 4: zip --------------------------------------------------------
    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    # ---- rule 5: names ------------------------------------------------------
    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    # ---- rule 6: updated ----------------------------------------------------
    def parse_updated(raw):
        u = raw.strip()
        for fmt in _UPDATED_FORMATS:
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                continue
        return datetime.min  # unparseable -> treated as oldest possible

    # ---- rules 7 & 8: opted_out, tags ---------------------------------------
    def parse_opted_out(raw):
        return raw.strip().lower() in _OPTED_OUT_VALUES

    def parse_tags(raw):
        return [t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()]

    # ---- normalize every row, drop rows with no valid contact point (rule 9) -
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(field(row, "email"))
        phone = norm_phone(field(row, "phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(field(row, "first_name")),
            "last_name": norm_name(field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(field(row, "zip")),
            "source": field(row, "source").strip().lower(),
            "tags": parse_tags(field(row, "tags")),
            "opted_out": parse_opted_out(field(row, "opted_out")),
            "updated": parse_updated(field(row, "updated")),
        })

    # ---- rule 10: transitive matching via union-find ------------------------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so roots reflect earliest rows
            if rb < ra:
                ra, rb = rb, ra
            parent[rb] = ra

    key_owner = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(key_owner[k], i)
            else:
                key_owner[k] = i

    # ---- rule 12: group, preserving order of each group's earliest row ------
    groups = {}
    order = []
    for i, r in enumerate(recs):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(r)

    # ---- rule 11: build one output dict per person --------------------------
    def pick(group, name):
        candidates = [r for r in group if r[name]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[name]

    output = []
    for root in order:
        group = groups[root]
        tags = sorted({t for r in group for t in r["tags"]})
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return output
```

A few notes on choices I made where the rules left room:

- **Invalid email/phone are stored as `""`**, so they never win the per-field pick (rule 11 says only valid values qualify) and never participate in matching. The output `phone` is the bare 10-digit string.
- **Per-field tie-breaking** uses `max` on `(is_manual, updated_datetime, -input_index)`, which gives exactly: manual first, then most recent, then earliest row.
- **Phone extension marker:** the leftmost of `x`, `ext`, `#` is cut, so `"617-555-0100 ext. 12"` and `"6175550100x12"` both work. Note a letter `x` anywhere in the field triggers this (e.g. an "x" typed as a placeholder), which matches the stated rule.
- **Names:** `str.isupper()` / `str.islower()` treat `"O'BRIEN"` and `"ann-marie"` as all-caps/all-lower, so they get title-cased to `O'Brien` / `Ann-Marie`; mixed-case entries like `McDonald` are left alone.
- **Unparseable `updated`** values fall back to the oldest possible datetime rather than crashing the run; per the spec that field should always be present, but a bad value in a CRM export shouldn't stop tomorrow's send. If you'd rather it fail loudly, change `return datetime.min` to `raise ValueError(...)`.
- Rows missing a key, or with `None` values, are treated as `""`.
