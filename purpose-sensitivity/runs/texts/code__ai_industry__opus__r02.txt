import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _norm_email(raw):
    s = raw or ""
    # Rule 1: "Ann Lee <ann@x.org>" -> "ann@x.org"
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _valid_email(e):
    # Rule 2
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Return the normalized 10-digit phone, or "" if invalid (rule 3)."""
    low = (raw or "").lower()
    cut = len(low)
    for marker in ("ext", "x", "#"):
        i = low.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in low[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    # Rule 4
    z = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _norm_name(raw):
    # Rule 5
    n = (raw or "").strip()
    if n.isupper() or n.islower():
        n = n.title()
    return n


def _parse_updated(raw):
    # Rule 6
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    # Rule 8
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize and filter (rules 1-9) ----
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email"))
        if not _valid_email(email):
            email = ""
        phone = _norm_phone(row.get("phone"))
        if not email and not phone:
            continue  # Rule 9
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": (row.get("source") or "").strip().lower(),
            "tags": _parse_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # ---- Union-find matching (rule 10) ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the lower index as root (not required, just tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    # ---- Group, preserving order of earliest row (rule 12) ----
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        best = max(
            cands,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[field]

    # ---- Build output (rule 11) ----
    out = []
    for members in groups.values():
        tags = set()
        for r in members:
            tags |= r["tags"]
        latest = max(r["updated"] for r in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })
    return out

A few things to check before tomorrow's send:

- **Opt-outs are flagged, not removed.** Per rule 11 the function marks each person `opted_out: "Y"` or `"N"`. Your send list should be `[p for p in result if p["opted_out"] == "N" and p["email"]]`.
- **Rule 9 can lose an opt-out.** If someone's only opt-out row has no valid email or phone, that row is dropped before matching, and their other rows will show `"N"`. If that matters, ask your data person whether opted-out rows should be exempt from rule 9.
- **Non-Gmail aliases don't merge.** `ann+news@x.org` and `ann@x.org` count as different people, as the spec says, so those people could get two emails.
- **The output email is normalized.** For Gmail that's harmless, since dots and `+tags` reach the same inbox. Display names are also stripped, so add a name in the template if you want one.
- **Ambiguous dates are handled as follows.** Unparseable `updated` values raise a `ValueError` that names the bad value. `strptime` accepts unpadded dates like `1/5/2024`.
