```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _is_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(idxs, records, field):
    # idxs is in input order. Stable sorts give the priority:
    # manual source first, then most recent updated, then earlier input row.
    cands = [i for i in idxs if records[i][field]]
    cands.sort(key=lambda i: records[i]["updated"], reverse=True)
    cands.sort(key=lambda i: records[i]["source"] != "manual")
    return records[cands[0]][field] if cands else ""


def dedupe_contacts(rows):
    # Normalize each row; drop rows with no valid email or phone.
    records = []
    for row in rows:
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone_digits = _normalize_phone(row.get("phone", ""))
        phone = phone_digits if _is_valid_phone(phone_digits) else ""
        if not email and not phone:
            continue
        records.append({
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over record indices (kept rows stay in input order).
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(records):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    # Group rows by component; dict insertion order = earliest row first.
    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(i)

    result = []
    for idxs in groups.values():
        tags = sorted({t for i in idxs for t in records[i]["tags"]})
        latest = max(records[i]["updated"] for i in idxs)
        result.append({
            "first_name": _pick(idxs, records, "first_name"),
            "last_name": _pick(idxs, records, "last_name"),
            "zip": _pick(idxs, records, "zip"),
            "email": _pick(idxs, records, "email"),
            "phone": _pick(idxs, records, "phone"),
            "source": _pick(idxs, records, "source"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(records[i]["opted_out"] for i in idxs) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Opt-outs:** Output rows with `opted_out == "Y"` are people who opted out in at least one row, so filter them out before sending: `recipients = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N"]`.
- **ZIP+4:** I read "looks like ZIP+4" as exactly five digits, a hyphen, and four digits. Something like `2139-4307` is left as invalid rather than fixed up. Tell me if you want that case handled.
- **Bad dates:** A value in `updated` that matches none of the three formats raises `ValueError` and names the bad value, so it gets fixed in the export rather than silently sorted to the wrong place.
