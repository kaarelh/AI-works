```python
import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DIGITS_RE = re.compile(r"[0-9]+")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _norm_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Return a valid 10-digit phone string, or "" if invalid."""
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[: m.start()]
    digits = "".join(_DIGITS_RE.findall(s))
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError("Unrecognized 'updated' value: %r" % raw)


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _norm_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with neither valid email nor phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _valid_email(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip", "")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _norm_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # ---- Union-find over records (rule 10, transitive)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the root as the smaller index for stability
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Build one output dict per person (rule 11)
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda r: (not r["is_manual"], -r["updated"].timestamp()
                           if False else _neg_dt_key(r["updated"]), r["index"]),
        )
        return best[field]

    out_groups = sorted(groups.values(), key=lambda g: min(r["index"] for r in g))

    result = []
    for group in out_groups:
        tags = set()
        for r in group:
            tags |= r["tags"]
        latest = max(r["updated"] for r in group)
        result.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return result


def _neg_dt_key(dt):
    """Sort key so that more recent datetimes sort first (no timezone/epoch issues)."""
    return (-dt.toordinal(), -(dt.hour * 3600 + dt.minute * 60 + dt.second),
            -dt.microsecond)
```

A few notes on how I interpreted the rules where they left room:

- **Display names**: if the field contains `<...>`, only the text inside the first pair of angle brackets is used.
- **Gmail handling** is applied only when the address has exactly one `@` (anything else is invalid anyway).
- **Extensions**: the line is cut at the first occurrence of `ext`, `x`, or `#` (case-insensitive), so `555-123-4567 ext. 12`, `x12`, and `#12` all work.
- **ZIP+4**: I also accept a 4-digit base like `2139-4307` and pad it to `02139`, since that is the same Excel leading-zero issue.
- **Digits**: only ASCII `0–9` count, so stray Unicode digits like `²` can't sneak through.
- **Phone output** is the bare 10-digit string (e.g. `7135550142`), since rule 11 asks for the normalized value.
- **Source** follows the same pick rule as the other fields (non-empty, manual first, then most recent, then earliest), and is output stripped and lower-cased.
- **Bad dates**: an unparseable `updated` value raises a `ValueError` instead of being silently guessed. Rule 6 says the field is always present, so a bad value probably means a bad export, which is worth knowing before a send.

Before sending, it's worth double-checking that your send list filters out every output row with `opted_out == "Y"`. Because opt-outs are merged across duplicates, one opted-out record makes the whole person opted out, which is the safe behavior you want.
