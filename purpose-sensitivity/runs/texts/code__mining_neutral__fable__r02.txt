```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, merge, and dedupe CRM contact rows per the data team's rules."""

    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- field normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        # Cut at the first extension marker: "x", "ext" (contains an x), or "#".
        s = re.split(r"[xX#]", raw or "", maxsplit=1)[0]
        digits = re.sub(r"\D", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                                 # invalid -> empty

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                              # ZIP+4 -> keep the ZIP5 part
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", s):             # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"\d{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s and (s.isupper() or s.islower()):
            return s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    def is_opted_out(raw):
        return (raw or "").strip().lower() in OPT_OUT_VALUES

    # ---------- normalize rows, drop unusable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "updated": parse_updated(row.get("updated", "")),
            "source": source,
            "is_manual": source == "manual",
        })

    # ---------- group rows into people with union-find (rule 10, transitive) ----------

    parent = list(range(len(recs)))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)   # root is always the earliest row

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)   # members stay in input order

    # ---------- build one output dict per person (rules 11-12) ----------

    def pick(members, field):
        candidates = [m for m in members if m[field]]
        if not candidates:
            return ""
        best = max(candidates, key=lambda m: (m["is_manual"], m["updated"], -m["idx"]))
        return best[field]

    out = []
    for root in sorted(groups):                 # earliest row first (rule 12)
        members = [recs[i] for i in groups[root]]
        tags = set()
        for m in members:
            tags |= m["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
        })
    return out
```

A few notes on choices I made where the rules left a little room:

- **Invalid emails/phones are treated as empty.** A row with an invalid email but valid phone is kept (rule 9), but that bad email is never used for matching or output.
- **Phone extensions:** I cut at the first `x`, `X`, or `#`. Since "ext" contains an x, that covers all three markers.
- **Phone output** is the normalized 10-digit string (e.g. `2025551234`), not reformatted — easy to change if your email tool wants dashes.
- **Field picking order** is manual source → most recent `updated` → earliest input row, applied independently per field, so a person's first name and ZIP may come from different rows if the best row for one field is empty for the other. That's what rule 11 describes.
- **Bad `updated` values raise a `ValueError`** naming the offending value rather than silently sorting wrong. If you'd rather it not stop the run, swap the `raise` for `return datetime.min`.
- Rows that drop out at rule 9 don't appear in the output at all, so if you want a count of discarded rows for a sanity check, compare `len(rows)` to the sum of group sizes before sending.

Before tomorrow's send I'd also run a quick spot check: filter the output for `opted_out == "Y"` and make sure your sending tool actually excludes those, since this function flags them but doesn't remove them.
