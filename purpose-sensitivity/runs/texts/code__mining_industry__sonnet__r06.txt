```python
import re
from datetime import datetime

_DIGITS = "0123456789"


def dedupe_contacts(rows):
    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"x|ext|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        digits = "".join(c for c in s if c in _DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
            s = s.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized updated value: %r" % raw)

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw or ""):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize rows and drop those with no valid email and no valid phone.
    recs = []
    for i, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": i,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": (row.get("source") or "").strip().lower(),
            "tags": parse_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower()
                         in ("y", "yes", "true", "1"),
            "updated": parse_updated(row.get("updated")),
        })

    # Union-find over the kept records.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for n, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(n, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = n
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(n, phone_last_seen[key])
            else:
                phone_last_seen[key] = n

    groups = {}
    for n in range(len(recs)):
        groups.setdefault(find(n), []).append(n)

    def pick(members, field):
        cands = [recs[n] for n in members if recs[n][field]]
        if not cands:
            return ""
        manual = [r for r in cands if r["source"] == "manual"]
        pool = manual if manual else cands
        best = pool[0]  # members are in input order
        for r in pool[1:]:
            if r["updated"] > best["updated"]:
                best = r
        return best[field]

    result = []
    # Order groups by earliest row in the input.
    for root in sorted(groups, key=lambda g: recs[groups[g][0]]["idx"]):
        members = groups[root]  # ascending, hence in input order
        tags = set()
        for n in members:
            tags |= recs[n]["tags"]
        latest = max(recs[n]["updated"] for n in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[n]["opted_out"] for n in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Some notes on how I read the rules:

- **Opt-outs:** If any row in a matched group has opted out, the whole person is `opted_out = "Y"`. When you send, filter on `opted_out == "N"` so nobody who opted out on any duplicate record gets the email.
- **Invalid emails:** An invalid email never participates in matching and is never output. That row can still survive and match on phone and last name if its phone is valid. If a person ends up with no valid email, `email` is `""`, so also skip empty emails when sending.
- **ZIP+4:** Only the exact 5-4 pattern counts as ZIP+4. Something like `2139-4307` becomes `""`.
- **Last-name matching:** Names are compared after normalization with `.lower()`, so `SMITH` and `smith` match.
- **Bad dates:** An unrecognized `updated` value raises `ValueError` rather than being silently guessed. If your export might contain other formats, you may want to catch that case.

I'd test it with a handful of rows covering a Gmail dot or plus variant, a transitive chain (A–B by email, B–C by phone and last name), a `manual` source row, and an opted-out duplicate before you run it on the full export.
