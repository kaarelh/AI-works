Here is the function. It uses only the standard library and implements your data person's rules in order.

```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_UPDATED_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _normalize_email(raw):
    """Rule 1 normalization; returns the normalized string (may be invalid)."""
    s = raw or ""
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
            s = local + "@" + domain
    return s


def _is_valid_email(s):
    """Rule 2."""
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Rule 3; returns a valid 10-digit phone or ''."""
    s = raw or ""
    m = re.search(r"(ext|x|#)", s, flags=re.IGNORECASE)
    if m:
        s = s[: m.start()]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]+)-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw, row_index):
    """Rule 6. Raises ValueError on an unrecognized format rather than guessing."""
    s = (raw or "").strip()
    for fmt in _UPDATED_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Row {row_index}: unrecognized 'updated' value {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    """Rule 8 (plus the tag normalization from rule 11)."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row ----
    recs = []
    for i, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(row.get("phone", ""))

        # Rule 9: drop rows with neither a valid email nor a valid phone.
        if not email and not phone:
            continue

        recs.append({
            "index": i,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row.get("zip", "")),
            "source": (row.get("source", "") or "").strip().lower(),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", ""), i),
        })

    # ---- Union-find over the kept records (rule 10) ----
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the root as the earlier record so roots stay stable.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for k, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(first_seen[key], k)
            else:
                first_seen[key] = k

    # ---- Group, preserving order of each person's earliest row (rule 12) ----
    groups = {}
    for k in range(len(recs)):
        groups.setdefault(find(k), []).append(k)
    ordered_groups = sorted(groups.values(), key=lambda members: recs[members[0]]["index"])

    # ---- Build one output dict per person (rule 11) ----
    def pick(members, field):
        best = None
        best_key = None
        for k in members:
            r = recs[k]
            if not r[field]:
                continue
            key = (r["source"] == "manual", r["updated"], -r["index"])
            if best_key is None or key > best_key:
                best_key, best = key, r[field]
        return best if best is not None else ""

    out = []
    for members in ordered_groups:
        tags = set()
        for k in members:
            tags |= recs[k]["tags"]
        latest = max(recs[k]["updated"] for k in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[k]["opted_out"] for k in members) else "N",
            "updated": latest.date().isoformat(),
            "source": pick(members, "source"),
        })
    return out
```

How I handled some ambiguous cases:

- **Invalid phones and emails** are treated as empty everywhere. They aren't used for matching, and they're never chosen as output values.
- **The phone extension marker** is matched as `ext`, `x`, or `#`, and everything from the first occurrence onward is cut. Because only digits are kept afterward, cutting at "ext" or at its "x" gives the same result.
- **ZIP+4** means digits, a hyphen, then four digits. A ZIP+4 that lost its leading zero (`2139-4307`) still becomes `02139`. Digits are checked as ASCII only.
- **A malformed `updated` value** raises a `ValueError` that names the row index. The function stops rather than guessing a date that could affect which values get picked.
- **The phone + last name match** uses the normalized last name compared with `casefold()`. A row without a last name can only match by email.

**Before you send:**

1. **Filter on `opted_out == "Y"` before sending.** This function flags opted-out people but doesn't remove them. If anyone in a merged group opted out, the whole person is marked `"Y"`, which errs on the side of not emailing.
2. **Rule 9 drops opted-out rows that have no valid email or phone.** For example, an opt-out recorded with the email `ann@x` (no dot in the domain) won't merge with Ann's good record, so the good record could still come through as `"N"`. If your CRM has opt-out rows like that, check them by hand or suppress them with a looser match before sending.
3. **Only one email is sent per merged person.** If two rows with different emails merge through phone + last name, the person gets the email only at the preferred address.
