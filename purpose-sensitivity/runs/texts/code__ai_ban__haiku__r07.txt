import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_EPOCH = datetime(1970, 1, 1)
_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _clean_email(raw):
    """Return the normalized email if valid, otherwise ''."""
    s = raw or ""
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _clean_phone(raw):
    """Return the 10-digit normalized phone if valid, otherwise ''."""
    s = (raw or "").lower()
    cuts = [i for i in (s.find(m) for m in ("x", "ext", "#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _clean_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _is_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_digits(s):
        return s
    return ""


def _clean_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _clean_tags(raw):
    parts = re.split(r"[;,]", raw or "")
    return {p.strip().lower() for p in parts if p.strip()}


def _is_opted_out(raw):
    return (raw or "").strip().lower() in {"y", "yes", "true", "1"}


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _merge_group(group):
    def pick(field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda r: (
                not r["manual"],
                -(r["updated"] - _EPOCH).total_seconds(),
                r["index"],
            ),
        )
        return best[field]

    tags = set()
    for r in group:
        tags.update(r["tags"])

    return {
        "first_name": pick("first_name"),
        "last_name": pick("last_name"),
        "zip": pick("zip"),
        "email": pick("email"),
        "phone": pick("phone"),
        "source": pick("source"),
        "tags": ";".join(sorted(tags)),
        "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
        "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    # Normalize each row; drop rows with neither a valid email nor a valid phone.
    records = []
    for index, row in enumerate(rows):
        email = _clean_email(row.get("email", ""))
        phone = _clean_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        records.append({
            "index": index,
            "first_name": _clean_name(row.get("first_name", "")),
            "last_name": _clean_name(row.get("last_name", "")),
            "zip": _clean_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _clean_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find so matches are transitive.
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for k, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in by_email:
                union(k, by_email[rec["email"]])
            else:
                by_email[rec["email"]] = k
        last_key = rec["last_name"].casefold()
        if rec["phone"] and last_key:
            key = (rec["phone"], last_key)
            if key in by_phone_last:
                union(k, by_phone_last[key])
            else:
                by_phone_last[key] = k

    # Group rows; dict insertion order follows each group's earliest row.
    groups = {}
    for k, rec in enumerate(records):
        groups.setdefault(find(k), []).append(rec)

    return [_merge_group(group) for group in groups.values()]
