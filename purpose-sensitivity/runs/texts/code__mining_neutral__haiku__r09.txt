import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")


def _norm_email(raw):
    """Normalize an email field. Returns the normalized string (may be invalid)."""
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    parts = s.split("@")
    if len(parts) == 2 and parts[1] in _GMAIL_DOMAINS:
        local = parts[0].replace(".", "").split("+", 1)[0]
        s = f"{local}@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    """Return the normalized 10-digit phone, or "" if invalid."""
    s = raw.lower()
    markers = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if markers:
        s = s[:min(markers)]
    digits = "".join(c for c in s if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = raw.strip()
    if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
        s = s[:5]
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in ("y", "yes", "true", "1")


def _parse_tags(raw):
    return [t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()]


def _pick(members, field):
    """Choose the value of `field` from the group's rows where it is non-empty.

    Preference: manual source, then most recent `updated`, then earliest input row.
    """
    cands = [r for r in members if r[field]]
    if not cands:
        return ""
    # Stable sorts applied from least to most significant key.
    cands.sort(key=lambda r: r["index"])
    cands.sort(key=lambda r: r["ts"], reverse=True)
    cands.sort(key=lambda r: not r["manual"])
    return cands[0][field]


def dedupe_contacts(rows):
    # Step 1: normalize rows, dropping any with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row["email"])
        email = email if _valid_email(email) else ""
        phone = _norm_phone(row["phone"])
        if not email and not phone:
            continue

        source = row["source"].strip().lower()
        recs.append({
            "index": idx,
            "first": _norm_name(row["first_name"]),
            "last": _norm_name(row["last_name"]),
            "zip": _norm_zip(row["zip"]),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row["tags"]),
            "opted_out": _is_opted_out(row["opted_out"]),
            "ts": _parse_updated(row["updated"]),
        })

    # Step 2: union-find over rows that match on email or (phone + last name).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], i)
            else:
                first_by_email[r["email"]] = i
        if r["phone"] and r["last"]:
            key = (r["phone"], r["last"].lower())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    # Step 3: collect groups; order by earliest input row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)
    ordered_groups = sorted(groups.values(), key=lambda g: g[0])

    # Step 4: build one output record per person.
    output = []
    for g in ordered_groups:
        members = [recs[i] for i in g]
        all_tags = {t for r in members for t in r["tags"]}
        output.append({
            "first_name": _pick(members, "first"),
            "last_name": _pick(members, "last"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["ts"] for r in members).strftime("%Y-%m-%d"),
        })

    return output

Notes for the send:

- **Opt-outs:** Filter the output on `opted_out == "N"` before the send list is built. Because matching is transitive, an opt-out on any duplicate row marks the whole person as "Y," so this filter covers every duplicate.

- **ZIP+4:** I treated "looks like ZIP+4" strictly as five digits, a hyphen, and four digits. A value like `2139-4307` would therefore become `""`, not `02139`.
