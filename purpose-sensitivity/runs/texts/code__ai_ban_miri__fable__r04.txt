```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, deduplicate, and merge CRM contact rows.

    See the data rules doc: emails/phones/zips/names are normalized, rows with
    neither a valid email nor a valid phone are dropped, rows are grouped
    transitively by (valid email) or (valid phone + non-empty last name), and
    one merged dict per person is returned, ordered by each person's earliest
    input row.
    """

    # ---------- field normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"x|ext|#", s, flags=re.IGNORECASE)   # earliest extension marker
        if m:
            s = s[: m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                               # ZIP+4 -> ZIP
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):           # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized `updated` value: {raw!r}")

    def is_opted_out(raw):
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):
        return [t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()]

    # ---------- normalize rows, drop unusable ones ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not phone_valid(phone):
            phone = ""
        if not email and not phone:
            continue                               # rule 9
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": split_tags(row.get("tags", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "updated": parse_updated(row.get("updated", "")),
            "source": (row.get("source", "") or "").strip().lower(),
        })

    # ---------- transitive matching (union-find) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so roots reflect earliest rows
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    key_owner = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(i, key_owner[k])
            else:
                key_owner[k] = i

    groups = {}                                    # root -> rows, in order of earliest row
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---------- merge each group ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(
            cands,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[field]

    out = []
    for root in sorted(groups, key=lambda rt: groups[rt][0]["idx"]):
        group = groups[root]
        tags = set()
        for r in group:
            tags.update(r["tags"])
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return out
```

A few notes on choices I made where the rules left room:

- **Unparseable `updated`** raises a `ValueError` naming the bad value rather than silently guessing, since a wrong date would silently change which row "wins" a merge. If you'd rather it never crash the morning send, swap the `raise` for `return datetime.min`.
- **Phone extension markers**: I cut at the *earliest* occurrence of `x`, `ext`, or `#` (case-insensitive), so `555-123-4567 ext. 12` and `555-123-4567x12` both work.
- **Gmail**: the `+` and dot rules only apply when the address has exactly one `@` and the domain is `gmail.com`/`googlemail.com`; malformed addresses fall through to the validity check and get treated as empty.
- **Phone matching** uses the normalized last name lowercased, so `SMITH` and `Smith` match, as rule 10 intends.
- **Field picking** uses the tiebreak order exactly as written: manual source → latest `updated` → earliest input row. Because `source` itself goes through this picker, any group containing a manual row will output `source: "manual"`.
- Rows missing a key entirely are treated as `""` (`row.get`) so a slightly ragged export won't crash the run.

Since this is going out to supporters tomorrow, I'd suggest a quick sanity pass before sending: check `len(output)` against the input, spot-check a handful of merged people, and confirm the count of `opted_out == "Y"` looks plausible — those are the ones your mailer must exclude.
