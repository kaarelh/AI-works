import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the data team's rules (1-12)."""

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)              # Rule 1: display name
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(e):                              # Rule 2
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):                             # Rule 3
        s = raw or ""
        m = re.search(r"ext|x|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                                    # "" means invalid

    def norm_zip(raw):                               # Rule 4
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):                              # Rule 5
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw, idx):                     # Rule 6
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Row {idx}: unrecognized 'updated' value {raw!r}")

    def is_opted_out(raw):                           # Rule 7
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):                             # Rule 8
        return [t.strip().lower() for t in re.split(r"[;,]", raw or "")]

    # ---------- normalize and filter (Rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(row.get("tags", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "updated": parse_updated(row.get("updated", ""), idx),
        })

    # ---------- matching via union-find (Rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output (Rules 11-12) ----------

    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        # manual first, then most recent updated, then earliest in input
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    ordered = sorted(groups.values(), key=lambda g: min(r["idx"] for r in g))

    out = []
    for members in ordered:
        tags = sorted({t for r in members for t in r["tags"] if t})
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "source": pick(members, "source"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        })
    return out

**Things to check before tomorrow's send:**

1. **Opted-out people are still in the output.** They appear with `opted_out: "Y"`, as the spec requires. You need to filter them out before mailing, for example with `[p for p in result if p["opted_out"] == "N" and p["email"]]`. If anyone in a merged group opted out, the whole person is marked "Y", which is the safe choice.
2. **Phone-only people have `email: ""`.** The filter above drops them from the email send.
3. **Unparseable `updated` values raise a `ValueError` that names the row.** I chose this over guessing so that bad data can't silently change which row wins.
4. **Extension markers.** Because "x" counts as a marker, any phone string containing the letter x is cut at that x. That's the rule as written, and it's unlikely to matter in real phone data.
5. **Shared phones merge people.** Two people who share a phone and a last name (e.g., a household landline) will be merged into one contact, and only the winning email is kept. Under rule 10 this is expected, but it means one of them won't get the alert. If that matters to you, consider matching on email alone for this send.

I'd suggest running it on the export and spot-checking a handful of merged groups and the final count before scheduling the email.
