import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EPOCH = datetime(1970, 1, 1)

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_NON_DIGIT_RE = re.compile(r"[^0-9]")          # ASCII digits only
_ZIP4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_TAG_SPLIT_RE = re.compile(r"[;,]")


def _norm_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Return the 10-digit phone if valid, else ""."""
    s = raw or ""
    m = _EXT_RE.search(s)  # leftmost extension marker
    if m:
        s = s[:m.start()]
    digits = _NON_DIGIT_RE.sub("", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "") if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with neither valid email nor phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip", "")),
            "source": source,
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # --- Union-find over normalized records (rule 10, transitive).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the earlier record (not required, just tidy).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # --- Field selection priority: manual source, then newest, then earliest.
    def priority(r):
        return (
            0 if r["source"] == "manual" else 1,
            -(r["updated"] - _EPOCH),   # negated timedelta: newer sorts first
            r["idx"],
        )

    out = []
    for members in groups.values():
        person = {}
        for field in ("first_name", "last_name", "zip", "email", "phone", "source"):
            candidates = [r for r in members if r[field]]
            person[field] = min(candidates, key=priority)[field] if candidates else ""

        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        person["tags"] = ";".join(sorted(all_tags))
        person["opted_out"] = "Y" if any(r["opted_out"] for r in members) else "N"
        person["updated"] = max(r["updated"] for r in members).strftime("%Y-%m-%d")
        person["_order"] = min(r["idx"] for r in members)
        out.append(person)

    # --- Rule 12: order by each person's earliest input row.
    out.sort(key=lambda p: p["_order"])
    for p in out:
        del p["_order"]
    return out

Some notes for tomorrow's send:

- **Opted-out people are still in the output**, marked `opted_out: "Y"`, as the rules specify. Filter them out (`[p for p in result if p["opted_out"] == "N"]`) before building the send list. Otherwise they will get the email.
- **One email per merged person.** If rows were merged by phone and last name but had different email addresses, only the chosen address (manual source first, then newest) is kept. The other addresses for that person won't receive the alert. That's the intended dedupe behavior, but it's worth knowing.
- **Bad dates fail loudly.** If an `updated` value doesn't match one of the three formats, the function raises a `ValueError` instead of guessing. Any mistake on the eve of a send should show up immediately.
- **Spec edge cases, implemented literally:**
  - The extension marker matches any `x` in the phone field.
  - A ZIP that lost two leading zeros (e.g. `501` for `00501`) becomes `""`.
  - `str.title()` turns `MCDONALD` into `Mcdonald`.
- Rows dropped under rule 9 never take part in matching. So an opt-out row with no valid email or phone can't flag anyone. Since such a row can't be contacted anyway, that should be harmless.
