```python
import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_DIGITS = "0123456789"
_TRUTHY = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _norm_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[: m.start()]
    d = "".join(c for c in s if c in _DIGITS)
    if len(d) == 11 and d[0] == "1":
        d = d[1:]
    return d


def _valid_phone(d):
    return len(d) == 10 and d[0] not in "01"


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and all(c in _DIGITS for c in s):
        s = "0" + s
    return s if _ZIP5_RE.fullmatch(s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw, index):
    s = (raw or "").strip()
    for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Row {index}: unrecognized 'updated' value {raw!r}")


def _split_tags(raw):
    out = set()
    for t in re.split(r"[;,]", raw or ""):
        t = t.strip().lower()
        if t:
            out.add(t)
    return out


def dedupe_contacts(rows):
    # Normalize rows and drop those with no valid email and no valid phone (rule 9)
    recs = []
    for i, row in enumerate(rows):
        email = _norm_email(row.get("email"))
        phone = _norm_phone(row.get("phone"))
        email_ok = _valid_email(email)
        phone_ok = _valid_phone(phone)
        if not email_ok and not phone_ok:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "index": i,
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "zip": _norm_zip(row.get("zip")),
            "source": source,
            "manual": source == "manual",
            "ts": _parse_updated(row.get("updated"), i),
            "tags": _split_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _TRUTHY,
        })

    # Union-find over the kept records (rule 10, transitive)
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the root at the smaller index (earliest row)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for k, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(k, first_seen[key])
            else:
                first_seen[key] = k

    groups = {}
    for k in range(len(recs)):
        groups.setdefault(find(k), []).append(recs[k])

    def pick(members, field):
        best = None
        best_key = None
        for r in members:
            if not r[field]:
                continue
            # manual first, then most recent, then earliest in input
            key = (r["manual"], r["ts"], -r["index"])
            if best_key is None or key > best_key:
                best, best_key = r, key
        return best[field] if best is not None else ""

    out = []
    # Output in order of earliest row (rule 12)
    for members in sorted(groups.values(), key=lambda m: min(r["index"] for r in m)):
        tags = set()
        for r in members:
            tags |= r["tags"]
        latest = max(r["ts"] for r in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })
    return out
```

**Choices I made where the rules leave room for interpretation:**
- **Extension markers:** The marker that appears first in the string wins. In "555-123-4567 ext. 9", the cut happens at "ext", which is fine because the same cut would happen at "x" anyway.
- **ZIP+4:** "2139-4307" (four digits before the hyphen, with the leading zero lost) becomes "02139". Anything else that doesn't fit the pattern is checked under the plain 5-digit rule.
- **Gmail:** If removing the dots and the "+" suffix leaves an empty local part (e.g. `+foo@gmail.com`), the address is treated as invalid.
- **Bad `updated` values:** These raise a `ValueError` that includes the row index, so bad data fails loudly instead of being silently misordered. The check only runs on rows that survive rule 9.
- **Digit checks:** These use ASCII 0–9 only, so non-ASCII digits like "٣" don't count.

**Before you send:**
- **Remove opted-out people yourself.** The function marks them `opted_out: "Y"`, but it doesn't remove them. Filter with something like `[p for p in result if p["opted_out"] == "N" and p["email"]]`.
- **Some opt-outs may not get applied.** Rows with no valid email or phone are dropped before matching (rule 9). Rows only merge on email, or on phone plus last name. So an opt-out recorded under an address that doesn't match the person's other rows won't carry over to them. If your email platform has its own suppression list, apply it as well.
- **Some people have no email.** Those who only have a valid phone will come out with `email: ""`, so they need the filter above too.
