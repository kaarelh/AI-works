from datetime import datetime

_DIGITS = "0123456789"


def dedupe_contacts(rows):
    def norm_email(raw):
        s = raw or ""
        lt = s.find("<")
        if lt != -1:
            gt = s.find(">", lt + 1)
            if gt != -1:
                s = s[lt + 1:gt]

            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                plus = local.find("+")
                if plus != -1:
                    local = local[:plus]
                domain = "gmail.com"
                s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"ext|x|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        d = "".join(c for c in s if c in _DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) != 10 or d[0] in "01":
            return ""
        return d

    def norm_zip(raw):
        s = (raw or "").strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
            s = s.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

        s = (raw or "").strip()
        if s.isupper() or s.islower():

        return s

            try:
                return datetime.strptime(s, fmt)

Notes on a few choices:

- **Opt-outs:** if any row in a person's group is opted out, the whole person is `"Y"`. Make sure your send step filters on `opted_out == "N"`.
- **Gmail normalization:** it only applies when the address has exactly one `@`. Otherwise the address fails validation anyway.
- **Extension markers:** the regex `ext|x|#` cuts at the earliest marker, so `555-123-4567 ext 12` and `555-123-4567x12` are both handled.
- **Bad `updated` values:** the function raises `ValueError` rather than silently guessing, since your rules say `updated` is always one of the three formats.
- **Ties on "most recent":** these fall back to input order, using the original row position, including rows that were dropped.
- **Source preference:** the `"manual"` preference is judged per row, so a manual row only wins a field if it has a non-empty value for that field.
