```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_FIVE_DIGITS_RE = re.compile(r"[0-9]{5}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _field(row, key):
    value = row.get(key)
    return "" if value is None else str(value)


def _normalize_email(raw):
    """Return the normalized email if valid, otherwise ''."""
    m = _ANGLE_RE.search(raw)
    if m:
        raw = m.group(1)
    email = raw.strip().lower()

    if email.count("@") != 1:
        return ""
    local, domain = email.split("@")

    if domain in _GMAIL_DOMAINS:
        local = local.split("+", 1)[0].replace(".", "")
        domain = "gmail.com"

    if not local:
        return ""
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return f"{local}@{domain}"


def _normalize_phone(raw):
    """Return the 10-digit phone if valid, otherwise ''."""
    m = _EXT_RE.search(raw)
    if m:
        raw = raw[:m.start()]
    digits = "".join(ch for ch in raw if "0" <= ch <= "9")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    z = raw.strip()
    m = _ZIP4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if len(z) == 4 and z.isascii() and z.isdigit():
        z = "0" + z
    return z if _FIVE_DIGITS_RE.fullmatch(z) else ""


def _normalize_name(raw):
    name = raw.strip()
    if name.isupper() or name.islower():
        name = name.title()
    return name


def _parse_updated(raw):
    text = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(text, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with no valid email and no valid phone.
    records = []
    for index, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        records.append({
            "index": index,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "updated": _parse_updated(_field(row, "updated")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
            "tags": _parse_tags(_field(row, "tags")),
        })

    # --- Union-find over record positions.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the earlier record as root (not required, but tidy).
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for pos, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(pos, first_seen[key])
            else:
                first_seen[key] = pos

    # --- Collect groups, ordered by earliest input row.
    groups = {}
    for pos, rec in enumerate(records):
        groups.setdefault(find(pos), []).append(rec)
    ordered_groups = sorted(groups.values(), key=lambda g: min(r["index"] for r in g))

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        # Prefer manual, then most recent updated, then earliest input row.
        best = max(candidates,
                   key=lambda r: (r["is_manual"], r["updated"], -r["index"]))
        return best[field]

    # --- Build output.
    output = []
    for group in ordered_groups:
        tags = set()
        for r in group:
            tags |= r["tags"]
        latest = max(r["updated"] for r in group)
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })
    return output
```

**How I read the ambiguous parts:**

- **Extension markers:** The code finds the leftmost marker. For "ext", the cut happens at the "e", so `555-123-4567 ext 12` and `555-123-4567x12` both reduce to the same 10 digits.
- **Gmail:** If an address like `+promo@gmail.com` ends up with nothing before the @ after normalization, it counts as invalid.
- **ZIP+4:** This also handles the case where Excel dropped the leading zero, e.g. `2139-4307` becomes `02139`. Only ASCII digits count, so oddities like full-width digits become `""`.
- **Phone matching:** It uses the normalized last name compared with `casefold()`. A phone match on a row with an empty last name never links that row to anything.
- **Bad `updated` values:** An unparseable `updated` raises a `ValueError` instead of being silently skipped. That way a malformed row can't quietly change which person's data wins. If you'd rather drop or default those rows, change `_parse_updated`.
- **Tie-break order:** For each field, a "manual" row always wins. After that, the most recent `updated` wins, and ties go to the earlier row in the input.

**Before you send:** filter the output to `opted_out == "N"`. The function marks a whole merged person as opted out if any of their rows opted out, but it doesn't remove them, so the send list still needs that filter.
