import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_PHONE_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_ZIP_PLUS4_RE = re.compile(r"[0-9]{4,5}-[0-9]{4}")
_ZIP4_RE = re.compile(r"[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _normalize_email(raw):
    s = raw
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    m = _PHONE_EXT_RE.search(raw)
    s = raw[:m.start()] if m else raw
    digits = _NON_DIGIT_RE.sub("", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    if _ZIP_PLUS4_RE.fullmatch(s):
        s = s.split("-", 1)[0]
    if _ZIP4_RE.fullmatch(s):
        s = "0" + s
    return s if _ZIP5_RE.fullmatch(s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _parse_tags(raw):
    tags = set()
    for part in _TAG_SPLIT_RE.split(raw):
        t = part.strip().lower()
        if t:
            tags.add(t)
    return tags


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with neither a valid email nor phone.
    kept = []
    for row in rows:
        email = _normalize_email(row.get("email") or "")
        phone = _normalize_phone(row.get("phone") or "")
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not email_ok and not phone_ok:
            continue

        source = (row.get("source") or "").strip().lower()
        kept.append({
            "first_name": _normalize_name(row.get("first_name") or ""),
            "last_name": _normalize_name(row.get("last_name") or ""),
            "zip": _normalize_zip(row.get("zip") or ""),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags") or ""),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated") or ""),
        })

    # Step 2: union-find over rows so matching is transitive.
    parent = list(range(len(kept)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the smaller index as root so groups sort by earliest row.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for j, r in enumerate(kept):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], j)
            else:
                by_email[r["email"]] = j
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(by_phone_last[key], j)
            else:
                by_phone_last[key] = j

    # Step 3: group rows. Dict insertion order follows each group's earliest row.
    groups = {}
    for j in range(len(kept)):
        groups.setdefault(find(j), []).append(j)

    def pick(members, field):
        cands = [j for j in members if kept[j][field]]
        if not cands:
            return ""
        # Stable sorts applied from least to most significant key:
        # input order -> most recent updated -> manual preference.
        cands.sort(key=lambda j: j)
        cands.sort(key=lambda j: kept[j]["updated"] or datetime.min, reverse=True)
        cands.sort(key=lambda j: 0 if kept[j]["manual"] else 1)
        return kept[cands[0]][field]

    # Step 4: build one output record per person.
    output = []
    for members in groups.values():
        all_tags = set()
        for j in members:
            all_tags |= kept[j]["tags"]

        dates = [kept[j]["updated"] for j in members if kept[j]["updated"] is not None]
        latest = max(dates).strftime("%Y-%m-%d") if dates else ""

        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(kept[j]["opted_out"] for j in members) else "N",
            "updated": latest,
        })

    return output

A few judgment calls where the rules were ambiguous, so your data person can confirm them:

- **ZIP+4 with a short prefix:** The function accepts `2139-4307` as well as `02139-4307`, so the Excel-stripped form is handled before the 4-digit padding step.
- **Unparseable or missing `updated`:** Such a row sorts as the oldest possible. If no row in a group has a parseable date, `updated` is output as `""`.
- **Name matching:** Last names are compared after the rule-5 normalization, lowercased.
- **Dropped rows:** Rows removed by rule 9 contribute nothing to tags, opt-out status, or matching.
