```python
import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_ANGLE_EMAIL = re.compile(r"<([^<>]*)>")
_EXT_MARKER = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4 = re.compile(r"[0-9]{4,5}-[0-9]{4}")


def _str(value):
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = _str(raw)
    m = _ANGLE_EMAIL.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = _EXT_MARKER.split(_str(raw), maxsplit=1)[0]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _normalize_zip(raw):
    z = _str(raw).strip()
    if _ZIP_PLUS4.fullmatch(z):
        z = z.split("-", 1)[0]
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    s = _str(raw).strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = _str(raw).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized updated value: {raw!r}")


def _is_opted_out(raw):
    return _str(raw).strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", _str(raw)) if t.strip()}


def dedupe_contacts(rows):
    # Pass 1: normalize each row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        phone = _normalize_phone(row.get("phone"))
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        source = _str(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Pass 2: union-find so matches are transitive.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_seen = {}
    for pos, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(email_seen[r["email"]], pos)
            else:
                email_seen[r["email"]] = pos
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in phone_seen:
                union(phone_seen[key], pos)
            else:
                phone_seen[key] = pos

    # Group rows by person, ordered by each person's earliest input row.
    groups = {}
    order = []
    for pos in range(len(recs)):
        root = find(pos)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[pos])

    def pick(members, field):
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        best = min(
            cands,
            key=lambda m: (not m["manual"], -m["updated"].timestamp(), m["idx"]),
        )
        return best[field]

    # Build one output record per person.
    result = []
    for root in order:
        members = groups[root]
        tags = set().union(*(m["tags"] for m in members))
        latest = max(m["updated"] for m in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Filter opt-outs before sending.** The function returns opted-out people with `opted_out == "Y"` rather than removing them. Filter those out before building the send list, e.g. `recipients = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N"]`. Because opt-out is group-level, if any duplicate row for a person is opted out, that whole person is marked `"Y"`, which is the conservative choice.
- **ZIP+4 with a dropped leading zero.** I treat `2139-4307` as ZIP+4 (keeping `02139`) because the 4-digit rule implies Excel-damaged ZIP+4 values are expected. If you want only exact 5-digit prefixes to count as ZIP+4, change `[0-9]{4,5}` to `[0-9]{5}` in `_ZIP_PLUS4`.
