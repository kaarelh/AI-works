import re
from datetime import datetime


def dedupe_contacts(rows):
    def s(v):
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"ext|x|#", p, re.IGNORECASE)
        if m:
            p = p[:m.start()]
        d = re.sub(r"[^0-9]", "", p)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        m = re.fullmatch(r"([0-9]{5})-([0-9]{4})", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        u = s(raw).strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                pass
        return datetime.min

    def is_opted_out(raw):
        return s(raw).strip().lower() in ("y", "yes", "true", "1")

    def parse_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize and filter ----------
    recs = []
    for i, r in enumerate(rows):
        email = norm_email(r.get("email"))
        phone = norm_phone(r.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": i,
            "first_name": norm_name(r.get("first_name")),
            "last_name": norm_name(r.get("last_name")),
            "zip": norm_zip(r.get("zip")),
            "email": email,
            "phone": phone,
            "source": s(r.get("source")).strip().lower(),
            "tags": parse_tags(r.get("tags")),
            "opted_out": is_opted_out(r.get("opted_out")),
            "updated": parse_updated(r.get("updated")),
        })

    # ---------- union-find ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for k, rec in enumerate(recs):
        if rec["email"]:
            key = rec["email"]
            if key in email_seen:
                union(k, email_seen[key])
            else:
                email_seen[key] = k
        last = rec["last_name"].strip().lower()
        if rec["phone"] and last:
            key = (rec["phone"], last)
            if key in phone_last_seen:
                union(k, phone_last_seen[key])
            else:
                phone_last_seen[key] = k

    groups = {}
    order = []
    for k in range(n):
        root = find(k)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(k)

    # ---------- build output ----------
    def pick(members, field):
        best = None
        best_rank = None
        for k in members:  # members are in input order
            rec = recs[k]
            if not rec[field]:
                continue
            manual = 1 if rec["source"] == "manual" else 0
            rank = (manual, rec["updated"])
            if best is None or rank > best_rank:  # strict: ties keep earlier row
                best = rec
                best_rank = rank
        return best[field] if best is not None else ""

    result = []
    for root in order:
        members = groups[root]
        tags = set()
        for k in members:
            tags.update(recs[k]["tags"])
        latest = max(recs[k]["updated"] for k in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[k]["opted_out"] for k in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result

Notes on a few judgment calls:

- **ZIP+4:** I only treat `NNNNN-NNNN` as ZIP+4, as in your example. A value like `2139-4307` would become `""`. If your data has Excel-mangled ZIP+4 values, change the regex to `([0-9]{4,5})-([0-9]{4})`.
- **Unparseable `updated`:** These are treated as the earliest possible date rather than raising an error.
- **Opt-outs:** Any opted-out row in a group marks the whole person as `Y`. For tomorrow's send, filter on `opted_out == "N"` and a non-empty `email`. Some people may have only a phone number.
- **Dropped rows:** Rows without a valid email or phone are dropped before matching, so they can't link groups together or affect output order.
