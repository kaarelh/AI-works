```python
import re
from datetime import datetime

_TRUTHY = {"y", "yes", "true", "1"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"^([0-9]+)-([0-9]{4})$")
_FIVE_DIGITS_RE = re.compile(r"^[0-9]{5}$")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = raw
        m = _EXT_RE.search(s)
        if m:
            s = s[:m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = _ZIP4_RE.match(z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if _FIVE_DIGITS_RE.match(z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable: treat as oldest

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize rows; drop those with no valid email and no valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in _TRUTHY,
            "updated": parse_updated(get(row, "updated")),
        })

    # Union-find over the records.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("e", r["email"]))
        last = r["last_name"].lower()
        if r["phone"] and last:
            keys.append(("p", r["phone"], last))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    ordered = sorted(groups.values(), key=lambda g: min(r["idx"] for r in g))

    result = []
    for g in ordered:
        tags = set()
        for r in g:
            tags |= r["tags"]
        result.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "zip": pick(g, "zip"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "source": pick(g, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["updated"] for r in g).strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Opt-outs:** a person is flagged `opted_out = "Y"` if any of their merged rows opted out. Before sending, filter on `opted_out == "N"` so the opt-out is honored.
- **Dropped rows:** rows with neither a valid email nor a valid phone are removed before matching, per rule 9. They won't appear in the output, so they can't receive the email anyway.
- **Unparseable dates:** if an `updated` value matches none of the three formats, it's treated as `datetime.min` (oldest) rather than raising an error. If you'd rather it fail loudly, replace that fallback with a `raise`.
- **Matching:** the last-name comparison for phone matches is case-insensitive via `.lower()`, applied after the title-casing in rule 5.
