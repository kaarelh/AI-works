```python
import re
from datetime import datetime

_FIELDS = ("first_name", "last_name", "email", "phone", "zip",
           "tags", "opted_out", "updated", "source")

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DT_MIN = datetime.min


def _normalize_email(raw):
    """Rule 1: extract from display name, strip, lowercase, canonicalize Gmail."""
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(s):
    """Rule 2."""
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _normalize_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, otherwise ""."""
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    s = (raw or "").strip()
    m = _ZIP4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6. Raises ValueError on an unrecognized date rather than guessing."""
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    """Rule 8 (plus rule 11's strip/lowercase/drop-empties)."""
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "") if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with neither valid email nor phone (rule 9).
    kept = []
    for idx, row in enumerate(rows):
        get = lambda k: row.get(k) or ""
        email = _normalize_email(get("email"))
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(get("phone"))
        if not email and not phone:
            continue
        source = get("source").strip().lower()
        kept.append({
            "idx": idx,
            "first_name": _normalize_name(get("first_name")),
            "last_name": _normalize_name(get("last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(get("zip")),
            "tags": _split_tags(get("tags")),
            "opted_out": _is_opted_out(get("opted_out")),
            "dt": _parse_updated(get("updated")),
            "source": source,
            "manual": source == "manual",
        })

    # --- Union-find for transitive matching (rule 10).
    parent = list(range(len(kept)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the earlier row as root (not required, just tidy).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(kept):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i, r in enumerate(kept):
        groups.setdefault(find(i), []).append(r)

    # --- Field selection (rule 11): manual first, then most recent, then earliest.
    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        best = min(candidates,
                   key=lambda r: (not r["manual"], -(r["dt"] - _DT_MIN), r["idx"]))
        return best[field]

    # --- Build output, ordered by each person's earliest input row (rule 12).
    output = []
    for members in sorted(groups.values(), key=lambda g: min(r["idx"] for r in g)):
        tags = set()
        for r in members:
            tags |= r["tags"]
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["dt"] for r in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return output
```

**Judgment calls (easy to change):**
- **Bad dates raise an error.** If a row's `updated` isn't in one of the three formats, the function stops with a `ValueError` instead of guessing. For a send like this, I think failing loudly is safer than quietly misordering rows.
- **ZIP+4 with a lost leading zero** (e.g. `2139-4307`) is treated as ZIP+4 and becomes `02139`. Plain `02139-4307` becomes `02139`, as specified.
- **Extension markers:** cutting starts at the *first* `x`, `ext`, or `#` anywhere in the phone string. So `617-555-1234 ext. 5` and `(617) 555-1234 #12` both work. Anything that happens to contain an `x` will also be cut there.
- **Last names match case-insensitively** for phone matching, using `casefold()`.
- **Missing keys or `None` values** are treated as empty strings (except `updated`, which is required).

**Things to know before tomorrow's send:**
1. **Opt-outs only carry over within a matched group.** If someone opted out as `ann@work.org` but is also on the list as `ann@gmail.com`, and nothing links the two rows (no shared phone plus last name), they become two separate people, and the second will still get the email. If you keep a separate suppression list, apply it to the output as a final step.
2. **Output emails are normalized.** Gmail addresses come out without dots or `+tags` and with `gmail.com` as the domain (e.g. `j.doe+news@googlemail.com` becomes `jdoe@gmail.com`). Those still deliver to the same inbox. Other addresses are lowercased, which in practice is fine for virtually every mail provider.
3. **The output email is filtered by validity, not by opt-out.** Your sending step should skip every row with `opted_out == "Y"` and every row with an empty `email` (people matched only by phone).
