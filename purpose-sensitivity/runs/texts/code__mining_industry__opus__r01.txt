```python
import re
from datetime import datetime

_DISPLAY_EMAIL_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _norm_email(raw):
    s = raw or ""
    m = _DISPLAY_EMAIL_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _email_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[: m.start()]
    digits = "".join(ch for ch in s if ch in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""  # invalid phone


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize and drop rows with neither a valid email nor a valid phone ---
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _email_valid(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        dt = _parse_updated(row.get("updated", ""))
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip", "")),
            "source": source,
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower() in _OPT_OUT_VALUES,
            "dt": dt,
            # Sort key: manual first, then most recent, then earliest in input
            "rank": (
                source != "manual",
                -dt.toordinal(),
                -(dt.hour * 3600 + dt.minute * 60 + dt.second),
                idx,
            ),
        })

    # --- Union-find over shared match keys (transitive grouping) ---
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(first_seen[k], i)
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # --- Build one output record per person ---
    out = []
    for members in groups.values():
        ranked = sorted(members, key=lambda r: r["rank"])

        def pick(field):
            for r in ranked:
                if r[field]:
                    return r[field]
            return ""

        tags = set()
        for r in members:
            tags |= r["tags"]

        out.append((
            min(r["idx"] for r in members),
            {
                "first_name": pick("first_name"),
                "last_name": pick("last_name"),
                "zip": pick("zip"),
                "email": pick("email"),
                "phone": pick("phone"),
                "source": pick("source"),
                "tags": ";".join(sorted(tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
                "updated": max(r["dt"] for r in members).strftime("%Y-%m-%d"),
            },
        ))

    out.sort(key=lambda t: t[0])
    return [rec for _, rec in out]
```

A few things to check before tomorrow's send:

1. **Filter out opt-outs before sending.** The function keeps opted-out people in the output, marked `opted_out == "Y"`, as the spec says. Your send list should be something like `[c for c in dedupe_contacts(rows) if c["opted_out"] == "N" and c["email"]]`.

2. **Some opt-outs may be lost (a gap in the spec).** Rule 9 drops rows that have neither a valid email nor a valid phone *before* matching. If someone's opt-out was recorded on such a row (for example, a malformed email like `ann@xorg`), that opt-out can't reach their other records, and they could still get the email. To be safe, you could also suppress anyone whose raw email, lowercased and with any display name removed, appears on an opted-out row, even if the row was dropped.

3. **Bad dates stop the run.** If an `updated` value isn't in one of the three formats, the function raises a `ValueError` and names the bad value. I did this on purpose so bad data gets noticed instead of silently misordered.

4. **Some behavior worth knowing:**
   - Gmail normalization is applied before the validity check, so `+tag@gmail.com` becomes `@gmail.com` and is treated as invalid.
   - Phone/last-name matching only happens when *both* rows have a valid phone and a non-empty last name.
   - ZIP+4 values with a 4-digit base, like `2139-4307`, become `02139`.
