```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _normalize_email(raw):
    s = (raw or "").strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    parts = s.split("@")
    if len(parts) == 2 and parts[1] in _GMAIL_DOMAINS:
        local = parts[0].replace(".", "").split("+", 1)[0]
        s = local + "@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Return the normalized 10-digit phone, or "" if invalid or empty."""
    s = raw or ""
    m = re.search(r"ext|x|#", s, re.IGNORECASE)
    if m:
        s = s[:m.start()]
    digits = "".join(ch for ch in s if ch in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    z = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _normalize_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # Step 1: normalize every row and drop rows with no valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(row.get("phone"))
        if not email and not phone:
            continue
        records.append({
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": (row.get("source") or "").strip().lower(),
            "tags": _normalize_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Step 2: link rows that share a valid email, or share a valid phone
    # plus the same non-empty last name. Union-find gives transitivity.
    n = len(records)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}

    def link(key, i):
        if key in first_seen:
            union(first_seen[key], i)
        else:
            first_seen[key] = i

    for i, r in enumerate(records):
        if r["email"]:
            link(("email", r["email"]), i)
        if r["phone"] and r["last_name"]:
            link(("phone_last", r["phone"], r["last_name"].lower()), i)

    # Group rows by root. Dict insertion order follows each group's earliest row.
    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    def ts_key(t):
        return t if t is not None else datetime.min

    # Step 3: build one output record per person.
    results = []
    for members in groups.values():
        # Preference order: manual first, then most recent updated, then earliest row.
        # Python's sort is stable and keeps original order for ties, even with reverse=True.
        order = sorted(members, key=lambda i: ts_key(records[i]["updated"]), reverse=True)
        order = sorted(order, key=lambda i: records[i]["source"] != "manual")

        def pick(field):
            for i in order:
                if records[i][field]:
                    return records[i][field]
            return ""

        tags = set()
        for i in members:
            tags |= records[i]["tags"]

        dates = [records[i]["updated"] for i in members if records[i]["updated"] is not None]
        latest = max(dates).strftime("%Y-%m-%d") if dates else ""

        results.append({
            "first_name": pick("first_name"),
            "last_name": pick("last_name"),
            "zip": pick("zip"),
            "email": pick("email"),
            "phone": pick("phone"),
            "source": pick("source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(records[i]["opted_out"] for i in members) else "N",
            "updated": latest,
        })

    return results
```

Notes on the implementation:

- **Matching:** Rows are linked through hash keys (shared email, or shared phone plus last name) rather than comparing every pair, so it stays fast on large exports.
- **ZIP+4 with a dropped leading zero:** An input like `2139-4307` is treated as ZIP+4 and becomes `02139`. The spec's wording could also be read as requiring five digits before the hyphen, so confirm with your data person if that matters.
- **Unparseable `updated` values:** These sort as the oldest. If no row in a group has a parseable date, the output `updated` is `""`.
- **Opt-outs:** Because `opted_out` is "Y" if any row in a group opted out, filter on `opted_out == "N"` when building the send list.
