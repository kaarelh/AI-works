```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows per the data team's rules.

    rows: list of dicts with string fields first_name, last_name, email, phone,
          zip, tags, opted_out, updated, source.
    Returns a list of merged dicts (same keys), one per person, ordered by the
    earliest input row belonging to that person.
    """
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- field normalizers ----------

    def text(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)        # display-name form: Ann Lee <ann@x.org>
        e = (m.group(1) if m else raw).strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+", 1)[0]
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        m = re.search(r"x|ext|#", raw, re.IGNORECASE)  # extension marker
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"\D", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:                                 # ZIP+4 -> keep the ZIP part
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):             # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n and (n.isupper() or n.islower()):
            return n.title()
        return n

    def parse_updated(raw, idx):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"row {idx}: unrecognized 'updated' value {raw!r}")

    def is_opted_out(raw):
        return raw.strip().lower() in OPT_OUT_VALUES

    def parse_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---------- normalize rows, drop unusable ones (rule 9) ----------

    prepared = []
    for idx, row in enumerate(rows):
        email = norm_email(text(row, "email"))
        phone = norm_phone(text(row, "phone"))
        email_ok = email_valid(email)
        phone_ok = phone_valid(phone)
        if not (email_ok or phone_ok):
            continue
        prepared.append({
            "idx": idx,
            "first_name": norm_name(text(row, "first_name")),
            "last_name": norm_name(text(row, "last_name")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "zip": norm_zip(text(row, "zip")),
            "tags": parse_tags(text(row, "tags")),
            "opted_out": is_opted_out(text(row, "opted_out")),
            "updated": parse_updated(text(row, "updated"), idx),
            "source": text(row, "source").strip().lower(),
        })

    # ---------- group rows into people with union-find (rule 10, transitive) ----------

    parent = list(range(len(prepared)))

    def find(i):
        root = i
        while parent[root] != root:
            root = parent[root]
        while parent[i] != root:          # path compression
            parent[i], i = root, parent[i]
        return root

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email, by_phone = {}, {}
    for i, r in enumerate(prepared):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], i)
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone:
                union(by_phone[key], i)
            else:
                by_phone[key] = i

    groups = {}                            # insertion order == order of earliest row (rule 12)
    for i, r in enumerate(prepared):
        groups.setdefault(find(i), []).append(r)

    # ---------- merge each group into one output dict (rule 11) ----------

    def pick(group, key):
        candidates = [r for r in group if r[key]]   # non-empty / valid normalized value
        if not candidates:
            return ""
        best = max(candidates, key=lambda r: (r["source"] == "manual",   # manual first
                                              r["updated"],              # then most recent
                                              -r["idx"]))                # then earliest row
        return best[key]

    output = []
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return output
```

A few notes on judgment calls, in case your data person wants to adjust them:

- **`updated`**: the spec says it's never empty, so an unparseable value raises a `ValueError` naming the row rather than silently sorting it wrong. If you'd rather it be lenient, swap the `raise` for `return datetime.min`.
- **Extension markers**: the regex finds the *earliest* of `x` / `ext` / `#` anywhere in the phone, so `"617-555-0100 ext. 12"` and `"6175550100x12"` both truncate correctly.
- **ZIP+4**: anything with a hyphen keeps the part before it, so `"2139-4307"` (hyphen plus eaten zero) also becomes `"02139"`.
- **`str.title()` quirk**: per rule 5, `"MCDONALD"` becomes `"Mcdonald"` and `"O'BRIEN"` becomes `"O'Brien"`; mixed-case input like `"McDonald"` is left alone.
- **Phone matching** uses the normalized last name compared case-insensitively, as specified; rows with a valid phone but a blank last name only merge via email.

Since the merged `opted_out` is `"Y"` if *any* row in the group opted out, you'll want to filter `opted_out == "Y"` out of the send list after deduping — the function keeps those people in its output so you have the record, but doesn't drop them for you.
