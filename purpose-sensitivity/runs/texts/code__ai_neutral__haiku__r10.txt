```python
import re
from datetime import datetime

_DIGITS = frozenset("0123456789")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = frozenset({"y", "yes", "true", "1"})
_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_PHONE_MARKER_RE = re.compile(r"x|ext|#")
_ZIP_PLUS4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_TAG_SPLIT_RE = re.compile(r"[;,]")


def _all_ascii_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw.strip()
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw.lower()
    m = _PHONE_MARKER_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    if _ZIP_PLUS4_RE.fullmatch(s):
        s = s.split("-")[0]
    if len(s) == 4 and _all_ascii_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_ascii_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    # Not covered by the spec; sorts as the oldest possible value.
    return datetime.min


def _parse_tags(raw):
    tags = set()
    for part in _TAG_SPLIT_RE.split(raw):
        t = part.strip().lower()
        if t:
            tags.add(t)
    return tags


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one output dict per person.

    See the module-level helpers for the per-field normalization rules.
    The input list is not modified.
    """
    # Normalize every row and drop those with no valid email or phone.
    records = []
    for row in rows:
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not _is_valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue

        updated = _parse_updated(row.get("updated", ""))
        opted_out = row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES

        records.append({
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": opted_out,
            "updated": updated,
        })

    # Union-find so that matches are transitive.
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    key_owner = {}
    for j, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in key_owner:
                union(j, key_owner[key])
            else:
                key_owner[key] = j

    # Collect groups. Iterating indices in order means dict order is
    # the order of each group's earliest row.
    groups = {}
    for j in range(len(records)):
        groups.setdefault(find(j), []).append(j)

    def pick(members, field):
        # members are in input order. Stable sorts give the priority:
        # manual first, then most recent updated, then earlier input row.
        cands = [records[j] for j in members if records[j][field]]
        cands.sort(key=lambda r: r["updated"], reverse=True)
        cands.sort(key=lambda r: r["source"] != "manual")
        return cands[0][field] if cands else ""

    output = []
    for members in groups.values():
        tags = set()
        opted_out = False
        latest = records[members[0]]["updated"]
        for j in members:
            rec = records[j]
            tags |= rec["tags"]
            opted_out = opted_out or rec["opted_out"]
            if rec["updated"] > latest:
                latest = rec["updated"]

        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if opted_out else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })

    return output
```

Notes on the implementation:

- **Matching:** Email and (phone + casefolded last name) are both used as union-find keys, so transitivity falls out automatically. Rows with an empty last name never match on phone alone.
- **Ordering:** Each group's output position is set by its earliest input row. Field selection uses two stable sorts, so ties resolve to the earlier input row.
- **Unparseable `updated`:** The spec says every `updated` value is one of the three formats, so this shouldn't happen. If it does, the row sorts as `datetime.min` instead of raising an error. Tell me if you'd prefer it to raise.
