import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"


def _is_ascii_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _normalize_email(raw):
    s = (raw or "").strip()
    # Display name form, e.g. "Ann Lee <ann@x.org>": keep only the angle-bracket part.
    i = s.find("<")
    if i != -1:
        j = s.find(">", i + 1)
        if j != -1:
            s = s[i + 1:j]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw or ""
    lower = s.lower()
    # Drop everything from the earliest extension marker onward.
    cuts = [i for i in (lower.find("x"), lower.find("ext"), lower.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    s = (raw or "").strip()
    # ZIP+4 such as "02139-4307" (or "2139-4307" after Excel dropped the zero):
    # keep the part before the hyphen.
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _is_ascii_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_ascii_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _normalize_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def _pick(members, field):
    """Choose the value for `field` among members where it is non-empty.
    Preference: source == manual, then most recent updated, then earliest input row."""
    cands = [m for m in members if m[field]]
    if not cands:
        return ""
    # Stable sorts applied from least to most significant key.
    cands.sort(key=lambda m: m["idx"])
    cands.sort(key=lambda m: m["updated"], reverse=True)
    cands.sort(key=lambda m: not m["manual"])
    return cands[0][field]


def dedupe_contacts(rows):
    # Normalize rows and drop those without a valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone"))
        if not (email or phone):
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _normalize_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower()
                         in ("y", "yes", "true", "1"),
            "updated": _parse_updated(row.get("updated")),
        })

    # Union-find over rows: shared valid email, or shared valid phone + last name.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    email_seen = {}
    phone_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in phone_seen:
                union(i, phone_seen[key])
            else:
                phone_seen[key] = i

    # Group rows by root. Dict insertion order follows each group's earliest row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    out = []
    for members in groups.values():
        out.append({
            "first_name": _pick(members, "first_name"),
            "last_name": _pick(members, "last_name"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(set().union(*(m["tags"] for m in members)))),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
        })
    return out

Because the merge is transitive, one opt-out row marks the whole person, which is the conservative choice.

- **Bad dates raise an error.** If an `updated` value in a kept row matches none of the three formats, the function raises `ValueError` and does not skip the row silently. Rows dropped for having no valid contact are never parsed, so bad dates in them are ignored.
- **Last-name matching uses `casefold()`** on the normalized name, so title-casing doesn't affect whether two rows match.
