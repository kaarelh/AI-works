import re
from datetime import datetime

_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    """Return the normalized email, or "" if it is not valid."""
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") != 1:
        return ""
    local, _, domain = s.partition("@")
    if domain in ("gmail.com", "googlemail.com"):
        local = local.replace(".", "").split("+", 1)[0]
        domain = "gmail.com"
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return f"{local}@{domain}"


def _normalize_phone(raw):
    """Return the normalized 10-digit phone, or "" if it is not valid."""
    s = raw.lower()
    cuts = [i for i in (s.find(marker) for marker in ("x", "ext", "#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Return the normalized 5-digit ZIP, or "" if it is not valid."""
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _all_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    parts = (t.strip().lower() for t in re.split(r"[;,]", raw))
    return [t for t in parts if t]


def _pick(group, field):
    """Choose the value for `field`: prefer manual source, then most recent
    `updated`, then the earliest input row. Returns "" if no row has a value."""
    cands = [r for r in group if r[field]]
    if not cands:
        return ""
    cands.sort(key=lambda r: r["index"])                    # earliest row first
    cands.sort(key=lambda r: r["updated"], reverse=True)    # most recent first (stable)
    cands.sort(key=lambda r: r["source"] != "manual")       # manual first (stable)
    return cands[0][field]


def _merge_group(group):
    tags = sorted({t for r in group for t in r["tags"]})
    return {
        "first_name": _pick(group, "first_name"),
        "last_name": _pick(group, "last_name"),
        "zip": _pick(group, "zip"),
        "email": _pick(group, "email"),
        "phone": _pick(group, "phone"),
        "source": _pick(group, "source"),
        "tags": ";".join(tags),
        "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
        "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    # 1. Normalize each row; drop rows with neither a valid email nor a valid phone.
    records = []
    for index, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        records.append({
            "index": index,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in ("y", "yes", "true", "1"),
            "updated": _parse_updated(row["updated"]),
        })

    # 2. Union-find over matching keys (transitive matching).
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen = {}

    def link(key, pos):
        if key in seen:
            union(seen[key], pos)
        else:
            seen[key] = pos

    for pos, rec in enumerate(records):
        if rec["email"]:
            link(("email", rec["email"]), pos)
        if rec["phone"] and rec["last_name"]:
            link(("phone", rec["phone"], rec["last_name"].casefold()), pos)

    # 3. Group rows. Dict insertion order follows each group's earliest row.
    groups = {}
    for pos in range(len(records)):
        groups.setdefault(find(pos), []).append(records[pos])

    # 4. Merge each group into one output dict.
    return [_merge_group(group) for group in groups.values()]
