import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_FOUR_DIGITS_RE = re.compile(r"[0-9]{4}")
_FIVE_DIGITS_RE = re.compile(r"[0-9]{5}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_EPOCH = datetime.min


def _text(value):
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = _text(raw)
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = _text(raw).lower()
    cuts = [s.find(marker) for marker in ("x", "ext", "#")]
    cuts = [i for i in cuts if i != -1]
    if cuts:
        s = s[: min(cuts)]
    digits = "".join(c for c in s if "0" <= c <= "9")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(phone):
    return len(phone) == 10 and phone[0] not in "01"


def _normalize_zip(raw):
    s = _text(raw).strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if _FOUR_DIGITS_RE.fullmatch(s):
        s = "0" + s
    return s if _FIVE_DIGITS_RE.fullmatch(s) else ""


def _normalize_name(raw):
    s = _text(raw).strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = _text(raw).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _is_opted_out(raw):
    return _text(raw).strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    tags = set()
    for part in _TAG_SPLIT_RE.split(_text(raw)):
        tag = part.strip().lower()
        if tag:
            tags.add(tag)
    return tags


def _age_seconds(record):
    dt = record["updated"]
    return 0.0 if dt is None else (dt - _EPOCH).total_seconds()


def _rank_key(record):
    # Manual rows first, then most recent updated, then earlier input row.
    return (not record["manual"], -_age_seconds(record), record["idx"])


def _first_nonempty(ranked, field):
    for record in ranked:
        if record[field]:
            return record[field]
    return ""


def dedupe_contacts(rows):
    # Normalize each row; drop rows with neither a valid email nor a valid phone.
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        phone = _normalize_phone(row.get("phone"))
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        source = _text(row.get("source")).strip().lower()
        records.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Union-find so that matching is transitive.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    email_seen = {}
    phone_seen = {}
    for pos, rec in enumerate(records):
        if rec["email"]:
            key = rec["email"]
            if key in email_seen:
                union(pos, email_seen[key])
            else:
                email_seen[key] = pos
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].lower())
            if key in phone_seen:
                union(pos, phone_seen[key])
            else:
                phone_seen[key] = pos

    # Group rows; dict order follows each person's earliest input row.
    groups = {}
    for pos in range(len(records)):
        groups.setdefault(find(pos), []).append(records[pos])

    people = []
    for members in groups.values():
        ranked = sorted(members, key=_rank_key)

        dates = [r["updated"] for r in members if r["updated"] is not None]
        latest = max(dates) if dates else None

        tags = set().union(*(r["tags"] for r in members))

        people.append({
            "first_name": _first_nonempty(ranked, "first_name"),
            "last_name": _first_nonempty(ranked, "last_name"),
            "zip": _first_nonempty(ranked, "zip"),
            "email": _first_nonempty(ranked, "email"),
            "phone": _first_nonempty(ranked, "phone"),
            "source": _first_nonempty(ranked, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": latest.strftime("%Y-%m-%d") if latest else "",
        })

    return people

Notes on interpretation and use:

- **ZIP+4:** I treat "looks like ZIP+4" as a 4- or 5-digit prefix, a hyphen, and 4 digits, so Excel-mangled values like `2139-4307` are handled.
- **Unparseable dates:** these sort as oldest. If no row in a group has a parseable date, `updated` is "".
- **Source:** it uses the same ranking as the other fields, so the output source comes from the highest-ranked row that has a non-empty source.
- **Opt-outs:** the function keeps opted-out people in the output with `opted_out` = "Y". Filter out every row where `opted_out == "Y"` before the send. Because any opted-out row marks the whole group, a person who opted out under any record is excluded.
