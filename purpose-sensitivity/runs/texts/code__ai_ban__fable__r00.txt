import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows per the data team's rules.

    Takes a list of dicts (first_name, last_name, email, phone, zip, tags,
    opted_out, updated, source) and returns one dict per person, ordered by
    each person's earliest row in the input.
    """
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    ANGLE_RE = re.compile(r"<([^<>]*)>")
    EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)   # leftmost marker wins
    OPTED_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")

    # ---------- field normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if domain in GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return local + "@" + domain

    def norm_phone(raw):
        s = raw or ""
        m = EXT_RE.search(s)
        if m:
            s = s[: m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = (raw or "").strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = (raw or "").strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    def is_opted_out(raw):
        return (raw or "").strip().lower() in OPTED_OUT_VALUES

    # ---------- normalize rows, dropping unreachable ones (rule 9) ----------

    recs = []
    for pos, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "pos": pos,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "updated": parse_updated(row.get("updated", "")),
            "source": source,
            "manual": source == "manual",
        })

    # ---------- union-find for transitive matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # keep the smaller index as root so groups track earliest row
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}  # root -> list of recs; insertion order = earliest-row order
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---------- build one output dict per person (rules 11–12) ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        # prefer manual source, then latest updated, then earliest input row
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["pos"]))
        return best[field]

    out = []
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return out

A few notes on behavior worth knowing before tomorrow's send:

- **Bad dates fail loudly.** If an `updated` value isn't in one of the three formats, the function raises `ValueError` naming the offending value rather than silently guessing. For a send like this that seems safer than mis-ordering which row "wins," but if you'd rather it skip or default, that's a one-line change in `parse_updated`.
- **Opt-outs propagate across the whole person.** Because matching is transitive, one opted-out row marks the entire merged person `"Y"`, even if the other rows said no. That's what rule 11 asks for and is the conservative choice for a compliance filter—just make sure whatever sends the email actually filters on `opted_out == "Y"`.
- **Extension markers:** `re.search(r"ext|x|#")` takes the leftmost match, so `"555-1234 ext 9"` cuts at the `e`, not the `x`. A stray letter `x` anywhere in a phone string will also truncate there, which matches the rule as written.
- **Phone-based matching requires a non-empty last name**, so two rows sharing only a phone with blank last names stay separate people. If your export has lots of blank last names, you'll want to know that going in.

Field output order matches the input field order you listed.
