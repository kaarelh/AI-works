```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one record per person.

    rows: list of dicts with string fields first_name, last_name, email,
    phone, zip, tags, opted_out, updated, source.
    Returns a list of dicts (one per person), ordered by each person's
    earliest row in the input.
    """

    # ---------- normalization helpers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # rule 1: display-name form
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        return s

    def email_valid(e):                           # rule 2
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            local != ""
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):                          # rule 3
        s = raw or ""
        m = re.search(r"ext|x|#", s, re.IGNORECASE)
        if m:
            s = s[: m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                                 # "" means invalid

    def norm_zip(raw):                            # rule 4
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):                           # rule 5
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw):                       # rule 6
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):                        # rule 7
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):                          # rule 8 (+ rule 11 cleanup)
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "")
                if t.strip()}

    # ---------- normalize rows, drop unreachable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "updated": parse_updated(row.get("updated", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "tags": split_tags(row.get("tags", "")),
        })

    # ---------- transitive matching via union-find (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the root at the smaller index (earliest row)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        last_key = r["last_name"].lower()
        if r["phone"] and last_key:
            keys.append(("phone_last", r["phone"], last_key))
        for k in keys:
            if k in first_seen:
                union(first_seen[k], i)
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output (rules 11 and 12) ----------

    def pick(members, field):
        candidates = [m for m in members if m[field]]
        if not candidates:
            return ""
        best = max(candidates,
                   key=lambda m: (m["manual"], m["updated"], -m["idx"]))
        return best[field]

    people = []
    for members in groups.values():
        all_tags = set()
        for m in members:
            all_tags |= m["tags"]
        people.append((
            min(m["idx"] for m in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "source": pick(members, "source"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
                "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
            },
        ))

    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]
```

Some notes for tomorrow's send:

- **Filter before sending.** The function keeps opted-out people in the output with `opted_out == "Y"`, as the spec requires. Your send list should be something like `[p for p in result if p["opted_out"] == "N" and p["email"]]`. Some people may be matched only by phone and have `email == ""`.
- **Gmail addresses are sent in normalized form.** For example, `Ann.Lee+news@googlemail.com` becomes `annlee@gmail.com`. Gmail delivers both to the same inbox, so this is fine.
- **One email per person.** If a group was merged through phone and last name and has two different valid emails, only the chosen one is in the output. That's what you want for "no one gets it twice." But if someone's opt-out was recorded under the other address, they're still flagged "Y", because opt-out applies to the whole group.
- **Bad dates raise an error.** An unrecognized `updated` value raises a `ValueError` naming the bad value, so you'll see the problem instead of getting a silently mis-sorted merge.
- **ZIP+4 with a dropped zero.** I treated `2139-4307` as ZIP+4, which becomes `2139` and then `02139`. The spec didn't say whether that case counts, so I read it as a likely Excel artifact.
- **Name casing.** `isupper()`/`islower()` ignore characters that have no case, so `O'BRIEN` becomes `O'Brien` and `ANN-MARIE` becomes `Ann-Marie`. Mixed-case names like `McDonald` are left alone.

Before the real send, I'd run it on a small sample with known duplicates and spot-check the output.
