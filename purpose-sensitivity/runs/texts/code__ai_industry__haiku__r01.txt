```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    """Step 1: extract, strip, lowercase, and canonicalize Gmail addresses."""
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if "@" not in s:
        return s
    local, _, domain = s.partition("@")
    if domain in _GMAIL_DOMAINS:
        local = local.replace(".", "")
        local = local.split("+", 1)[0]
        domain = "gmail.com"
    return f"{local}@{domain}"


def _valid_email(e):
    """Step 2: exactly one @, non-empty local part, dotted domain not starting/ending with a dot."""
    if e.count("@") != 1:
        return False
    local, _, domain = e.partition("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    """Step 3: cut at extension marker, keep digits, drop leading 1, validate. Returns '' if invalid."""
    s = raw.lower()
    markers = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if markers:
        s = s[:min(markers)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Step 4: strip ZIP+4, restore a dropped leading zero, validate 5 digits."""
    s = raw.strip()
    m = re.fullmatch(r"([0-9]+)-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _is_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_digits(s):
        return s
    return ""


def _norm_name(raw):
    """Step 5: strip, and title-case if entirely upper or lower case."""
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    """Step 6: parse one of the three supported formats. Raises ValueError if none match."""
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _norm_tags(raw):
    """Step 8: split on ';' or ',', strip, lowercase, drop empties."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _is_opted_out(raw):
    """Step 7."""
    return raw.strip().lower() in _OPT_OUT_VALUES


def _best(members, field):
    """
    Step 11 selection: among members with a non-empty normalized value for `field`,
    prefer source == 'manual', then the latest `updated`, then the earliest input row.
    """
    cands = [r for r in members if r[field]]
    if not cands:
        return ""
    # Stable sorts applied from least to most significant key.
    cands.sort(key=lambda r: r["idx"])
    cands.sort(key=lambda r: r["updated"], reverse=True)
    cands.sort(key=lambda r: r["source"] != "manual")
    return cands[0][field]


def dedupe_contacts(rows):
    # Normalize every row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        updated = _parse_updated(row.get("updated", ""))

        email = _norm_email(row.get("email", ""))
        email = email if _valid_email(email) else ""
        phone = _norm_phone(row.get("phone", ""))

        if not email and not phone:
            continue

        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _norm_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": updated,
        })

    # Step 10: union-find over matching rules (transitive).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    # Group rows; order groups by their earliest row in the input.
    order = []
    groups = {}
    for i in range(len(recs)):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[i])

    # Step 11: build one output record per person.
    output = []
    for root in order:
        members = groups[root]
        tags = sorted(set().union(*(r["tags"] for r in members)))
        output.append({
            "first_name": _best(members, "first_name"),
            "last_name": _best(members, "last_name"),
            "zip": _best(members, "zip"),
            "email": _best(members, "email"),
            "phone": _best(members, "phone"),
            "source": _best(members, "source"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        })

    return output
```

Notes for the send:

- **Filter before sending.** The function sets `opted_out` to "Y" but does not drop those people. Remove every record where `opted_out == "Y"` before the email goes out. Under rule 11, one opted-out row suppresses the whole person, including their other duplicates.
- **Send to the output address.** The `email` field is the normalized form. For Gmail, dots and `+tags` are removed, which Gmail still delivers to, but it may look different from what the customer has on file.
- **Bad dates raise an error.** If any `updated` value doesn't match the three formats, the function raises `ValueError` and nothing is returned. I chose this over silently guessing, so a bad export doesn't quietly change which row wins.
- **Rule 9 applies before matching.** Rows with no valid email or phone are dropped before duplicate matching, so they cannot link two other rows together.
