import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_UPDATED_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    s = (raw or "").strip()
    start = s.find("<")
    if start != -1:
        end = s.find(">", start + 1)
        if end != -1:
            s = s[start + 1:end]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:

def _is_valid_email(s):
    if s.count("@") != 1:

    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))

def _is_valid_phone(p):

def dedupe_contacts(rows):
    """Normalize, validate, and dedupe CRM contact rows.

    Returns one dict per person, ordered by each person's earliest input row.
    Keys: first_name, last_name, zip, email, phone, source, tags, opted_out
    ("Y"/"N"), updated ("YYYY-MM-DD").
    """
    # Step 1: normalize and validate; drop rows with no valid email or phone.
    people = []
    for row in rows:
        email = _norm_email(row.get("email"))
        if not _is_valid_email(email):
            email = ""
        phone = _norm_phone(row.get("phone"))
        if not _is_valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue
        people.append({
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": (row.get("source") or "").strip().lower(),
            "tags": _norm_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # Step 2: union-find so matching is transitive.
    parent = list(range(len(people)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for i, p in enumerate(people):
        if p["email"]:
            if p["email"] in first_by_email:
                union(first_by_email[p["email"]], i)
            else:
                first_by_email[p["email"]] = i
        last_key = p["last_name"].casefold()
        if p["phone"] and last_key:
            key = (p["phone"], last_key)
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    # Group members; iterating in input order gives groups ordered by earliest row.
    groups = {}
    order = []
    for i in range(len(people)):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    def pick(members, field):
        # Candidates: rows where the field is non-empty (email/phone are already valid).
        cands = [i for i in members if people[i][field]]
        if not cands:
            return ""
        # Stable sorts: first by updated desc, then manual first.
        # Remaining ties fall back to input order (members are ascending).
        cands.sort(key=lambda i: people[i]["updated"], reverse=True)
        cands.sort(key=lambda i: people[i]["source"] != "manual")
        return people[cands[0]][field]

        latest = max(people[i]["updated"] for i in members)

            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
