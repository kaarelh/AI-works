```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL = {"gmail.com", "googlemail.com"}
    DIGITS = "0123456789"
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        d = "".join(c for c in raw if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]{5})-([0-9]{4})", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_date(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable: treated as oldest

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---- normalize rows, drop those with no valid email/phone ----
    recs = []
    for i, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": i,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_date(get(row, "updated")),
        })

    # ---- union-find ----
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for k, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(k, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = k
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(k, phone_last_seen[key])
            else:
                phone_last_seen[key] = k

    groups = {}
    order = []
    for k in range(n):
        root = find(k)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(k)

    # ---- build output ----
    result = []
    for root in order:
        members = [recs[k] for k in groups[root]]  # already in input order
        # preference order: manual first, then most recent, then earlier in input
        ranked = sorted(members, key=lambda r: r["idx"])
        ranked.sort(key=lambda r: r["updated"], reverse=True)  # stable
        ranked.sort(key=lambda r: 0 if r["manual"] else 1)     # stable

        def pick(field):
            for r in ranked:
                if r[field]:
                    return r[field]
            return ""

        tags = set()
        for r in members:
            tags |= r["tags"]
        latest = max(r["updated"] for r in members)

        result.append({
            "first_name": pick("first_name"),
            "last_name": pick("last_name"),
            "zip": pick("zip"),
            "email": pick("email"),
            "phone": pick("phone"),
            "source": pick("source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })

    return result
```

Notes:

- **Opt-outs:** The function merges and flags opted-out people (`opted_out == "Y"`) but doesn't remove them. Before sending, filter with `[p for p in people if p["opted_out"] == "N" and p["email"]]`. Phone-only contacts will have an empty `email`, so they need to be excluded from the email send.
- **Unparseable dates:** The spec says `updated` is always one of the three formats. If one isn't, the function treats it as the oldest possible date rather than raising an error.
- **Gmail normalization:** Dots and `+tags` are stripped only when the address has exactly one `@`. An address like `+promo@gmail.com` becomes invalid because the local part ends up empty.
