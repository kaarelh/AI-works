import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DIGITS = "0123456789"

    def s(value):
        return "" if value is None else str(value)

    def norm_email(raw):
        raw = s(raw)
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        raw = raw.strip().lower()
        if raw.count("@") == 1:
            local, domain = raw.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+")[0].replace(".", "")
                domain = "gmail.com"
            raw = local + "@" + domain
        # validity
        if raw.count("@") != 1:
            return ""
        local, domain = raw.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return raw

    def norm_phone(raw):
        raw = s(raw)
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        digits = "".join(ch for ch in raw if ch in DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        m = re.fullmatch(r"([0-9]+)-([0-9]{4})", z)
        if m:
            z = m.group(1)
        if len(z) == 4 and all(ch in DIGITS for ch in z):
            z = "0" + z
        if len(z) == 5 and all(ch in DIGITS for ch in z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        raw = s(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(raw, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable: treated as oldest

    def parse_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # Normalize rows; drop those with neither valid email nor valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": s(row.get("source")).strip().lower(),
            "tags": parse_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated")),
        })

    # Union-find over the kept records.
    parent = list(range(len(recs)))

    def find(x):
        root = x
        while parent[root] != root:
            root = parent[root]
        while parent[x] != root:
            parent[x], x = root, parent[x]
        return root

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            key = r["email"]
            if key in seen_email:
                union(i, seen_email[key])
            else:
                seen_email[key] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(
            cands,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[field]

    result = []
    for group in sorted(groups.values(), key=lambda g: min(r["idx"] for r in g)):
        tags = set()
        for r in group:
            tags.update(r["tags"])
        latest = max(r["updated"] for r in group)
        result.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": latest.date().isoformat(),
        })
    return result

Things to know before tomorrow's send:

- **The function doesn't remove opted-out people.** It returns them with `opted_out == "Y"`, so filter before sending: `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`. The `email` check matters because phone-only people have no email to send to.
- **Opt-outs only propagate within a matched group.** If someone opted out under one address and appears elsewhere with a different email, a different last name, or a phone that doesn't match, the rules won't link them. Those cases are worth a manual spot check.
- **Unparseable `updated` values** are treated as the oldest possible date rather than raising an error, so a bad value can't crash the run. If every row in a group has one, the output date will be `0001-01-01`.
- **ZIP+4 detection** accepts any digits, a hyphen, then 4 digits. So `2139-4307` becomes `02139`, and `123-4567` becomes `""`.
- **Gmail normalization** only applies when the address has exactly one `@`. Anything else fails validation anyway.

I haven't run this against your data, so I'd test it on a small sample with known duplicates, such as `Ann.Lee+cas@Gmail.com` vs `annlee@gmail.com`, and a phone like `(202) 555-0100 x12`, before the real export.
