```python
import re
from datetime import datetime

_EPOCH = datetime(1970, 1, 1)
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, flags=re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return None

    def split_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize rows, drop those with no valid email/phone ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        upd = parse_updated(get(row, "updated"))
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": get(row, "source").strip().lower(),
            "tags": split_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "updated": upd,
        })

    # ---------- union-find matching ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            k = r["email"]
            if k in email_seen:
                union(i, email_seen[k])
            else:
                email_seen[k] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            k = (r["phone"], last)
            if k in phone_last_seen:
                union(i, phone_last_seen[k])
            else:
                phone_last_seen[k] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def sort_key(r):
        ts = (r["updated"] - _EPOCH).total_seconds() if r["updated"] else float("-inf")
        return (0 if r["source"] == "manual" else 1, -ts, r["idx"])

    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        return min(cands, key=sort_key)[field]

    result = []
    # group roots are the smallest member index, and recs are in input order,
    # so sorting by root gives order of earliest row in the input.
    for root in sorted(groups):
        members = groups[root]
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        dates = [recs[i]["updated"] for i in members if recs[i]["updated"]]
        latest = max(dates) if dates else None
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": (f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}"
                        if latest else ""),
        })
    return result
```

Notes:

- **Opt-outs:** The function doesn't remove opted-out people, it flags them with `opted_out == "Y"`. Before sending, filter to `opted_out == "N"`, and also filter out anyone with an empty `email`, since some people will be phone-only.
- **Opt-out scope:** If any row in a matched group opted out, the whole person is marked "Y". This is the cautious reading of rule 11.
- **Dropped rows:** Per rule 9, rows with no valid email or phone are dropped before matching, so an opt-out on such a row is ignored. Those rows can't be linked to anyone anyway.
- **Gmail edge case:** An address like `+news@gmail.com` normalizes to an empty local part and is treated as invalid.
- **Bad dates:** The spec says `updated` is always one of the three formats. If a value doesn't parse, it's treated as oldest, and the output `updated` is `""` if nothing in the group parses.
- **ZIP+4:** I accept `digits-4digits` before the hyphen, so a ZIP+4 whose first part lost its leading zero (e.g. `2139-4307`) still becomes `02139`.
