```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, match, and merge CRM contact rows per the data team's rules."""

    OPTED_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- field normalizers ----------

    def norm_email(raw):
        """Return the normalized email if valid, else ''."""
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # 'Ann Lee <ann@x.org>' -> 'ann@x.org'
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return local + "@" + domain

    def norm_phone(raw):
        """Return the 10-digit normalized phone if valid, else ''."""
        s = raw or ""
        m = re.search(r"ext|x|#", s, flags=re.IGNORECASE)   # extension marker
        if m:
            s = s[: m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = (raw or "").strip()
        if "-" in z:                              # ZIP+4 -> keep the 5-digit part
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):          # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = (raw or "").strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable/missing: treat as oldest possible

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    def norm_opted_out(raw):
        return (raw or "").strip().lower() in OPTED_OUT_VALUES

    # ---------- normalize rows, dropping unreachable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip")),
            "tags": norm_tags(row.get("tags")),
            "opted_out": norm_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
            "source": (row.get("source") or "").strip().lower(),
        })

    # ---------- group rows into people with union-find (rule 10, transitive) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    key_owner = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(i, key_owner[k])
            else:
                key_owner[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])
    # recs are in input order, so each group's first member is its earliest row (rule 12)
    ordered_groups = sorted(groups.values(), key=lambda g: g[0]["idx"])

    # ---------- merge each group into one output dict (rule 11) ----------

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[field]

    output = []
    for g in ordered_groups:
        all_tags = set().union(*(r["tags"] for r in g))
        output.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "zip": pick(g, "zip"),
            "source": pick(g, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["updated"] for r in g).strftime("%Y-%m-%d"),
        })

    return output
```

A few notes on choices I made where the rules left room:

- **Invalid emails/phones become `""`** at normalization time, so they're never used for matching and never selected for output. A row whose only contact info is invalid is dropped (rule 9).
- **Extension markers**: the regex finds the *earliest* occurrence of `ext`, `x`, or `#` (case-insensitive) and cuts there, so `555-123-4567 Ext. 12` → `5551234567`.
- **Gmail local part that empties out** (e.g. `...@gmail.com`) is treated as invalid.
- **Unparseable `updated`** values don't crash the run; they're treated as the oldest possible date (`datetime.min`), so a row with a bad date will only win a field if no better-dated row has it. If you'd rather it fail loudly so you can fix the export, swap the `return datetime.min` for `raise ValueError(...)`.
- **Picking fields**: `max` over the key `(is_manual, updated, -index)` implements "manual first, then most recent, then earliest in input."
- Since the output `opted_out` is `"Y"` if *any* merged row opted out, make sure your send list filters on that column before sending — that's what guarantees nobody who opted out on any duplicate record gets the alert.
