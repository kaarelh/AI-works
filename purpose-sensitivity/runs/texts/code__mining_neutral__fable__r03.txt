import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- normalizers ----------
    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        # validity check -> return "" if invalid
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = (raw or "").lower()
        cut = len(s)
        for marker in ("ext", "x", "#"):          # ignore everything from the earliest marker on
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                              # ZIP+4 -> keep the part before the hyphen
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):          # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    # ---------- normalize rows, drop unusable ones (rule 9) ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        tags = {t.strip().lower() for t in re.split(r"[;,]", row.get("tags") or "")}
        tags.discard("")
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "zip": norm_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": (row.get("opted_out") or "").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", "")),
        })

    # ---------- transitive matching via union-find (rule 10) ----------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    key_owner = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(key_owner[k], i)
            else:
                key_owner[k] = i

    groups = {}                                   # insertion order == earliest row order (rule 12)
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---------- build one output dict per person (rule 11) ----------
    def pick(members, field):
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        # prefer manual source, then most recent updated, then earliest in input
        best = max(cands, key=lambda m: (m["manual"], m["updated"], -m["idx"]))
        return best[field]

    out = []
    for members in groups.values():
        all_tags = set()
        for m in members:
            all_tags |= m["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return out

Two things worth knowing before you run it on the real export:

- **Bad dates fail loudly.** If an `updated` value doesn't match one of the three formats, the function raises `ValueError` naming the offending value rather than guessing. Given this gates a send to subscribers, I'd rather it stop than silently mis-order a merge; if you'd prefer a fallback, say so and I'll add one.
- **Phones are output as 10 bare digits** (e.g. `2025551234`), since that's the normalized form the rules define. If the email tool needs a formatted number, that's a one-line change.
