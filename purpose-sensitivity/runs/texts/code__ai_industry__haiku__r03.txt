```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _field(row, key):
    return row.get(key) or ""


def _normalize_email(raw):
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Return the digit string (possibly invalid) after extension and 1-prefix handling."""
    s = raw
    m = re.search(r"ext|x|#", s, re.IGNORECASE)
    if m:
        s = s[:m.start()]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _normalize_tags(raw):
    parts = re.split(r"[;,]", raw)
    return {p.strip().lower() for p in parts if p.strip()}


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(members, field):
    """Choose the value for `field` from the group, using the preference order in the rules."""
    cands = [r for r in members if r[field]]
    if not cands:
        return ""
    cands.sort(key=lambda r: r["idx"])                    # earliest input row
    cands.sort(key=lambda r: r["ts"], reverse=True)       # most recent updated
    cands.sort(key=lambda r: r["source"] != "manual")     # manual rows first
    return cands[0][field]


def dedupe_contacts(rows):
    # Normalize each row and drop rows with no valid email or phone.
    recs = []
    for row in rows:
        email = _normalize_email(_field(row, "email"))
        if not _is_valid_email(email):
            email = ""
        digits = _normalize_phone(_field(row, "phone"))
        phone = digits if _is_valid_phone(digits) else ""
        if not email and not phone:
            continue

        last_name = _normalize_name(_field(row, "last_name"))
        recs.append({
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": last_name,
            "last_key": last_name.casefold(),
            "zip": _normalize_zip(_field(row, "zip")),
            "email": email,
            "phone": phone,
            "source": _field(row, "source").strip().lower(),
            "tags": _normalize_tags(_field(row, "tags")),
            "opted_out": _field(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "ts": _parse_updated(row.get("updated")),
            "idx": len(recs),  # position among kept rows; preserves input order
        })

    # Union-find: rows sharing a valid email, or a valid phone plus the same
    # last name, belong to the same person. Unions are transitive.
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_with_key = {}
    for p, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_key"]:
            keys.append(("phone", r["phone"], r["last_key"]))
        for key in keys:
            if key in first_with_key:
                union(first_with_key[key], p)
            else:
                first_with_key[key] = p

    # Groups are inserted in order of their earliest row.
    groups = {}
    for p in range(n):
        groups.setdefault(find(p), []).append(recs[p])

    out = []
    for members in groups.values():
        tags = set().union(*(r["tags"] for r in members))
        out.append({
            "first_name": _pick(members, "first_name"),
            "last_name": _pick(members, "last_name"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["ts"] for r in members).strftime("%Y-%m-%d"),
        })
    return out
```

Notes:

- **Opt-outs are flagged, not removed.** Before sending, filter the output to `opted_out == "N"`. Because a person is one group, anyone who opted out on any row is excluded entirely, which is what you want.
- **ZIP+4 with a dropped leading zero:** I treat `2139-4307` as ZIP+4 too (4 or 5 digits before the hyphen, then 4 digits), so it becomes `02139`. Tell me if you want that stricter.
- **Bad dates raise `ValueError`** instead of being guessed. The rules say `updated` is always one of the three formats, so a failure means the export has a problem you should look at.
