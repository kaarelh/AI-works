import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_TAG_SPLIT_RE = re.compile(r"[;,]")
_ZIP4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_ZIP_4DIGIT_RE = re.compile(r"[0-9]{4}")
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key)
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                s = local + "@gmail.com"
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = raw
        m = _EXT_RE.search(s)
        if m:
            s = s[:m.start()]
        digits = _NON_DIGIT_RE.sub("", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) != 10 or digits[0] in "01":
            return ""
        return digits

    def norm_zip(raw):
        s = raw.strip()
        if _ZIP4_RE.fullmatch(s):
            s = s.split("-", 1)[0]
        if _ZIP_4DIGIT_RE.fullmatch(s):
            s = "0" + s
        return s if _ZIP5_RE.fullmatch(s) else ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_dt(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized 'updated' value: %r" % raw)

    def dt_value(dt):
        return dt.toordinal() * 86400 + dt.hour * 3600 + dt.minute * 60 + dt.second

    # ---- normalize rows, dropping those with no valid email or phone ----
    recs = []
    for row in rows:
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        tags = set()
        for t in _TAG_SPLIT_RE.split(get(row, "tags")):
            t = t.strip().lower()
            if t:
                tags.add(t)
        dt = parse_dt(get(row, "updated"))
        recs.append({
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": tags,
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "dt": dt,
            "dtv": dt_value(dt),
        })

    n = len(recs)

    # ---- union-find ----
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    # ---- group, ordered by earliest row ----
    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # ---- build output ----
    def pick(indices, field):
        best = None
        best_key = None
        for i in indices:
            r = recs[i]
            if not r[field]:

            key = (0 if r["source"] == "manual" else 1, -r["dtv"], i)
            if best_key is None or key < best_key:
                best_key = key
                best = r[field]
        return best if best is not None else ""

    for root in order:

        tags = set()
        for i in idxs:

Notes on choices and edge cases:

- **Opt-outs are group-wide.** If any duplicate row has opted out, the merged person gets `opted_out = "Y"`. When you build the send list, filter on `opted_out == "N"`, and also drop rows with an empty `email`, since a person matched by phone only may have no valid email.
- **Dropped rows.** Rows with no valid email and no valid phone are removed before matching, per rule 9. An opt-out recorded only on such a row therefore can't be applied to anyone.
- **Gmail normalization happens before validation.** An address like `+news@gmail.com` becomes `@gmail.com`, which is invalid.
- **Unparseable `updated` values raise a `ValueError`.** I chose that over guessing, since the spec says the field is always present and in one of three formats.
- **Last-name matching.** The comparison uses `.lower()` on the already-normalized last name.
- **Phone extensions.** The `ext|x|#` search takes the leftmost marker, so "ext" is handled correctly rather than being cut at its "x".
