```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _get(row, key):
    return str(row.get(key) or "")


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    """Rule 1. Returns the normalized address (may still be invalid)."""
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(e):
    """Rule 2."""
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Rule 3. Returns the 10-digit phone, or "" if invalid."""
    s = raw.lower()
    cut = len(s)
    for marker in ("x", "ext", "#"):
        i = s.find(marker)
        if i != -1:
            cut = min(cut, i)
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Rule 4. Returns a 5-digit ZIP, or "" if invalid."""
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _all_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_digits(s):
        return s
    return ""


def _norm_name(raw):
    """Rule 5."""
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6. Returns a datetime, or None if unparseable."""
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _tags(raw):
    """Rule 8. Returns a set of stripped, lowercased, non-empty tags."""
    out = set()
    for t in re.split(r"[;,]", raw):
        t = t.strip().lower()
        if t:
            out.add(t)
    return out


def _ts_key(rec):
    return rec["ts"] if rec["ts"] is not None else datetime.min


def dedupe_contacts(rows):
    # Normalize each row and drop rows with neither a valid email nor a valid phone (rules 1-9).
    recs = []
    for row in rows:
        email = _norm_email(_get(row, "email"))
        email = email if _valid_email(email) else ""
        phone = _norm_phone(_get(row, "phone"))
        if not email and not phone:
            continue
        source = _get(row, "source").strip().lower()
        recs.append({
            "email": email,
            "phone": phone,
            "first": _norm_name(_get(row, "first_name")),
            "last": _norm_name(_get(row, "last_name")),
            "zip": _norm_zip(_get(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "ts": _parse_updated(_get(row, "updated")),
            "tags": _tags(_get(row, "tags")),
            "opted": _get(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
        })

    # Rule 10: group rows with union-find so matching is transitive.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last"]:
            keys.append(("phone", r["phone"], r["last"].lower()))
        for key in keys:
            if key in first_seen:
                a, b = find(i), find(first_seen[key])
                if a != b:
                    parent[b] = a
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)
    # Rule 12: order people by their earliest input row.
    ordered = sorted(groups.values(), key=lambda m: m[0])

    def pick(members, field):
        # Rule 11: non-empty candidates, manual first, then most recent updated,
        # then earliest input row. Stable sorts apply the keys in reverse priority.
        cands = [recs[i] for i in members if recs[i][field]]
        cands.sort(key=_ts_key, reverse=True)
        cands.sort(key=lambda r: not r["manual"])
        return cands[0][field] if cands else ""

    out = []
    for members in ordered:
        group = [recs[i] for i in members]
        tags = set()
        for r in group:
            tags |= r["tags"]
        dated = [r["ts"] for r in group if r["ts"] is not None]
        out.append({
            "first_name": pick(members, "first"),
            "last_name": pick(members, "last"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted"] for r in group) else "N",
            "updated": max(dated).strftime("%Y-%m-%d") if dated else "",
        })
    return out
```

Notes for the send:

- **Filter opt-outs before sending.** The function keeps opted-out people and marks them `opted_out == "Y"`, so the send step has to exclude them. Because of the transitive merge, if any duplicate of a person opted out, the merged record is "Y" and they should be excluded. That's the conservative result.
- **ZIP+4 with a dropped leading zero.** I read "looks like ZIP+4" as 4–5 digits, a hyphen, and 4 digits, so `2139-4307` becomes `02139`. If the data person only means strict 5-digit ZIP+4, change the regex in `_norm_zip`.
- **Shared household phones.** Two different people with the same phone and last name will merge into one record, which is what rule 10 specifies. Expect a few of these in household data.
