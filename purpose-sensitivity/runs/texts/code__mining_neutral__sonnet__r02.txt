import re
from datetime import datetime


def dedupe_contacts(rows):
    # ---------- helpers ----------
    def s(v):
        return "" if v is None else str(v)

    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^<>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        return e

    def valid_email(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        p = s(raw)
        p = re.split(r"(?i)ext|x|#", p, maxsplit=1)[0]
        digits = re.sub(r"[^0-9]", "", p)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = s(raw).strip()
        if n and (n.isupper() or n.islower()):
            n = n.title()
        return n

    def parse_date(raw):
        d = s(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(d, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated value: %r" % raw)

    def parse_tags(raw):
        return [t.strip().lower() for t in re.split(r"[;,]", s(raw)) if t.strip()]

    # ---------- normalize and filter ----------
    recs = []
    for idx, r in enumerate(rows):
        email = norm_email(r.get("email"))
        email = email if valid_email(email) else ""
        phone = norm_phone(r.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(r.get("first_name")),
            "last_name": norm_name(r.get("last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(r.get("zip")),
            "source": s(r.get("source")).strip().lower(),
            "tags": parse_tags(r.get("tags")),
            "opted_out": s(r.get("opted_out")).strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_date(r.get("updated")),
        })

    # ---------- union-find ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen_email = {}
    seen_pl = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in seen_email:
                union(i, seen_email[rec["email"]])
            else:
                seen_email[rec["email"]] = i
        last = rec["last_name"].lower()
        if rec["phone"] and last:
            key = (rec["phone"], last)
            if key in seen_pl:
                union(i, seen_pl[key])
            else:
                seen_pl[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        cands.sort(key=lambda r: r["idx"])
        cands.sort(key=lambda r: r["updated"], reverse=True)  # stable
        cands.sort(key=lambda r: 0 if r["source"] == "manual" else 1)  # stable
        return cands[0][field]

    ordered = sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m))

    out = []
    for members in ordered:
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["updated"] for i in members)
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return out

Some notes on how I read the rules:

- **Opt-outs:** if any row in a merged person is opted out, the whole person is `"Y"`. Filter on `opted_out == "N"` before sending tomorrow's email.
- **Email brackets:** if a field has a `<...>` pair, I use only what's inside. A stray `<` with no closing `>` is left alone and will most likely fail validation.
- **Phone extensions:** I split on the first `ext`, `x`, or `#`, case-insensitive, and digits are ASCII `0-9` only.
- **ZIP+4:** only a 5-digit, hyphen, 4-digit form is treated as ZIP+4. A value like `2139-4307` becomes `""`.
- **Unparseable `updated`:** the function raises `ValueError` rather than guessing. If your export may contain bad dates, you may prefer to catch that per row.
- **`source`:** it is picked using the same manual-first, then newest, then earliest ordering as the other fields. Rows with an empty source are skipped for that field.
- **Dropped rows:** they never participate in matching, so they can't bridge two groups.
