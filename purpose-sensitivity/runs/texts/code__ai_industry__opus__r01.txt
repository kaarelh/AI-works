```python
import re
from datetime import datetime

_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the data team's rules."""

    # ---------- normalizers ----------
    def norm_email(raw):
        s = raw or ""
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_is_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        s = raw or ""
        m = _EXT_RE.search(s)
        if m:
            s = s[: m.start()]
        digits = "".join(c for c in s if c in "0123456789")
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""  # invalid

    def norm_zip(raw):
        s = (raw or "").strip()
        m = _ZIP_PLUS4_RE.fullmatch(s)
        if m:
            s = m.group(1)
        if len(s) == 4 and s.isascii() and s.isdigit():
            s = "0" + s
        return s if _ZIP5_RE.fullmatch(s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s and (s.isupper() or s.islower()):
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def parse_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize rows, drop unreachable ones (rule 9) ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_is_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "tags": parse_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", "")),
            "manual": source == "manual",
        })

    # ---------- union-find matching (rule 10) ----------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the earlier row as root (not required, just tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output (rules 11, 12) ----------
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    people = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        people.append((
            min(r["idx"] for r in group),
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "zip": pick(group, "zip"),
                "source": pick(group, "source"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            },
        ))

    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]
```

Some notes on how I handled cases the rules leave open:

- **Phone extensions:** The function cuts at the leftmost of "ext", "x", or "#". So `555-123-4567 ext. 12` and `555-123-4567x12` both become `5551234567`.
- **Invalid emails and phones** are treated as empty. They are never used for matching or output.
- **ZIP+4 with a missing leading zero:** `2139-4307` is also accepted. The function keeps `2139` and pads it to `02139`.
- **Last-name matching** compares the normalized names with `casefold()`.
- **Bad `updated` values:** An `updated` value that isn't in one of the three formats raises a `ValueError` naming the bad value. This stops the job instead of guessing at a date.
- **Opt-outs:** The output still contains people who opted out, marked `opted_out="Y"`. Before you send, filter on `opted_out == "N"`. Because opt-out status is merged across the whole group, anyone who opted out under any of their addresses or phone numbers will be excluded.

I'd suggest running it on a small sample with known duplicates and comparing the counts before the full send.
