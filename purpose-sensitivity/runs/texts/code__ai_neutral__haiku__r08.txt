import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _normalize_email(raw):
    s = raw.strip()
    # Use only the part inside angle brackets, if present.
    if "<" in s and ">" in s and s.index("<") < s.rindex(">"):
        s = s[s.index("<") + 1 : s.rindex(">")]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw.lower()
    cuts = [i for i in (s.find(m) for m in ("x", "ext", "#")) if i != -1]
    if cuts:
        s = s[: min(cuts)]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    # ZIP+4, including Excel-mangled forms like "2139-4307": keep the part before the hyphen.
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    # Excel dropped the leading zero.
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None  # Not expected per the spec; treated as oldest.


def _is_opted_out(raw):
    return raw.strip().lower() in {"y", "yes", "true", "1"}


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _best(cands):
    """Preference: manual source, then most recent updated, then earliest input row."""
    ordered = sorted(cands, key=lambda r: r["idx"])  # 1st: earliest row
    ordered.sort(key=lambda r: r["updated"] or datetime.min, reverse=True)  # 2nd: newest
    ordered.sort(key=lambda r: r["source"] != "manual")  # 3rd: manual first
    return ordered[0]


def dedupe_contacts(rows):
    # Normalize every row and drop rows with no valid email or phone.
    recs = []
    for row in rows:
        email = _normalize_email(row.get("email", ""))
        email_ok = _is_valid_email(email)
        phone = _normalize_phone(row.get("phone", ""))
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        recs.append({
            "idx": len(recs),
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": row.get("source", "").strip().lower(),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Union-find over rows: same valid email, or same valid phone + same last name.
    n = len(recs)
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], i)
            else:
                first_by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)
    # Members are appended in increasing index order, so g[0] is the earliest row.
    ordered_groups = sorted(groups.values(), key=lambda g: g[0])

    # Build one output record per person.
    output = []
    for members in ordered_groups:
        group = [recs[i] for i in members]
        person = {}
        for field in ("first_name", "last_name", "zip", "email", "phone", "source"):
            cands = [r for r in group if r[field]]
            person[field] = _best(cands)[field] if cands else ""

        tags = set()
        for r in group:
            tags |= r["tags"]
        person["tags"] = ";".join(sorted(tags))

        person["opted_out"] = "Y" if any(r["opted_out"] for r in group) else "N"

        dates = [r["updated"] for r in group if r["updated"]]
        person["updated"] = max(dates).strftime("%Y-%m-%d") if dates else ""

        output.append(person)

    return output
