```python
import re
from datetime import datetime

_EPOCH = datetime(1970, 1, 1)
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_PHONE_MARKER_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _norm_email(raw):
    s = raw
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = f"{local}@gmail.com"
    return s


def _email_is_valid(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw
    m = _PHONE_MARKER_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    z = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    if re.fullmatch(r"[0-9]{5}", z):
        return z
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _norm_tags(raw):
    tags = set()
    for t in _TAG_SPLIT_RE.split(raw):
        t = t.strip().lower()
        if t:
            tags.add(t)
    return tags


def _rank(r):
    # Lower sorts first: manual sources, then most recent updated, then earliest input row.
    return (not r["is_manual"], -(r["updated"] - _EPOCH).total_seconds(), r["idx"])


def _pick(members, field):
    # Normalized values are "" when empty or invalid, so truthiness means "qualifies".
    cands = [r for r in members if r[field]]
    if not cands:
        return ""
    return min(cands, key=_rank)[field]


def dedupe_contacts(rows):
    # Normalize rows and drop those without a valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        email = _norm_email(row["email"])
        email = email if _email_is_valid(email) else ""
        phone = _norm_phone(row["phone"])
        phone = phone if _phone_is_valid(phone) else ""
        if not email and not phone:
            continue
        source = row["source"].strip().lower()
        records.append({
            "idx": idx,
            "first_name": _norm_name(row["first_name"]),
            "last_name": _norm_name(row["last_name"]),
            "zip": _norm_zip(row["zip"]),
            "email": email,
            "phone": phone,
            "source": source,
            "is_manual": source == "manual",
            "tags": _norm_tags(row["tags"]),
            "opted_out": row["opted_out"].strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over record indices.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # Same valid email -> same person.
    email_seen = {}
    # Same valid phone and same non-empty last name (case-insensitive) -> same person.
    phone_seen = {}
    for i, r in enumerate(records):
        if r["email"]:
            if r["email"] in email_seen:
                union(email_seen[r["email"]], i)
            else:
                email_seen[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in phone_seen:
                union(phone_seen[key], i)
            else:
                phone_seen[key] = i

    # Group record indices by person; members stay in input order.
    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(i)
    ordered = sorted(groups.values(), key=lambda g: g[0])

    result = []
    for member_ids in ordered:
        members = [records[i] for i in member_ids]
        tags = set().union(*(r["tags"] for r in members))
        latest = max(r["updated"] for r in members)
        result.append({
            "first_name": _pick(members, "first_name"),
            "last_name": _pick(members, "last_name"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Opt-outs are flagged, not removed.** Per rule 11, the function returns opted-out people with `opted_out="Y"`. Filter them out before sending: `recipients = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N"]`. If a person has any opted-out row, the merged record is marked "Y", so a duplicate entry can't re-enable their email.
- **ZIP+4 with a dropped leading zero** (e.g., `2139-4307`) is treated as ZIP+4 and becomes `02139`. The rules didn't explicitly cover this case, so confirm with your data person.
- **Unparseable `updated` values raise `ValueError`** rather than being silently sorted. Only rows that survive the email/phone check are parsed, so bad dates in dropped rows won't cause errors.
