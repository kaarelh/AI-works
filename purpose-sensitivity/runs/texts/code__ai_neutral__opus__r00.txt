```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the CPR dedupe rules.

    Returns a list of dicts, one per person, with keys:
    first_name, last_name, email, phone, zip, tags, opted_out, updated, source.
    """

    def field(row, name):
        value = row.get(name)
        return value if isinstance(value, str) else ""

    # Rule 1: email normalization
    def norm_email(raw):
        s = raw
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        parts = s.split("@")
        if len(parts) == 2 and parts[1] in _GMAIL_DOMAINS:
            local = parts[0].split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
        return s

    # Rule 2: email validity
    def email_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return (
            local != ""
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    # Rule 3: phone normalization + validity ("" if invalid)
    def norm_phone(raw):
        s = raw
        m = re.search(r"ext|x|#", s, flags=re.IGNORECASE)
        if m:
            s = s[: m.start()]
        digits = "".join(ch for ch in s if ch in "0123456789")
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    # Rule 4: ZIP normalization
    def norm_zip(raw):
        s = raw.strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    # Rule 5: name normalization
    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    # Rule 6: parse `updated`
    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    # Rule 7: opt-out
    def is_opted_out(raw):
        return raw.strip().lower() in _OPT_OUT_VALUES

    # Rule 8: tags
    def split_tags(raw):
        return [t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()]

    # Normalize every row; Rule 9: drop rows with neither valid email nor phone
    records = []
    for idx, row in enumerate(rows):
        email = norm_email(field(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(field(row, "phone"))
        if not email and not phone:
            continue
        source = field(row, "source").strip().lower()
        records.append({
            "idx": idx,
            "first_name": norm_name(field(row, "first_name")),
            "last_name": norm_name(field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(field(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "tags": split_tags(field(row, "tags")),
            "opted_out": is_opted_out(field(row, "opted_out")),
            "updated": parse_updated(field(row, "updated")),
        })

    # Rule 10: union-find over shared email / (phone + last name)
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i, rec in enumerate(records):
        groups.setdefault(find(i), []).append(rec)

    # Rule 11: build one output dict per person
    def pick(group, name):
        candidates = [r for r in group if r[name]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda r: (
                0 if r["is_manual"] else 1,
                -(r["updated"] - datetime.min),  # most recent first
                r["idx"],                        # earlier in input first
            ),
        )
        return best[name]

    output = []
    # Rule 12: order by each person's earliest input row
    for group in sorted(groups.values(), key=lambda g: min(r["idx"] for r in g)):
        latest = max(r["updated"] for r in group)
        tags = sorted({t for r in group for t in r["tags"]})
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "source": pick(group, "source"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
        })

    return output
```

**Before you send:**

- **The function doesn't remove opted-out people.** Per the spec, it keeps them and sets `opted_out: "Y"`. Filter your send list with `[p for p in result if p["opted_out"] == "N" and p["email"]]`. The `p["email"]` check matters because some people are kept on a valid phone alone and have no email.
- **Opt-outs follow merges.** If any row in a merged group opted out, the whole person is marked `"Y"`, including their other addresses. This is the safe behavior.
- **Opt-outs can be lost in one case.** A row with no valid email and no valid phone is dropped by rule 9, opt-out flag and all. That row couldn't be matched to anyone anyway, so it shouldn't cause a bad send. Still, if your CRM keeps a separate suppression list, apply it as well.
- **Interpretation choices** where the spec left room:
  - Extension markers match the earliest of `ext`, `x`, or `#` anywhere in the phone string.
  - ZIP+4 is recognized as 4 or 5 digits, a hyphen, then 4 digits. The 4-digit case covers Excel-stripped values like `2139-4307`.
  - An unparseable `updated` value raises a `ValueError` instead of being silently guessed. That way a bad row shows up before the send.
