```python
import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _s(row, key):
    """Fetch a field as a string, treating missing/None as ''."""
    v = row.get(key)
    return "" if v is None else str(v)


# Rule 1
def _norm_email(raw):
    s = raw
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


# Rule 2
def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


# Rule 3 (returns the normalized digits, valid or not)
def _norm_phone(raw):
    m = _EXT_RE.search(raw)
    if m:
        raw = raw[: m.start()]
    digits = "".join(c for c in raw if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(digits):
    return len(digits) == 10 and digits[0] not in "01"


# Rule 4
def _norm_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


# Rule 5
def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


# Rule 6
def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


# Rule 7
def _is_opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


# Rule 8
def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email or phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(_s(row, "email"))
        email = email if _email_is_valid(email) else ""
        phone = _norm_phone(_s(row, "phone"))
        phone = phone if _phone_is_valid(phone) else ""
        if not email and not phone:
            continue
        source = _s(row, "source").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _norm_name(_s(row, "first_name")),
            "last_name": _norm_name(_s(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(_s(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _split_tags(_s(row, "tags")),
            "opted_out": _is_opted_out(_s(row, "opted_out")),
            "updated": _parse_updated(_s(row, "updated")),
        })

    # ---- Union-find over matching keys (rule 10)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the earliest record (helps ordering, not required)
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Build one output dict per person (rule 11)
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["is_manual"], r["updated"], -r["index"]),
        )
        return best[field]

    out = []
    # Rule 12: order by earliest input row in each group
    for group in sorted(groups.values(), key=lambda g: min(r["index"] for r in g)):
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return out
```

1. **The function does not remove opted-out people.** It flags them, following the spec. Filter the output before mailing:

   The `p["email"]` check also drops phone-only contacts, who can't receive an email anyway.

2. **Opt-outs can be missed when a person can't be matched.** A row is dropped under rule 9 if it has no valid email or phone, and that includes its opt-out flag. Likewise, if someone opted out under one address and is subscribed under another, with no shared phone and last name, the two rows won't merge. If your CRM keeps a separate suppression list, run the final send list against it as well.

3. **Shared phones merge households.** Under rule 10, two people with the same last name and the same phone number (for example, spouses with one landline) become one person. Only one email address is kept, so the other spouse won't get the alert. If one of them opted out, both are marked "Y". Given the opt-out concern, that's the conservative failure mode, but you should know it happens.

4. **Bad dates stop the run.** An `updated` value in any other format raises a `ValueError` that names the value. The alternative was to guess at it silently.

5. **Edge cases handled:**
   - Gmail addresses like `+tag@gmail.com` have an empty local part after normalization, so they're treated as invalid.
   - Extension markers are matched anywhere in the phone string, so `555-123-4567 x12`, `ext. 12`, and `#12` are all trimmed.
   - A 4-digit ZIP+4 such as `2139-4307` becomes `02139`.
