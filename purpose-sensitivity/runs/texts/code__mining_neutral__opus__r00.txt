import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EXT_MARKER = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DISPLAY_NAME = re.compile(r"<([^<>]*)>")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_EPOCH = datetime(1, 1, 1)


def _field(row, key):
    value = row.get(key)
    return value if isinstance(value, str) else ("" if value is None else str(value))


def _norm_email(raw):
    s = raw
    m = _DISPLAY_NAME.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Return the 10-digit phone if valid, else ""."""
    s = raw
    m = _EXT_MARKER.search(s)
    if m:
        s = s[: m.start()]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(_field(row, "email"))
        email = email if _email_is_valid(email) else ""
        phone = _norm_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(_field(row, "first_name")),
            "last_name": _norm_name(_field(row, "last_name")),
            "zip": _norm_zip(_field(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": _split_tags(_field(row, "tags")),
            "opted_out": _field(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(_field(row, "updated")),
            "is_manual": source == "manual",
        })

    # ---- Union-find for transitive matching (rule 10).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the earlier record as root (not required, but tidy).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    seen_keys = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in seen_keys:
                union(i, seen_keys[key])
            else:
                seen_keys[key] = i

    # ---- Group records; dict insertion order == order of each person's earliest row (rule 12).
    groups = {}
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    def preference(r):
        # manual first, then most recent updated, then earliest in input
        return (
            0 if r["is_manual"] else 1,
            -(r["updated"] - _EPOCH).total_seconds(),
            r["idx"],
        )

    out = []
    for members in groups.values():
        ranked = sorted(members, key=preference)
        person = {}
        for f in ("first_name", "last_name", "zip", "email", "phone", "source"):
            person[f] = next((r[f] for r in ranked if r[f]), "")
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        person["tags"] = ";".join(sorted(all_tags))
        person["opted_out"] = "Y" if any(r["opted_out"] for r in members) else "N"
        latest = max(r["updated"] for r in members)
        person["updated"] = f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}"
        out.append(person)

    return out

**Notes on interpretation choices**

- **Opt-outs are flagged, not removed.** Per the spec, opted-out people still appear in the output with `opted_out: "Y"`. Before sending, filter with `[p for p in result if p["opted_out"] == "N"]`. Because a single opted-out row marks the whole merged person, this is the safe direction.
- **ZIP+4 with a lost leading zero** (e.g. `"2139-4307"`) is also accepted and becomes `"02139"`. If you want only the strict 5+4 form, change `{4,5}` to `{5}` in `_ZIP_PLUS4`.
- **Extension markers:** the earliest marker in the string wins, so `"555-123-4567 ext. 12"` and `"555-123-4567x12"` both work. Any letter "x" counts as a marker, as the rules say.
- **Gmail rewriting** only happens when the address has exactly one `@`. Addresses with zero or several `@` are invalid anyway.
- **Bad `updated` values** raise a `ValueError` that names the offending value. This stops the run instead of silently guessing. The rules say `updated` is always present, so this should only trigger on genuinely malformed data.
- Digits are matched as ASCII `0-9` only, so stray Unicode digits won't sneak into phones or ZIPs.
- **Missing emails:** each person gets exactly one email, the preferred one. If someone has two different addresses that merged via phone and last name, only the chosen address receives the newsletter.
