```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, filter, and merge duplicate contact rows into one dict per person."""

    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    TRUE_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        s = raw
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, flags=re.IGNORECASE)
        if m:
            raw = raw[: m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_dt(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized updated format: %r" % raw)

    def parse_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize rows, drop those with no valid email/phone ----------
    recs = []  # kept rows, in input order
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in TRUE_VALUES,
            "updated": parse_dt(get(row, "updated")),
        })

    n = len(recs)

    # ---------- union-find ----------
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root (not required, just tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def sort_key(r):
        secs = int((r["updated"] - datetime.min).total_seconds())
        return (0 if r["manual"] else 1, -secs, r["idx"])

    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        return min(cands, key=sort_key)[field]

    people = []
    for members_idx in groups.values():
        members = [recs[i] for i in members_idx]  # already in input order
        tags = set()
        for r in members:
            tags.update(r["tags"])
        person = {
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        }
        people.append((min(r["idx"] for r in members), person))

    people.sort(key=lambda t: t[0])
    return [p for _, p in people]
```

Notes on a few interpretation choices:

- **Opt-outs:** the group is marked `"Y"` if any merged row opted out. Your send step should skip every person with `opted_out == "Y"`.
- **Gmail rules:** dot-removal and `+` stripping only apply when the address has exactly one `@`. Anything else is invalid anyway.
- **ZIP+4:** only a strict `#####-####` pattern is treated as ZIP+4. Something like `02139-43` becomes `""`.
- **Phone extensions:** the marker search finds the leftmost `ext`, `x`, or `#` and discards everything from there on.
- **Last-name matching:** it uses the normalized (stripped, title-cased where applicable) last name, compared with `.lower()`.
- **Output order:** people are ordered by their earliest *kept* row. Dropped rows don't affect ordering.
- **Dates:** an unparseable `updated` value raises `ValueError` rather than being silently guessed. If you'd rather skip those rows, wrap `parse_dt` in a try/except.
