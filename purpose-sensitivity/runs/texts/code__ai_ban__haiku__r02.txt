import re
from datetime import datetime

_DIGITS = "0123456789"


def _is_digits(s):
    return s != "" and all(c in _DIGITS for c in s)


def _norm_email(raw):
    s = raw.strip()
    start = s.find("<")
    if start != -1:
        end = s.find(">", start + 1)
        if end != -1:
            s = s[start + 1:end]
    s = s.strip().lower()
    parts = s.split("@")
    if len(parts) == 2 and parts[1] in ("gmail.com", "googlemail.com"):
        local = parts[0].replace(".", "").split("+", 1)[0]
        s = local + "@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _norm_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = raw.strip()
    if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
        s = s.split("-", 1)[0]
    if len(s) == 4 and _is_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_digits(s):
        return s
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _norm_tags(raw):
    tags = set()
    for t in re.split(r"[;,]", raw):
        t = t.strip().lower()
        if t:
            tags.add(t)
    return tags


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        updated = _parse_updated(row["updated"])
        email = _valid_email(_norm_email(row.get("email", "")))
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = row.get("source", "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "updated": updated,
            "manual": source == "manual",
            "opted_out": row.get("opted_out", "").strip().lower() in ("y", "yes", "true", "1"),
            "tags": _norm_tags(row.get("tags", "")),
        })

    # Step 2: group rows into people (union-find handles transitivity).
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], i)
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    # Groups are ordered by their earliest input row.
    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(recs[i])

    # Step 3: pick the representative value for each field.
    def better(a, b):
        if a["manual"] != b["manual"]:
            return a["manual"]
        if a["updated"] != b["updated"]:
            return a["updated"] > b["updated"]
        return a["idx"] < b["idx"]

    def pick(group, field):
        best = None
        for r in group:
            if r[field] and (best is None or better(r, best)):
                best = r
        return best[field] if best else ""

    results = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        results.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return results

Notes before you use this:

- **Opt-outs aren't removed.** The function sets `opted_out` to "Y" for any person with an opted-out row, but it still returns them. Filter out `opted_out == "Y"` before building the send list.
- **I haven't run this.** I can't execute code here. Before tomorrow's send, run it on a copy of the export and check the counts, and spot-check a few known duplicates, opt-outs, and gmail variants.
- **Judgment calls:**
  - An unparseable `updated` value raises a `ValueError` rather than being silently sorted to the bottom, so bad data surfaces early.
  - "Entirely upper-case" and "entirely lower-case" use `str.isupper()` and `str.islower()`. So "ABC123" is treated as upper-case, and "123" is left unchanged.
  - The gmail rules apply only when the address has exactly one "@".
  - A ZIP like "1234-5678" becomes "" because it doesn't match the ZIP+4 pattern.
