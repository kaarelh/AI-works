import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_UPDATED_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_ZIP4_DIGITS_RE = re.compile(r"[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DIGITS = "0123456789"


def _normalize_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw or ""
    m = _EXT_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    z = (raw or "").strip()
    if _ZIP4_RE.fullmatch(z):
        z = z[:5]
    if _ZIP4_DIGITS_RE.fullmatch(z):
        z = "0" + z
    if not _ZIP5_RE.fullmatch(z):
        return ""
    return z


def _normalize_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _UPDATED_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None  # unparseable: treated as oldest


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _normalize_tags(raw):
    return {p.strip().lower() for p in _TAG_SPLIT_RE.split(raw or "") if p.strip()}


def _pick(group, field):
    """Pick a field value: prefer manual source, then most recent updated, then earliest row."""
    cands = [x for x in group if x[field]]  # rows are already in input order
    if not cands:
        return ""
    # Stable sorts: the last sort is the primary key.
    cands.sort(key=lambda x: x["updated"] if x["updated"] is not None else datetime.min,
               reverse=True)
    cands.sort(key=lambda x: x["source"] != "manual")
    return cands[0][field]


def _latest_date(group):
    dates = [x["updated"] for x in group if x["updated"] is not None]
    return max(dates).strftime("%Y-%m-%d") if dates else ""


def dedupe_contacts(rows):
    # Step 1: normalize and drop rows with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone"))
        if not _is_valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue
        last_name = _normalize_name(row.get("last_name"))
        recs.append({
            "index": idx,
            "email": email,
            "phone": phone,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": last_name,
            "zip": _normalize_zip(row.get("zip")),
            "source": (row.get("source") or "").strip().lower(),
            "tags": _normalize_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Step 2: union-find over matching keys (transitive matching).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_key = {}
    for i, rec in enumerate(recs):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_by_key:
                union(i, first_by_key[key])
            else:
                first_by_key[key] = i

    # Group rows by root, ordered by each group's earliest input row.
    order = []
    members = {}
    for i, rec in enumerate(recs):
        root = find(i)
        if root not in members:
            members[root] = []
            order.append(root)
        members[root].append(rec)

    # Step 3: build one output record per person.
    result = []
    for root in order:
        group = members[root]
        result.append({
            "first_name": _pick(group, "first_name"),
            "last_name": _pick(group, "last_name"),
            "zip": _pick(group, "zip"),
            "email": _pick(group, "email"),
            "phone": _pick(group, "phone"),
            "source": _pick(group, "source"),
            "tags": ";".join(sorted(set().union(*(x["tags"] for x in group)))),
            "opted_out": "Y" if any(x["opted_out"] for x in group) else "N",
            "updated": _latest_date(group),
        })
    return result

**Notes on interpretation:**

- **ZIP+4:** Only a strict `12345-6789` pattern is reduced to its first five digits. Anything else that isn't 5 digits (after the 4-digit leading-zero fix) becomes `""`.
- **Names:** Title-casing applies when `str.isupper()` or `str.islower()` is true, so names with no cased letters (for example, `"123"`) are left alone.
- **Unparseable `updated`:** Your rules say every value is in one of the three formats, but if one isn't, the row sorts as oldest and is ignored for the group's latest date. Tell me if you'd rather it raise an error.
- **Opt-outs before sending:** Output `opted_out == "Y"` means the person opted out in at least one row. Filter those out in the send step. Rows with neither a valid email nor a valid phone are dropped, so they can't be emailed anyway.
