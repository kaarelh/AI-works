import re
from datetime import datetime


def dedupe_contacts(rows):
    _ANGLE = re.compile(r"<([^>]*)>")
    _NON_DIGIT = re.compile(r"[^0-9]")
    _ZIP5 = re.compile(r"[0-9]{5}")
    _TAG_SPLIT = re.compile(r"[;,]")
    _DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- field normalizers ----------
    def norm_email(raw):
        s = str(raw or "")
        m = _ANGLE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") != 1:
            return ""                      # invalid
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            plus = local.find("+")
            if plus != -1:
                local = local[:plus]
            domain = "gmail.com"
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""                      # invalid
        return local + "@" + domain

    def norm_phone(raw):
        s = str(raw or "").lower()
        cut = len(s)
        for marker in ("x", "ext", "#"):
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = _NON_DIGIT.sub("", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                          # invalid

    def norm_zip(raw):
        z = str(raw or "").strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if len(z) == 4 and _ZIP5.fullmatch("0" + z):
            z = "0" + z
        return z if _ZIP5.fullmatch(z) else ""

    def norm_name(raw):
        s = str(raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = str(raw or "").strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return str(raw or "").strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):
        out = set()
        for t in _TAG_SPLIT.split(str(raw or "")):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize every row ----------
    recs = []
    for row in rows:
        source = str(row.get("source", "") or "").strip().lower()
        recs.append({
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": norm_email(row.get("email", "")),     # "" if invalid
            "phone": norm_phone(row.get("phone", "")),     # "" if invalid
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "dt": parse_updated(row.get("updated", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "tags": split_tags(row.get("tags", "")),
        })

    # Rule 9: drop rows with neither a valid email nor a valid phone
    kept = [i for i, r in enumerate(recs) if r["email"] or r["phone"]]

    # ---------- union-find for transitive matching ----------
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i in kept:
        r = recs[i]
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    # Group members in input order; dict insertion order = order of each group's earliest row
    groups = {}
    for i in kept:
        groups.setdefault(find(i), []).append(i)

    # ---------- build one output dict per person ----------
    def pick(idxs, field):
        cands = [i for i in idxs if recs[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda i: (recs[i]["manual"], recs[i]["dt"], -i))
        return recs[best][field]

    output = []
    for idxs in groups.values():
        tags = set()
        for i in idxs:
            tags |= recs[i]["tags"]
        output.append({
            "first_name": pick(idxs, "first_name"),
            "last_name": pick(idxs, "last_name"),
            "email": pick(idxs, "email"),
            "phone": pick(idxs, "phone"),
            "zip": pick(idxs, "zip"),
            "source": pick(idxs, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in idxs) else "N",
            "updated": max(recs[i]["dt"] for i in idxs).strftime("%Y-%m-%d"),
        })
    return output

A few judgment calls worth knowing before you run it on the real export:

- **Unparseable `updated` values raise a `ValueError`** rather than being silently guessed, since a bad guess could change which row's email/phone wins. If the export has blanks or odd formats there, tell me and I'll add a fallback.
- **ZIP with a hyphen:** anything before the first hyphen is kept, then the 4-digit fix is applied, so `"2139-4307"` becomes `"02139"` (Excel-eaten zero plus ZIP+4).
- **Invalid emails/phones are stored as `""`**, so they can never win the per-field selection or be used for matching — which is what rules 9–11 require.
- The function reads fields with `.get(..., "")`, so a row missing a key won't crash it (except `updated`, per the first bullet).
