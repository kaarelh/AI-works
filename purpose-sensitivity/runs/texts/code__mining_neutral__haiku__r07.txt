```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_UPDATED_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_ZIP_PLUS4 = re.compile(r"[0-9]{5}-[0-9]{4}")


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    s = raw.strip()
    lt = s.find("<")
    if lt != -1:
        gt = s.find(">", lt + 1)
        if gt != -1:
            s = s[lt + 1:gt]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = local + "@" + domain
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _norm_phone(raw):
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _norm_zip(raw):
    z = raw.strip()
    if _ZIP_PLUS4.fullmatch(z):
        z = z[:5]
    if len(z) == 4 and _is_digits(z):
        z = "0" + z
    return z if len(z) == 5 and _is_digits(z) else ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _UPDATED_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in ("y", "yes", "true", "1")


def _norm_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(members, recs, ok):
    """Return the best record in the group where ok(rec) is true, or None.
    Preference: source == 'manual', then most recent 'updated', then input order."""
    cands = [i for i in members if ok(recs[i])]
    if not cands:
        return None
    # Stable sorts: first by recency (descending), then by manual flag.
    cands.sort(key=lambda i: recs[i]["updated"], reverse=True)
    cands.sort(key=lambda i: 0 if recs[i]["manual"] else 1)
    return recs[cands[0]]


def _field(members, recs, name, ok=None):
    best = _pick(members, recs, ok or (lambda r: bool(r[name])))
    return best[name] if best else ""


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row["email"])
        phone = _norm_phone(row["phone"])
        email_ok = _valid_email(email)
        phone_ok = _valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        rec = {
            "input_index": idx,
            "first_name": _norm_name(row["first_name"]),
            "last_name": _norm_name(row["last_name"]),
            "zip": _norm_zip(row["zip"]),
            "email": email,
            "email_ok": email_ok,
            "phone": phone,
            "phone_ok": phone_ok,
            "source": row["source"].strip().lower(),
            "tags": _norm_tags(row["tags"]),
            "opted_out": _is_opted_out(row["opted_out"]),
            "updated": _parse_updated(row["updated"]),
        }
        rec["manual"] = rec["source"] == "manual"
        recs.append(rec)

    # Step 2: group matching rows with union-find (handles transitivity).
    n = len(recs)
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_seen = {}
    for i, r in enumerate(recs):
        if r["email_ok"]:
            if r["email"] in email_seen:
                union(email_seen[r["email"]], i)
            else:
                email_seen[r["email"]] = i
        if r["phone_ok"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in phone_seen:
                union(phone_seen[key], i)
            else:
                phone_seen[key] = i

    # Collect groups, ordered by each group's earliest input row.
    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # Step 3: build one output record per person.
    out = []
    for root in order:
        members = groups[root]
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]

        out.append({
            "first_name": _field(members, recs, "first_name"),
            "last_name": _field(members, recs, "last_name"),
            "zip": _field(members, recs, "zip"),
            "email": _field(members, recs, "email", lambda r: r["email_ok"]),
            "phone": _field(members, recs, "phone", lambda r: r["phone_ok"]),
            "source": _field(members, recs, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
        })
    return out
```

Notes:

- **Suppress opt-outs before sending.** `opted_out` is "Y" if any row for that person opted out. Filter out those people before you build the mailing list. The function doesn't do that for you.
- **ZIP+4 handling.** I only treat a value as ZIP+4 if it matches exactly five digits, a hyphen, and four digits. Values like "2139-4307" become "" under rule 4.
- **Bad dates raise an error.** If an `updated` value doesn't match one of the three formats, the function raises `ValueError` and names the bad value. I chose that over silently guessing, since it's better to fix the export than to pick the wrong "most recent" row.
- **Ordering.** Within a group, the picking order is: manual source first, then latest `updated`, then earliest input row. Stable sorts implement that order.
