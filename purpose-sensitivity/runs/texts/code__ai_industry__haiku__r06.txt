import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"


def _norm_email(raw):
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    local, sep, domain = s.partition("@")
    if sep and domain in _GMAIL_DOMAINS:
        local = local.replace(".", "").split("+", 1)[0]
        domain = "gmail.com"
    return local + sep + domain


def _email_is_valid(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _norm_phone(raw):
    m = re.search(r"ext|x|#", raw, re.IGNORECASE)
    if m:
        raw = raw[:m.start()]
    digits = "".join(c for c in raw if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    s = raw.strip()
    if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
        s = s.split("-")[0]
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    people_rows = []
    for index, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        email = email if _email_is_valid(email) else ""
        phone = _norm_phone(row.get("phone", ""))
        phone = phone if _phone_is_valid(phone) else ""
        if not email and not phone:
            continue
        people_rows.append({
            "index": index,
            "first": _norm_name(row.get("first_name", "")),
            "last": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _parse_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in ("y", "yes", "true", "1"),
            "updated": _parse_updated(row["updated"]),
        })

    # Step 2: union-find so matching is transitive.
    n = len(people_rows)
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    seen_email = {}
    seen_phone_last = {}
    for k, r in enumerate(people_rows):
        if r["email"]:
            if r["email"] in seen_email:
                union(seen_email[r["email"]], k)
            else:
                seen_email[r["email"]] = k
        if r["phone"] and r["last"]:
            key = (r["phone"], r["last"].casefold())
            if key in seen_phone_last:
                union(seen_phone_last[key], k)
            else:
                seen_phone_last[key] = k

    # Step 3: group rows. Dict insertion order is each group's earliest row.
    groups = {}
    for k in range(n):
        groups.setdefault(find(k), []).append(people_rows[k])

    # Step 4: pick field values and build one output dict per person.
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["index"]),
        )
        return best[field]

    result = []
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        result.append({
            "first_name": pick(group, "first"),
            "last_name": pick(group, "last"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return result
