```python
import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"(ext|x|#)", re.IGNORECASE)
_ZIP4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP5_RE = re.compile(r"^[0-9]{5}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the agreed matching rules."""

    # ---------- normalization helpers ----------

    def norm_email(raw):
        s = raw or ""
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_is_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        s = raw or ""
        m = _EXT_RE.search(s)
        if m:
            s = s[: m.start()]
        digits = "".join(ch for ch in s if "0" <= ch <= "9")
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_is_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        m = _ZIP4_RE.match(s)
        if m:
            s = m.group(1)
        if len(s) == 4 and s.isascii() and s.isdigit():
            s = "0" + s
        return s if _ZIP5_RE.match(s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return (raw or "").strip().lower() in _OPT_OUT_VALUES

    def parse_tags(raw):
        return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "") if t.strip()}

    # ---------- normalize rows, drop unreachable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        email_ok = email_is_valid(email)
        phone_ok = phone_is_valid(phone)
        if not email_ok and not phone_ok:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "zip": norm_zip(row.get("zip", "")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "updated": parse_updated(row.get("updated", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "tags": parse_tags(row.get("tags", "")),
        })

    # ---------- union-find matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the lower index as root (earliest row in input)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone_last", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in first_seen:
                union(first_seen[k], i)
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output (rules 11, 12) ----------

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    def fmt_date(d):
        return f"{d.year:04d}-{d.month:02d}-{d.day:02d}"

    output = []
    for group in sorted(groups.values(), key=lambda g: min(r["idx"] for r in g)):
        tags = set()
        for r in group:
            tags |= r["tags"]
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": fmt_date(max(r["updated"] for r in group)),
        })
    return output
```

Notes on a few judgment calls:

- **Extension markers:** The function cuts at the earliest of `ext`, `x`, or `#`. For example, `555-123-4567 ext. 89` and `555-123-4567 x89` both become `5551234567`.
- **Gmail normalization** only runs when the address has exactly one `@`. Addresses with more than one `@` are invalid anyway.
- **ZIP+4 with a lost leading zero** (e.g. `2139-4307`) is also handled. It becomes `02139`.
- **Digits:** Only ASCII digits 0–9 are counted, so odd Unicode digits can't sneak into phones or ZIPs.
- **Bad dates:** An `updated` value that matches none of the three formats raises `ValueError`. That way bad data fails loudly instead of quietly affecting which row wins.
- **Matching uses only kept rows.** Rows dropped under rule 9 never link other rows together.

**Before tomorrow's send:** filter the output to drop everyone with `opted_out == "Y"`. The function flags opted-out people but doesn't remove them. This is deliberate: if someone opted out under one email, the whole merged person is marked "Y". Excluding them is what keeps the send compliant.
