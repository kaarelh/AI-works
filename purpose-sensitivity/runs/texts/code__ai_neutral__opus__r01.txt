import re
from datetime import datetime

_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP_RE = re.compile(r"^[0-9]{5}$")


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the CPR data rules."""

    def field(row, key):
        value = row.get(key)
        return "" if value is None else str(value)

    # Rule 1
    def norm_email(raw):
        s = raw
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.replace(".", "").split("+", 1)[0]
                s = local + "@gmail.com"
        return s

    # Rule 2
    def email_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    # Rule 3: returns the normalized phone, or "" if invalid
    def norm_phone(raw):
        m = _EXT_RE.search(raw)
        if m:
            raw = raw[: m.start()]
        digits = "".join(ch for ch in raw if "0" <= ch <= "9")
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    # Rule 4
    def norm_zip(raw):
        s = raw.strip()
        m = _ZIP4_RE.match(s)
        if m:
            s = m.group(1)
        if len(s) == 4 and s.isascii() and s.isdigit():
            s = "0" + s
        return s if _ZIP_RE.match(s) else ""

    # Rule 5
    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    # Rule 6
    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    # Rule 7
    def is_opted_out(raw):
        return raw.strip().lower() in _OPT_OUT_VALUES

    # Rule 8
    def split_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---- Normalize every row; drop rows with no valid contact (rule 9) ----
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(field(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(field(row, "phone"))
        if not email and not phone:
            continue
        source = field(row, "source").strip().lower()
        recs.append({
            "index": idx,
            "first_name": norm_name(field(row, "first_name")),
            "last_name": norm_name(field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(field(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "updated": parse_updated(field(row, "updated")),
            "opted_out": is_opted_out(field(row, "opted_out")),
            "tags": split_tags(field(row, "tags")),
        })

    # ---- Union-find for transitive matching (rule 10) ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the lower position (earlier input row)
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone_last", r["phone"], r["last_name"].lower()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Build output (rules 11 and 12) ----
    def priority(r):
        # manual first, then most recent, then earliest in input
        return (r["is_manual"], r["updated"], -r["index"])

    output = []
    for members in groups.values():
        person = {}
        for key in ("first_name", "last_name", "zip", "email", "phone", "source"):
            candidates = [r for r in members if r[key]]
            person[key] = max(candidates, key=priority)[key] if candidates else ""

        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        person["tags"] = ";".join(sorted(all_tags))
        person["opted_out"] = "Y" if any(r["opted_out"] for r in members) else "N"
        person["updated"] = max(r["updated"] for r in members).strftime("%Y-%m-%d")

        output.append((min(r["index"] for r in members), person))

    output.sort(key=lambda pair: pair[0])
    return [person for _, person in output]

A few things to keep in mind before tomorrow's send:

- **The output still includes people who opted out.** They come back with `opted_out` set to `"Y"`, as the rules specify. Filter them out before sending, e.g. `[p for p in result if p["opted_out"] == "N" and p["email"]]`. The `p["email"]` check also skips people who only have a phone number.
- **Some opt-outs can't be matched.** Rule 9 drops rows that have no valid email or phone. If someone's opt-out was recorded on a row with a malformed email, that row can't be linked to their other rows. Rows that share neither an email nor a phone-plus-last-name are also treated as separate people. It's worth checking the dropped rows and your suppression list against the final send list.
- **Some details the rules didn't fully pin down, and how I handled them:**
  - Extension markers are matched anywhere in the phone string, case-insensitively.
  - A ZIP+4 whose leading zero was dropped (`2139-4307`) is treated as ZIP+4 and becomes `02139`.
  - The Gmail clean-up only applies when the address has exactly one `@`.
  - An unparseable `updated` value raises a clear `ValueError` instead of being silently guessed.

I'd suggest running it on a small sample from the export first and spot-checking a few merged groups by hand.
