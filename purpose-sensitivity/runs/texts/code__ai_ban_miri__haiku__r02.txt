import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_TRUE_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")


def _field(row, key):
    return row.get(key) or ""


def _is_ascii_digits(s):
    return s != "" and all("0" <= c <= "9" for c in s)


def _normalize_email(raw):
    s = raw
    start = s.find("<")
    if start != -1:
        end = s.find(">", start + 1)
        if end != -1:
            s = s[start + 1:end]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (local != ""
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith("."))


def _normalize_phone(raw):
    s = raw.lower()
    markers = [p for p in (s.find("x"), s.find("ext"), s.find("#")) if p != -1]
    if markers:
        s = s[:min(markers)]
    digits = "".join(c for c in s if "0" <= c <= "9")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in ("0", "1")


def _normalize_zip(raw):
    z = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if len(z) == 4 and _is_ascii_digits(z):
        z = "0" + z
    if len(z) == 5 and _is_ascii_digits(z):
        return z
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def dedupe_contacts(rows):
    # Step 1: normalize each row; drop rows with neither a valid email nor phone.
    recs = []
    for row in rows:
        email = _normalize_email(_field(row, "email"))
        email = email if _valid_email(email) else ""

        phone = _normalize_phone(_field(row, "phone"))
        phone = phone if _valid_phone(phone) else ""

        if not email and not phone:
            continue

        source = _field(row, "source").strip().lower()
        ts = _parse_updated(_field(row, "updated"))
        tags = {t for t in (x.strip().lower()
                            for x in re.split(r"[;,]", _field(row, "tags")))
                if t}

        recs.append({
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "zip": _normalize_zip(_field(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": _field(row, "opted_out").strip().lower() in _TRUE_VALUES,
            "ts": ts,
            "ts_key": ts if ts is not None else datetime.min,
        })

    # Step 2: union-find so matching is transitive.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], i)
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    # Group rows; dict insertion order follows each group's earliest row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    def pick(members, field):
        # Prefer manual, then most recent updated, then earliest input row.
        cands = [i for i in members if recs[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda i: (recs[i]["manual"],
                                         recs[i]["ts_key"],
                                         -i))
        return recs[best][field]

    # Step 3: build one output dict per person.
    result = []
    for members in groups.values():
        ts_values = [recs[i]["ts"] for i in members if recs[i]["ts"] is not None]
        all_tags = set()
        for i in members:
            all_tags |= recs[i]["tags"]

        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(ts_values).strftime("%Y-%m-%d") if ts_values else "",
        })

    return result

Notes on interpretation and before the send:

- **Filter before sending.** The function flags opted-out people with `opted_out == "Y"` but does not remove them. Drop those rows before building the send list. Because a merged person is flagged "Y" if any of their rows opted out, the flag applies to the whole person, which is the conservative choice.
- **ZIP+4:** I accept 4 or 5 digits before the hyphen so that Excel-damaged forms like `2139-4307` are repaired to `02139`.
- **Names:** "entirely upper-case or lower-case" is implemented with `str.isupper()` / `str.islower()`, so mixed-case names such as "McDonald" are left alone.
- **Email validity:** I applied only the rules you listed. The spec doesn't reject characters such as spaces or stray brackets, so a malformed address with exactly one `@` and a dotted domain counts as valid.
