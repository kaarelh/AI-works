```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one dict per person.

    Rows with no valid email and no valid phone are dropped, and rows are
    then merged (transitively) when they share a valid normalized email, or a
    valid normalized phone plus the same non-empty last name.
    """

    # ---------- normalizers ----------
    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"ext|x|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", s):
            s = s.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated value: %r" % (raw,))

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw or ""):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize and filter ----------
    recs = []
    for orig_idx, r in enumerate(rows):
        email = norm_email(r.get("email", ""))
        if not email_valid(email):
            email = ""
        phone = norm_phone(r.get("phone", ""))
        if not email and not phone:
            continue
        source = (r.get("source", "") or "").strip().lower()
        recs.append({
            "idx": orig_idx,
            "first_name": norm_name(r.get("first_name", "")),
            "last_name": norm_name(r.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(r.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(r.get("tags", "")),
            "opted_out": (r.get("opted_out", "") or "").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_updated(r.get("updated", "")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # ---------- matching ----------
    by_email = {}
    by_phone_last = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in by_email:
                union(i, by_email[rec["email"]])
            else:
                by_email[rec["email"]] = i
        last = rec["last_name"].casefold()
        if rec["phone"] and last:
            key = (rec["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def pick(members, field):
        best = None
        best_key = None
        for i in members:  # members are in input order
            rec = recs[i]
            if not rec[field]:
                continue
            key = (rec["manual"], rec["updated"])
            if best is None or key > best_key:  # strict: ties keep earlier row
                best, best_key = rec, key
        return best[field] if best is not None else ""

    ordered = sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m))

    result = []
    for members in ordered:
        members = sorted(members, key=lambda i: recs[i]["idx"])
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
        })
    return result
```

Notes on a few judgment calls:

- **Opt-outs:** if any row in a merged person has opted out, the output is `opted_out: "Y"`. Before sending, filter out everyone with `"Y"`.
- **Invalid emails and phones:** these are blanked on the row, so they never trigger a match and are never output. A row with a bad email but a valid phone still survives and can match by phone and last name.
- **Phone matching:** it needs a non-empty last name. Two rows with the same phone but no last name are not merged unless they share an email.
- **Extension markers:** the regex `ext|x|#` takes the leftmost match, so "555-123-4567 ext 12" is cut at "ext" rather than at the "x" inside it.
- **Ties:** the manual preference comes first, then the latest `updated`, then the earlier input row. The comparison is strict, so a later row only wins if it is strictly better.
- **Bad dates:** an `updated` value in an unexpected format raises `ValueError` instead of being silently guessed. You'd rather find out before a send than quietly mis-merge a record.
