import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ZIP4_RE = re.compile(r"^[0-9]{5}-[0-9]{4}$")
_ZIP5_RE = re.compile(r"^[0-9]{5}$")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _s(value):
    """Coerce a possibly missing/None field to a string."""
    return "" if value is None else str(value)


def _normalize_email(raw):
    """Return the normalized email if valid, else ''."""
    e = _s(raw)
    lt = e.find("<")
    if lt != -1:
        gt = e.find(">", lt + 1)
        if gt != -1:
            e = e[lt + 1:gt]
    e = e.strip().lower()

    if e.count("@") == 1:
        local, domain = e.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            plus = local.find("+")

            domain = "gmail.com"
            e = local + "@" + domain

        return ""
    return e


def _normalize_phone(raw):
    """Return the 10-digit normalized phone if valid, else ''."""
    p = _s(raw).lower()
    cut = len(p)

    if len(digits) == 11 and digits[0] == "1":

    return ""


def _normalize_zip(raw):

        z = z.split("-")[0]

def _normalize_name(raw):

def _parse_updated(raw):

def dedupe_contacts(rows):
    # ---- Step 1: normalize each row, drop rows with no valid email/phone ----
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        phone = _normalize_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = _s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _s(row.get("opted_out")).strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        root = x
        while parent[root] != root:
            root = parent[root]
        while parent[x] != root:
            parent[x], x = root, parent[x]
        return root

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so order is stable
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # ---- Step 2: match (transitively) ----
    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---- Step 3: merge each group ----
    epoch = datetime.min

    def pick(members, field):
        best = None
        best_key = None
        for i in members:
            r = recs[i]
            if not r[field]:
                continue
            key = (
                0 if r["source"] == "manual" else 1,
                -(r["updated"] - epoch).total_seconds(),
                r["idx"],
            )
            if best_key is None or key < best_key:
                best_key = key
                best = r[field]
        return best if best is not None else ""

    results = []
    for members in groups.values():
        members.sort(key=lambda i: recs[i]["idx"])
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        results.append((
            recs[members[0]]["idx"],
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "zip": pick(members, "zip"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "source": pick(members, "source"),
                "tags": ";".join(sorted(tags)),
                "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
                "updated": latest.strftime("%Y-%m-%d"),
            },
        ))

    # ---- Step 4: order by earliest row in input ----
    results.sort(key=lambda t: t[0])
    return [d for _, d in results]

Notes on judgment calls:

- **Opt-outs:** Because opted_out is "Y" if *any* row in a merged group opted out, a person who opted out under one email and appears under another matched record stays opted out. For tomorrow's send, filter the output on `opted_out == "N"` as well as deduping.
- **Extension markers:** Searching for `x` also covers `ext`, since it contains an `x`. The cut happens at the earliest marker found.
- **Gmail:** Dot and plus handling only applies when the address has exactly one `@`. Something like `+promo@gmail.com` becomes `@gmail.com`, which fails validity (empty local part).
- **Unparseable `updated`:** The function raises `ValueError` rather than silently guessing. Better to catch a bad date than to mis-order records.
- **Tie-breaking:** The most recent `updated` is compared as a full datetime, so `2025-03-01T14:00:00` beats `2025-03-01`.
- **Name matching:** Last names are compared after normalization (strip and title-casing), lowercased. Rows with an empty last name never match by phone.
