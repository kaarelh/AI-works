```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the data team's rules."""

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"ext|x|#", s, flags=re.IGNORECASE)
        if m:
            s = s[: m.start()]
        digits = "".join(ch for ch in s if "0" <= ch <= "9")  # ASCII digits only
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize & filter (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        e_ok = email_valid(email)
        p_ok = phone_valid(phone)
        if not e_ok and not p_ok:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email if e_ok else "",
            "phone": phone if p_ok else "",
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "updated": parse_updated(row.get("updated", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "tags": split_tags(row.get("tags", "")),
        })

    # ---------- union-find matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the earlier record as root (not required, just tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output (rules 11 & 12) ----------

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    out = []
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append((
            min(r["idx"] for r in group),
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "zip": pick(group, "zip"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "source": pick(group, "source"),
                "tags": ";".join(sorted(tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            },
        ))

    out.sort(key=lambda t: t[0])
    return [d for _, d in out]
```

**How some ambiguous cases are handled:**
- **Extension markers:** The regex checks `ext` before `x`, so in `555-1234 ext 9` the cut happens at `ext`. Note that any `x` counts as a marker, so a phone like `"Ph: 555..."` won't be affected, but a stray `x` anywhere will truncate the number.
- **ZIP+4:** The 4-digit form (`2139-4307`) is also accepted and then padded with a leading zero, in case Excel stripped the zero there too.
- **Digits:** Only ASCII `0`–`9` are counted, so Unicode digits like `²` are ignored.
- **Unparseable dates:** An `updated` value that doesn't match any of the three formats raises `ValueError` rather than being silently guessed. Wrap the call if you'd rather skip those rows.
- **Ties:** When choosing between rows, the code prefers manual source, then the newest `updated`, then the earliest position in the input.

**Opt-out checks before the send:**
- **Filter the output:** The function marks people as `opted_out: "Y"` but doesn't remove them. Filter with `[c for c in result if c["opted_out"] == "N"]` before mailing.
- **Opt-outs that rule 9 drops:** A row with no valid email or phone is discarded before matching, so its opt-out flag is lost. That's harmless in practice, since such a row can't match anyone anyway.
- **Opt-outs that don't merge:** An opt-out row that shares only a phone number with another row, but has a different or blank last name, won't merge with it. If you want to be conservative, also suppress any outgoing email whose address appears on *any* opted-out row in the raw export.
