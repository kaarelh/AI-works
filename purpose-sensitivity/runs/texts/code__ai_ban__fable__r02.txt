```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    PICKED_FIELDS = ("first_name", "last_name", "zip", "email", "phone", "source")

    # ---------- field normalizers ----------

    def text(row, key):
        v = row.get(key)
        if v is None:
            return ""
        return v if isinstance(v, str) else str(v)

    def norm_email(raw):
        # Rule 1
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        # Rule 2
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        # Rule 3 -- returns the 10-digit string if valid, else ""
        s = raw.lower()
        cut = len(s)
        for marker in ("x", "ext", "#"):
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        # Rule 4
        z = raw.strip()
        if "-" in z:
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        # Rule 5
        n = raw.strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    def parse_updated(raw):
        # Rule 6
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        # Rule 7
        return raw.strip().lower() in OPT_OUT_VALUES

    def split_tags(raw):
        # Rule 8
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---------- normalize every row, drop rows with no valid contact (rule 9) ----------

    records = []
    for idx, row in enumerate(rows):
        email = norm_email(text(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(text(row, "phone"))
        if not email and not phone:
            continue
        records.append(
            {
                "idx": idx,
                "first_name": norm_name(text(row, "first_name")),
                "last_name": norm_name(text(row, "last_name")),
                "email": email,
                "phone": phone,
                "zip": norm_zip(text(row, "zip")),
                "tags": split_tags(text(row, "tags")),
                "opted_out": is_opted_out(text(row, "opted_out")),
                "updated": parse_updated(text(row, "updated")),
                "source": text(row, "source").strip().lower(),
            }
        )

    # ---------- union-find for transitive matching (rule 10) ----------

    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(records):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(records[i])

    # ---------- build one output row per person (rules 11 & 12) ----------

    def pick(group, field):
        # Prefer source == "manual", then latest updated, then earliest input row.
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[field]

    output = []
    for group in sorted(groups.values(), key=lambda g: min(r["idx"] for r in g)):
        person = {field: pick(group, field) for field in PICKED_FIELDS}
        tags = set()
        for r in group:
            tags |= r["tags"]
        person["tags"] = ";".join(sorted(tags))
        person["opted_out"] = "Y" if any(r["opted_out"] for r in group) else "N"
        person["updated"] = max(r["updated"] for r in group).strftime("%Y-%m-%d")
        output.append(person)

    return output
```

A few notes on judgment calls, so you can confirm they match what your data person intended:

- **`updated` in an unknown format raises `ValueError`** rather than guessing. The rules say it's always one of three formats, so a bad value likely signals a broken export—better to fail loudly tonight than quietly mis-rank rows. If you'd rather it skip or default, that's a one-line change in `parse_updated`.
- **Phone output is the 10-digit string** (e.g. `2025551234`), since that's the "normalized value" from rule 3. Say the word if you want it reformatted for display.
- **ZIP with a hyphen** keeps whatever precedes the first hyphen, so `2139-4307` → `02139`. Anything non-digit left over becomes `""`.
- **Matching uses post-normalization last names** compared case-insensitively, so `SMITH` and `Smith` with the same phone merge.
- Rows that get dropped under rule 9 (no valid email or phone) still can't receive the email, so dropping them doesn't risk a missed opt-out—but any row that *does* merge carries its opt-out into the group, so one opted-out duplicate marks the whole person `Y`.
