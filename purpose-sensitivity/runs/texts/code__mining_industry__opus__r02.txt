import re
from datetime import datetime

_DISPLAY_EMAIL_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"(ext|x|#)", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP5_RE = re.compile(r"^[0-9]{5}$")
_ZIP4_RE = re.compile(r"^[0-9]{4}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _field(row, key):
    value = row.get(key, "")
    return "" if value is None else str(value)


def _normalize_email(raw):
    """Return the normalized email, or '' if it is not valid."""
    s = raw
    m = _DISPLAY_EMAIL_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"

    # Validity check (rule 2)
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local:
        return ""
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _normalize_phone(raw):
    """Return the normalized 10-digit phone, or '' if it is not valid."""
    s = raw
    m = _EXT_RE.search(s)
    if m:
        s = s[: m.start()]
    digits = "".join(ch for ch in s if ch in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) != 10 or digits[0] in "01":
        return ""
    return digits


def _normalize_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4_RE.match(s)
    if m:
        s = m.group(1)
    if _ZIP4_RE.match(s):
        s = "0" + s
    return s if _ZIP5_RE.match(s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email or phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": source,
            "tags": _parse_tags(_field(row, "tags")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
            "updated": _parse_updated(_field(row, "updated")),
            "is_manual": source == "manual",
        })

    # ---- Union-find over records (rule 10, transitive)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the root as the smaller position for stability
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone_last", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    # ---- Group records
    groups = {}
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---- Build one output dict per person (rule 11)
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["is_manual"], r["updated"], -r["index"]),
        )
        return best[field]

    people = []
    for group in groups.values():
        latest = max(r["updated"] for r in group)
        tags = set()
        for r in group:
            tags |= r["tags"]
        people.append((
            min(r["index"] for r in group),
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "zip": pick(group, "zip"),
                "tags": ";".join(sorted(tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
                "source": pick(group, "source"),
            },
        ))

    # ---- Order by earliest input row (rule 12)
    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

**How I read a few points the rules leave open:**

- **Extension markers:** The function cuts the phone at the first `x`, `ext`, or `#`, ignoring case. For example, `555-123-4567 ext. 12` becomes `5551234567`.
- **ZIP+4 with a dropped zero:** `2139-4307` is treated as ZIP+4 and padded to `02139`.
- **Gmail rules:** These apply only when the address has exactly one `@`. Addresses with more than one `@` are invalid anyway.
- **Bad `updated` values:** An unparseable date raises a `ValueError` naming the bad value. The rules say `updated` is always present, so I didn't drop the row silently.
- **Opt-outs:** If any row in a merged group has opted out, the whole person is marked `opted_out = "Y"`. Before you send, filter the output to `opted_out == "N"`. This function marks opt-outs but doesn't remove them.
