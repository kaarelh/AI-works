import re
from datetime import datetime


def dedupe_contacts(rows):
    DIGITS = "0123456789"

    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = re.search(r"<([^>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        m = re.search(r"x|ext|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        d = "".join(c for c in raw if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = raw.strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_dt(raw):
        s = raw.strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        try:
            return datetime.fromisoformat(s)
        except ValueError:
            return datetime.min

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize and drop rows with neither valid email nor valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        email = email if email_valid(email) else ""
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": get(row, "source").strip().lower(),
            "tags": parse_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_dt(get(row, "updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:

            else:
                email_seen[r["email"]] = i

            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # Groups ordered by earliest row (lists are already in input order).
    ordered = sorted(groups.values(), key=lambda g: g[0])

    for g in ordered:

        def pick(field):
            for r in ranked:
                if r[field]:

Notes on a few judgment calls:

- **Opt-outs:** a person is opted out if any of their merged rows is opted out. Before sending, filter the output to `opted_out == "N"`.
- **Dropped rows:** these don't count toward output order, since rule 9 applies before matching.
- **Unparseable `updated`:** a value that matches none of the three formats is treated as the earliest possible date rather than raising an error.
- **Phone extensions:** I look for the first "x", "ext", or "#" anywhere in the string, per rule 3. A phone containing a stray letter "x" would be truncated at that point.
- **Transitive matching:** this uses union-find, so chains like A–B–C collapse into one person.
