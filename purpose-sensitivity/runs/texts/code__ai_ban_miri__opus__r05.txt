```python
import re
from datetime import datetime

_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_EXT_MARKER = re.compile(r"ext|x|#", re.IGNORECASE)
_ANGLE_EMAIL = re.compile(r"<([^>]*)>")
_ZIP_PLUS4 = re.compile(r"^(\d{4,5})-\d{4}$")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _norm_email(raw):
    s = raw or ""
    m = _ANGLE_EMAIL.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
            s = local + "@" + domain
    return s


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw or ""
    m = _EXT_MARKER.search(s)
    if m:
        s = s[: m.start()]
    digits = "".join(ch for ch in s if ch.isdigit())
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(phone):
    return len(phone) == 10 and phone[0] not in "01"


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4.match(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and s.isdigit():
        s = "0" + s
    if len(s) == 5 and s.isdigit():
        return s
    return ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError("Unrecognized 'updated' value: %r" % raw)


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # Normalize every row and drop those with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        phone = _norm_phone(row.get("phone", ""))
        email_ok = _email_is_valid(email)
        phone_ok = _phone_is_valid(phone)
        if not (email_ok or phone_ok):
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Union-find over the surviving records (positions in recs).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the root as the smaller position (earlier in input).
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_key = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_by_key:
                union(first_by_key[key], i)
            else:
                first_by_key[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    def pick(members, field):
        candidates = [m for m in members if m[field]]
        if not candidates:
            return ""
        best = max(candidates, key=lambda m: (m["manual"], m["updated"], -m["idx"]))
        return best[field]

    output = []
    # Each group's members are in input order, so members[0] is its earliest row.
    for members in sorted(groups.values(), key=lambda ms: ms[0]["idx"]):
        tags = set()
        for m in members:
            tags |= m["tags"]
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(m["updated"] for m in members).strftime("%Y-%m-%d"),
        })
    return output
```

**How I handled the ambiguous parts:**
- **Extension markers:** The function cuts the phone number at the earliest of `ext`, `x`, or `#`. For example, `555-123-4567 ext. 89` becomes `5551234567`.
- **ZIP+4:** It accepts `12345-6789` and also `2139-4307`, where Excel dropped the leading zero. Both are trimmed to the part before the hyphen and then padded to 5 digits.
- **Gmail cleanup that leaves nothing:** If removing dots and the `+` suffix leaves an empty local part, as with `+promo@gmail.com`, the email counts as invalid.
- **Emails with more than one @:** These skip the Gmail cleanup and are treated as invalid.
- **Unparseable dates:** If an `updated` value matches none of the three formats, the function raises an error rather than guessing.
- **Last names:** Matching compares normalized last names with `casefold()`.

**Before you send:**
1. **The output still includes people who opted out.** They're marked `opted_out: "Y"`, so you need to filter to `opted_out == "N"` before building the send list.
2. **Some people don't have an email.** People who matched only by phone may have `email == ""`. Remove them from the email send too.
3. **Some opt-outs may not be caught.** Rule 9 drops a row that has no valid email or phone, even if it says the person opted out. If that row's raw email is just slightly malformed, the person's other rows could still go out as "N". If you have a separate suppression list, check the final list against it as well.
4. **Gmail addresses are rewritten.** They come out in cleaned-up form, e.g. `j.doe+miri@googlemail.com` becomes `jdoe@gmail.com`. Gmail delivers these the same way, but your email platform may not match them to its own unsubscribe records. If it keeps a separate suppression list, it's worth checking against that too.
