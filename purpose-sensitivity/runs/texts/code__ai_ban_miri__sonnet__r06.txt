import re
from datetime import datetime


def dedupe_contacts(rows):
    def s(v):
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        e = s(raw)
        if "<" in e:
            after = e.split("<", 1)[1]
            if ">" in after:
                e = after.split(">", 1)[0]
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        p = s(raw)
        m = re.search(r"ext|x|#", p, re.IGNORECASE)
        if m:
            p = p[:m.start()]
        digits = re.sub(r"[^0-9]", "", p)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-", 1)[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n and (n.isupper() or n.islower()):
            n = n.title()
        return n

    def parse_dt(raw):
        u = s(raw).strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                pass
        return datetime.min

    def is_opted_out(raw):
        return s(raw).strip().lower() in ("y", "yes", "true", "1")

    def parse_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize rows, drop those with no valid contact ----------
    recs = []
    for idx, r in enumerate(rows):
        email = norm_email(r.get("email"))
        phone = norm_phone(r.get("phone"))
        if not email and not phone:
            continue
        source = s(r.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(r.get("first_name")),
            "last_name": norm_name(r.get("last_name")),
            "zip": norm_zip(r.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "dt": parse_dt(r.get("updated")),
            "tags": parse_tags(r.get("tags")),
            "opted_out": is_opted_out(r.get("opted_out")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        root = x
        while parent[root] != root:
            root = parent[root]
        while parent[x] != root:
            parent[x], x = root, parent[x]
        return root

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # ---------- matching ----------
    by_email = {}
    by_phone_last = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            k = rec["email"]
            if k in by_email:
                union(i, by_email[k])
            else:
                by_email[k] = i
        if rec["phone"] and rec["last_name"]:
            k = (rec["phone"], rec["last_name"].lower())
            if k in by_phone_last:
                union(i, by_phone_last[k])
            else:
                by_phone_last[k] = i

    groups = {}
    order = []
    for i in range(n):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(i)

    # ---------- build output ----------
    def pick(members, field):
        best = None
        best_key = None
        for i in members:
            rec = recs[i]
            if not rec[field]:
                continue
            key = (rec["manual"], rec["dt"], -rec["idx"])
            if best_key is None or key > best_key:
                best, best_key = rec[field], key
        return best if best is not None else ""

    result = []
    for root in order:  # recs are in input order, so first-seen = earliest row
        members = groups[root]
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["dt"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.date().isoformat(),
        })
    return result

Notes on interpretation choices:

- **Opt-outs:** If any row in a merged group is opted out, the whole person is marked `Y`. When you build the send list, filter on `opted_out == "N"`.
- **ZIP+4:** I treated "looks like ZIP+4" strictly as 5 digits, a hyphen, then 4 digits. Something like `2139-4307` becomes `""`.
- **Gmail:** If stripping dots and the `+` suffix leaves an empty local part (e.g. `+news@gmail.com`), the email is invalid.
- **Extensions:** The marker search cuts at the earliest `ext`, `x`, or `#`, so a phone like `617-555-0100 ext. 12` is read as `6175550100`.
- **Unparseable `updated` values:** The spec says these shouldn't occur. If one does, it is treated as the oldest possible date rather than raising an error.
