import re
from datetime import datetime


def dedupe_contacts(rows):
    # ---------- normalizers ----------
    def norm_email(raw):
        s = (raw or "")
        m = re.search(r"<([^>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"x|ext|#", s, re.IGNORECASE)
        if m:
            s = s[:m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]{5})-([0-9]{4})", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_dt(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized updated format: %r" % raw)

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", raw or ""):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize rows, drop rows with no valid contact ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_dt(row.get("updated")),
        })

    # ---------- union-find over recs ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("e", r["email"]))
        last = r["last_name"].lower()
        if r["phone"] and last:
            keys.append(("p", r["phone"], last))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------- build output ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    ordered = sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m))

    result = []
    for members in ordered:
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result

Notes on a few interpretation choices:

- **Opted-out people are kept in the output** with `opted_out == "Y"`, per rule 11. Before sending, filter them out, e.g. `[p for p in people if p["opted_out"] == "N" and p["email"]]`. People with no valid email (phone only) also appear, so you'll want that email filter.
- **Opt-outs are group-wide.** If any row matched to a person opted out, the whole person is flagged, which is the safe behavior for your concern.
- **Dropped rows (rule 9) are removed before matching**, so an opt-out on a row with no valid email or phone can't propagate. If you want opt-outs on such rows honored, tell me and I can handle that separately.
- **ZIP+4** is recognized strictly as 5 digits, a hyphen, then 4 digits. Something like `2139-4307` becomes `""`.
- **Extension markers** are matched anywhere in the phone string, as the rules say, so a stray "x" or "ext" cuts the number at that point.
- **Bad `updated` values** raise a `ValueError` rather than being silently guessed at. This is deliberate, so you'd notice a malformed export before sending.
