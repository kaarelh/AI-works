import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EXT_MARKERS = ("x", "ext", "#")
_TRUE_VALUES = {"y", "yes", "true", "1"}
_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"[0-9]{4,5}-[0-9]{4}")


def _normalize_email(raw):
    s = raw.strip()
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = local + "@" + domain
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return None
    local, domain = s.split("@")
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return None
    return s


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in _EXT_MARKERS:
        i = s.find(marker)
        if i != -1:
            cut = min(cut, i)
    digits = "".join(c for c in s[:cut] if "0" <= c <= "9")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return None


def _normalize_zip(raw):
    s = raw.strip()
    if _ZIP_PLUS4_RE.fullmatch(s):
        s = s.split("-")[0]
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    if re.fullmatch(r"[0-9]{5}", s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return datetime.min  # should not happen per the spec; sorts as oldest


def dedupe_contacts(rows):
    # 1. Normalize each row and drop rows with no valid email or phone.
    recs = []
    for row in rows:
        email = _valid_email(_normalize_email(row.get("email", "")))
        phone = _valid_phone(_normalize_phone(row.get("phone", "")))
        if email is None and phone is None:
            continue
        first = _normalize_name(row.get("first_name", ""))
        last = _normalize_name(row.get("last_name", ""))
        source = row.get("source", "").strip().lower()
        recs.append({
            "email": email or "",
            "phone": phone or "",
            "first_name": first,
            "last_name": last,
            "last_key": last.lower(),
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "updated": _parse_updated(row.get("updated", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _TRUE_VALUES,
            "tags": [t.strip().lower()
                     for t in re.split(r"[;,]", row.get("tags", ""))
                     if t.strip()],
            "pos": len(recs),  # position in input order (after dropped rows)
        })

    # 2. Union-find so that matching is transitive.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], i)
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_key"]:
            key = (r["phone"], r["last_key"])
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    # 3. Group rows. Dict insertion order = order of each person's earliest row.
    groups = {}
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # 4. Build one output record per person.
    result = []
    for members in groups.values():
        out = {}
        for field in ("first_name", "last_name", "zip", "email", "phone", "source"):
            cands = [r for r in members if r[field]]
            if cands:
                best = min(
                    cands,
                    key=lambda r: (not r["manual"],
                                   -(r["updated"] - datetime.min),
                                   r["pos"]),
                )
                out[field] = best[field]
            else:
                out[field] = ""

        out["tags"] = ";".join(sorted({t for r in members for t in r["tags"]}))
        out["opted_out"] = "Y" if any(r["opted_out"] for r in members) else "N"
        out["updated"] = max(r["updated"] for r in members).strftime("%Y-%m-%d")
        result.append(out)

    return result

Notes on a few interpretation choices:

- **ZIP+4:** I accept 4 or 5 digits before the hyphen, so an Excel-mangled value like `2139-4307` becomes `02139`. Otherwise the leading-zero fix would never apply to ZIP+4 values.
- **Extension markers:** the cut happens at the earliest marker, so `555 ext 123` and `555x123` both stop at the marker.
- **Last-name matching:** last names are compared case-insensitively. Contacts without a last name can still match on email, but never on phone alone.
- **Unparseable `updated`:** the spec says it's always valid, so I treat any unparseable value as the oldest possible date rather than raising an error.
