```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows per the data team's normalization/matching rules."""

    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    EPOCH = datetime.min

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        """Returns the 10-digit string if valid, else ''."""
        s = (raw or "").lower()
        cut = len(s)
        for marker in ("ext", "x", "#"):
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                               # ZIP+4 -> keep the 5-digit part
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):           # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return (raw or "").strip().lower() in {"y", "yes", "true", "1"}

    def split_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize rows, drop unusable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        dt = parse_updated(row.get("updated", ""))
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": split_tags(row.get("tags", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "dt": dt,
            "ts": (dt - EPOCH).total_seconds(),
            "source": (row.get("source", "") or "").strip().lower(),
        })

    # ---------- union-find for transitive matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)   # keep the earliest row as root

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}          # root -> list of recs, in input order
    order = []           # roots in order of earliest row (rule 12)
    for i, r in enumerate(recs):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(r)

    # ---------- build one output dict per person (rule 11) ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = min(cands, key=lambda r: (0 if r["source"] == "manual" else 1,
                                         -r["ts"],
                                         r["idx"]))
        return best[field]

    out = []
    for root in order:
        g = groups[root]
        all_tags = set().union(*(r["tags"] for r in g))
        out.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "zip": pick(g, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["dt"] for r in g).strftime("%Y-%m-%d"),
            "source": pick(g, "source"),
        })
    return out
```

**Judgment calls worth a quick check with your data person:**

1. **ZIP+4 detection** – I treat *any* hyphen as the ZIP+4 separator and keep what's before it. A value like "0213-9" would become "02139"; the rule doesn't say to reject that, so I left it permissive.
2. **Unparseable `updated`** – the spec says it's never empty and always one of three formats, so I raise a `ValueError` with the offending value rather than silently guessing. Better to crash the dry run tonight than send duplicates tomorrow.
3. **Phone output** – the "normalized value" is the bare 10-digit string (e.g. `2025551234`), not reformatted.
4. **Names** – `str.title()` has known quirks on names with apostrophes/particles (`"MCDONALD"` → `"Mcdonald"`), but that's what the rule specifies.

A quick sanity test before the send: run it on a few hand-picked rows you know are duplicates (gmail dots/plus, display-name emails, `1-xxx` phones with extensions) and confirm each collapses to one record with `opted_out` set if any copy was opted out.
