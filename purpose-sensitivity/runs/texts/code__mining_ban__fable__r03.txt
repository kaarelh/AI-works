import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _s(row, key):
    """Field as a string ('' if missing/None)."""
    return row.get(key) or ""


def _normalize_email(raw):
    s = raw
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("x", "#"):          # "x" also catches "ext"
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = re.sub(r"[^0-9]", "", s[:cut])
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_valid(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    if "-" in s:
        s = s.split("-", 1)[0].strip()
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        return s.title()
    return s


def _parse_updated(raw, idx):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"row {idx}: unrecognized 'updated' value {raw!r}")


def _normalize_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- 1. Normalize every row, dropping rows with no valid email or phone ----
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_s(row, "email"))
        phone = _normalize_phone(_s(row, "phone"))
        email_ok = _email_valid(email)
        phone_ok = _phone_valid(phone)
        if not email_ok and not phone_ok:
            continue
        source = _s(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(_s(row, "first_name")),
            "last_name": _normalize_name(_s(row, "last_name")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "zip": _normalize_zip(_s(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "updated": _parse_updated(_s(row, "updated"), idx),
            "opted_out": _s(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "tags": _normalize_tags(_s(row, "tags")),
        })

    # ---- 2. Union-find to group rows into people (transitive matching) ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)   # keep the earliest row as root

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    # Groups in order of earliest row (recs is already in input order)
    groups = {}
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---- 3. Build one output dict per person ----
    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        # prefer manual source, then latest updated, then earliest input row
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    out = []
    for group in groups.values():
        tags = set().union(*(r["tags"] for r in group))
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return out

A few notes on choices where the rules left a little room:

- **Unparseable `updated`** raises a `ValueError` naming the row index rather than silently guessing, since you said that field is always present. If the export turns out to have blanks, tell me and I'll make it fall back instead.
- **Extension markers:** truncating at the first `x` or `#` also covers `ext` (its `x`), so `555-123-4567 ext 12` → `5551234567`.
- **ZIP hyphens:** anything before the first hyphen is kept, then the 4-digit/5-digit checks apply, so `2139-4307` → `02139` and junk like `abc-1` → `""`.
- **Rule 9 happens first**, so a row with no valid email/phone never contributes tags, opt-out status, or dates to anyone — matching the spec's "before matching."
- Since you're sending to people who haven't opted out, filter the result on `opted_out == "N"` before loading it into your email tool.
