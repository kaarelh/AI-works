import re
from datetime import datetime

_DIGITS = "0123456789"
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP4_RE = re.compile(r"^([0-9]{5})-[0-9]{4}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _s(value):
    return "" if value is None else str(value)


def _norm_email(raw):
    s = _s(raw)
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        s = local + "@" + domain
    # validity
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local:
        return ""
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _norm_phone(raw):
    s = _s(raw)
    m = _EXT_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = _s(raw).strip()
    m = _ZIP4_RE.match(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and all(c in _DIGITS for c in s):
        s = "0" + s
    if len(s) == 5 and all(c in _DIGITS for c in s):
        return s
    return ""


def _norm_name(raw):
    s = _s(raw).strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = _s(raw).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError("Unrecognized updated value: %r" % raw)


def _is_opted_out(raw):

def _parse_tags(raw):

        if t:

def dedupe_contacts(rows):
    # Normalize rows; drop those with neither valid email nor valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email"))
        phone = _norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = _s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip")),
            "source": source,
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Union-find over the kept records.
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            key = r["email"]
            if key in email_seen:
                union(i, email_seen[key])
            else:
                email_seen[key] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    epoch = datetime.min

    def pick(members, field):
        best = None
        best_key = None
        for i in members:
            r = recs[i]
            if not r[field]:
                continue
            key = (
                0 if r["source"] == "manual" else 1,
                -(r["updated"] - epoch).total_seconds(),
                r["idx"],
            )
            if best_key is None or key < best_key:
                best_key = key
                best = r[field]
        return best if best is not None else ""

    # Members are in input order (indices ascending), so the first member is
    # the earliest row of the group; order groups by that.
    ordered = sorted(groups.values(), key=lambda m: recs[m[0]]["idx"])

    result = []
    for members in ordered:
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),

            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result

Notes on judgment calls:

- **Opt-outs:** If any row in a merged person has opted out, the whole person is `"Y"`. Filter on `opted_out == "N"` before sending tomorrow's alert.
- **Merging:** Matching is transitive via union-find, so chains like A–B by email and B–C by phone+last name collapse into one person.
- **Dropped rows:** Rows with no valid email or phone are discarded before matching and contribute nothing (not even tags or opt-out flags). They have nothing to match on anyway.
- **Gmail rule:** It only applies when the address has exactly one `@`. An address like `+foo@gmail.com` normalizes to an empty local part and is invalid.
- **Extension markers:** The earliest of `x`, `ext`, `#` is the cut point, so `"555-123-4567 ext 12"` cuts at the `e`.
- **Bad dates:** An `updated` value that matches none of the three formats raises `ValueError` rather than being silently guessed. If your export may contain junk there, tell me how you'd like it handled.
- **Fields outside the picker:** `source` is picked among rows with a non-empty source, using the same manual > most recent > earliest ordering.
