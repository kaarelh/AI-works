```python
import re
from datetime import datetime

_EMAIL_BRACKET_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_TRUE_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}


def _normalize_email(raw):
    """Return the normalized email, or "" if it is not valid."""
    s = raw.strip()
    m = _EMAIL_BRACKET_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = f"{local}@gmail.com"

    # Validity: exactly one "@", non-empty local part, and a domain
    # that contains a dot and does not start or end with one.
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _normalize_phone(raw):
    """Return the 10-digit normalized phone, or "" if it is not valid."""
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Return the normalized 5-digit ZIP, or "" if it cannot be normalized."""
    z = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def dedupe_contacts(rows):
    # Step 1: normalize every row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email") or "")
        phone = _normalize_phone(row.get("phone") or "")
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        tags_raw = row.get("tags") or ""
        tags = {t.strip().lower() for t in re.split(r"[;,]", tags_raw) if t.strip()}
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name") or ""),
            "last_name": _normalize_name(row.get("last_name") or ""),
            "zip": _normalize_zip(row.get("zip") or ""),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": (row.get("opted_out") or "").strip().lower() in _TRUE_VALUES,
            "updated": _parse_updated(row.get("updated") or ""),
        })

    # Step 2: union-find so that matches are transitive.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the earlier row as the root so group order is stable.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_index = {}
    phone_index = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_index:
                union(i, email_index[r["email"]])
            else:
                email_index[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in phone_index:
                union(i, phone_index[key])
            else:
                phone_index[key] = i

    # Group members, in input order; groups are ordered by first member.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)
    ordered_groups = sorted(groups.values(), key=lambda members: members[0])

    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        # cands is already in input order. Sort stably by each criterion from
        # least to most significant: input order, then most recent updated,
        # then manual-first.
        cands.sort(
            key=lambda r: r["updated"] if r["updated"] is not None else datetime.min,
            reverse=True,
        )
        cands.sort(key=lambda r: not r["manual"])
        return cands[0][field]

    # Step 3: build one output row per person.
    result = []
    for members in ordered_groups:
        dates = [recs[i]["updated"] for i in members if recs[i]["updated"] is not None]
        all_tags = set().union(*(recs[i]["tags"] for i in members))
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(dates).strftime("%Y-%m-%d") if dates else "",
        })
    return result
```

Notes on the implementation:

- **Opt-outs:** `opted_out` is "Y" if any row in a person's group has opted out. Filter `opted_out == "Y"` out of the send list, not only the rows that opted out.
- **ZIP+4 pattern:** I accept a 4- or 5-digit prefix before the hyphen, so `2139-4307` becomes `02139`. The spec doesn't say whether that case should count as ZIP+4. Tell me if you want it limited to exactly five digits.
- **Missing or unparseable `updated`:** The spec says every row has a valid `updated`, so I didn't add a fallback value. If a row has an unparseable date, it's treated as the oldest, and a group with no parseable dates gets `""`.
