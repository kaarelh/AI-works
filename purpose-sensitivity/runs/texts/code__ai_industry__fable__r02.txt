```python
import re
from datetime import datetime

_UPDATED_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_EXT_MARKER_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_FIVE_DIGITS_RE = re.compile(r"[0-9]{5}")


def dedupe_contacts(rows):
    # ---------- field normalizers ----------
    def norm_email(raw):
        e = raw or ""
        m = _DISPLAY_NAME_RE.search(e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if (not local or "." not in domain
                or domain.startswith(".") or domain.endswith(".")):
            return ""
        return f"{local}@{domain}"

    def norm_phone(raw):
        p = _EXT_MARKER_RE.split(raw or "", maxsplit=1)[0]
        digits = re.sub(r"[^0-9]", "", p)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = (raw or "").strip()
        if "-" in z:  # ZIP+4 -> keep the part before the hyphen
            z = z.split("-", 1)[0].strip()
        if len(z) == 4 and z.isdigit():
            z = "0" + z
        return z if _FIVE_DIGITS_RE.fullmatch(z) else ""

    def norm_name(raw):
        n = (raw or "").strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in _UPDATED_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def norm_tags(raw):
        return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw or "") if t.strip()}

    # ---------- normalize every row; drop rows with no valid email or phone ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", "")),
            "source": (row.get("source", "") or "").strip().lower(),
        })

    # ---------- union-find for transitive matching ----------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in seen:
                union(seen[k], i)
            else:
                seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # ---------- build one output row per person ----------
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]))
        return best[field]

    out = []
    for members in sorted(groups.values(), key=lambda m: recs[m[0]]["idx"]):
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
        })
    return out
```

A few notes on how I resolved details the rules left open:

- **Phone extensions:** I split at the first occurrence of `x`, `ext`, or `#` (case-insensitive). Since only digits are kept afterward, `ext` vs `x` ordering doesn't matter.
- **ZIP+4:** anything containing a hyphen is treated as ZIP+4 and truncated at the hyphen, so `"2139-4307"` (leading zero eaten) still becomes `"02139"`.
- **Bad `updated` values:** the function raises `ValueError` naming the offending value rather than silently guessing, since a wrong date would silently change which row "wins" for each field.
- **Missing fields / `None`:** treated as `""`, so a slightly ragged export won't crash it.
- **Field selection:** `pick` only considers rows where the normalized value is non-empty (valid, for email/phone), then prefers `source == "manual"`, then latest `updated`, then earliest input row.

Because the function keeps any opt-out from any row in a person's group, you'll want to filter on `opted_out == "N"` before sending.
