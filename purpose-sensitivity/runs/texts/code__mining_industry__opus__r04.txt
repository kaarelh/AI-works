import re
from datetime import datetime

_FIELDS = ("first_name", "last_name", "email", "phone", "zip",
           "tags", "opted_out", "updated", "source")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _normalize_email(raw):
    """Rule 1: extract from display name, strip, lowercase, canonicalize Gmail."""
    s = raw or ""
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    """Rule 2."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _normalize_phone(raw):
    """Rule 3. Returns the 10-digit phone if valid, else ""."""
    s = raw or ""
    # Earliest extension marker; "ext" is tried before "x" at the same position.
    m = re.search(r"ext|x|#", s, re.IGNORECASE)
    if m:
        s = s[:m.start()]
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6."""
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    """Rule 8 (plus the normalization from rule 11)."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row -------------------------------------------
    records = []
    for idx, row in enumerate(rows):
        get = lambda k: row.get(k) or ""
        email = _normalize_email(get("email"))
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(get("phone"))

        # Rule 9: drop rows with neither a valid email nor a valid phone.
        if not email and not phone:
            continue

        source = get("source").strip().lower()
        records.append({
            "idx": idx,
            "first_name": _normalize_name(get("first_name")),
            "last_name": _normalize_name(get("last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(get("zip")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _parse_tags(get("tags")),
            "opted_out": _is_opted_out(get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # ---- Union-find over records (rule 10, transitive) ------------------
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the lower index as root (convenient, not required).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].lower()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(records[i])

    # ---- Build one output dict per person (rule 11) ---------------------
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates,
                   key=lambda r: (r["is_manual"], r["updated"], -r["idx"]))
        return best[field]

    people = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        people.append((
            min(r["idx"] for r in group),
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "zip": pick(group, "zip"),
                "source": pick(group, "source"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            },
        ))

    # ---- Rule 12: order by earliest input row ---------------------------
    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

A few things to check before tomorrow's send:

- **This function marks opt-outs but doesn't remove them.** Following rule 11, opted-out people stay in the output with `opted_out: "Y"`. Filter them out before sending, for example `[p for p in result if p["opted_out"] == "N"]`.
- **Opt-outs only carry over through matching.** Suppose someone opted out under `ann@work.org` and also appears as `ann@home.org` with no shared phone and last name. Those records won't be linked, so the second one would still get the email. If you have a separate suppression list, apply it to the final emails as well.
- **Gmail rule side effect.** `j.smith@gmail.com` and `jsmith@gmail.com` are treated as the same person. That's correct for Gmail, but the email goes to whichever address the pick rules choose.
- **Bad dates stop the run.** The function raises a `ValueError` on an `updated` value it can't parse instead of guessing, so one malformed row will stop it with a clear message.
- **Spot-check the results.** Run it on the export and review a handful of merged groups by hand, especially phone and last-name merges, which are the most likely to join two different people.
