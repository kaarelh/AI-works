```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    # ---------------- field normalizers ----------------
    def norm_email(raw):
        s = str(raw or "")
        m = re.search(r"<([^<>]*)>", s)          # display-name form: use inside of <...>
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        s = str(raw or "")
        low = s.lower()
        cuts = [i for i in (low.find(mk) for mk in ("ext", "x", "#")) if i != -1]
        if cuts:
            s = s[:min(cuts)]                     # drop extension marker and everything after
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = str(raw or "").strip()
        s = s.split("-", 1)[0].strip()            # ZIP+4 -> keep part before hyphen
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s                           # restore leading zero eaten by Excel
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = str(raw or "").strip()
        if s and (s.isupper() or s.islower()):
            s = s.title()
        return s

    def parse_updated(raw):
        s = str(raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def norm_opted_out(raw):
        return str(raw or "").strip().lower() in ("y", "yes", "true", "1")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", str(raw or "")) if t.strip()}

    def norm_source(raw):
        return str(raw or "").strip().lower()

    # ---------------- normalize rows, drop unusable ones ----------------
    recs = []
    for row in rows:
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        email = email if email_valid(email) else ""
        phone = phone if phone_valid(phone) else ""
        if not email and not phone:
            continue                              # rule 9
        recs.append({
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": norm_opted_out(row.get("opted_out", "")),
            "updated": parse_updated(row.get("updated", "")),
            "source": norm_source(row.get("source", "")),
        })

    # ---------------- group with union-find (transitive matching) ----------------
    n = len(recs)
    parent = list(range(n))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen_email, seen_phone_last = {}, {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}                                   # insertion order = earliest row order
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---------------- build one output record per person ----------------
    def pick(members, field):
        cands = [i for i in members if recs[i][field]]
        if not cands:
            return ""
        best = min(cands, key=lambda i: (
            0 if recs[i]["source"] == "manual" else 1,   # prefer manual
            datetime.min - recs[i]["updated"],           # then most recent (negated timedelta)
            i,                                           # then earliest in input
        ))
        return recs[best][field]

    out = []
    for members in groups.values():
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return out
```

A few notes on choices where the rules left wiggle room:

- **Extension markers:** the cut point is the earliest occurrence of `ext`, `x`, or `#` (case-insensitive), so `555-1234 EXT 9` and `555-1234x9` both work. A letter "x" anywhere in the phone string will trigger a cut, which matches the rule as written.
- **`updated`:** your spec says it's always one of the three formats, so an unparseable value raises a `ValueError` naming the bad value rather than silently guessing. If you'd rather the job keep running on bad dates, swap the `raise` for `return datetime.min`.
- **Transitivity:** union-find makes rule 10 hold across chains (A↔B by email, B↔C by phone+last name ⇒ one person), and groups are emitted in order of their earliest input row, as in rule 12.
- **Missing keys:** `row.get(...)` is used throughout, so a row lacking a field is treated as an empty string instead of crashing.
