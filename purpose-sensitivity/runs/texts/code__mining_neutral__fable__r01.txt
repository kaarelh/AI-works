```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, merge, and deduplicate CRM contact rows (see rules 1-12)."""

    def text(v):
        return "" if v is None else str(v)

    # ---- Rule 1 & 2: email -------------------------------------------------
    def norm_email(raw):
        e = text(raw)
        m = re.search(r"<([^<>]*)>", e)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    # ---- Rule 3: phone -----------------------------------------------------
    def norm_phone(raw):
        p = text(raw).lower()
        cut = len(p)
        for marker in ("x", "ext", "#"):          # ignore from the earliest marker on
            i = p.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", p[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                                 # invalid -> empty

    # ---- Rule 4: zip -------------------------------------------------------
    def norm_zip(raw):
        z = text(raw).strip()
        if "-" in z:                               # ZIP+4 -> keep the part before the hyphen
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):           # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    # ---- Rule 5: names -----------------------------------------------------
    def norm_name(raw):
        n = text(raw).strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    # ---- Rule 6: updated ---------------------------------------------------
    def parse_updated(raw):
        u = text(raw).strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                pass
        return datetime.min                        # unparseable -> treated as oldest

    # ---- Rule 7 & 8: opted_out, tags --------------------------------------
    def is_opted_out(raw):
        return text(raw).strip().lower() in ("y", "yes", "true", "1")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", text(raw)) if t.strip()}

    # ---- Normalize every row; drop rows with no valid email or phone (Rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip")),
            "source": text(row.get("source")).strip().lower(),
            "tags": norm_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
        })

    # ---- Rule 10: transitive matching via union-find -----------------------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            parent[max(ri, rj)] = min(ri, rj)

    seen_email, seen_phone = {}, {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone:
                union(i, seen_phone[key])
            else:
                seen_phone[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Rule 11: build one output dict per person -------------------------
    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]))
        return best[field]

    people = []
    for group in groups.values():
        group.sort(key=lambda r: r["idx"])
        tags = set().union(*(r["tags"] for r in group))
        people.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })

    # ---- Rule 12: order by earliest input row ------------------------------
    return people
```

A few notes on judgment calls, in case your data person wants to adjust:

- **Output order (rule 12):** groups are keyed by their earliest row index and `groups` is built by scanning rows in order, so the output comes out ordered by earliest row without an extra sort.
- **Phone output:** the normalized phone is the bare 10-digit string (e.g. `2025551234`). If you'd rather have a formatted version, change the return in `norm_phone`.
- **Unparseable `updated`:** the spec says it's always one of the three formats, so I treat anything else as the oldest possible date rather than crashing the run the night before send. If you'd prefer a hard failure so bad dates get caught, swap `return datetime.min` for `raise ValueError(...)`.
- **ZIP:** a 9-digit ZIP without a hyphen (e.g. `021394307`) becomes `""` per rule 4; only the hyphenated ZIP+4 form is trimmed.
- **Rows with no valid email or phone** are dropped entirely before matching, so they don't contribute tags, opt-outs, or dates to any group — that's how rule 9 reads, but it means an opt-out on a row with a bad email and bad phone won't carry over. Worth a quick look at those dropped rows before sending.

For the send itself, you'd filter the result to `opted_out == "N"` and non-empty `email`.
