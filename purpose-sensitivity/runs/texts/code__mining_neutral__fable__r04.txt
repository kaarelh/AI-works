import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- field normalizers ----------
    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # display name: Ann Lee <ann@x.org>
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local)
                and "." in domain
                and not domain.startswith(".")
                and not domain.endswith("."))

    def norm_phone(raw):
        s = re.split(r"(?i)ext|x|#", raw or "", maxsplit=1)[0]   # cut at extension marker
        digits = re.sub(r"\D", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = (raw or "").strip()
        if "-" in z:                               # ZIP+4 -> keep part before hyphen
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", z):              # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"\d{5}", z) else ""

    def norm_name(raw):
        n = (raw or "").strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized 'updated' value: %r" % raw)

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    def opted_out(raw):
        return (raw or "").strip().lower() in OPT_OUT_VALUES

    # ---------- normalize every row, drop unusable ones (rule 9) ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        e_ok, p_ok = email_valid(email), phone_valid(phone)
        if not (e_ok or p_ok):
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email if e_ok else "",
            "phone": phone if p_ok else "",
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": opted_out(row.get("opted_out", "")),
            "updated": parse_updated(row.get("updated", "")),
            "source": source,
            "is_manual": source == "manual",
        })

    # ---------- union-find grouping (rule 10, transitive) ----------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    key_to_rec = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_to_rec:
                union(i, key_to_rec[k])
            else:
                key_to_rec[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build one output dict per person (rule 11) ----------
    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["is_manual"], r["updated"], -r["idx"]))
        return best[field]

    out = []
    for root in sorted(groups, key=lambda root: min(r["idx"] for r in groups[root])):
        g = groups[root]
        tags = set()
        for r in g:
            tags |= r["tags"]
        out.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "zip": pick(g, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["updated"] for r in g).strftime("%Y-%m-%d"),
            "source": pick(g, "source"),
        })
    return out

A few notes on judgment calls, so you can adjust before tomorrow's send:

- **`updated` parsing is strict.** Your spec says it's never empty and always in one of the three formats, so the function raises a `ValueError` naming the bad value rather than silently guessing (a wrong guess could make the "most recent" row win the wrong field). If the export turns out to have blanks, tell me and I'll add a fallback.
- **Phone matching requires a non-empty last name** on both rows, per rule 10; two rows sharing a phone but with no last name are not merged unless they also share an email.
- **Rows dropped under rule 9** contribute nothing to any group (no tags, no opt-out flag). If someone opted out on a row that has neither a valid email nor phone, that opt-out is lost — which is fine for emailing purposes since that row couldn't be reached anyway, but worth knowing.
- ZIP+4 is detected by the presence of a hyphen (so `"2139-4307"` → `"02139"` also works).
