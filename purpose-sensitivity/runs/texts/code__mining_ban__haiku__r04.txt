```python
import re
from datetime import datetime

_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP4_RE = re.compile(r"[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EXT_MARKERS = ("x", "ext", "#")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_TAG_SPLIT_RE = re.compile(r"[;,]")


def _norm_email(raw):
    s = raw
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw.lower()
    cut = min(
        (i for i in (s.find(mk) for mk in _EXT_MARKERS) if i != -1),
        default=len(s),
    )
    s = s[:cut]
    digits = "".join(ch for ch in s if "0" <= ch <= "9")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(d):
    return len(d) == 10 and d[0] not in "01"


def _norm_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if _ZIP4_RE.fullmatch(s):
        s = "0" + s
    if _ZIP5_RE.fullmatch(s):
        return s
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _norm_tags(raw):
    tags = set()
    for t in _TAG_SPLIT_RE.split(raw):
        t = t.strip().lower()
        if t:
            tags.add(t)
    return tags


def _pick_key(rec):
    # Lower key wins: manual first, then most recent updated, then earliest input row.
    ts = rec["updated"]
    return (
        rec["source"] != "manual",
        ts is None,
        -ts.timestamp() if ts is not None else 0.0,
        rec["idx"],
    )


def dedupe_contacts(rows):
    # Step 1: normalize every row and drop those with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _valid_email(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))
        if not _valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "zip": _norm_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": row.get("source", "").strip().lower(),
            "tags": _norm_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Step 2: union-find so matching is transitive.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], i)
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    # Group members; dict order is order of each group's earliest row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        return min(cands, key=_pick_key)[field]

    # Step 3: build one output record per person.
    output = []
    for members in groups.values():
        group = [recs[i] for i in members]

        tags = set()
        for r in group:
            tags |= r["tags"]

        updates = [r["updated"] for r in group if r["updated"] is not None]
        latest = max(updates).strftime("%Y-%m-%d") if updates else ""

        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": latest,
        })

    return output
```

Notes on interpretation, since the rules leave a few cases open:

- **ZIP+4:** I accept a 4- or 5-digit prefix before the hyphen (so `2139-4307` becomes `02139`). If the data person meant only exact 5-digit prefixes, change `{4,5}` to `{5}` in `_ZIP_PLUS4_RE`.
- **Unparseable `updated`:** a value that matches none of the three formats is treated as older than any dated row. If every row in a group is unparseable, the output `updated` is `""`.
- **Empty `last_name`:** a phone-only match requires a non-empty last name, per rule 10. Rows that share a phone but have no last name stay separate unless they also share an email.
