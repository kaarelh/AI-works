```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP_RE = re.compile(r"^[0-9]{5}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        m = _ANGLE_RE.search(raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        m = _EXT_RE.search(raw)
        if m:
            raw = raw[:m.start()]
        digits = "".join(c for c in raw if c in _DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = _ZIP4_RE.match(z)
        if m:
            z = m.group(1)
        if len(z) == 4 and all(c in _DIGITS for c in z):
            z = "0" + z
        return z if _ZIP_RE.match(z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_date(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min

    def fmt_date(d):
        return "%04d-%02d-%02d" % (d.year, d.month, d.day)

    # Normalize rows and drop those with neither valid email nor valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        tags = set()
        for t in _TAG_SPLIT_RE.split(get(row, "tags")):
            t = t.strip().lower()
            if t:
                tags.add(t)
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_date(get(row, "updated")),
        })

    # Union-find over the kept records.
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # recs are in input order, so the earliest member is the first index.
    ordered = sorted(groups.values(), key=lambda g: g[0])

    out = []
    for members in ordered:
        group = [recs[i] for i in members]
        person = {}
        for field in ("first_name", "last_name", "zip", "email", "phone", "source"):
            best = None
            for r in group:
                if not r[field]:
                    continue
                if best is None:
                    best = r
                elif (r["manual"], r["updated"]) > (best["manual"], best["updated"]):
                    best = r  # strict, so ties keep the earlier row
            person[field] = best[field] if best else ""
        tags = set()
        for r in group:
            tags |= r["tags"]
        person["tags"] = ";".join(sorted(tags))
        person["opted_out"] = "Y" if any(r["opted_out"] for r in group) else "N"
        person["updated"] = fmt_date(max(r["updated"] for r in group))
        out.append(person)

    return out
```

Notes:
- **Opt-outs:** the function only flags a person as `opted_out == "Y"` if any of their duplicate rows opted out. It doesn't remove them, so filter on `"N"` before sending.
- **Unparseable dates:** an empty or malformed `updated` is treated as the oldest possible date, so it never wins a tie-break. It would show as `0001-01-01` only if every row in a group lacked a date.
- **ZIP+4:** I accept `NNNNN-NNNN` and also `NNNN-NNNN`, so an Excel-stripped zero before the hyphen still gets repaired.
- **Gmail:** if stripping dots and the `+` suffix leaves an empty local part (for example `+promo@gmail.com`), the email is treated as invalid.
