import re
from datetime import datetime

_DIGITS = "0123456789"
_BRACKET_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"^[0-9]{5}-[0-9]{4}$")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_TRUE_VALUES = {"y", "yes", "true", "1"}


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = _BRACKET_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        m = _EXT_RE.search(raw)
        if m:
            raw = raw[:m.start()]
        digits = "".join(c for c in raw if c in _DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = raw.strip()
        if _ZIP4_RE.match(s):
            s = s.split("-")[0]
        if len(s) == 4 and all(c in _DIGITS for c in s):
            s = "0" + s
        if len(s) == 5 and all(c in _DIGITS for c in s):
            return s
        return ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return None

        for part in re.split(r"[;,]", raw):

    def find(a):

Notes on choices:

- **Opt-outs:** If any row in a merged person is opted out, the whole person is `"Y"`. Filter on `opted_out == "N"` before sending the alert.
- **Row dropping:** Rows with no valid email and no valid phone are dropped first, so they never contribute tags, opt-outs, or dates to a group. A dropped row marked opted out therefore can't suppress anyone. If you want that, tell me and I'll change it.
- **Source field:** For the `source` field, "non-empty" means non-empty after stripping. The "manual" preference is judged on each row's own source.
- **Bad dates:** An unparseable `updated` is treated as the oldest possible date for ranking, rather than raising an error. If no row in a group parses, the output `updated` is `""`.
- **Transitive matching:** Union-find handles this, so chains like A–B via email and B–C via phone+last name collapse into one person.
- **Phone matching:** Phone matching requires a non-empty last name. Two rows with the same phone but blank last names are not merged unless they share an email.
