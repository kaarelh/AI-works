import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"
_OPTED_OUT_VALUES = {"y", "yes", "true", "1"}


def _get(row, key):
    return row.get(key) or ""


def _norm_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    local, at, domain = s.partition("@")
    if at and domain in _GMAIL_DOMAINS:
        local = local.replace(".", "")
        local = local.split("+", 1)[0]
        s = local + "@gmail.com"
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, _, domain = e.partition("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    z = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if len(z) == 4 and all(c in _DIGITS for c in z):
        z = "0" + z
    if len(z) == 5 and all(c in _DIGITS for c in z):
        return z
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(members, field):
    # Candidates are rows where the normalized field is non-empty.
    # Preference: source == "manual", then most recent updated, then earliest input row.
    cands = [r for r in members if r[field]]
    cands.sort(key=lambda r: r["idx"])                    # tiebreak: earlier input row
    cands.sort(key=lambda r: r["updated"], reverse=True)  # then most recent updated
    cands.sort(key=lambda r: not r["manual"])             # then manual rows first
    return cands[0][field] if cands else ""


def _merge(members):
    tags = set()
    for r in members:
        tags |= r["tags"]
    return {
        "first_name": _pick(members, "first_name"),
        "last_name": _pick(members, "last_name"),
        "zip": _pick(members, "zip"),
        "email": _pick(members, "email"),
        "phone": _pick(members, "phone"),
        "source": _pick(members, "source"),
        "tags": ";".join(sorted(tags)),
        "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
        "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        raw_email = _get(row, "email")
        email = _norm_email(raw_email)
        if not _valid_email(email):
            email = ""
        phone = _norm_phone(_get(row, "phone"))
        if not email and not phone:
            continue

        source_raw = _get(row, "source").strip().lower()
        records.append({
            "idx": idx,
            "first_name": _norm_name(_get(row, "first_name")),
            "last_name": _norm_name(_get(row, "last_name")),
            "zip": _norm_zip(_get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source_raw,
            "manual": source_raw == "manual",
            "tags": {t.strip().lower()
                     for t in re.split(r"[;,]", _get(row, "tags"))
                     if t.strip()},
            "opted_out": _get(row, "opted_out").strip().lower() in _OPTED_OUT_VALUES,
            "updated": _parse_updated(row["updated"]),
        })

    # Step 2: union-find. Same valid email, or same valid phone plus same
    # non-empty last name (case-insensitive), means same person.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the smaller index as root so each group's root is its earliest row.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for k, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in first_by_email:
                union(k, first_by_email[rec["email"]])
            else:
                first_by_email[rec["email"]] = k
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].casefold())
            if key in first_by_phone_last:
                union(k, first_by_phone_last[key])
            else:
                first_by_phone_last[key] = k

    # Step 3: group rows. Groups are inserted in order of their earliest row.
    groups = {}
    for k in range(len(records)):
        groups.setdefault(find(k), []).append(records[k])

    return [_merge(members) for members in groups.values()]
