import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _norm_email(raw):
    s = (raw or "").strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _norm_phone(raw):
    s = raw or ""
    low = s.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        pos = low.find(marker)
        if pos != -1:
            cut = min(cut, pos)
    digits = "".join(c for c in s[:cut] if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    z = (raw or "").strip()
    if re.fullmatch(r"[0-9]{4,5}-[0-9]{4}", z):
        z = z.split("-", 1)[0]
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    return None


def _is_opted_out(raw):
    return (raw or "").strip().lower() in ("y", "yes", "true", "1")


def _parse_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def _ts_key(dt):
    # Sort key: rows with a date rank above rows without one.
    return (dt is not None, dt if dt is not None else datetime.min)


def _pick(members, recs, field):
    cands = [i for i in members if recs[i][field]]
    if not cands:
        return ""
    # Stable sorts: first by most recent updated, then manual-first.
    # Ties keep input order because the original list is in index order.
    cands.sort(key=lambda i: _ts_key(recs[i]["updated"]), reverse=True)
    cands.sort(key=lambda i: recs[i]["source"] != "manual")
    return recs[cands[0]][field]


def dedupe_contacts(rows):
    recs = []
    for row in rows:
        email = _norm_email(row.get("email"))
        phone = _norm_phone(row.get("phone"))
        recs.append({
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email if _valid_email(email) else "",
            "phone": phone if _valid_phone(phone) else "",
            "source": (row.get("source") or "").strip().lower(),
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # Rows with neither a valid email nor a valid phone are dropped.
    valid = [i for i, r in enumerate(recs) if r["email"] or r["phone"]]

    # Union-find so matches are transitive.
    parent = {i: i for i in valid}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    email_first = {}
    phone_first = {}
    for i in valid:
        r = recs[i]
        if r["email"]:
            if r["email"] in email_first:
                union(i, email_first[r["email"]])
            else:
                email_first[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in phone_first:
                union(i, phone_first[key])
            else:
                phone_first[key] = i

    # Group by root. Dict insertion order follows the earliest row in each group.
    groups = {}
    for i in valid:
        groups.setdefault(find(i), []).append(i)

    result = []
    for members in groups.values():
        dates = [recs[i]["updated"] for i in members if recs[i]["updated"] is not None]
        tags = set().union(*(recs[i]["tags"] for i in members))
        result.append({
            "first_name": _pick(members, recs, "first_name"),
            "last_name": _pick(members, recs, "last_name"),
            "zip": _pick(members, recs, "zip"),
            "email": _pick(members, recs, "email"),
            "phone": _pick(members, recs, "phone"),
            "source": _pick(members, recs, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(dates).strftime("%Y-%m-%d") if dates else "",
        })
    return result
