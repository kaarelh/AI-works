```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the data team's rules (1-12)."""

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)              # rule 1: display name
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                domain = "gmail.com"
                s = local + "@" + domain
        return s

    def email_is_valid(e):                            # rule 2
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):                              # rule 3
        s = raw or ""
        low = s.lower()
        cut = len(s)
        for marker in ("x", "ext", "#"):
            i = low.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = "".join(ch for ch in s[:cut] if "0" <= ch <= "9")
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                                     # invalid

    def norm_zip(raw):                                # rule 4
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):                               # rule 5
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw, idx):                      # rule 6
        s = (raw or "").strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Row {idx}: unrecognized 'updated' value {raw!r}")

    def is_opted_out(raw):                            # rule 7
        return (raw or "").strip().lower() in _OPT_OUT_VALUES

    def split_tags(raw):                              # rule 8
        return [t.strip().lower() for t in re.split(r"[;,]", raw or "")
                if t.strip()]

    # ---------- normalize & filter (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        email = email if email_is_valid(email) else ""
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated"), idx),
        })

    # ---------- union-find matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output (rules 11-12) ----------

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(candidates,
                   key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    people = []
    for group in groups.values():
        tags = sorted({t for r in group for t in r["tags"]})
        people.append((min(r["idx"] for r in group), {
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        }))

    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]
```

Notes on a few judgment calls:

- **Extension markers (rule 3):** the phone is cut at whichever marker appears first. Since "ext" contains "x", "555-123-4567 ext 9" is cut at the "e", as intended.
- **ZIP+4 (rule 4):** a 4-digit ZIP+4 like `2139-4307` also has its prefix kept and is then zero-padded to `02139`, which handles Excel damage to ZIP+4 values too.
- **Bad dates:** an `updated` value in none of the three formats raises a `ValueError` naming the row. I did this instead of silently dropping the row, because a dropped row could take an opt-out with it.
- **Last-name matching** uses `casefold()`, which is a slightly stronger version of case-insensitive comparison.
- **Ties** under the "manual → most recent → earliest" rule go to the earliest row, via `-idx` inside `max`.

Before sending, check these:

1. **Filter the output.** The function returns everyone, including opted-out people, with `opted_out: "Y"`. The send list should be `[p for p in result if p["opted_out"] == "N" and p["email"]]`.
2. **Unmatched opt-outs can still slip through.** If someone opted out under one email and appears elsewhere under a different email with no matching phone and last name, the rules treat them as two people. If you keep a separate suppression list, apply it to the final send list as well.
3. **Rule 9 drops rows with no valid email or phone before matching.** Such rows couldn't be emailed anyway, but any opt-out flag on them is ignored. That's another reason to run a suppression-list check at the end.
