import re
from datetime import datetime

_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _normalize_email(raw):
    s = raw.strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = "".join(c for c in s[:cut] if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    if re.fullmatch(r"[0-9]{5}", s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _normalize_tags(raw):
    tags = set()
    for t in re.split(r"[;,]", raw):
        t = t.strip().lower()
        if t:
            tags.add(t)
    return tags


def _is_opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    # Unparseable dates are treated as the oldest possible value.
    return datetime.min


def _pick(group, field):
    """Choose the best non-empty value for `field` from a group of records.

    Preference: manual source first, then most recent `updated`, then earliest input row.
    """
    cands = [r for r in group if r[field]]
    if not cands:
        return ""
    # Stable sorts applied from least to most significant key.
    cands.sort(key=lambda r: r["idx"])
    cands.sort(key=lambda r: r["updated"], reverse=True)
    cands.sort(key=lambda r: not r["manual"])
    return cands[0][field]


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one output dict per person.

    Rows without a valid email or valid phone are dropped. Rows match if they share
    a valid normalized email, or a valid normalized phone plus the same non-empty
    normalized last name (case-insensitive). Matching is transitive.

    Returns a list of dicts ordered by each person's earliest input row.
    """
    # Step 1: normalize and filter rows.
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", "") or "")
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone", "") or "")
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        last_name = _normalize_name(row.get("last_name", "") or "")
        records.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "") or ""),
            "last_name": last_name,
            "zip": _normalize_zip(row.get("zip", "") or ""),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _normalize_tags(row.get("tags", "") or ""),
            "opted_out": _is_opted_out(row.get("opted_out", "") or ""),
            "updated": _parse_updated(row.get("updated", "") or ""),
        })

    # Step 2: union-find over matching keys (transitive grouping).
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].lower()))
        for key in keys:
            if key in first_seen:
                union(first_seen[key], i)
            else:
                first_seen[key] = i

    # Group records; dict insertion order follows each group's earliest row.
    groups = {}
    for i, rec in enumerate(records):
        groups.setdefault(find(i), []).append(rec)

    # Step 3: build one output row per person.
    result = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        latest = max(r["updated"] for r in group)
        result.append({
            "first_name": _pick(group, "first_name"),
            "last_name": _pick(group, "last_name"),
            "zip": _pick(group, "zip"),
            "email": _pick(group, "email"),
            "phone": _pick(group, "phone"),
            "source": _pick(group, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": "" if latest == datetime.min else latest.strftime("%Y-%m-%d"),
        })
    return result

- **Unparseable `updated` values:** The rules say every value is in one of three formats, so I sort any value that fails to parse as oldest instead of raising an error. If nothing in a group parses, `updated` is output as "".
- **Matching and opt-outs:** Rows dropped under rule 9 don't contribute tags, opt-out status, or any other field. So a row with no valid contact info can't mark someone as opted out.
