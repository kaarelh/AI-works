import re
from datetime import datetime

_DIGITS = "0123456789"
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_FOUR_DIGITS_RE = re.compile(r"[0-9]{4}")
_FIVE_DIGITS_RE = re.compile(r"[0-9]{5}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _text(value):
    return "" if value is None else str(value)


def _norm_email(raw):
    s = _text(raw)
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = _text(raw).lower()
    cut_points = [i for i in (s.find("x"), s.find("#"), s.find("ext")) if i != -1]
    if cut_points:
        s = s[:min(cut_points)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _norm_zip(raw):
    s = _text(raw).strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if _FOUR_DIGITS_RE.fullmatch(s):
        s = "0" + s
    return s if _FIVE_DIGITS_RE.fullmatch(s) else ""


def _norm_name(raw):
    s = _text(raw).strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _norm_tags(raw):
    parts = re.split(r"[;,]", _text(raw))
    return {p.strip().lower() for p in parts if p.strip()}


def _parse_updated(raw):
    s = _text(raw).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    return None


def _pick(members, field, usable):
    """Choose a value: manual source first, then most recent updated,
    then earliest input row. Returns "" if no row qualifies."""
    candidates = [m for m in members if usable(m)]
    # Stable sorts: newest first, then move "manual" rows to the front.
    # Equal keys keep input order, so the earliest row wins ties.
    candidates.sort(
        key=lambda m: m["updated"] if m["updated"] is not None else datetime.min,
        reverse=True,
    )
    candidates.sort(key=lambda m: m["source"] != "manual")
    return candidates[0][field] if candidates else ""


def dedupe_contacts(rows):
    # Normalize each row and drop rows with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email"))
        phone = _norm_phone(row.get("phone"))
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        recs.append({
            "index": idx,
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": _text(row.get("source")).strip().lower(),
            "tags": _norm_tags(row.get("tags")),
            "opted_out": _text(row.get("opted_out")).strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # Union-find: same valid email, or same valid phone + same last name.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], i)
            else:
                first_by_email[r["email"]] = i
        last_key = r["last_name"].casefold()
        if r["phone"] and last_key:
            key = (r["phone"], last_key)
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    # Group rows. Dict insertion order follows each group's earliest row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    result = []
    for members in groups.values():
        all_tags = set()
        for m in members:
            all_tags |= m["tags"]
        dated = [m["updated"] for m in members if m["updated"] is not None]

        result.append({
            "first_name": _pick(members, "first_name", lambda m: m["first_name"] != ""),
            "last_name": _pick(members, "last_name", lambda m: m["last_name"] != ""),
            "zip": _pick(members, "zip", lambda m: m["zip"] != ""),
            "email": _pick(members, "email", lambda m: m["email"] != ""),
            "phone": _pick(members, "phone", lambda m: m["phone"] != ""),
            "source": _pick(members, "source", lambda m: m["source"] != ""),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(dated).date().isoformat() if dated else "",
        })
    return result

Notes for the send:

- **The function keeps opted-out people and marks them `"Y"`; it does not remove them.** Before the send, filter with `opted_out == "N"`. Because opt-out is checked across the whole group, a person who opted out on any duplicate row is marked `"Y"`.
- **The alert is an email, so you probably also want `email != ""`.** Phone-only contacts come through with an empty email and would not receive the message.
- **The spec says `updated` is always present, so the function should not hit unparseable dates.** If one does appear, that row is ranked last for picking and ignored for the group's `updated` value. I'd rather you see that than have the function crash mid-export.
