import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # Ann Lee <ann@x.org>  ->  ann@x.org
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if domain in GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return local + "@" + domain

    def norm_phone(raw):
        s = (raw or "").lower()
        cut = len(s)
        for marker in ("x", "ext", "#"):          # ignore everything from the extension on
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"\D", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                              # ZIP+4 -> keep the ZIP5 part
            s = s.split("-", 1)[0].strip()
        if len(s) == 4 and s.isdigit():           # Excel ate the leading zero
            s = "0" + s
        return s if (len(s) == 5 and s.isdigit()) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s and (s.isupper() or s.islower()):
            return s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError("Unrecognized 'updated' value: %r" % (raw,))

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize rows, dropping unreachable ones ----------

    records = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:               # rule 9
            continue
        records.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", "")),
            "source": (row.get("source", "") or "").strip().lower(),
        })

    # ---------- union-find for transitive matching ----------

    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen = {}
    for i, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].lower()))
        for key in keys:
            if key in seen:
                union(i, seen[key])
            else:
                seen[key] = i

    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(records[i])

    # ---------- build one output row per person ----------

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda r: (r["source"] != "manual", -r["updated"].timestamp(), r["idx"]),
        )
        return best[field]

    out = []
    for root in sorted(groups, key=lambda r: groups[r][0]["idx"]):   # rule 12
        group = groups[root]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(set().union(*(r["tags"] for r in group)))),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return out

A few notes on how I resolved spots where the rules left room for interpretation:

- **Invalid emails/phones** are treated as `""` everywhere (matching, field selection, and output), so an invalid email never contributes to matching or to the output email even if it's on a "manual" row.
- **Phone matching** uses the normalized (title-cased) last name, compared lowercase, so `SMITH` and `Smith` match. Rows with a valid phone but an empty last name only match other rows via email.
- **Unparseable `updated` values raise** rather than silently guessing, since a bad date would quietly change which row's data "wins." If the export can contain blanks or other formats, let me know and I'll add a fallback.
- The `updated` comparison uses `.timestamp()` for the sort key; if any dates predate 1970 on Windows that can error, in which case swap it for `-r["updated"].toordinal()` plus seconds — unlikely for a CRM export, but flagging it.
