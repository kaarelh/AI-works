```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    # ---------- field normalizers ----------
    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)          # display name form: Ann Lee <ann@x.org>
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return bool(local) and "." in domain \
            and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        s = (raw or "").lower()
        s = re.split(r"[x#]", s, maxsplit=1)[0]  # "x", "ext", "#" all cut at the x/# position
        d = re.sub(r"\D", "", s)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                               # ZIP+4
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", s):             # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"\d{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # fallback; spec says one of the formats above always applies

    def norm_opted_out(raw):
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize rows, drop unmatched ones ----------
    recs = []
    for idx, row in enumerate(rows):
        g = lambda k: row.get(k, "") or ""
        email = norm_email(g("email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(g("phone"))
        if not email and not phone:                # rule 9
            continue
        source = g("source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(g("first_name")),
            "last_name": norm_name(g("last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(g("zip")),
            "tags": norm_tags(g("tags")),
            "opted_out": norm_opted_out(g("opted_out")),
            "updated": parse_updated(g("updated")),
            "source": source,
            "is_manual": source == "manual",
        })

    # ---------- union-find for transitive matching ----------
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email, by_phone_last = {}, {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])
    # recs are in input order, so group members (and thus first member) are already ordered
    ordered_groups = sorted(groups.values(), key=lambda grp: grp[0]["idx"])

    # ---------- build one output dict per person ----------
    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["is_manual"], r["updated"], -r["idx"]))
        return best[field]

    out = []
    for group in ordered_groups:
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return out
```

A few notes on how I interpreted the rules where they left a little room:

- **Phone output** is the normalized 10-digit string (e.g. `"7135550123"`), since that's what the matching uses. Say the word if you'd rather have it formatted.
- **Gmail normalization** only runs when the address has exactly one `@`; malformed emails are left alone and then fail validation anyway.
- **ZIP**: anything with a hyphen is treated as ZIP+4 and cut at the hyphen, then the 4-digit/5-digit checks apply, so `"2139-4307"` still becomes `"02139"`.
- **Tie-breaking** in `pick()` follows rule 11 exactly: manual source first, then latest `updated`, then earlier input row.
- For the send itself: filter on `opted_out == "N"` after deduping, since a group counts as opted out if *any* of its rows did — that's the whole point of merging before you check.
