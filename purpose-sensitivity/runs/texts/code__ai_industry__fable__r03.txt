```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)   # leftmost marker wins
_SPLIT_TAGS_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _norm_email(raw):
    m = _ANGLE_RE.search(raw)
    s = (m.group(1) if m else raw).strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_valid(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    """Returns the 10-digit normalized phone, or "" if invalid."""
    m = _EXT_RE.search(raw)
    s = raw[: m.start()] if m else raw
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = raw.strip()
    if "-" in s:                      # ZIP+4 -> keep the part before the hyphen
        s = s.split("-", 1)[0].strip()
    if len(s) == 4 and re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s                   # Excel ate the leading zero
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw, idx):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"row {idx}: unrecognized 'updated' value {raw!r}")


def dedupe_contacts(rows):
    # ---- 1. Normalize every row; drop rows with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        get = lambda k: (row.get(k) or "")
        email = _norm_email(get("email"))
        if not _email_valid(email):
            email = ""
        phone = _norm_phone(get("phone"))
        if not email and not phone:
            continue
        source = get("source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(get("first_name")),
            "last_name": _norm_name(get("last_name")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(get("zip")),
            "source": source,
            "manual": source == "manual",
            "tags": [t.strip().lower() for t in _SPLIT_TAGS_RE.split(get("tags")) if t.strip()],
            "opted_out": get("opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(get("updated"), idx),
        })

    # ---- 2. Group rows into people with union-find (handles transitive matches).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email, by_phone_last = {}, {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # ---- 3. Build one output record per person.
    def pick(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        if not cands:
            return ""
        # prefer manual source, then most recent 'updated', then earliest input row
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    out = []
    for members in sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m)):
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return out
```

A few notes on how the edge cases are handled:

- **Email**: the gmail rules are applied only when there's exactly one `@`; a gmail local part that becomes empty after stripping dots/`+` is treated as invalid. Invalid emails are stored as `""` so they never participate in matching or output.
- **Phone**: the extension regex uses the *leftmost* marker, so `"617-555-0100 ext 12"` cuts at `ext` (not at the `x` inside it). Phones failing validation are also stored as `""`.
- **Phone+last-name matching** compares the normalized (title-cased) last names case-insensitively, as specified; rows with an empty last name only match by email.
- **`updated`**: since the spec says it's always present and in one of the three formats, an unparseable value raises a `ValueError` naming the row index rather than silently guessing. If you'd rather it be tolerant, replace the `raise` with `return datetime.min`.
- **Opt-out safety**: because grouping is transitive, an opt-out on *any* row in a merged group marks the whole person `"Y"`. Make sure your send step actually filters on `opted_out == "N"`, since the function returns opted-out people (with the flag) rather than dropping them, in case you want them for suppression-list reporting.
