import re
from datetime import datetime

_DISPLAY_NAME_RE = re.compile(r"<([^<>]*)>")
_EXT_MARKER_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_ZIP_PLUS4_RE = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_ZIP5_RE = re.compile(r"^[0-9]{5}$")
_TAG_SPLIT_RE = re.compile(r"[;,]")

_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _field(row, key):
    value = row.get(key)
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = raw
    m = _DISPLAY_NAME_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    # Only apply Gmail rules to a well-formed single-@ address, so that
    # dropping "+..." can never turn an invalid address into a valid one.
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Return the valid 10-digit phone, or "" if invalid."""
    s = raw
    m = _EXT_MARKER_RE.search(s)  # leftmost marker wins
    if m:
        s = s[: m.start()]
    digits = _NON_DIGIT_RE.sub("", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4_RE.match(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and s.isascii() and s.isdigit():
        s = "0" + s
    return s if _ZIP5_RE.match(s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone.
    recs = []
    for index, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue  # rule 9
        recs.append({
            "index": index,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": _field(row, "source").strip().lower(),
            "tags": _parse_tags(_field(row, "tags")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
            "updated": _parse_updated(_field(row, "updated")),
        })

    # ---- Union-find over record positions (transitive matching, rule 10).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the smaller position as root (not required, just tidy).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_with_key = {}
    for pos, rec in enumerate(recs):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_with_key:
                union(pos, first_with_key[key])
            else:
                first_with_key[key] = pos

    # ---- Group records, ordered by each person's earliest input row (rule 12).
    groups = {}
    order = []
    for pos, rec in enumerate(recs):  # recs are already in input order
        root = find(pos)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(rec)

    # ---- Build one output dict per person (rule 11).
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["index"]),
        )
        return best[field]

    result = []
    for root in order:
        group = groups[root]
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        latest = max(r["updated"] for r in group)
        result.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": latest.date().isoformat(),
        })
    return result

A few notes before tomorrow's send:

- **Filter out opt-outs before sending.** The function keeps opted-out people in its output and marks them `opted_out == "Y"`. That way, a person who opted out under one email address stays flagged even if they're merged with another record. Build the send list as `[p for p in result if p["opted_out"] == "N" and p["email"]]`.
- **One email per person.** If records were merged through a matching phone and last name, the group can contain several different email addresses. Only the one chosen by the rule-11 preference order is sent, which is what you want for "nobody gets it twice."
- **Bad dates raise an error.** If an `updated` value doesn't match any of the three formats, the function raises `ValueError` and names the bad value. I did this instead of guessing or skipping the row, because a skipped row could hide an opt-out.
- **Interpretation choices** where the rules were open:
  - The Gmail cleanup only applies to addresses with exactly one `@`. This keeps something like `a+x@b@gmail.com` from becoming valid.
  - A ZIP+4 with a missing leading zero (`2139-4307`) is padded to `02139`.
  - Phone and ZIP checks count only ASCII digits.
