```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_TRUE_VALUES = {"y", "yes", "true", "1"}
_EXT_MARKERS = ("x", "ext", "#")
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4_RE = re.compile(r"[0-9]{4,5}-[0-9]{4}")
_TAG_SPLIT_RE = re.compile(r"[;,]")


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in _EXT_MARKERS:
        i = s.find(marker)
        if i != -1:
            cut = min(cut, i)
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    if _ZIP_PLUS4_RE.fullmatch(s):
        s = s.split("-", 1)[0]
    if len(s) == 4 and _is_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def dedupe_contacts(rows):
    # Normalize each row and drop rows with neither a valid email nor a valid phone.
    kept = []
    for row in rows:
        email = _normalize_email(row.get("email", ""))
        phone = _normalize_phone(row.get("phone", ""))
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        kept.append({
            "first": _normalize_name(row.get("first_name", "")),
            "last": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "email_ok": email_ok,
            "phone": phone,
            "phone_ok": phone_ok,
            "source": row.get("source", "").strip().lower(),
            "tags": row.get("tags", ""),
            "opted_out": row.get("opted_out", "").strip().lower() in _TRUE_VALUES,
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over kept rows, so matching is transitive.
    m = len(kept)
    parent = list(range(m))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for p, r in enumerate(kept):
        if r["email_ok"]:
            key = r["email"]
            if key in by_email:
                union(p, by_email[key])
            else:
                by_email[key] = p
        if r["phone_ok"] and r["last"]:
            key = (r["phone"], r["last"].lower())
            if key in by_phone_last:
                union(p, by_phone_last[key])
            else:
                by_phone_last[key] = p

    # Groups are ordered by their earliest row, since p is scanned in input order.
    groups = {}
    for p in range(m):
        groups.setdefault(find(p), []).append(p)

    def rank(p):
        # Manual rows first, then most recent updated, then earliest in input.
        r = kept[p]
        return (r["source"] != "manual", -(r["updated"] - datetime.min), p)

    def choose(members, field, ok_key=None):
        if ok_key:
            cands = [p for p in members if kept[p][ok_key]]
        else:
            cands = [p for p in members if kept[p][field]]
        if not cands:
            return ""
        return kept[min(cands, key=rank)][field]

    result = []
    for members in groups.values():
        tag_set = set()
        for p in members:
            for t in _TAG_SPLIT_RE.split(kept[p]["tags"]):
                t = t.strip().lower()
                if t:
                    tag_set.add(t)

        result.append({
            "first_name": choose(members, "first"),
            "last_name": choose(members, "last"),
            "zip": choose(members, "zip"),
            "email": choose(members, "email", "email_ok"),
            "phone": choose(members, "phone", "phone_ok"),
            "source": choose(members, "source"),
            "tags": ";".join(sorted(tag_set)),
            "opted_out": "Y" if any(kept[p]["opted_out"] for p in members) else "N",
            "updated": max(kept[p]["updated"] for p in members).strftime("%Y-%m-%d"),
        })

    return result
```

Notes on a few judgment calls:

- **ZIP+4:** I treat any `dddd-dddd` or `ddddd-dddd` as ZIP+4, so an Excel-mangled `2139-4307` becomes `02139`. If your data person wants strict 5-digit matching only, change `[0-9]{4,5}` to `[0-9]{5}`.
- **Bad `updated` values:** the function raises a `ValueError` naming the bad value instead of guessing. One malformed date will stop the run, which is safer than silently mis-ranking records before a send.
- **Source selection:** the manual-first rule is applied to the `source` field too, so a `manual` row's source wins when there's a choice.
- **Opt-outs:** an opt-out on any row marks the whole person `opted_out = "Y"`. For tomorrow's send, filter the output to `opted_out == "N"` and `email != ""`.
