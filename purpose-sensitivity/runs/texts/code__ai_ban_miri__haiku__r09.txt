import re
from datetime import datetime

_EMAIL_BRACKET = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")


def _normalize_email(raw):
    s = raw
    m = _EMAIL_BRACKET.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _normalize_zip(raw):
    z = raw.strip()
    m = _ZIP_PLUS4.fullmatch(z)
    if m:
        z = m.group(1)
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    if re.fullmatch(r"[0-9]{5}", z):
        return z
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _opted_out(raw):
    return raw.strip().lower() in ("y", "yes", "true", "1")


def _normalize_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(group, field):
    """Pick the normalized value of `field` from the group's rows where it is non-empty.
    Preference: source "manual", then most recent `updated`, then earliest input row."""
    cands = [r for r in group if r[field]]
    if not cands:
        return ""
    # Stable multi-pass sort; the last pass is the primary key.
    cands.sort(key=lambda r: r["idx"])                          # earliest input row
    cands.sort(key=lambda r: r["updated"], reverse=True)        # most recent updated
    cands.sort(key=lambda r: r["source"] != "manual")           # manual first
    return cands[0][field]


def dedupe_contacts(rows):
    # Normalize each row and drop rows with neither a valid email nor a valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row["email"])
        phone = _normalize_phone(row["phone"])
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not email_ok and not phone_ok:
            continue
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row["first_name"]),
            "last_name": _normalize_name(row["last_name"]),
            "zip": _normalize_zip(row["zip"]),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": row["source"].strip().lower(),
            "tags": _normalize_tags(row["tags"]),
            "opted_out": _opted_out(row["opted_out"]),
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over rows, so matching is transitive.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(by_email[r["email"]], i)
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(by_phone_last[key], i)
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # Output people ordered by their earliest input row.
    ordered = sorted(groups.values(), key=lambda members: members[0])

    out = []
    for members in ordered:
        grp = [recs[i] for i in members]
        all_tags = set().union(*(r["tags"] for r in grp))
        out.append({
            "first_name": _pick(grp, "first_name"),
            "last_name": _pick(grp, "last_name"),
            "zip": _pick(grp, "zip"),
            "email": _pick(grp, "email"),
            "phone": _pick(grp, "phone"),
            "source": _pick(grp, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in grp) else "N",
            "updated": max(r["updated"] for r in grp).strftime("%Y-%m-%d"),
        })
    return out

Notes on a few choices where the rules were open to reading:

- **ZIP+4:** I accept a 4- or 5-digit prefix before the hyphen (e.g., `02139-4307` and `2139-4307`), so a ZIP+4 whose leading zero Excel dropped still gets the zero restored.
- **Names:** `isupper()` and `islower()` ignore digits and punctuation, so `"JOHN 3"` counts as all upper-case.
- **Unparseable `updated`:** a row with a date outside the three formats raises `ValueError`, and only if the row survives the validity check. I'd rather fail loudly than silently guess a date for a send.
- **Source:** a row with an empty source is never picked for the `source` field, but its email, phone, and other data still count toward the group.
