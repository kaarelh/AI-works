import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EPOCH = datetime(1, 1, 1)
_DIGITS = set("0123456789")


def _field(row, key):
    v = row.get(key)
    return v if isinstance(v, str) else ("" if v is None else str(v))


def _norm_email(raw):
    """Rules 1 and 2. Returns the normalized email, or "" if invalid."""
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if domain in ("gmail.com", "googlemail.com"):
        local = local.replace(".", "").split("+", 1)[0]
        domain = "gmail.com"
    if not local:
        return ""
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return local + "@" + domain


def _norm_phone(raw):
    """Rule 3. Returns 10 digits, or "" if invalid."""
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i >= 0]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Rule 4."""
    s = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    """Rule 5."""
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    """Rule 6."""
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return raw.strip().lower() in ("y", "yes", "true", "1")


def _split_tags(raw):
    """Rule 8 (plus the cleanup from rule 11)."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(_field(row, "email"))
        phone = _norm_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        recs.append({
            "index": idx,
            "first_name": _norm_name(_field(row, "first_name")),
            "last_name": _norm_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(_field(row, "zip")),
            "source": source,
            "manual": source == "manual",
            "tags": _split_tags(_field(row, "tags")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
            "updated": _parse_updated(_field(row, "updated")),
        })

    # ---- Union-find for transitive matching (rule 10).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the root as the lower position so roots stay stable.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(first_seen[k], i)
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Build one output dict per person (rule 11).
    def priority(r):
        # manual first, then most recent updated, then earliest in input
        return (0 if r["manual"] else 1, -(r["updated"] - _EPOCH), r["index"])

    people = []
    for members in groups.values():
        ranked = sorted(members, key=priority)
        out = {}
        for fld in ("first_name", "last_name", "zip", "email", "phone", "source"):
            out[fld] = next((r[fld] for r in ranked if r[fld]), "")
        tags = set()
        for r in members:
            tags |= r["tags"]
        out["tags"] = ";".join(sorted(tags))
        out["opted_out"] = "Y" if any(r["opted_out"] for r in members) else "N"
        latest = max(r["updated"] for r in members)
        out["updated"] = f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}"
        people.append((min(r["index"] for r in members), out))

    # ---- Order by each person's earliest row in the input (rule 12).
    people.sort(key=lambda p: p[0])
    return [out for _, out in people]

**Notes on interpretation:**
- **ZIP+4:** I also accept a 4-digit ZIP+4 like `2139-4307` (Excel dropped the zero), which becomes `02139`.
- **Phone extensions:** The phone is cut at the earliest marker found.
- **Matching:** Last names are compared after normalization, using `casefold()`.
- **Bad dates:** If an `updated` value doesn't match any of the three formats, the function raises a `ValueError` instead of guessing. I'd rather you find out before the send.
- **Ties:** For field selection, ties are broken by input order, as specified.

**Before you send:** This function only dedupes. It doesn't filter anyone out. When you build the send list from the output, keep only people with `opted_out == "N"` **and** a non-empty `email`. Some people will be merged on phone alone and have no valid email.
