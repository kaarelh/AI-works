import re
from datetime import datetime


def dedupe_contacts(rows):
    _DIGITS = "0123456789"
    _ANGLE = re.compile(r"<([^>]*)>")
    _EXT = re.compile(r"ext|x|#", re.IGNORECASE)
    _ZIP4 = re.compile(r"^[0-9]{5}-[0-9]{4}$")
    _TAG_SPLIT = re.compile(r"[;,]")
    _DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    _OPT_YES = {"y", "yes", "true", "1"}

    def s(value):
        return "" if value is None else str(value)

    def norm_email(raw):
        raw = s(raw)
        m = _ANGLE.search(raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "")
                local = local.split("+", 1)[0]
                domain = "gmail.com"
                e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        p = s(raw)
        m = _EXT.search(p)
        if m:
            p = p[:m.start()]
        digits = "".join(ch for ch in p if ch in _DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) != 10 or digits[0] in "01":
            return ""
        return digits

    def norm_zip(raw):
        z = s(raw).strip()
        if _ZIP4.match(z):
            z = z.split("-", 1)[0]
        if len(z) == 4 and all(ch in _DIGITS for ch in z):
            z = "0" + z
        if len(z) == 5 and all(ch in _DIGITS for ch in z):
            return z
        return ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        t = s(raw).strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(t, fmt)
            except ValueError:
                continue
        return datetime.min  # unparseable: treated as oldest

    def parse_tags(raw):
        out = []
        for t in _TAG_SPLIT.split(s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---- normalize and filter ----
    recs = []
    for row in rows:
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = s(row.get("source")).strip().lower()
        recs.append({
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": parse_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower() in _OPT_YES,
            "updated": parse_updated(row.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # ---- matching ----
    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---- build output ----
    def pick(members, field):
        best = None
        best_manual = False
        best_updated = None
        for i in members:  # members are in input order
            r = recs[i]
            if not r[field]:
                continue
            manual = r["source"] == "manual"
            if best is None:
                better = True
            elif manual != best_manual:
                better = manual
            else:
                better = r["updated"] > best_updated
            if better:
                best, best_manual, best_updated = i, manual, r["updated"]
        return recs[best][field] if best is not None else ""

    result = []
    for root in sorted(groups, key=lambda k: min(groups[k])):
        members = sorted(groups[root])
        tags = set()
        for i in members:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": "%04d-%02d-%02d" % (latest.year, latest.month, latest.day),
        })
    return result

Notes:

- **Opted-out people are still in the output**, flagged `opted_out == "Y"`, as your rules specify. Filter them out before sending, e.g. `[p for p in people if p["opted_out"] == "N"]`.
- **Unparseable `updated` values**: the rules say they won't occur, so I treat one as the oldest possible date rather than raising an error.
- **Ties**: with equal manual status and equal `updated`, the earlier input row wins, per rule 11.
- **Transitive matching** uses union-find, so chains like A–B–C collapse into one person.
- **Ordering**: people appear in order of their earliest surviving row. Rows dropped under rule 9 don't count.
