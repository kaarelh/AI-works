```python
import re
from datetime import datetime

_EMAIL_BRACKET_RE = re.compile(r"<([^<>]*)>")
_PHONE_MARKER_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]+)-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _normalize_email(raw):
    s = raw
    m = _EMAIL_BRACKET_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = f"{local}@gmail.com"
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    m = _PHONE_MARKER_RE.search(raw)
    s = raw[:m.start()] if m else raw
    digits = re.sub(r"[^0-9]", "", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    if re.fullmatch(r"[0-9]{5}", s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    # Not expected per the spec (updated is never empty); treat unparseable as oldest.
    return datetime.min


def _normalize_tags(raw):
    parts = re.split(r"[;,]", raw)
    return sorted({p.strip().lower() for p in parts if p.strip()})


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        email_ok = _valid_email(email)
        phone = _normalize_phone(row.get("phone", ""))
        phone_ok = _valid_phone(phone)
        if not email_ok and not phone_ok:
            continue
        records.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": row.get("source", "").strip().lower(),
            "tags": _normalize_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated", "")),
        })

    # Step 2: union-find so matching is transitive.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_owner = {}
    phone_owner = {}
    for pos, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in email_owner:
                union(email_owner[rec["email"]], pos)
            else:
                email_owner[rec["email"]] = pos
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].casefold())
            if key in phone_owner:
                union(phone_owner[key], pos)
            else:
                phone_owner[key] = pos

    # Group members. Records are in input order, so groups appear in order
    # of their earliest row.
    groups = {}
    for pos in range(len(records)):
        groups.setdefault(find(pos), []).append(records[pos])

    # Step 3: build one output dict per person.
    results = []
    for members in groups.values():
        # Preference: most recent updated first (ties keep input order),
        # then manual sources ahead of all others (stable sort).
        ordered = sorted(members, key=lambda r: r["updated"], reverse=True)
        ordered = sorted(ordered, key=lambda r: r["source"] != "manual")

        def pick(field):
            for r in ordered:
                if r[field]:
                    return r[field]
            return ""

        all_tags = set()
        for r in members:
            all_tags.update(r["tags"])

        results.append({
            "first_name": pick("first_name"),
            "last_name": pick("last_name"),
            "zip": pick("zip"),
            "email": pick("email"),
            "phone": pick("phone"),
            "source": pick("source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        })

    return results
```

Notes on using it:

- **Opt-outs are flagged, not removed.** The function sets `opted_out` to `"Y"` but still returns those people. Filter with `opted_out == "N"` before building the send list, or the email will go to them.
- **Opt-outs spread across a match group.** Under rule 11, if any row for a person is opted out, the whole merged person is flagged `"Y"`. That is the conservative choice for suppression, but it means you may suppress someone who has a valid, non-opted-out row elsewhere in the export. Check a sample of flagged people before the send.
- **Dropped rows are gone from the output.** Rows with no valid email or phone are removed before matching, so they can't contribute opt-out flags or tags to anyone. If you need them for suppression, keep the original export.
