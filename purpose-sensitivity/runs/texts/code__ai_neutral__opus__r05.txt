```python
import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"


def _normalize_email(raw):
    """Rule 1: extract from <...>, strip, lowercase, canonicalize Gmail."""
    m = _ANGLE_RE.search(raw)
    s = m.group(1) if m else raw
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    """Rule 2."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Rule 3: returns the 10-digit phone if valid, else ""."""
    m = _EXT_RE.search(raw)
    if m:
        raw = raw[: m.start()]
    digits = "".join(c for c in raw if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    """Rule 4."""
    s = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and all(c in _DIGITS for c in s):
        s = "0" + s
    return s if _ZIP5_RE.fullmatch(s) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6: a date alone means midnight."""
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return raw.strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    """Rule 8 (plus the cleanup from rule 11)."""
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize each row, dropping rows with no valid email or phone (rule 9)
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = row.get("source", "").strip().lower()
        records.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "is_manual": source == "manual",
            "dt": _parse_updated(row["updated"]),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "tags": _parse_tags(row.get("tags", "")),
        })

    # ---- Union-find over records (rule 10, transitive)
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the earlier record as root (not required, just tidy).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}  # match key -> record position
    for pos, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(pos, first_seen[key])
            else:
                first_seen[key] = pos

    # ---- Group records; dict insertion order = order of earliest row (rule 12)
    groups = {}
    for pos, rec in enumerate(records):
        groups.setdefault(find(pos), []).append(rec)

    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        # Prefer manual source, then most recent updated, then earliest input row.
        best = max(candidates, key=lambda r: (r["is_manual"], r["dt"], -r["idx"]))
        return best[field]

    # ---- Build output (rule 11)
    output = []
    for members in groups.values():
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["dt"] for r in members).date().isoformat(),
            "source": pick(members, "source"),
        })
    return output
```

**Things to check before the send:**

1. **Opted-out people are still in the output.** They come out marked `opted_out: "Y"`, as the spec asks. Filter them out when building the send list, for example `[c for c in result if c["opted_out"] == "N"]`. Because opt-out is "any row in the group," someone who opted out under one address is suppressed for all of their matched records. That's what you want.
2. **One email per person.** If two rows join only by phone and last name but have different emails, the person gets one email (the "best" one). That prevents a double send. It also means the other address won't get the newsletter.
3. **Rows with no contact info disappear.** Rows with neither a valid email nor a valid phone are dropped. They can't affect anyone's opt-out status.
4. **Places where I had to interpret the spec:**
   - **ZIP+4:** this also accepts a 4-digit base (`2139-4307` → `02139`), on the theory that Excel can strip the leading zero there too.
   - **Phone extensions:** the extension cut happens at the *first* marker found. Note that any letter "x" counts, so `555-CALL-XRAY`-style vanity numbers would be truncated. Real data rarely has these.
   - **Gmail addresses:** a Gmail address whose local part becomes empty (e.g. `+news@gmail.com`) is treated as invalid.
   - **Bad dates:** an `updated` value in an unrecognized format raises a `ValueError`. Better to fail loudly than to mis-sort silently.
5. **Sending to normalized addresses** is safe. Gmail ignores dots and `+tags`, and `googlemail.com` delivers to the same inbox as `gmail.com`.
