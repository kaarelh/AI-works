import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ZIP4_RE = re.compile(r"([0-9]{4,5})-([0-9]{4})")


def _all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    s = raw.strip()
    lt = s.find("<")
    if lt != -1:
        gt = s.find(">", lt + 1)
        if gt != -1:
            s = s[lt + 1:gt]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _normalize_phone(raw):
    s = raw.lower()
    cuts = [i for i in (s.find("ext"), s.find("x"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    m = _ZIP4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _all_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None  # unparseable: treated as oldest, never output


def _split_tags(raw):
    out = []
    for t in re.split(r"[;,]", raw):
        t = t.strip().lower()
        if t:
            out.append(t)
    return out


def _pick(members, field):
    # Members are already in input order, so stable sorts preserve it.
    cands = [m for m in members if m[field]]
    cands.sort(key=lambda m: m["updated"] or datetime.min, reverse=True)  # most recent first
    cands.sort(key=lambda m: m["source"] != "manual")                     # manual first
    return cands[0][field] if cands else ""


def _merge(members):
    tags = set()
    for m in members:
        tags.update(m["tags"])
    dates = [m["updated"] for m in members if m["updated"]]
    return {
        "first_name": _pick(members, "first_name"),
        "last_name": _pick(members, "last_name"),
        "zip": _pick(members, "zip"),
        "email": _pick(members, "email"),
        "phone": _pick(members, "phone"),
        "source": _pick(members, "source"),
        "tags": ";".join(sorted(tags)),
        "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
        "updated": max(dates).strftime("%Y-%m-%d") if dates else "",
    }


def dedupe_contacts(rows):
    # 1-9: normalize each row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        phone = _normalize_phone(row.get("phone", ""))
        email_ok = _is_valid_email(email)
        phone_ok = _is_valid_phone(phone)
        if not (email_ok or phone_ok):
            continue
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": row.get("source", "").strip().lower(),
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": row.get("opted_out", "").strip().lower()
                         in {"y", "yes", "true", "1"},
            "updated": _parse_updated(row.get("updated", "")),
        })

    # 10: union-find over shared keys (transitive matching).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in first_seen:
                union(first_seen[k], i)
            else:
                first_seen[k] = i

    # 11-12: group rows; dict order is each group's earliest input row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    return [_merge(members) for members in groups.values()]
