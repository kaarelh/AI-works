```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EPOCH = datetime(1970, 1, 1)


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        e = raw
        start = e.find("<")
        if start != -1:
            end = e.find(">", start + 1)
            if end != -1:
                e = e[start + 1:end]
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                domain = "gmail.com"
            e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        m = re.search(r"x|ext|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        digits = "".join(c for c in raw if c in _DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_dt(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.fromisoformat(s)

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # Normalize rows, dropping those with no valid email and no valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "tags": split_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": parse_dt(get(row, "updated")),
        })

    # Union-find over the kept records.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    def sort_key(r):
        return (
            0 if r["source"] == "manual" else 1,
            -(r["updated"] - _EPOCH).total_seconds(),
            r["idx"],
        )

    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        return min(cands, key=sort_key)[field]

    result = []
    for root in sorted(groups, key=lambda k: min(recs[i]["idx"] for i in groups[k])):
        members = [recs[i] for i in groups[root]]
        tags = set()
        for r in members:
            tags.update(r["tags"])
        latest = max(r["updated"] for r in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes on a few judgment calls:

- **Opt-outs:** if any row in a merged group has opted out, the whole person is marked `"Y"`. Filter on `opted_out == "N"` before sending tomorrow's email.
- **ZIP+4:** I accept any digits before the hyphen followed by 4 digits, so `"2139-4307"` becomes `"02139"`. Other shapes become `""`.
- **Phone extensions:** the marker search looks for the first `x`, `ext`, or `#`, case-insensitive, and cuts there.
- **Dropped rows:** rows with no valid email or phone are dropped before matching, so they don't affect output order or merged fields like tags and opt-out status. If a dropped row was an opt-out, it won't carry over. You may want to check your export for that case separately.
- **Unparseable `updated` values:** these raise an exception rather than being silently guessed, since the spec says only three formats occur.
