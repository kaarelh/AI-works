import re
from datetime import datetime


def dedupe_contacts(rows):
    """Collapse duplicate CRM rows into one record per person (rules 1-12)."""

    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")

    # ---------- field normalizers ----------

    def text(row, key):
        value = row.get(key, "")
        return value if isinstance(value, str) else ("" if value is None else str(value))

    def normalize_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        email = (m.group(1) if m else raw).strip().lower()
        if email.count("@") != 1:
            return ""
        local, domain = email.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""                      # invalid -> treated as absent
        return local + "@" + domain

    def normalize_phone(raw):
        # "ext" begins with "x"-containing text; cutting at the first x/X/# covers x, ext, #.
        cuts = [i for i in (raw.find("x"), raw.find("X"), raw.find("#")) if i != -1]
        if cuts:
            raw = raw[:min(cuts)]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                          # invalid -> treated as absent

    def normalize_zip(raw):
        z = raw.strip()
        if "-" in z:                       # ZIP+4
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):   # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def normalize_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min                # unparseable: loses every recency comparison

    def is_opted_out(raw):
        return raw.strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---------- normalize every row, drop rows with no usable contact info (rule 9) ----------

    records = []
    for idx, row in enumerate(rows):
        rec = {
            "index": idx,
            "first_name": normalize_name(text(row, "first_name")),
            "last_name": normalize_name(text(row, "last_name")),
            "email": normalize_email(text(row, "email")),
            "phone": normalize_phone(text(row, "phone")),
            "zip": normalize_zip(text(row, "zip")),
            "tags": split_tags(text(row, "tags")),
            "opted_out": is_opted_out(text(row, "opted_out")),
            "updated": parse_updated(text(row, "updated")),
            "source": text(row, "source").strip().lower(),
        }
        if rec["email"] or rec["phone"]:
            records.append(rec)

    # ---------- transitive matching via union-find (rule 10) ----------

    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen_email = {}
    seen_phone_last = {}
    for i, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in seen_email:
                union(i, seen_email[rec["email"]])
            else:
                seen_email[rec["email"]] = i
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}
    for i, rec in enumerate(records):
        groups.setdefault(find(i), []).append(rec)

    # ---------- build one output dict per person (rules 11-12) ----------

    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        # prefer manual source, then latest updated, then earliest input row
        best = max(candidates,
                   key=lambda r: (r["source"] == "manual", r["updated"], -r["index"]))
        return best[field]

    output = []
    for group in sorted(groups.values(), key=lambda g: min(r["index"] for r in g)):
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).date().isoformat(),
            "source": pick(group, "source"),
        })
    return output

A few notes on how I resolved details the rules leave open:

- **Invalid email/phone are stored as `""`** after normalization, so they never match, never get output, and count as "absent" for the rule 9 drop. A row with a malformed email but a good phone is kept.
- **Phone extensions:** cutting at the first `x`/`X`/`#` handles `ext`/`EXT` too (the leading `e` is discarded with the other non-digits).
- **ZIP:** any hyphen is treated as a ZIP+4 separator; whatever is left that isn't 5 digits becomes `""`. `"2139-4307"` → `"02139"`.
- **Names:** `isupper()`/`islower()` ignore punctuation, so `"O'BRIEN"` → `"O'Brien"` and `"de la cruz"` → `"De La Cruz"`; `"McDonald"` is left alone. Phone+last-name matching compares the normalized last name lowercased.
- **Field selection** is done independently per field, exactly as rule 11 says — so the winning `first_name` and `email` can come from different rows in the group.
- **Unparseable `updated`** values (not expected per the rules) fall back to `datetime.min` so they lose every recency comparison rather than crashing the export.
