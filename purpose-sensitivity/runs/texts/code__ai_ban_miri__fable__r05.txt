```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows per the data team's rules.

    Takes a list of dicts (first_name, last_name, email, phone, zip, tags,
    opted_out, updated, source) and returns one merged dict per person,
    ordered by each person's earliest row in the input.
    """
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")

    # ---------- normalizers ----------

    def parse_updated(s):
        s = (s or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        raise ValueError(f"Unrecognized 'updated' value: {s!r}")

    def norm_email(s):
        m = re.search(r"<([^<>]*)>", s)      # display-name form: Ann Lee <ann@x.org>
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(s):
        low = s.lower()
        cut = len(low)
        for marker in ("ext", "x", "#"):     # ignore everything from an extension marker on
            i = low.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", low[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(s):
        s = s.strip()
        if "-" in s:                          # ZIP+4 -> keep the 5-digit part
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):      # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(s):
        s = s.strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def norm_opted_out(s):
        return s.strip().lower() in ("y", "yes", "true", "1")

    def norm_tags(s):
        return {t.strip().lower() for t in re.split(r"[;,]", s) if t.strip()}

    # ---------- normalize + filter (rule 9) ----------

    kept = []
    for idx, row in enumerate(rows):
        g = lambda k: row.get(k, "") or ""
        email = norm_email(g("email"))
        phone = norm_phone(g("phone"))
        if not email_valid(email):
            email = ""
        if not phone_valid(phone):
            phone = ""
        if not email and not phone:
            continue
        kept.append({
            "idx": idx,
            "first_name": norm_name(g("first_name")),
            "last_name": norm_name(g("last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(g("zip")),
            "tags": norm_tags(g("tags")),
            "opted_out": norm_opted_out(g("opted_out")),
            "updated": parse_updated(g("updated")),
            "source": g("source").strip().lower(),
        })

    # ---------- union-find for transitive matching (rule 10) ----------

    parent = list(range(len(kept)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so root == earliest row
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(kept):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for key in keys:
            if key in seen:
                union(i, seen[key])
            else:
                seen[key] = i

    groups = {}  # root -> list of rows, insertion order = earliest row order (rule 12)
    for i, r in enumerate(kept):
        groups.setdefault(find(i), []).append(r)

    # ---------- merge each group (rule 11) ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(
            cands,
            key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
        )
        return best[field]

    out = []
    for group in groups.values():
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return out
```

A few notes on how I resolved spots the spec leaves open:

- **Invalid emails/phones are treated as empty** after normalization, so they never participate in matching and are never selected for output (per rule 11's "valid" requirement). If a row has an invalid email but a valid phone, it's kept and its email is just `""`.
- **Extension markers**: the cut happens at the earliest of `ext`, `x`, or `#`. Because `x` is inside `ext`, both give the same digits; I listed both for clarity.
- **ZIP**: anything with a hyphen is treated as ZIP+4 and truncated at the first hyphen; anything not 5 digits after that becomes `""`, so garbage like `"abc-123"` becomes `""` rather than raising.
- **Tie-breaking in `pick`**: the key is `(is_manual, updated, -index)` under `max`, which gives manual > non-manual, then latest `updated`, then earliest input row — exactly rule 11's order.
- **Phone matching uses the normalized (title-cased) last name, lowercased**, so `"SMITH"` and `"Smith"` match.
- **`updated`** raises `ValueError` if a row's value matches none of the three formats, rather than silently dropping the row — I'd rather you see the bad row before tomorrow's send than lose a supporter quietly.
