import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EPOCH = datetime(1970, 1, 1)
_DIGITS = "0123456789"
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _get(row, name):
    return row.get(name) or ""


def _is_all_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _norm_email(raw):
    """Return (normalized_email, is_valid)."""
    s = raw.strip()
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"

    valid = False
    if s.count("@") == 1:
        local, domain = s.split("@")
        valid = (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )
    return s, valid


def _norm_phone(raw):
    """Return (digits, is_valid). Digits are the 10-digit number when valid."""
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    valid = len(digits) == 10 and digits[0] not in "01"
    return digits, valid


def _norm_zip(raw):
    s = raw.strip()
    m = _ZIP_PLUS4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _is_all_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_all_digits(s):
        return s
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _norm_tags(raw):
    tags = set()
    for part in _TAG_SPLIT_RE.split(raw):
        tag = part.strip().lower()
        if tag:
            tags.add(tag)
    return tags


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(members, field):
    """Pick the field value from the best row that has it non-empty.

    Preference: source 'manual', then most recent 'updated', then earliest input row.
    """
    candidates = [r for r in members if r[field]]
    if not candidates:
        return ""

    def key(r):
        return (
            r["source"] != "manual",
            -(r["updated"] - _EPOCH).total_seconds(),
            r["idx"],
        )

    return min(candidates, key=key)[field]


def _merge_group(members):
    tags = set()
    for r in members:
        tags |= r["tags"]
    latest = max(r["updated"] for r in members)
    return {
        "first_name": _pick(members, "first_name"),
        "last_name": _pick(members, "last_name"),
        "zip": _pick(members, "zip"),
        "email": _pick(members, "email"),
        "phone": _pick(members, "phone"),
        "source": _pick(members, "source"),
        "tags": ";".join(sorted(tags)),
        "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
        "updated": latest.strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    # Normalize every row and drop rows with no valid email or phone.
    kept = []
    for idx, row in enumerate(rows):
        email, email_ok = _norm_email(_get(row, "email"))
        phone, phone_ok = _norm_phone(_get(row, "phone"))
        if not email_ok and not phone_ok:
            continue
        kept.append({
            "idx": idx,
            "first_name": _norm_name(_get(row, "first_name")),
            "last_name": _norm_name(_get(row, "last_name")),
            "zip": _norm_zip(_get(row, "zip")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": _get(row, "source").strip().lower(),
            "tags": _norm_tags(_get(row, "tags")),
            "opted_out": _get(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(_get(row, "updated")),
        })

    # Union-find so that matches are transitive.
    parent = list(range(len(kept)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    # Link rows sharing a valid email, or a valid phone plus the same last name.
    email_first_seen = {}
    phone_first_seen = {}
    for i, r in enumerate(kept):
        if r["email"]:
            if r["email"] in email_first_seen:
                union(email_first_seen[r["email"]], i)
            else:
                email_first_seen[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in phone_first_seen:
                union(phone_first_seen[key], i)
            else:
                phone_first_seen[key] = i

    # Group rows by root. Dict insertion order follows each person's earliest row.
    groups = {}
    for i, r in enumerate(kept):
        groups.setdefault(find(i), []).append(r)

    return [_merge_group(members) for members in groups.values()]

Notes on the implementation:

- **Opt-outs are flagged, not removed.** The function returns `opted_out` as "Y" or "N" and does not filter those people out. Your send step should exclude any row where `opted_out == "Y"`.
- **Opt-outs only apply through matches.** An opt-out record merges with a contact only when it shares a valid email or a valid phone plus last name. If an opt-out row shares neither, the same person could still appear as a separate sendable row. It's worth checking this before the send.
- **ZIP+4 handling:** I read "looks like ZIP+4" as a 4- or 5-digit prefix, a hyphen, and 4 digits. This lets the leading-zero rule apply to inputs like "2139-4307".
- **Unparseable `updated` values** raise a `ValueError` rather than being silently ordered. I chose this so bad dates surface instead of quietly changing which row wins.
- **Phone extension markers:** the cut happens at the earliest of "x", "ext", or "#". The digits kept are the same either way.
