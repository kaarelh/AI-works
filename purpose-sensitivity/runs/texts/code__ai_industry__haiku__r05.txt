import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_TAG_SEP = re.compile(r"[;,]")
_ANGLE_ADDR = re.compile(r"<([^<>]*)>")
_ZIP_PLUS4 = re.compile(r"^([0-9]{4,5})-[0-9]{4}$")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _all_digits(s):
    return bool(s) and all("0" <= c <= "9" for c in s)


def _norm_email(raw):
    s = (raw or "").strip()
    m = _ANGLE_ADDR.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        local != ""
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = (raw or "").lower()
    markers = [i for i in (s.find("x"), s.find("ext"), s.find("#")) if i != -1]
    if markers:
        s = s[: min(markers)]
    digits = "".join(c for c in s if "0" <= c <= "9")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(d):
    return len(d) == 10 and d[0] not in "01"


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP_PLUS4.match(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and _all_digits(s):
        s = "0" + s
    if len(s) == 5 and _all_digits(s):
        return s
    return ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _parse_tags(raw):
    return {t.strip().lower() for t in _TAG_SEP.split(raw or "") if t.strip()}


def _ts(dt):
    return dt if dt is not None else datetime.min


def dedupe_contacts(rows):
    # 1. Normalize each row; drop rows with neither a valid email nor a valid phone.
    recs = []
    for row in rows:
        email = _norm_email(row.get("email"))
        if not _is_valid_email(email):
            email = ""
        phone = _norm_phone(row.get("phone"))
        if not _is_valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _parse_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated")),
        })

    # 2. Group rows that refer to the same person (transitive, via union-find).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_by_email = {}
    first_by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in first_by_email:
                union(first_by_email[r["email"]], i)
            else:
                first_by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in first_by_phone_last:
                union(first_by_phone_last[key], i)
            else:
                first_by_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)
    # Member indices are ascending, so members[0] is the group's earliest row.
    ordered = sorted(groups.values(), key=lambda members: members[0])

    # 3. Pick each field's value from the group.
    # Stable sorts: first by most recent updated (desc), then manual rows first.
    # Ties keep input order.
    def best(members, field):
        cands = [recs[i] for i in members if recs[i][field]]
        cands.sort(key=lambda r: _ts(r["updated"]), reverse=True)
        cands.sort(key=lambda r: not r["manual"])
        return cands[0][field] if cands else ""

    result = []
    for members in ordered:
        group = [recs[i] for i in members]

        tags = set()
        for r in group:
            tags |= r["tags"]

        dates = [r["updated"] for r in group if r["updated"] is not None]

        result.append({
            "first_name": best(members, "first_name"),
            "last_name": best(members, "last_name"),
            "zip": best(members, "zip"),
            "email": best(members, "email"),
            "phone": best(members, "phone"),
            "source": best(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(dates).strftime("%Y-%m-%d") if dates else "",
        })

    return result
