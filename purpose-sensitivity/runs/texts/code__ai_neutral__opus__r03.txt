import re
from datetime import datetime

_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def _normalize_email(raw):
    s = raw or ""
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Return the 10-digit phone if valid, else ""."""
    s = raw or ""
    m = re.search(r"ext|x|#", s, flags=re.IGNORECASE)
    if m:
        s = s[: m.start()]
    digits = "".join(re.findall(r"[0-9]", s))
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with no valid email and no valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        updated = _parse_updated(row.get("updated", ""))
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": updated,
            # Integer sort key; avoids naive-datetime timestamp()/timezone issues.
            "updated_key": updated.toordinal() * 86400
                           + updated.hour * 3600 + updated.minute * 60 + updated.second,
        })

    # --- Union-find over records (transitive matching).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the lower (earlier) record index.
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # --- Build output records.
    def preference(r):
        return (
            0 if r["source"] == "manual" else 1,  # manual first
            -r["updated_key"],                    # most recent first
            r["idx"],                             # earlier in input first
        )

    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        return min(candidates, key=preference)[field]

    output = []
    for members in sorted(groups.values(), key=lambda g: min(r["idx"] for r in g)):
        tags = set()
        for r in members:
            tags |= r["tags"]
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return output

A few things worth knowing before the send:

- **Judgment calls on ambiguous spec points:**
  - **ZIP+4 with a dropped leading zero:** I treated a ZIP like `2139-4307` as ZIP+4 too, so it becomes `02139`. If you want only strict `NNNNN-NNNN` accepted, change `{4,5}` to `{5}`.
  - **Extension markers:** `ext` is checked so that `555-123-4567 ext 12` cuts at the `e`. Any `x` anywhere also counts as a marker, as the rules say.
- **Bad dates:** An `updated` value in none of the three formats raises a `ValueError` naming the bad value. I did this instead of guessing so that bad data doesn't silently change which row wins.
- **Opt-out gaps:**
  - **Different contact details:** If someone opted out under one email and appears elsewhere with a different email and no matching phone + last name, the rules treat them as two people. The second record would still get the newsletter.
  - **Dropped rows:** An opted-out row with no valid email or phone is dropped before matching (rule 9). Its opt-out can't carry over to other rows.
  - **Suggestion:** If you keep a separate suppression list, check it against the output before sending.
- **Filter the output yourself:** The function returns everyone, including people with `opted_out: "Y"` and people with no email (phone-only). Exclude both when building the send list.
