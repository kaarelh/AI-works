import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)   # "ext" listed first so it wins over bare "x"
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_TAG_SPLIT_RE = re.compile(r"[;,]")


def _text(row, key):
    """Field as a string; missing/None fields are treated as ''."""
    value = row.get(key, "")
    return "" if value is None else str(value)


def _normalize_email(raw):
    m = _ANGLE_RE.search(raw)
    email = (m.group(1) if m else raw).strip().lower()
    if email.count("@") == 1:
        local, domain = email.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            email = local + "@gmail.com"
    return email


def _email_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    base = _EXT_RE.split(raw, maxsplit=1)[0]   # drop extension (x / ext / #) onward
    digits = _NON_DIGIT_RE.sub("", base)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_valid(phone):
    return len(phone) == 10 and phone[0] not in "01"


def _normalize_zip(raw):
    z = raw.strip()
    if "-" in z:                                # ZIP+4 -> keep the 5-digit part
        z = z.split("-", 1)[0].strip()
    if re.fullmatch(r"[0-9]{4}", z):            # Excel ate the leading zero
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    name = raw.strip()
    if name.isupper() or name.islower():
        return name.title()
    return name


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    return None  # caller treats an unparseable value as the oldest possible date


def _normalize_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def _pick(group, field):
    """Best row's value for `field`: non-empty, prefer source 'manual',
    then latest `updated`, then earliest position in the input."""
    candidates = [r for r in group if r[field]]
    if not candidates:
        return ""
    best = max(
        candidates,
        key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
    )
    return best[field]


def dedupe_contacts(rows):
    # ---- 1. Normalize every row; drop rows with neither a valid email nor phone ----
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_text(row, "email"))
        phone = _normalize_phone(_text(row, "phone"))
        email_ok = _email_valid(email)
        phone_ok = _phone_valid(phone)
        if not (email_ok or phone_ok):
            continue
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(_text(row, "first_name")),
            "last_name": _normalize_name(_text(row, "last_name")),
            "zip": _normalize_zip(_text(row, "zip")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "source": _text(row, "source").strip().lower(),
            "tags": _normalize_tags(_text(row, "tags")),
            "opted_out": _text(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(_text(row, "updated")) or datetime.min,
        })

    # ---- 2. Union-find to merge rows into people (transitive matching) ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)   # root is always the earliest row

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for key in keys:
            if key in seen:
                union(i, seen[key])
            else:
                seen[key] = i

    groups = {}
    for i, r in enumerate(recs):               # input order => groups ordered by earliest row
        groups.setdefault(find(i), []).append(r)

    # ---- 3. Build one output dict per person ----
    out = []
    for group in groups.values():
        tags = set().union(*(r["tags"] for r in group))
        out.append({
            "first_name": _pick(group, "first_name"),
            "last_name": _pick(group, "last_name"),
            "zip": _pick(group, "zip"),
            "email": _pick(group, "email"),
            "phone": _pick(group, "phone"),
            "source": _pick(group, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).date().isoformat(),
        })
    return out

A few judgment calls to flag, in case the data person wants them changed:

- **Unparseable `updated`**: the spec says it's always one of the three formats, so I didn't make this crash the whole run; a bad value is treated as the oldest possible date (so it never wins a tiebreak, and a group where *every* row is bad would show `updated` = `0001-01-01`). If you'd rather fail loudly, replace the `or datetime.min` with a `raise`.
- **ZIP hyphens**: anything after the first hyphen is dropped, not just a strict `NNNNN-NNNN` pattern, so `"2139-4307"` still becomes `"02139"`. Whatever is left must be exactly 5 digits or it's blanked.
- **Phone output** is the bare 10 digits (e.g. `2025551234`), since that's the normalized value; format it for display downstream if needed.
- **Missing/None fields** are treated as `""` rather than raising, since CRM exports are often ragged.
- Gmail addresses are detected only when the address has exactly one `@`; malformed ones are left untouched and then rejected by the validity check.
