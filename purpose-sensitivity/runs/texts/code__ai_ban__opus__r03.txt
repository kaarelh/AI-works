import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the CAS data rules.

    Returns a list of dicts with keys: first_name, last_name, email, phone,
    zip, tags, opted_out, updated, source. Opted-out people are still
    included (with opted_out == "Y"), so filter them out before sending.
    """

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^<>]*)>", s)            # "Ann Lee <ann@x.org>"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+", 1)[0]
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"ext|x|#", s, re.IGNORECASE)  # extension marker
        if m:
            s = s[:m.start()]
        digits = "".join(re.findall(r"[0-9]", s))
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", s)  # ZIP+4
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):            # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y"):
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def is_opted_out(raw):
        return (raw or "").strip().lower() in ("y", "yes", "true", "1")

    def split_tags(raw):
        return [t.strip().lower() for t in re.split(r"[;,]", raw or "")
                if t.strip()]

    # ---------- normalize rows; drop rows with no valid email or phone ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        ev = email_valid(email)
        pv = phone_valid(phone)
        if not ev and not pv:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "email": email if ev else "",
            "phone": phone if pv else "",
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "updated": parse_updated(row.get("updated", "")),
            "opted_out": is_opted_out(row.get("opted_out", "")),
            "tags": split_tags(row.get("tags", "")),
        })

    # ---------- union-find matching (transitive) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the root as the earlier record so order is easy to track
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("e", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("p", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build output ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        # prefer manual, then most recent updated, then earliest in input
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    people = []
    for group in groups.values():
        latest = max(r["updated"] for r in group)
        tags = sorted({t for r in group for t in r["tags"]})
        people.append((min(r["idx"] for r in group), {
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(tags),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": f"{latest.year:04d}-{latest.month:02d}-{latest.day:02d}",
            "source": pick(group, "source"),
        }))

    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

Before you send tomorrow:

- **Opted-out people stay in the output.** Rule 11 marks them with `opted_out: "Y"` but doesn't remove them, so filter the result before mailing:
  ```python
  send_list = [p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]
  ```
  The `p["email"]` check also skips people who matched only on phone and have no valid email.
- **Opt-outs carry over across matches.** If someone opted out under one address and later signed up again under another address that links to them by email or by phone + last name, the whole merged person is marked opted out. That's the safe behavior, but it may shrink your list a bit.
- **Gmail addresses get rewritten.** For example, `Jane.Doe+cas@googlemail.com` becomes `janedoe@gmail.com`. Gmail delivers both to the same inbox, so this is fine, but the output email won't always match what's in the CRM character-for-character. Keep that in mind if you sync the results back.
- **Bad dates stop the run.** An `updated` value in none of the three formats raises a `ValueError` naming the bad value, rather than being silently misordered.
- **Extension markers are matched loosely.** Any "x" in the phone field is treated as an extension marker, as the rules specify. So text like "fax 202…" would be cut off at the "x". That's unlikely in a phone column, but worth knowing.

I'd run it once on the real export and spot-check a few merged groups before the send.
