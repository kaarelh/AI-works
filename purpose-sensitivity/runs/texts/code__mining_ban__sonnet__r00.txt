import re
from datetime import datetime


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one dict per person (see rules 1-12)."""

    # ---------- helpers ----------
    def s(value):
        return "" if value is None else str(value)

    def norm_email(raw):
        raw = s(raw)
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+", 1)[0]
                domain = "gmail.com"
            e = local + "@" + domain
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        raw = s(raw)
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = s(raw).strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if not re.fullmatch(r"[0-9]{5}", z):
            z = ""
        return z

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw, idx):
        raw = s(raw).strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(raw, fmt)
            except ValueError:
                pass
        raise ValueError(f"Row {idx}: unrecognized 'updated' value {raw!r}")

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize & filter ----------
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        e_ok = email_valid(email)
        p_ok = phone_valid(phone)
        if not (e_ok or p_ok):
            continue
        source = s(row.get("source")).strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email if e_ok else "",
            "phone": phone if p_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_updated(row.get("updated"), idx),
        })

    # ---------- union-find ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(recs[i])  # members stay in input order

    # ---------- build output ----------
    def pick(members, field):
        best = None
        for r in members:  # input order => earlier row wins full ties
            if not r[field]:
                continue
            if best is None:
                best = r
                continue
            if r["manual"] != best["manual"]:
                if r["manual"]:
                    best = r
            elif r["updated"] > best["updated"]:
                best = r
        return best[field] if best else ""

    ordered = sorted(groups.values(), key=lambda g: g[0]["idx"])

    result = []
    for members in ordered:
        tags = set()
        for r in members:
            tags.update(r["tags"])
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        })
    return result

Notes on judgment calls:

- **Opt-outs:** Rule 9 drops rows with no valid email or phone before matching. If someone opted out on a row like that, the opt-out is lost, because that row can't be linked to anyone. Before sending, it's worth checking your raw export for opted-out rows that were dropped.
- **Bad dates:** If an `updated` value isn't in one of the three formats, the function raises a `ValueError` naming the row index instead of guessing. I'd rather you see that than have a wrong date quietly decide which record wins.
- **ZIP+4:** Only the strict `#####-####` pattern is treated as ZIP+4. Anything else that isn't 4 or 5 digits becomes `""`.
- **Email and phone fields:** These are output as `""` if no row in the group has a valid value. Before sending, filter the output on `opted_out == "N"` and a non-empty `email`.
