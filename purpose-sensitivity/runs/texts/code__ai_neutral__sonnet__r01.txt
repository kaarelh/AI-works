```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    DIGITS = "0123456789"

    def s(v):
        return "" if v is None else str(v)

    def norm_email(raw):
        e = s(raw)
        m = re.search(r"<([^<>]*)>", e)
        if m:
            e = m.group(1)
        e = e.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
                e = local + "@" + domain
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        p = s(raw)
        low = p.lower()
        cut = len(p)
        for marker in ("x", "ext", "#"):
            i = low.find(marker)
            if i != -1 and i < cut:
                cut = i
        p = p[:cut]
        d = "".join(c for c in p if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        if re.fullmatch(r"[0-9]{5}-[0-9]{4}", z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if not re.fullmatch(r"[0-9]{5}", z):
            return ""
        return z

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        u = s(raw).strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(u, fmt)
            except ValueError:
                pass
        return datetime.min

    def is_opted_out(raw):
        return s(raw).strip().lower() in ("y", "yes", "true", "1")

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize and filter
    recs = []
    for row in rows:
        email = norm_email(row.get("email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": s(row.get("source")).strip().lower(),
            "tags": parse_tags(row.get("tags")),
            "opted_out": is_opted_out(row.get("opted_out")),
            "updated": parse_updated(row.get("updated")),
        })

    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    def pick(members, field):
        best = None
        for i in members:  # input order, so ties keep the earlier row
            r = recs[i]
            if not r[field]:
                continue
            if best is None:
                best = r
                continue
            r_manual = r["source"] == "manual"
            b_manual = best["source"] == "manual"
            if r_manual and not b_manual:
                best = r
            elif r_manual == b_manual and r["updated"] > best["updated"]:
                best = r
        return best[field] if best else ""

    result = []
    for root in sorted(groups, key=lambda k: min(groups[k])):
        members = groups[root]
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        latest = max(recs[i]["updated"] for i in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes on a few interpretation choices:

- **Dropped rows:** rows with no valid email or phone are removed before matching, so they don't contribute tags or opt-outs to any group. Such a row has nothing to match on anyway.
- **Opt-outs:** the function doesn't remove opted-out people; it marks them `opted_out = "Y"`. Filter on that field before sending, e.g. `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`.
- **People without an email:** a person who matched only by phone may have `email == ""`, so they can't receive the newsletter.
- **Extension markers:** the earliest of "x", "ext" or "#" in the phone string is where the text is cut.
- **Tie-breaking:** the "manual, then most recent, then earlier row" preference is done by scanning rows in input order and replacing the current pick only on a strictly better candidate.
- **Unparseable `updated` values:** these are treated as `datetime.min` rather than raising an error. If you'd rather fail loudly, replace that fallback with a `raise`.
