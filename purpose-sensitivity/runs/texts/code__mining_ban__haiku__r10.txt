import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_PHONE_MARKERS = ("x", "ext", "#")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_TRUTHY_OPT_OUT = {"y", "yes", "true", "1"}


def _get(row, key):
    value = row.get(key)
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in _PHONE_MARKERS:
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    s = s[:cut]
    digits = "".join(c for c in s if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _is_valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    s = raw.strip()
    # Treat "NNNNN-NNNN" (and "NNNN-NNNN", where Excel dropped the leading zero) as ZIP+4.
    if re.fullmatch(r"[0-9]{4,5}-[0-9]{4}", s):
        s = s.split("-")[0]
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(members, field):
    """Pick the best non-empty normalized value for `field` from a group.

    Preference: source == "manual", then latest `updated`, then earliest input row.
    """
    candidates = [r for r in members if r[field]]
    if not candidates:
        return ""
    best = max(
        candidates,
        key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]),
    )
    return best[field]


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    records = []
    for idx, row in enumerate(rows):
        email = _normalize_email(_get(row, "email"))
        if not _is_valid_email(email):
            email = ""

        phone = _normalize_phone(_get(row, "phone"))
        if not _is_valid_phone(phone):
            phone = ""

        if not email and not phone:
            continue

        tags = {
            t.strip().lower()
            for t in re.split(r"[;,]", _get(row, "tags"))
            if t.strip()
        }

        records.append({
            "idx": idx,
            "first": _normalize_name(_get(row, "first_name")),
            "last": _normalize_name(_get(row, "last_name")),
            "zip": _normalize_zip(_get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": _get(row, "source").strip().lower(),
            "tags": tags,
            "opted_out": _get(row, "opted_out").strip().lower() in _TRUTHY_OPT_OUT,
            "updated": _parse_updated(_get(row, "updated")),
        })

    # Step 2: union-find to group rows that refer to the same person.
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    seen_keys = {}
    for i, r in enumerate(records):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last"]:
            keys.append(("phone", r["phone"], r["last"].lower()))
        for key in keys:
            if key in seen_keys:
                union(seen_keys[key], i)
            else:
                seen_keys[key] = i

    # Group members, ordered by each group's earliest input row.
    groups = {}
    order = []
    for i in range(len(records)):
        root = find(i)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(records[i])

    # Step 3: build one output dict per person.
    result = []
    for root in order:
        members = groups[root]
        tag_union = set()
        for r in members:
            tag_union |= r["tags"]

        result.append({
            "first_name": _pick(members, "first"),
            "last_name": _pick(members, "last"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(tag_union)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
        })

    return result

Notes on interpretation:

- **ZIP+4:** I treated both `02139-4307` and `2139-4307` as ZIP+4 (a 4- or 5-digit prefix followed by a 4-digit suffix), so the rule about adding a leading zero applies to the prefix.
- **Bad dates:** An unrecognized `updated` value raises a `ValueError` rather than being silently ordered. Dates are only parsed for rows that survive the email/phone check, so dropped rows can't cause an error. Since this list goes out tomorrow, I'd rather the run fail loudly than send to the wrong people.
- **Matching:** Rows are merged with union-find, so transitive matches (A↔B, B↔C) end up in one group even when A and C share nothing directly.
