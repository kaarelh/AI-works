```python
import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_ANGLE_RE = re.compile(r"<([^>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_OUTPUT_FIELDS = ("first_name", "last_name", "zip", "email", "phone", "source")


def _norm_email(raw):
    m = _ANGLE_RE.search(raw)
    s = (m.group(1) if m else raw).strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")


def _norm_phone(raw):
    s = _EXT_RE.split(raw, maxsplit=1)[0]
    digits = _NON_DIGIT_RE.sub("", s)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    s = raw.strip()
    if "-" in s:
        s = s.split("-", 1)[0].strip()
    if len(s) == 4 and s.isascii() and s.isdigit():
        s = "0" + s
    return s if _ZIP5_RE.fullmatch(s) else ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _norm_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _opted_out(raw):
    return raw.strip().lower() in _OPT_OUT_VALUES


def _pick(group, field):
    """Best row for a field: manual source > latest updated > earliest input row."""
    candidates = [r for r in group if r[field]]
    if not candidates:
        return ""
    best = max(candidates, key=lambda r: (r["source"] == "manual", r["updated_dt"], -r["idx"]))
    return best[field]


def dedupe_contacts(rows):
    # --- Normalize every row, dropping rows with no valid email and no valid phone ---
    recs = []
    for idx, row in enumerate(rows):
        get = lambda f: row.get(f) or ""
        email = _norm_email(get("email"))
        if not _email_valid(email):
            email = ""
        phone = _norm_phone(get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "idx": idx,
            "first_name": _norm_name(get("first_name")),
            "last_name": _norm_name(get("last_name")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(get("zip")),
            "tags": _norm_tags(get("tags")),
            "opted_out": _opted_out(get("opted_out")),
            "updated_dt": _parse_updated(get("updated")),
            "source": get("source").strip().lower(),
        })

    # --- Union-find to group rows into people (transitive matching) ---
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller root so groups stay anchored to their earliest row
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(i, by_email[r["email"]])
            else:
                by_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in by_phone_last:
                union(i, by_phone_last[key])
            else:
                by_phone_last[key] = i

    groups = {}  # insertion order == order of each group's earliest row
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # --- Build one output dict per person ---
    output = []
    for group in groups.values():
        person = {field: _pick(group, field) for field in _OUTPUT_FIELDS}
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        person["tags"] = ";".join(sorted(all_tags))
        person["opted_out"] = "Y" if any(r["opted_out"] for r in group) else "N"
        person["updated"] = max(r["updated_dt"] for r in group).strftime("%Y-%m-%d")
        output.append(person)

    return output
```

Notes on a few judgment calls:

- **Matching uses normalized values**: email matching only on valid normalized emails; phone matching on valid normalized phone + non-empty normalized last name (lowercased). Union-find makes the matching transitive, and each group is anchored to its earliest input row so output order follows rule 12.
- **Field selection** (rule 11) treats "manual" as the source after stripping/lowercasing, so `" Manual "` counts. Because the same preference applies when picking `source` itself, any group containing a manual row will output `"manual"`.
- **Phone extensions**: the regex cuts at the first `x`, `ext`, or `#` (case-insensitive), so `"555-123-4567 ext. 12"`, `"...x12"`, and `"...#12"` all reduce to the 10-digit number.
- **ZIP**: the hyphen split happens before the leading-zero fix, so `"2139-4307"` → `"2139"` → `"02139"`.
- **Bad `updated` values raise `ValueError`** rather than silently guessing, since the spec says the field is always present and in one of the three formats. If you'd rather skip or default such rows, that's a one-line change in `_parse_updated`.
- `None` values are tolerated (treated as `""`), even though the spec says fields are strings.

Quick sanity check: `[{"email": "Ann Lee <A.N.N+news@GoogleMail.com>", "last_name": "LEE", ...}, {"email": "ann@gmail.com", "last_name": "Lee", "opted_out": "Yes", ...}]` collapses to one person with `email="ann@gmail.com"`, `last_name="Lee"`, `opted_out="Y"`, which is the behavior you want for suppressing opt-outs across duplicates.
