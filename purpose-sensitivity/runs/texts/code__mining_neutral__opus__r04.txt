import re
from datetime import datetime

_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_EXT_MARKERS = ("ext", "x", "#")


def _s(value):
    """Treat None as an empty string so one missing value doesn't crash the run."""
    return "" if value is None else str(value)


def _normalize_email(raw):
    s = _s(raw)
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Return the 10-digit phone if valid, otherwise ""."""
    s = _s(raw)
    lowered = s.lower()
    cut = len(s)
    for marker in _EXT_MARKERS:
        idx = lowered.find(marker)
        if idx != -1 and idx < cut:
            cut = idx
    digits = "".join(re.findall(r"[0-9]", s[:cut]))
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    s = _s(raw).strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    s = _s(raw).strip()
    if s and (s.isupper() or s.islower()):
        return s.title()
    return s


def _parse_updated(raw, row_index):
    s = _s(raw).strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(
        f"Row {row_index}: unrecognized 'updated' value {raw!r} "
        f"(expected YYYY-MM-DD, MM/DD/YYYY, or YYYY-MM-DDTHH:MM:SS)"
    )


def _is_opted_out(raw):
    return _s(raw).strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", _s(raw)) if t.strip()}


def _format_date(dt):
    return f"{dt.year:04d}-{dt.month:02d}-{dt.day:02d}"


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for index, row in enumerate(rows):
        email = _normalize_email(row.get("email"))
        if not _email_is_valid(email):
            email = ""
        phone = _normalize_phone(row.get("phone"))
        if not email and not phone:
            continue
        recs.append({
            "index": index,
            "first_name": _normalize_name(row.get("first_name")),
            "last_name": _normalize_name(row.get("last_name")),
            "zip": _normalize_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": _s(row.get("source")).strip().lower(),
            "tags": _split_tags(row.get("tags")),
            "opted_out": _is_opted_out(row.get("opted_out")),
            "updated": _parse_updated(row.get("updated"), index),
        })

    # ---- Union-find over the kept rows (rule 10, transitive).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the root as the smaller position (stable, not required for correctness).
            if rb < ra:
                ra, rb = rb, ra
            parent[rb] = ra

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Build one output record per person (rule 11).
    def priority(r):
        # max() of this key: manual first, then latest updated, then earliest input row.
        return (r["source"] == "manual", r["updated"], -r["index"])

    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        return max(candidates, key=priority)[field]

    people = []
    for members in groups.values():
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        people.append((
            min(r["index"] for r in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
                "updated": _format_date(max(r["updated"] for r in members)),
                "source": pick(members, "source"),
            },
        ))

    # ---- Order by each person's earliest input row (rule 12).
    people.sort(key=lambda p: p[0])
    return [person for _, person in people]

- **Name casing quirks.** `str.title()` turns `MCDONALD` into `Mcdonald` and `o'brien` into `O'Brien`. That is how the rules are written, but it may matter if the email greets people by name.
- **Implementation details:**

  - Rows with opted-out status but no valid email or phone are dropped. This doesn't lose an opt-out, because such a row can't match anyone anyway.
