import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}

_OUTPUT_KEYS = ("first_name", "last_name", "email", "phone", "zip",
                "tags", "opted_out", "updated", "source")


def _norm_email(raw):
    s = raw or ""
    m = _ANGLE_RE.search(s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_is_valid(e):
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _norm_phone(raw):
    s = raw or ""
    m = _EXT_RE.search(s)  # leftmost marker; "ext" is tried before "x" at each position
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in "0123456789")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(p):
    return len(p) == 10 and p[0] not in "01"


def _norm_zip(raw):
    s = (raw or "").strip()
    m = _ZIP4_RE.fullmatch(s)
    if m:
        s = m.group(1)
    if len(s) == 4 and s.isascii() and s.isdigit():
        s = "0" + s
    return s if _ZIP5_RE.fullmatch(s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw, row_index):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Row {row_index}: unrecognized 'updated' value {raw!r}")


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone (rule 9)
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        phone = _norm_phone(row.get("phone", ""))
        email_ok = _email_is_valid(email)
        phone_ok = _phone_is_valid(phone)
        if not (email_ok or phone_ok):
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email if email_ok else "",
            "phone": phone if phone_ok else "",
            "zip": _norm_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", ""), idx),
        })

    # ---- Union-find over records (rule 10, transitive)
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # keep the root as the earliest record so ordering stays simple
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(i, first_seen[k])
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # ---- Build one output dict per person
    def preference(i):
        r = recs[i]
        age = (r["updated"] - datetime.min).total_seconds()
        return (0 if r["manual"] else 1, -age, r["idx"])

    def pick(members, field):
        candidates = [i for i in members if recs[i][field]]
        if not candidates:
            return ""
        return recs[min(candidates, key=preference)][field]

    output = []
    # recs are in input order, so sorting by the smallest member gives rule 12 ordering
    for members in sorted(groups.values(), key=lambda m: min(m)):
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        person = {
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["updated"] for i in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        }
        output.append({k: person[k] for k in _OUTPUT_KEYS})
    return output

A few things to check before tomorrow's send:

- **Filter out opted-out people.** The function only *marks* them, per the spec. You still need to exclude them before sending, e.g. `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`. That filter also drops phone-only people, who can't receive the email anyway. I'd also keep your email platform's own suppression/unsubscribe list turned on as a backstop, since an opt-out recorded only there wouldn't appear in this export.
- **Gmail addresses change form.** `J.Smith+cas@googlemail.com` comes out as `jsmith@gmail.com`. That reaches the same inbox, but if your email platform matches opt-outs by exact address, a person who unsubscribed under the original form might not be recognized. You could keep the original address for sending and use the normalized one only for matching.
- **Bad dates stop the run.** If any `updated` value isn't in one of the three formats, the function raises a `ValueError` naming the row. I chose that over silently guessing, but it's easy to change if you'd prefer a default.
- **One address per person.** Someone with two different emails in the group gets the message once, at the address picked by the manual, most-recent, earliest-row rule.
