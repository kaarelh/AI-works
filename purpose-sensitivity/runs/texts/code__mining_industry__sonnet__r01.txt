```python
import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ANGLE_RE = re.compile(r"<([^>]*)>")
_ZIP4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def dedupe_contacts(rows):
    # ---------- helpers ----------
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in _GMAIL_DOMAINS:
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        m = _EXT_RE.search(raw)
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        if _ZIP4_RE.fullmatch(z):
            z = z.split("-")[0]
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if re.fullmatch(r"[0-9]{5}", z):
            return z
        return ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        return datetime.min

    def split_tags(raw):
        out = []
        for t in re.split(r"[;,]", raw):
            t = t.strip().lower()
            if t:
                out.append(t)
        return out

    # ---------- normalize rows ----------
    recs = []  # kept rows, in input order
    for row in rows:
        email = norm_email(get(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        recs.append({
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(get(row, "zip")),
            "source": get(row, "source").strip().lower(),
            "tags": split_tags(get(row, "tags")),
            "opted_out": get(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_updated(get(row, "updated")),
        })

    n = len(recs)
    if n == 0:
        return []

    # ---------- union-find ----------
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("e", r["email"]))
        last = r["last_name"].lower()
        if r["phone"] and last:
            keys.append(("p", r["phone"], last))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)  # indices ascending

    # ---------- build output ----------
    def pick(indices, field):
        cands = [i for i in indices if recs[i][field]]
        if not cands:
            return ""
        # indices are already in input order; stable sorts keep that as final tiebreak
        cands.sort(key=lambda i: recs[i]["updated"], reverse=True)
        cands.sort(key=lambda i: 0 if recs[i]["source"] == "manual" else 1)
        return recs[cands[0]][field]

    result = []
    for root in sorted(groups, key=lambda r: groups[r][0]):
        idx = groups[root]
        tags = set()
        for i in idx:
            tags.update(recs[i]["tags"])
        latest = max(recs[i]["updated"] for i in idx)
        result.append({
            "first_name": pick(idx, "first_name"),
            "last_name": pick(idx, "last_name"),
            "zip": pick(idx, "zip"),
            "email": pick(idx, "email"),
            "phone": pick(idx, "phone"),
            "source": pick(idx, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in idx) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes on judgment calls:

- **Opt-outs propagate to the whole person.** If any row in a merged group has opted out, the output is `"Y"`. Filter on `opted_out == "N"` before sending tomorrow.
- **ZIP+4:** I only treat it as ZIP+4 when it's exactly 5 digits, a hyphen, then 4 digits, as in your example. Something like `2139-4307` (Excel ate the zero in a ZIP+4) becomes `""`. If you want that rescued to `02139`, change `_ZIP4_RE` to `[0-9]{4,5}-[0-9]{4}`.
- **Digits:** I use `[0-9]` rather than `str.isdigit`, so non-ASCII digits aren't counted.
- **Unparseable `updated` values:** these are treated as the earliest possible date rather than raising an error. The spec says that shouldn't happen.
- **Source selection:** a row with an empty source can't supply the `source` field, but it can still supply other fields.
- **Transitive merging:** a union-find handles it, so chains like A–B–C collapse into one person.

I'd spot-check the output on a handful of known duplicates before you load it into the send tool.
