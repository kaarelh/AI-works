import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_TRUE_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
_ASCII_DIGITS = "0123456789"


def _normalize_email(raw):
    s = raw or ""
    # Rule 1: "Ann Lee <ann@x.org>" -> "ann@x.org"
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _is_valid_email(email):
    # Rule 2
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    """Return the normalized 10-digit phone, or "" if invalid (rule 3)."""
    s = raw or ""
    lower = s.lower()
    cut_points = [i for i in (lower.find(mk) for mk in ("x", "ext", "#")) if i >= 0]
    if cut_points:
        s = s[: min(cut_points)]
    digits = "".join(c for c in s if c in _ASCII_DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    # Rule 4
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _normalize_name(raw):
    # Rule 5
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    # Rule 6
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    # Rule 7
    return (raw or "").strip().lower() in _TRUE_VALUES


def _split_tags(raw):
    # Rule 8 (+ normalization from rule 11)
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # --- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _is_valid_email(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # --- Union-find over records (rule 10, transitive).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the root as the lower index for stable ordering.
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # --- Build one output dict per person (rule 11).
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = min(
            candidates,
            key=lambda r: (not r["is_manual"], -r["updated"].timestamp(), r["idx"]),
        )
        return best[field]

    people = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        people.append((
            min(r["idx"] for r in group),
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "zip": pick(group, "zip"),
                "source": pick(group, "source"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            },
        ))

    # --- Order by earliest input row (rule 12).
    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

A few notes before you send tomorrow:

- **The function doesn't remove opted-out people. It flags them.** Per the spec, a merged person gets `opted_out: "Y"` if *any* of their rows opted out. Your send list must then exclude `opted_out == "Y"`.
- **Phone-only contacts come out with `email: ""`.** Filter those out of the email send as well.
- **Some opt-outs can still slip through.** If someone opted out under one address but appears elsewhere under a different email (with no matching phone and last name), the rules can't link those rows, so they stay as two people. That's inherent to the spec. If you keep a separate suppression list, apply it to the final output too.
- **Gmail addresses get rewritten.** `J.Doe+news@googlemail.com` becomes `jdoe@gmail.com`. Mail sent there reaches the same inbox, but if your email platform matches unsubscribes by the exact address string, check the rewritten addresses against its suppression list.
- **Bad dates stop the run.** An unparseable `updated` value raises a `ValueError` naming that value, rather than being silently sorted somewhere arbitrary.
- **Details on edge cases:**
  - The extension cut happens at the earliest marker found, so in `555-123-4567 ext. 9` the cut is at "ext", not at its "x".
  - Digit checks accept only ASCII 0–9.
  - When picking a value, ties on `updated` fall back to the earlier input row, as specified.

I'd suggest running it on a small sample with a few known duplicates and one known opt-out, and eyeballing the result before the full send.
