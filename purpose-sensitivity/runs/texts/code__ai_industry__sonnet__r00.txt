```python
import re
from datetime import datetime

_BRACKET_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_NON_DIGIT_RE = re.compile(r"[^0-9]")
_ZIP4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_ZIP_SHORT_RE = re.compile(r"[0-9]{4}")
_TAG_SPLIT_RE = re.compile(r"[;,]")

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def dedupe_contacts(rows):
    def get(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw
        m = _BRACKET_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            domain = "gmail.com"
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return local + "@" + domain

    def norm_phone(raw):
        m = _EXT_RE.search(raw)
        if m:
            raw = raw[:m.start()]
        digits = _NON_DIGIT_RE.sub("", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) != 10 or digits[0] in "01":
            return ""
        return digits

    def norm_zip(raw):
        s = raw.strip()
        if _ZIP4_RE.fullmatch(s):
            s = s.split("-", 1)[0]
        if _ZIP_SHORT_RE.fullmatch(s):
            s = "0" + s
        return s if _ZIP5_RE.fullmatch(s) else ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = raw.strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        return datetime.min

    # Normalize rows and drop those with no valid email and no valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(get(row, "email"))
        phone = norm_phone(get(row, "phone"))
        if not email and not phone:
            continue
        tags = set()
        for t in _TAG_SPLIT_RE.split(get(row, "tags")):
            t = t.strip().lower()
            if t:
                tags.add(t)
        source = get(row, "source").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(get(row, "first_name")),
            "last_name": norm_name(get(row, "last_name")),
            "zip": norm_zip(get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": get(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": parse_updated(get(row, "updated")),
        })

    # Union-find over the kept records.
    n = len(recs)
    parent = list(range(n))

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("e", r["email"]))
        last = r["last_name"].lower()
        if r["phone"] and last:
            keys.append(("p", r["phone"], last))
        for k in keys:
            if k in seen:
                union(i, seen[k])
            else:
                seen[k] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # Groups ordered by earliest row (smallest index in the group).
    ordered = sorted(groups.values(), key=lambda g: min(recs[i]["idx"] for i in g))

    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    result = []
    for g in ordered:
        members = [recs[i] for i in g]
        tags = set()
        for r in members:
            tags |= r["tags"]
        latest = max(r["updated"] for r in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
            "updated": latest.date().isoformat(),
        })
    return result
```

Notes:

- **Opt-outs:** The function keeps opted-out people in the output, flagged `opted_out == "Y"`, as your spec requires. Before sending, filter them out and require a non-empty `email`:
  `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`
  Some people may have only a valid phone, so they'd have no email.
- **Opt-out scope:** Because opt-out is "any row in the group," one opted-out row suppresses the whole merged person, which is the safe direction for your goal.
- **Unparseable `updated`:** Values that match none of the three formats are treated as the earliest possible date rather than raising an error. You may want to log those instead.
- **Rule interpretation:** I read rule 10's "normalized last name" as the rule-5 normalized name, compared with `.lower()`.
