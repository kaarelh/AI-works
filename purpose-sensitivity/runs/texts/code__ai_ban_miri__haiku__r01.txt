import re
from datetime import datetime

_DIGITS = "0123456789"
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ZIP_PLUS4 = re.compile(r"([0-9]{4,5})-[0-9]{4}")


def _get(row, key):
    value = row.get(key)
    return "" if value is None else str(value)


def _is_ascii_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    email = raw
    lt = email.find("<")
    if lt != -1:
        gt = email.find(">", lt + 1)
        if gt != -1:
            email = email[lt + 1:gt]
    email = email.strip().lower()
    if email.count("@") == 1:
        local, domain = email.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "").split("+", 1)[0]
            email = f"{local}@gmail.com"
    return email


def _valid_email(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _normalize_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        pos = s.find(marker)
        if pos != -1 and pos < cut:
            cut = pos
    digits = "".join(c for c in s[:cut] if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(digits):
    return len(digits) == 10 and digits[0] not in "01"


def _normalize_zip(raw):
    z = raw.strip()
    m = _ZIP_PLUS4.fullmatch(z)
    if m:
        z = m.group(1)
    if _is_ascii_digits(z) and len(z) == 4:
        z = "0" + z
    return z if _is_ascii_digits(z) and len(z) == 5 else ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return None


def _normalize_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def _pick(members, field):
    """Non-empty value from the best row: manual first, then most recent
    updated, then earliest input row (members are already in input order)."""
    cands = [m for m in members if m[field]]
    if not cands:
        return ""
    cands.sort(key=lambda m: m["updated"] if m["updated"] is not None else datetime.min,
               reverse=True)  # stable, so input order breaks ties
    cands.sort(key=lambda m: m["source"] != "manual")  # stable; manual first
    return cands[0][field]


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    records = []
    for row in rows:
        email = _normalize_email(_get(row, "email"))
        if not _valid_email(email):
            email = ""
        phone = _normalize_phone(_get(row, "phone"))
        if not _valid_phone(phone):
            phone = ""
        if not email and not phone:
            continue
        records.append({
            "first_name": _normalize_name(_get(row, "first_name")),
            "last_name": _normalize_name(_get(row, "last_name")),
            "zip": _normalize_zip(_get(row, "zip")),
            "email": email,
            "phone": phone,
            "source": _get(row, "source").strip().lower(),
            "tags": _normalize_tags(_get(row, "tags")),
            "opted_out": _get(row, "opted_out").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(_get(row, "updated")),
        })

    # Step 2: union-find over records (positions are in input order).
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for pos, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in by_email:
                union(pos, by_email[rec["email"]])
            else:
                by_email[rec["email"]] = pos
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].lower())
            if key in by_phone_last:
                union(pos, by_phone_last[key])
            else:
                by_phone_last[key] = pos

    # Step 3: group rows; dict order follows each group's earliest row.
    groups = {}
    for pos in range(len(records)):
        groups.setdefault(find(pos), []).append(records[pos])

    # Step 4: build one output dict per person.
    output = []
    for members in groups.values():
        tags = set()
        for m in members:
            tags |= m["tags"]
        dates = [m["updated"] for m in members if m["updated"] is not None]
        output.append({
            "first_name": _pick(members, "first_name"),
            "last_name": _pick(members, "last_name"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": max(dates).strftime("%Y-%m-%d") if dates else "",
        })
    return output

- **Interpretation choices I made:**
