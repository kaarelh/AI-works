```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    """Normalize, merge, and deduplicate CRM contact rows.

    Takes a list of dicts (first_name, last_name, email, phone, zip, tags,
    opted_out, updated, source) and returns one merged dict per person,
    ordered by each person's earliest row in the input.
    """
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    EXT_MARKERS = ("ext", "x", "#")

    # ------------------------------------------------------------------ #
    # Field normalizers
    # ------------------------------------------------------------------ #
    def text(row, key):
        v = row.get(key)
        return "" if v is None else str(v)

    def norm_email(raw):
        s = raw.strip()
        m = re.search(r"<([^<>]*)>", s)          # "Ann Lee <ann@x.org>"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_is_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        """Returns the 10-digit normalized phone, or "" if invalid."""
        s = raw.lower()
        cut = len(s)
        for marker in EXT_MARKERS:
            i = s.find(marker)
            if i != -1:
                cut = min(cut, i)
        s = s[:cut]
        digits = re.sub(r"\D", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = raw.strip()
        if "-" in s:                               # ZIP+4 -> keep ZIP5 part
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"\d{4}", s):              # Excel ate the leading 0
            s = "0" + s
        return s if re.fullmatch(r"\d{5}", s) else ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                continue
        return datetime.min  # unparsable -> treated as oldest

    def is_opted_out(raw):
        return raw.strip().lower() in ("y", "yes", "true", "1")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ------------------------------------------------------------------ #
    # Normalize every row; drop rows with no valid email and no valid phone
    # ------------------------------------------------------------------ #
    normed = {}  # original index -> normalized record
    for idx, row in enumerate(rows):
        email = norm_email(text(row, "email"))
        if not email_is_valid(email):
            email = ""
        phone = norm_phone(text(row, "phone"))
        if not email and not phone:
            continue  # rule 9
        source = text(row, "source").strip().lower()
        normed[idx] = {
            "first_name": norm_name(text(row, "first_name")),
            "last_name": norm_name(text(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(text(row, "zip")),
            "tags": norm_tags(text(row, "tags")),
            "opted_out": is_opted_out(text(row, "opted_out")),
            "updated_dt": parse_updated(text(row, "updated")),
            "source": source,
            "is_manual": source == "manual",
        }

    # ------------------------------------------------------------------ #
    # Union-find to group rows into people (transitive matching)
    # ------------------------------------------------------------------ #
    parent = {i: i for i in normed}

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the smaller index as root so groups stay anchored early
            if rb < ra:
                ra, rb = rb, ra
            parent[rb] = ra

    seen_email = {}
    seen_phone_last = {}
    for idx in sorted(normed):
        rec = normed[idx]
        if rec["email"]:
            if rec["email"] in seen_email:
                union(idx, seen_email[rec["email"]])
            else:
                seen_email[rec["email"]] = idx
        if rec["phone"] and rec["last_name"]:
            key = (rec["phone"], rec["last_name"].lower())
            if key in seen_phone_last:
                union(idx, seen_phone_last[key])
            else:
                seen_phone_last[key] = idx

    groups = {}  # root -> list of member indices in input order
    for idx in sorted(normed):
        groups.setdefault(find(idx), []).append(idx)

    # ------------------------------------------------------------------ #
    # Build one output record per person
    # ------------------------------------------------------------------ #
    def pick(members, field):
        """Best non-empty value: manual source > latest updated > earliest row."""
        best_val, best_key = "", None
        for i in members:
            val = normed[i][field]
            if not val:
                continue
            key = (normed[i]["is_manual"], normed[i]["updated_dt"], -i)
            if best_key is None or key > best_key:
                best_key, best_val = key, val
        return best_val

    output = []
    # dict preserves insertion order; groups were inserted by earliest member
    for members in groups.values():
        tags = set()
        for i in members:
            tags |= normed[i]["tags"]
        latest = max(normed[i]["updated_dt"] for i in members)
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(normed[i]["opted_out"] for i in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })

    return output
```

A few notes on choices where the spec left a little room:

- **Phone extensions:** I cut at the earliest of `ext`, `x`, or `#`. Since `ext` contains `x`, cutting at the `x` would give the same digits anyway, but checking all three is explicit.
- **ZIP+4:** anything before the first hyphen is kept (so `2139-4307` → `02139`). Anything that isn't 5 digits after the 4-digit fix becomes `""`.
- **Unparsable `updated`:** the spec says it's always one of the three formats; just in case, a bad value is treated as `datetime.min` (oldest) rather than crashing the run.
- **Normalized phone output** is the bare 10-digit string (e.g. `5105551234`).
- **Email matching keys are case-insensitive** by construction (everything's lowercased), and phone+last-name matching lowercases the normalized last name as rule 10 asks.
- Missing keys or `None` values are treated as `""`, so a slightly ragged export won't blow up.
