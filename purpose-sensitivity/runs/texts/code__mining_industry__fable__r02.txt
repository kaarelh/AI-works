```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    # ---------- field normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        return s

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        s = raw or ""
        m = re.search(r"ext|[x#]", s, flags=re.IGNORECASE)
        if m:
            s = s[: m.start()]
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(p):
        return bool(re.fullmatch(r"[0-9]{10}", p)) and p[0] not in "01"

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:  # ZIP+4 -> keep the part before the hyphen
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):  # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min  # unparseable -> treated as oldest possible

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize rows, drop unusable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        ev, pv = email_valid(email), phone_valid(phone)
        if not ev and not pv:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "zip": norm_zip(row.get("zip", "")),
            "email": email if ev else "",   # only valid emails are kept/used
            "phone": phone if pv else "",   # only valid phones are kept/used
            "source": source,
            "manual": source == "manual",
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", "")),
        })

    # ---------- group matching rows with union-find (rule 10, transitive) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the root with the smaller index so order is stable
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    seen_email = {}
    seen_phone = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone:
                union(i, seen_phone[key])
            else:
                seen_phone[key] = i

    groups = {}  # root -> list of recs; insertion order == earliest row order (rule 12)
    for i, r in enumerate(recs):
        groups.setdefault(find(i), []).append(r)

    # ---------- build one output dict per person (rule 11) ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        # prefer: manual source, then most recent updated, then earliest input row
        best = max(cands, key=lambda r: (r["manual"], r["updated"], -r["idx"]))
        return best[field]

    output = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        output.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "zip": pick(group, "zip"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "source": pick(group, "source"),
            "tags": ";".join(sorted(all_tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
        })
    return output
```

A few judgment calls worth knowing about:

- **Invalid emails/phones are discarded entirely.** A row with a malformed email but a valid phone is kept, but its email is treated as "" for both matching and output, so a bad email can never win the "pick" step.
- **Gmail handling only applies when there's exactly one `@`.** Anything else fails validation anyway.
- **ZIP hyphen handling** splits on the first hyphen regardless of what follows; anything that isn't 5 digits afterward (optionally after re-adding a leading zero) becomes "". So `"2139-4307"` → `"02139"`.
- **Phone extension markers:** the regex cuts at the earliest `x`, `#`, or `ext` (case-insensitive). Since "ext" contains no digits, cutting at the "e" vs the "x" gives the same result.
- **Unparseable `updated` values** fall back to `datetime.min` rather than raising, so one bad row won't crash the whole run. If you'd rather it fail loudly, change the `return datetime.min` to `raise ValueError(...)`.
- **The input dicts are not mutated**; all normalization is done on copies.

One reminder since this is an advocacy send: `opted_out` only reflects what's in the CRM export, so make sure your email platform's own suppression list is still applied on send — this function won't know about unsubscribes that happened after the export.
