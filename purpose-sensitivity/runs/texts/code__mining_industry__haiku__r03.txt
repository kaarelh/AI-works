```python
import re
from datetime import datetime

_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _norm_email(raw):
    s = raw.strip()
    lt = s.find("<")
    if lt != -1:
        gt = s.find(">", lt + 1)
        if gt != -1:
            s = s[lt + 1:gt]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"
    return s


def _is_valid_email(s):
    if s.count("@") != 1:
        return False
    local, domain = s.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw.lower()
    cut = len(s)
    for marker in ("x", "ext", "#"):
        i = s.find(marker)
        if i != -1 and i < cut:
            cut = i
    digits = [c for c in s[:cut] if c in _DIGITS]
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return "".join(digits)
    return ""


def _norm_zip(raw):
    z = raw.strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", z)
    if m:
        z = m.group(1)
    if len(z) == 4 and all(c in _DIGITS for c in z):
        z = "0" + z
    if len(z) == 5 and all(c in _DIGITS for c in z):
        return z
    return ""


def _norm_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_ts(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return datetime.min


def _pick(members, field):
    """Pick the field's value from the group, preferring manual rows, then newest, then earliest input."""
    cands = [r for r in members if r[field]]
    if not cands:
        return ""
    cands.sort(key=lambda r: r["ts"], reverse=True)   # newest first; ties keep input order
    cands.sort(key=lambda r: not r["manual"])          # manual rows first (stable)
    return cands[0][field]


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    records = []
    for row in rows:
        email = _norm_email(row.get("email") or "")
        if not _is_valid_email(email):
            email = ""
        phone = _norm_phone(row.get("phone") or "")
        if not email and not phone:
            continue

        source = (row.get("source") or "").strip().lower()
        tags = []
        for t in re.split(r"[;,]", row.get("tags") or ""):
            t = t.strip().lower()
            if t:
                tags.append(t)

        records.append({
            "first": _norm_name(row.get("first_name") or ""),
            "last": _norm_name(row.get("last_name") or ""),
            "zip": _norm_zip(row.get("zip") or ""),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "ts": _parse_ts(row.get("updated") or ""),
            "opted": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "tags": tags,
        })

    # Step 2: union-find so matching is transitive.
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    email_owner = {}
    phone_owner = {}
    for pos, r in enumerate(records):
        if r["email"]:
            if r["email"] in email_owner:
                union(pos, email_owner[r["email"]])
            else:
                email_owner[r["email"]] = pos
        if r["phone"] and r["last"]:
            key = (r["phone"], r["last"].lower())
            if key in phone_owner:
                union(pos, phone_owner[key])
            else:
                phone_owner[key] = pos

    # Step 3: group rows; groups are ordered by their earliest input row.
    groups = {}
    order = []
    for pos, r in enumerate(records):
        root = find(pos)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(r)

    # Step 4: build one output dict per person.
    people = []
    for root in order:
        members = groups[root]
        people.append({
            "first_name": _pick(members, "first"),
            "last_name": _pick(members, "last"),
            "zip": _pick(members, "zip"),
            "email": _pick(members, "email"),
            "phone": _pick(members, "phone"),
            "source": _pick(members, "source"),
            "tags": ";".join(sorted({t for r in members for t in r["tags"]})),
            "opted_out": "Y" if any(r["opted"] for r in members) else "N",
            "updated": max(r["ts"] for r in members).strftime("%Y-%m-%d"),
        })
    return people
```

Notes on the implementation:

- **Opt-outs:** A person with `opted_out == "Y"` is still returned, as the spec requires. Filter them out before you build the send list, e.g. `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N"]`.
- **Judgment calls where the spec was silent:**
  - ZIP+4 is recognized as 4–5 digits, a hyphen, and 4 digits. That way "2139-4307" becomes "02139" rather than being rejected.
  - An `updated` value that fails to parse is treated as the oldest possible date rather than raising an error.
