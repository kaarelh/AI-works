```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")
    EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
    ZIP4_RE = re.compile(r"([0-9]{5})-[0-9]{4}")

    def text(row, key):
        value = row.get(key)
        return "" if value is None else str(value)

    def norm_email(raw):
        s = raw
        lt = s.find("<")
        if lt != -1:
            gt = s.find(">", lt + 1)
            if gt != -1:
                s = s[lt + 1:gt]
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.replace(".", "")
                plus = local.find("+")
                if plus != -1:
                    local = local[:plus]
                domain = "gmail.com"
                s = local + "@" + domain
        # validity
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local:
            return ""
        if "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        m = EXT_RE.search(raw)
        s = raw[:m.start()] if m else raw
        digits = re.sub(r"[^0-9]", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) != 10 or digits[0] in "01":
            return ""
        return digits

    def norm_zip(raw):
        s = raw.strip()
        m = ZIP4_RE.fullmatch(s)
        if m:
            s = m.group(1)
        if re.fullmatch(r"[0-9]{4}", s):
            s = "0" + s
        if re.fullmatch(r"[0-9]{5}", s):
            return s
        return ""

    def norm_name(raw):
        s = raw.strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_date(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError("Unrecognized 'updated' value: %r" % raw)

    def split_tags(raw):
        out = []
        for part in re.split(r"[;,]", raw):
            part = part.strip().lower()
            if part:
                out.append(part)
        return out

    # ---- Normalize rows, dropping those with no valid email or phone ----
    recs = []
    for row in rows:
        email = norm_email(text(row, "email"))
        phone = norm_phone(text(row, "phone"))
        if not email and not phone:
            continue
        source = text(row, "source").strip().lower()
        recs.append({
            "first_name": norm_name(text(row, "first_name")),
            "last_name": norm_name(text(row, "last_name")),
            "zip": norm_zip(text(row, "zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": split_tags(text(row, "tags")),
            "opted_out": text(row, "opted_out").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_date(text(row, "updated")),
        })

    n = len(recs)
    if n == 0:
        return []

    # ---- Union-find ----
    parent = list(range(n))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(i, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = i
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    groups = {}
    for i in range(n):
        groups.setdefault(find(i), []).append(i)

    # ---- Build output ----
    def pick(members, field):
        best = None
        for idx in members:  # members are in input order
            r = recs[idx]
            if not r[field]:
                continue
            rank = 0 if r["manual"] else 1
            if best is None:
                best = (rank, r["updated"], r[field])
            else:
                b_rank, b_updated, _ = best
                if rank < b_rank or (rank == b_rank and r["updated"] > b_updated):
                    best = (rank, r["updated"], r[field])
        return best[2] if best else ""

    result = []
    for members in sorted(groups.values(), key=lambda m: m[0]):
        tags = set()
        for idx in members:
            tags.update(recs[idx]["tags"])
        latest = max(recs[idx]["updated"] for idx in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[idx]["opted_out"] for idx in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes:

- **Opted-out people stay in the output.** Per the rules, they're flagged `opted_out == "Y"` rather than removed. Before tomorrow's send, filter them out, e.g. `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`. Also, people with only a valid phone have `email == ""`, so the email list needs that second check.
- **Opt-outs propagate across a group.** If any duplicate row opted out, the whole person is marked `Y`. That's the conservative behavior you want.
- **Unparseable `updated` values raise a `ValueError`.** The spec says only three formats occur, so I chose a loud failure over silently mis-ordering records.
- **Tie-breaks and edge cases:** "Entirely upper/lower-case" uses `str.isupper()`/`islower()`. ZIP+4 is recognized strictly as 5 digits, a hyphen, then 4 digits. The name comparison for matching uses `.lower()` on the already-normalized last name.
