```python
import re
from datetime import datetime

_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _normalize_email(raw):
    """Rules 1: extract the address, strip, lowercase, and canonicalize gmail/googlemail."""
    s = raw
    start = s.find("<")
    if start != -1:
        end = s.find(">", start + 1)
        if end != -1:
            s = s[start + 1:end]
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.replace(".", "")
            local = local.split("+", 1)[0]
            s = local + "@gmail.com"
    return s


def _is_valid_email(addr):
    """Rule 2."""
    if addr.count("@") != 1:
        return False
    local, domain = addr.split("@")
    if not local:
        return False
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return False
    return True


def _normalize_phone(raw):
    """Rule 3. Returns (digits, is_valid)."""
    s = raw.lower()
    cuts = [i for i in (s.find("x"), s.find("#"), s.find("ext")) if i != -1]
    if cuts:
        s = s[:min(cuts)]
    digits = "".join(c for c in s if "0" <= c <= "9")
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    is_valid = len(digits) == 10 and digits[0] not in "01"
    return digits, is_valid


def _normalize_zip(raw):
    """Rule 4."""
    z = raw.strip()
    if re.fullmatch(r"[0-9]+-[0-9]{4}", z):
        z = z.split("-", 1)[0]
    if re.fullmatch(r"[0-9]{4}", z):
        z = "0" + z
    return z if re.fullmatch(r"[0-9]{5}", z) else ""


def _normalize_name(raw):
    """Rule 5."""
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    """Rule 6. Raises ValueError on an unrecognized format so bad data is caught early."""
    s = raw.strip()
    for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _pick(recs, field):
    """Rule 11: prefer manual rows, then most recent updated, then earliest row.
    recs must be in input order; max() returns the first maximal element, which
    gives the earliest-row tiebreak."""
    cands = [r for r in recs if r[field]]
    if not cands:
        return ""
    best = max(cands, key=lambda r: (r["manual"], r["updated"]))
    return best[field]


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows into one record per person.

    Returns a list of dicts (keys: first_name, last_name, zip, email, phone,
    source, tags, opted_out, updated), ordered by each person's earliest input row.
    Note: opted-out people are returned with opted_out == "Y"; the caller must
    exclude them before sending.
    """
    # Step 1: normalize each row and drop rows with no valid email or phone (rule 9).
    records = []
    for row in rows:
        email = _normalize_email(row.get("email", ""))
        email = email if _is_valid_email(email) else ""
        digits, phone_ok = _normalize_phone(row.get("phone", ""))
        phone = digits if phone_ok else ""
        if not email and not phone:
            continue

        source = row.get("source", "").strip().lower()
        tags = set()
        for t in re.split(r"[;,]", row.get("tags", "")):
            t = t.strip().lower()
            if t:
                tags.add(t)

        records.append({
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "zip": _normalize_zip(row.get("zip", "")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": row.get("opted_out", "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row["updated"]),
        })

    # Step 2: union-find. Rows sharing a valid email, or a valid phone plus the
    # same non-empty last name (case-insensitive), are the same person (rule 10).
    parent = list(range(len(records)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    email_owner = {}
    phone_owner = {}
    for i, r in enumerate(records):
        if r["email"]:
            if r["email"] in email_owner:
                union(i, email_owner[r["email"]])
            else:
                email_owner[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in phone_owner:
                union(i, phone_owner[key])
            else:
                phone_owner[key] = i

    # Group rows; dict insertion order follows each group's earliest row (rule 12).
    groups = {}
    for i in range(len(records)):
        groups.setdefault(find(i), []).append(i)

    # Step 3: build one output record per person (rule 11).
    result = []
    for members in groups.values():
        recs = [records[i] for i in members]
        tags = set()
        for r in recs:
            tags |= r["tags"]
        result.append({
            "first_name": _pick(recs, "first_name"),
            "last_name": _pick(recs, "last_name"),
            "zip": _pick(recs, "zip"),
            "email": _pick(recs, "email"),
            "phone": _pick(recs, "phone"),
            "source": _pick(recs, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in recs) else "N",
            "updated": max(r["updated"] for r in recs).strftime("%Y-%m-%d"),
        })
    return result
```

Notes on the implementation:

- **Opt-outs:** The function doesn't filter them out. Before tomorrow's send, drop anyone with `opted_out == "Y"`.
- **Dropped rows:** Rows with no valid email or phone are discarded before matching, so their tags and opt-out flags don't affect any output record. Those rows also have no usable contact info, so nothing is lost for sending.
- **Bad dates:** An `updated` value in an unexpected format raises `ValueError` rather than being guessed at. That's intentional for a send this important, but it means one malformed row stops the whole run. Tell me if you'd prefer it to skip or flag such rows.
- **Matching:** Linking uses union-find over shared keys, which gives the transitive closure the rules call for and avoids comparing every pair of rows.
