import re
from datetime import datetime

_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_ZIP_PLUS4_RE = re.compile(r"[0-9]{4,5}-[0-9]{4}")
_EXT_MARKER_RE = re.compile(r"ext|x|#")
_DIGITS = "0123456789"


def _is_digits(s):
    return bool(s) and all(c in _DIGITS for c in s)


def _normalize_email(raw):
    """Return the normalized email, or "" if it is not valid."""
    s = raw
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()

    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.replace(".", "").split("+", 1)[0]
            domain = "gmail.com"
        s = f"{local}@{domain}"

    if s.count("@") != 1:
        return ""
    local, domain = s.split("@")
    if not local:
        return ""
    if "." not in domain or domain.startswith(".") or domain.endswith("."):
        return ""
    return s


def _normalize_phone(raw):
    """Return the normalized 10-digit phone, or "" if it is not valid."""
    s = raw.lower()
    m = _EXT_MARKER_RE.search(s)
    if m:
        s = s[:m.start()]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _normalize_zip(raw):
    s = raw.strip()
    if _ZIP_PLUS4_RE.fullmatch(s):
        s = s.split("-", 1)[0]
    if len(s) == 4 and _is_digits(s):
        s = "0" + s
    if len(s) == 5 and _is_digits(s):
        return s
    return ""


def _normalize_name(raw):
    s = raw.strip()
    if s.isupper() or s.islower():
        s = s.title()
    return s


def _parse_updated(raw):
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def dedupe_contacts(rows):
    # Normalize each row and drop rows with neither a valid email nor a valid phone.
    recs = []
    for row in rows:
        email = _normalize_email(row["email"])
        phone = _normalize_phone(row["phone"])
        if not email and not phone:
            continue
        source = row["source"].strip().lower()
        recs.append({
            "first_name": _normalize_name(row["first_name"]),
            "last_name": _normalize_name(row["last_name"]),
            "zip": _normalize_zip(row["zip"]),
            "email": email,
            "phone": phone,
            "source": source,
            "is_manual": source == "manual",
            "tags": {t.strip().lower() for t in re.split(r"[;,]", row["tags"]) if t.strip()},
            "opted_out": row["opted_out"].strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row["updated"]),
        })

    # Union-find over row indices, so matches are transitive.
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    email_seen = {}
    phone_seen = {}
    for i, rec in enumerate(recs):
        if rec["email"]:
            if rec["email"] in email_seen:
                union(email_seen[rec["email"]], i)
            else:
                email_seen[rec["email"]] = i
        last_cf = rec["last_name"].casefold()
        if rec["phone"] and last_cf:
            key = (rec["phone"], last_cf)
            if key in phone_seen:
                union(phone_seen[key], i)
            else:
                phone_seen[key] = i

    # Group rows by person. Dict insertion order follows each group's earliest row.
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    def pick(idxs, field):
        candidates = [i for i in idxs if recs[i][field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda i: (recs[i]["is_manual"], recs[i]["updated"], -i),
        )
        return recs[best][field]

    people = []
    for idxs in groups.values():
        people.append({
            "first_name": pick(idxs, "first_name"),
            "last_name": pick(idxs, "last_name"),
            "zip": pick(idxs, "zip"),
            "email": pick(idxs, "email"),
            "phone": pick(idxs, "phone"),
            "source": pick(idxs, "source"),
            "tags": ";".join(sorted(set().union(*(recs[i]["tags"] for i in idxs)))),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in idxs) else "N",
            "updated": max(recs[i]["updated"] for i in idxs).strftime("%Y-%m-%d"),
        })
    return people

- **Interpretation choices:**
