```python
import re
from datetime import datetime

_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP_PLUS4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_FIVE_DIGITS_RE = re.compile(r"[0-9]{5}")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")


def _field(row, key):
    value = row.get(key)
    return value if isinstance(value, str) else ("" if value is None else str(value))


def _normalize_email(raw):
    # Rule 1: use only the part inside angle brackets, if present.
    match = _ANGLE_RE.search(raw)
    email = match.group(1) if match else raw
    email = email.strip().lower()
    if email.count("@") == 1:
        local, domain = email.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            email = local + "@gmail.com"
    return email


def _is_valid_email(email):
    # Rule 2
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    # Rule 3: cut at the first extension marker, keep digits, drop leading 1.
    match = _EXT_RE.search(raw)
    if match:
        raw = raw[: match.start()]
    digits = re.sub(r"[^0-9]", "", raw)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""  # invalid


def _normalize_zip(raw):
    # Rule 4
    z = raw.strip()
    match = _ZIP_PLUS4_RE.fullmatch(z)
    if match:
        z = match.group(1)
    if len(z) == 4 and z.isascii() and z.isdigit():
        z = "0" + z
    return z if _FIVE_DIGITS_RE.fullmatch(z) else ""


def _normalize_name(raw):
    # Rule 5
    name = raw.strip()
    if name and (name.isupper() or name.islower()):
        name = name.title()
    return name


def _parse_updated(raw):
    # Rule 6
    s = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    # Rule 7
    return raw.strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    # Rule 8 (plus normalization for rule 11)
    return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    records = []
    for index, row in enumerate(rows):
        email = _normalize_email(_field(row, "email"))
        email = email if _is_valid_email(email) else ""
        phone = _normalize_phone(_field(row, "phone"))
        if not email and not phone:
            continue
        source = _field(row, "source").strip().lower()
        records.append({
            "index": index,
            "first_name": _normalize_name(_field(row, "first_name")),
            "last_name": _normalize_name(_field(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(_field(row, "zip")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _split_tags(_field(row, "tags")),
            "opted_out": _is_opted_out(_field(row, "opted_out")),
            "updated": _parse_updated(_field(row, "updated")),
        })

    # ---- Union-find for transitive matching (rule 10).
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the lower position (earlier in input).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for pos, rec in enumerate(records):
        keys = []
        if rec["email"]:
            keys.append(("email", rec["email"]))
        if rec["phone"] and rec["last_name"]:
            keys.append(("phone_last", rec["phone"], rec["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(pos, first_seen[key])
            else:
                first_seen[key] = pos

    groups = {}
    for pos in range(len(records)):
        groups.setdefault(find(pos), []).append(records[pos])

    # ---- Build one output dict per person (rule 11).
    def pick(group, field):
        candidates = [r for r in group if r[field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda r: (r["is_manual"], r["updated"], -r["index"]),
        )
        return best[field]

    people = []
    for group in groups.values():
        all_tags = set()
        for r in group:
            all_tags |= r["tags"]
        people.append((
            min(r["index"] for r in group),
            {
                "first_name": pick(group, "first_name"),
                "last_name": pick(group, "last_name"),
                "email": pick(group, "email"),
                "phone": pick(group, "phone"),
                "zip": pick(group, "zip"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
                "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
                "source": pick(group, "source"),
            },
        ))

    # ---- Order by each person's earliest row in the input (rule 12).
    people.sort(key=lambda item: item[0])
    return [person for _, person in people]
```
