import re
from datetime import datetime

_ASCII_DIGITS = "0123456789"
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_EXT_RE = re.compile(r"ext|x|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"([0-9]{4,5})-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_TAG_SPLIT_RE = re.compile(r"[;,]")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}
_EPOCH = datetime(1, 1, 1)


def dedupe_contacts(rows):
    """Deduplicate CRM contact rows according to the data team's rules.

    Returns a list of dicts (one per person) with keys:
    first_name, last_name, email, phone, zip, tags, opted_out, updated, source.
    """

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = _ANGLE_RE.search(s)
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+", 1)[0]
                s = local + "@gmail.com"
        return s

    def email_is_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return (
            bool(local)
            and "." in domain
            and not domain.startswith(".")
            and not domain.endswith(".")
        )

    def norm_phone(raw):
        """Return the 10-digit phone if valid, else ""."""
        s = raw or ""
        m = _EXT_RE.search(s)
        if m:
            s = s[: m.start()]
        digits = "".join(c for c in s if c in _ASCII_DIGITS)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        s = (raw or "").strip()
        m = _ZIP4_RE.fullmatch(s)
        if m:
            s = m.group(1)
        if len(s) == 4 and all(c in _ASCII_DIGITS for c in s):
            s = "0" + s
        return s if _ZIP5_RE.fullmatch(s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            s = s.title()
        return s

    def parse_updated(raw, idx):
        s = (raw or "").strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Row {idx}: unrecognized 'updated' value {raw!r}")

    def parse_tags(raw):
        out = set()
        for t in _TAG_SPLIT_RE.split(raw or ""):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # ---------- normalize rows, drop unreachable ones (rule 9) ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        if not email_is_valid(email):
            email = ""
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue

        dt = parse_updated(row.get("updated", ""), idx)
        delta = dt - _EPOCH
        source = (row.get("source", "") or "").strip().lower()

        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "source": source,
            "tags": parse_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower()
                         in _OPT_OUT_VALUES,
            "dt": dt,
            # integer seconds since 0001-01-01; used for "most recent" ranking
            "secs": delta.days * 86400 + delta.seconds,
            "manual": source == "manual",
        })

    # ---------- union-find matching (rule 10) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the earlier row as root (not required, just tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}  # match key -> record position
    for pos, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(pos, first_seen[k])
            else:
                first_seen[k] = pos

    groups = {}
    for pos in range(len(recs)):
        groups.setdefault(find(pos), []).append(recs[pos])

    # ---------- build output (rules 11 & 12) ----------

    def rank(r):
        # manual first, then most recent, then earliest in input
        return (not r["manual"], -r["secs"], r["idx"])

    def pick(members, field):
        cands = [r for r in members if r[field]]
        if not cands:
            return ""
        return min(cands, key=rank)[field]

    people = []
    for members in groups.values():
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        people.append((
            min(r["idx"] for r in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
                "updated": max(r["dt"] for r in members).strftime("%Y-%m-%d"),
                "source": pick(members, "source"),
            },
        ))

    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

A few judgment calls and things to check before tomorrow's send:

1. **Filter out opt-outs before sending.** The function still returns opted-out people, marked `opted_out: "Y"`, as the spec asks. Make sure the send list keeps only `opted_out == "N"`.
2. **Opt-outs only carry over through a match.** If someone opted out on a row with just a phone and no last name, and is subscribed on another row with a different email, the two rows won't merge and the opt-out won't carry over. The same applies to rows dropped by rule 9. If you have a separate suppression list, apply it to the output as well.
3. **Gmail cleanup applies only to well-formed addresses.** It runs only when the address has exactly one `@`. An address like `+foo@gmail.com` normalizes to an empty local part and is treated as invalid.
4. **Extension markers.** The function cuts the phone number at the first `x`, `ext`, or `#`, ignoring case.
5. **ZIP+4.** I also accept a 4-digit ZIP+4 like `2139-4307`, since Excel strips leading zeros there too. It becomes `02139`. Only ASCII digits count, for both ZIPs and phones.
6. **Names.** "Entirely upper/lower-case" uses Python's `isupper()`/`islower()`, so `O'BRIEN` becomes `O'Brien`. Names with no letters are left as is.
7. **Bad dates stop the run.** An unparseable `updated` value raises a `ValueError` that names the row, instead of being skipped silently. I'd rather the run fail loudly than quietly mis-rank rows the night before a send.
8. **Tie-breaking.** Ties on "most recent" compare full datetimes, so `2025-03-01T09:00:00` beats `2025-03-01`.

I'd suggest running it on a small hand-built sample first. Include a gmail dot/plus variant, a display-name email, a `1-`prefixed phone with an extension, and an opted-out duplicate, and check the output by eye before running the full export.
