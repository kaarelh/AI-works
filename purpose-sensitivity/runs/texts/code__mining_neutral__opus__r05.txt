import re
from datetime import datetime

_FIELDS = ("first_name", "last_name", "email", "phone", "zip",
           "tags", "opted_out", "updated", "source")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_DIGITS = "0123456789"


def _norm_email(raw):
    s = raw or ""
    # Rule 1: display name -> keep only the part inside the angle brackets
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in ("gmail.com", "googlemail.com"):
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_is_valid(e):
    # Rule 2
    if e.count("@") != 1:
        return False
    local, domain = e.split("@")
    return (bool(local) and "." in domain
            and not domain.startswith(".") and not domain.endswith("."))


def _norm_phone(raw):
    s = raw or ""
    # Rule 3: cut at the first extension marker (x, ext, #), case-insensitive
    m = re.search(r"ext|x|#", s, re.IGNORECASE)
    if m:
        s = s[:m.start()]
    digits = "".join(ch for ch in s if ch in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""  # invalid


def _norm_zip(raw):
    s = (raw or "").strip()
    # Rule 4: ZIP+4 -> keep the part before the hyphen
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    # Rule 5
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    # Rule 6
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    # Rule 7
    return (raw or "").strip().lower() in ("y", "yes", "true", "1")


def _split_tags(raw):
    # Rule 8 (plus rule 11 cleanup: strip, lowercase, drop empties)
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row ----
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _email_is_valid(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))

        # Rule 9: need a valid email or a valid phone
        if not email and not phone:
            continue

        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip", "")),
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
            "source": source,
        })

    # ---- Rule 10: transitive matching via union-find ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # keep the earlier row as root (not required, just tidy)
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for k in keys:
            if k in first_seen:
                union(first_seen[k], i)
            else:
                first_seen[k] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Rule 11: build one output dict per person ----
    def priority(r):
        # manual source first, then most recent update, then earlier input row
        return (r["source"] == "manual", r["updated"], -r["idx"])

    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        return max(candidates, key=priority)[field]

    people = []
    for members in groups.values():
        tags = set()
        for r in members:
            tags |= r["tags"]
        people.append((
            min(r["idx"] for r in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "tags": ";".join(sorted(tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
                "updated": max(r["updated"] for r in members).strftime("%Y-%m-%d"),
                "source": pick(members, "source"),
            },
        ))

    # ---- Rule 12: order by each person's earliest input row ----
    people.sort(key=lambda p: p[0])
    return [p[1] for p in people]

Notes on how I handled the gray areas:

- **The output still includes opted-out people.** They come back with `opted_out: "Y"`, as the rules specify. Before sending, filter them out, for example with `[p for p in result if p["opted_out"] == "N"]`. Otherwise they will get the email.
- **Some opt-outs can't be linked.** Rule 9 drops rows with no valid email or phone before matching. If such a row was an opt-out, it gets dropped, and the person's other rows can't inherit its opt-out. The same applies to an opt-out row whose email or phone doesn't match the person's other rows. You may want to check opt-outs against the raw export as well.
- **Bad dates stop the run.** If `updated` isn't in one of the three formats, the function raises a `ValueError` naming the bad value. I chose that over guessing silently.
- **Extension markers:** the function cuts at the first `x`, `ext`, or `#` it finds, so `555-123-4567 Ext. 12` becomes `5551234567`.
- **ZIP+4 with a missing leading zero:** `2139-4307` is accepted as ZIP+4 and padded to `02139`.
- **Phone + last name matching** uses the normalized last name, compared with `casefold()`.
- **Gmail handling** only applies when the address has exactly one `@`. Addresses with more than one `@` are invalid anyway.
- **Source** is chosen like the other fields: rows with a non-empty source are ranked manual first, then most recent, then earliest in the input.
