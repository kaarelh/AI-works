import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _norm_email(raw):
    s = (raw or "").strip()
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    local, sep, domain = s.partition("@")
    if sep and domain in _GMAIL_DOMAINS:
        local = local.replace(".", "").split("+", 1)[0]
        s = local + "@gmail.com"
    return s


def _valid_email(e):
    if e.count("@") != 1:
        return False
    local, _, domain = e.partition("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    s = raw or ""
    low = s.lower()
    cut = len(s)
    for marker in ("ext", "x", "#"):
        i = low.find(marker)
        if i != -1:
            cut = min(cut, i)
    digits = re.sub(r"[^0-9]", "", s[:cut])
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _valid_phone(p):
    return len(p) == 10 and p[0] not in "01"


def _norm_zip(raw):
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _norm_tags(raw):
    return [t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()]


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    return datetime.min  # unparseable dates sort as oldest


def _pick(members, field):
    cands = [r for r in members if r[field]]
    # Stable sorts applied last-to-first give the precedence:
    # manual first, then most recent updated, then earliest input row.
    cands.sort(key=lambda r: r["idx"])
    cands.sort(key=lambda r: r["updated"], reverse=True)
    cands.sort(key=lambda r: not r["manual"])
    return cands[0][field] if cands else ""


def _merge(members):
    tags = sorted({t for r in members for t in r["tags"]})
    latest = max(r["updated"] for r in members)
    return {
        "first_name": _pick(members, "first_name"),
        "last_name": _pick(members, "last_name"),
        "zip": _pick(members, "zip"),
        "email": _pick(members, "email"),
        "phone": _pick(members, "phone"),
        "source": _pick(members, "source"),
        "tags": ";".join(tags),
        "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
        "updated": latest.strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    # Step 1: normalize each row and drop rows with no valid email or phone.
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email"))
        email = email if _valid_email(email) else ""
        phone = _norm_phone(row.get("phone"))
        phone = phone if _valid_phone(phone) else ""
        if not email and not phone:
            continue
        source = (row.get("source") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name")),
            "last_name": _norm_name(row.get("last_name")),
            "zip": _norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": _norm_tags(row.get("tags")),
            "opted_out": (row.get("opted_out") or "").strip().lower() in _OPT_OUT_VALUES,
            "updated": _parse_updated(row.get("updated")),
        })

    # Step 2: union-find over matching rows (transitive).
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    by_email = {}
    by_phone_last = {}
    for p, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(p, by_email[r["email"]])
            else:
                by_email[r["email"]] = p
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].casefold())
            if key in by_phone_last:
                union(p, by_phone_last[key])
            else:
                by_phone_last[key] = p

    # Step 3: group rows; dict insertion order follows each person's earliest row.
    groups = {}
    for p in range(len(recs)):
        groups.setdefault(find(p), []).append(recs[p])

    # Step 4: merge each group into one output record.
    return [_merge(members) for members in groups.values()]
