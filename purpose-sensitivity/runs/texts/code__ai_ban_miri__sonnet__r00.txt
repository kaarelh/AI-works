import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL = {"gmail.com", "googlemail.com"}
_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_ZIP4_RE = re.compile(r"^([0-9]{5})-[0-9]{4}$")
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")

    def get(row, key):
        v = row.get(key, "")

        m = re.search(r"<([^<>]*)>", s)
        if m:
            s = m.group(1)
        s = s.strip().lower()

            local, domain = s.split("@")
            if domain in _GMAIL:
                local = local.split("+")[0].replace(".", "")
                domain = "gmail.com"
            s = local + "@" + domain
        return s

    def email_valid(s):
        if s.count("@") != 1:
            return False
        local, domain = s.split("@")
        return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")

        if m:

        d = "".join(c for c in raw if c in _DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = raw.strip()

        if m:
            z = m.group(1)

            z = "0" + z
        if len(z) == 5 and all(c in _DIGITS for c in z):

        return ""

    # Normalize rows, dropping those with no valid email or phone.
    recs = []

    parent = list(range(len(recs)))

    def find(i):

Notes:

- **Opt-outs:** The function doesn't remove opted-out people. It returns them with `opted_out == "Y"` (set if any duplicate row opted out). Before sending, filter with `[p for p in dedupe_contacts(rows) if p["opted_out"] == "N" and p["email"]]`. People with no valid email can't receive the email anyway.
- **ZIP+4:** I read "looks like ZIP+4" literally as 5 digits, a hyphen, then 4 digits. Something like `2139-4307` (Excel-eaten zero plus a +4) becomes `""`. If you want that case rescued to `02139`, loosen the regex to `[0-9]{4,5}-[0-9]{4}`.
- **Bad dates:** An `updated` value in an unexpected format raises `ValueError` rather than being silently guessed. That is deliberate, so a bad export gets caught before the send.
