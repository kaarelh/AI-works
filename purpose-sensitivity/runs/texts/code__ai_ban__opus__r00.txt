import re
from datetime import datetime

_DIGITS = "0123456789"
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_OPT_OUT_VALUES = {"y", "yes", "true", "1"}


def _norm_email(raw):
    """Rule 1: extract from display name, strip, lowercase, canonicalize Gmail."""
    s = raw or ""
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            domain = "gmail.com"
        s = local + "@" + domain
    return s


def _email_is_valid(email):
    """Rule 2."""
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _norm_phone(raw):
    """Rule 3: returns the 10-digit phone if valid, else ''."""
    s = (raw or "").lower()
    cut_points = [i for i in (s.find("ext"), s.find("x"), s.find("#")) if i >= 0]
    if cut_points:
        s = s[:min(cut_points)]
    digits = "".join(c for c in s if c in _DIGITS)
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    if len(digits) == 10 and digits[0] not in "01":
        return digits
    return ""


def _norm_zip(raw):
    """Rule 4."""
    s = (raw or "").strip()
    m = re.fullmatch(r"([0-9]{4,5})-[0-9]{4}", s)
    if m:
        s = m.group(1)
    if re.fullmatch(r"[0-9]{4}", s):
        s = "0" + s
    return s if re.fullmatch(r"[0-9]{5}", s) else ""


def _norm_name(raw):
    """Rule 5."""
    s = (raw or "").strip()
    if s.isupper() or s.islower():
        return s.title()
    return s


def _parse_updated(raw):
    """Rule 6: parse to datetime (date-only values mean midnight)."""
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    """Rule 7."""
    return (raw or "").strip().lower() in _OPT_OUT_VALUES


def _split_tags(raw):
    """Rule 8 (plus the stripping/lowercasing from rule 11)."""
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone (rule 9).
    recs = []
    for idx, row in enumerate(rows):
        email = _norm_email(row.get("email", ""))
        if not _email_is_valid(email):
            email = ""
        phone = _norm_phone(row.get("phone", ""))
        if not email and not phone:
            continue
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _norm_name(row.get("first_name", "")),
            "last_name": _norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _norm_zip(row.get("zip", "")),
            "source": source,
            "is_manual": source == "manual",
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
            "updated": _parse_updated(row.get("updated", "")),
        })

    # ---- Union-find for transitive matching (rule 10).
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(i, j):
        ri, rj = find(i), find(j)
        if ri != rj:
            # Keep the root as the earlier record (not required, just tidy).
            if ri < rj:
                parent[rj] = ri
            else:
                parent[ri] = rj

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---- Build one output record per person (rule 11).
    epoch = datetime(1, 1, 1)

    def preference(r):
        # manual first, then most recent updated, then earliest in input
        return (0 if r["is_manual"] else 1, -(r["updated"] - epoch), r["idx"])

    def pick(members, field):
        candidates = [r for r in members if r[field]]
        if not candidates:
            return ""
        return min(candidates, key=preference)[field]

    people = []
    for members in groups.values():
        all_tags = set()
        for r in members:
            all_tags |= r["tags"]
        latest = max(r["updated"] for r in members)
        people.append((
            min(r["idx"] for r in members),
            {
                "first_name": pick(members, "first_name"),
                "last_name": pick(members, "last_name"),
                "email": pick(members, "email"),
                "phone": pick(members, "phone"),
                "zip": pick(members, "zip"),
                "source": pick(members, "source"),
                "tags": ";".join(sorted(all_tags)),
                "opted_out": "Y" if any(r["opted_out"] for r in members) else "N",
                "updated": latest.date().isoformat(),
            },
        ))

    # ---- Order by each person's earliest row in the input (rule 12).
    people.sort(key=lambda p: p[0])
    return [person for _, person in people]
