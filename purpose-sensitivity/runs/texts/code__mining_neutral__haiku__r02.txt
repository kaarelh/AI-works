import re
from datetime import datetime

_EMAIL_DISPLAY_RE = re.compile(r"<([^<>]*)>")
_GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
_DIGITS = "0123456789"
_ZIP_PLUS4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_FOUR_DIGITS_RE = re.compile(r"[0-9]{4}")
_FIVE_DIGITS_RE = re.compile(r"[0-9]{5}")
_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_TRUE_VALUES = {"y", "yes", "true", "1"}
_TAG_SPLIT_RE = re.compile(r"[;,]")


def _normalize_email(raw):

    m = _EMAIL_DISPLAY_RE.search(value)
    if m:

    value = value.strip().lower()

        return False
    local, domain = email.split("@")

        bool(local)

def _normalize_phone(raw):
    value = raw.lower()
    cut = len(value)

def _is_valid_phone(digits):

def _normalize_zip(raw):
    value = raw.strip()
    if _ZIP_PLUS4_RE.fullmatch(value):
        value = value[:5]
    if _FOUR_DIGITS_RE.fullmatch(value):
        value = "0" + value

def _normalize_name(raw):
    value = raw.strip()
    if value.isupper() or value.islower():
        value = value.title()
    return value


def _parse_updated(raw):
    value = raw.strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(value, fmt)
        except ValueError:
            pass
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _normalize_tags(raw):
    return {t.strip().lower() for t in _TAG_SPLIT_RE.split(raw) if t.strip()}


def _to_record(row):
    """Normalize one input row. Returns None if it has no valid email or phone."""
    email = _normalize_email(row.get("email", ""))
    email = email if _is_valid_email(email) else ""

    phone = _normalize_phone(row.get("phone", ""))
    phone = phone if _is_valid_phone(phone) else ""

    if not email and not phone:
        return None

    last_name = _normalize_name(row.get("last_name", ""))
    source = row.get("source", "").strip().lower()

    return {
        "first_name": _normalize_name(row.get("first_name", "")),
        "last_name": last_name,
        "last_key": last_name.lower(),
        "zip": _normalize_zip(row.get("zip", "")),
        "email": email,
        "phone": phone,
        "source": source,
        "manual": source == "manual",
        "tags": _normalize_tags(row.get("tags", "")),
        "opted_out": row.get("opted_out", "").strip().lower() in _TRUE_VALUES,
        "updated": _parse_updated(row["updated"]),
    }


def _pick(group, field):
    """Pick the best non-empty value for a field: manual first, then most recent
    updated, then earliest input row (group is already in input order, and
    Python's sort is stable)."""
    candidates = [r for r in group if r[field]]
    candidates.sort(key=lambda r: r["updated"], reverse=True)
    candidates.sort(key=lambda r: not r["manual"])
    return candidates[0][field] if candidates else ""


def _merge_group(group):
    tags = set().union(*(r["tags"] for r in group))
    latest = max(r["updated"] for r in group)
    return {
        "first_name": _pick(group, "first_name"),
        "last_name": _pick(group, "last_name"),
        "zip": _pick(group, "zip"),
        "email": _pick(group, "email"),
        "phone": _pick(group, "phone"),
        "source": _pick(group, "source"),
        "tags": ";".join(sorted(tags)),
        "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
        "updated": latest.strftime("%Y-%m-%d"),
    }


def dedupe_contacts(rows):
    records = []
    for row in rows:
        rec = _to_record(row)
        if rec is not None:
            records.append(rec)

    # Union-find over record indexes. Two records are linked if they share a
    # valid email, or share a valid phone plus a non-empty last name.
    parent = list(range(len(records)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, rec in enumerate(records):
        if rec["email"]:
            if rec["email"] in email_seen:
                union(i, email_seen[rec["email"]])
            else:
                email_seen[rec["email"]] = i
        if rec["phone"] and rec["last_key"]:
            key = (rec["phone"], rec["last_key"])
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    # Group records by root. Dict insertion order follows each group's
    # earliest input row, which gives the required output order.
    groups = {}
    for i, rec in enumerate(records):
        groups.setdefault(find(i), []).append(rec)

    return [_merge_group(group) for group in groups.values()]
