```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL = {"gmail.com", "googlemail.com"}
    DIGITS = "0123456789"

    def s(row, key):
        v = row.get(key, "")
        return "" if v is None else str(v)

    # ---------- normalizers ----------
    def norm_email(raw):
        m = re.search(r"<([^<>]*)>", raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL:
                local = local.split("+", 1)[0].replace(".", "")
                domain = "gmail.com"
            e = local + "@" + domain
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return bool(local) and "." in domain and not domain.startswith(".") and not domain.endswith(".")

    def norm_phone(raw):
        m = re.search(r"ext|x|#", raw, re.IGNORECASE)
        if m:
            raw = raw[:m.start()]
        d = "".join(c for c in raw if c in DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        return d

    def phone_valid(p):
        return len(p) == 10 and p[0] not in "01"

    def norm_zip(raw):
        z = raw.strip()
        m = re.fullmatch(r"([0-9]+)-[0-9]{4}", z)
        if m:
            z = m.group(1)
        if re.fullmatch(r"[0-9]{4}", z):
            z = "0" + z
        if not re.fullmatch(r"[0-9]{5}", z):
            return ""
        return z

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        v = raw.strip()
        for fmt in ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                return datetime.strptime(v, fmt)
            except ValueError:
                pass
        return datetime.min

    # ---------- normalize rows, drop unusable ones ----------
    recs = []
    for i, row in enumerate(rows):
        email = norm_email(s(row, "email"))
        phone = norm_phone(s(row, "phone"))
        e_ok = email_valid(email)
        p_ok = phone_valid(phone)
        if not e_ok and not p_ok:
            continue
        source = s(row, "source").strip().lower()
        tags = set()
        for t in re.split(r"[;,]", s(row, "tags")):
            t = t.strip().lower()
            if t:
                tags.add(t)
        recs.append({
            "idx": i,
            "first_name": norm_name(s(row, "first_name")),
            "last_name": norm_name(s(row, "last_name")),
            "zip": norm_zip(s(row, "zip")),
            "email": email if e_ok else "",
            "phone": phone if p_ok else "",
            "source": source,
            "manual": source == "manual",
            "tags": tags,
            "opted_out": s(row, "opted_out").strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_updated(s(row, "updated")),
        })

    # ---------- union-find matching ----------
    n = len(recs)
    parent = list(range(n))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    by_email = {}
    by_phone_last = {}
    for k, r in enumerate(recs):
        if r["email"]:
            if r["email"] in by_email:
                union(k, by_email[r["email"]])
            else:
                by_email[r["email"]] = k
        last = r["last_name"].lower()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in by_phone_last:
                union(k, by_phone_last[key])
            else:
                by_phone_last[key] = k

    groups = {}
    for k in range(n):
        groups.setdefault(find(k), []).append(k)

    # recs are in input order, so the group's first member is its earliest row;
    # ordering groups by that member gives the required output order.
    ordered = sorted(groups.values(), key=lambda members: members[0])

    def pick(members, field):
        best = None
        for k in members:  # input order, so ties keep the earlier row
            r = recs[k]
            if not r[field]:
                continue
            if best is None:
                best = r
            elif r["manual"] and not best["manual"]:
                best = r
            elif r["manual"] == best["manual"] and r["updated"] > best["updated"]:
                best = r
        return best[field] if best else ""

    out = []
    for members in ordered:
        rs = [recs[k] for k in members]
        tags = set()
        for r in rs:
            tags |= r["tags"]
        out.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in rs) else "N",
            "updated": max(r["updated"] for r in rs).strftime("%Y-%m-%d"),
        })
    return out
```

Notes on interpretation:

- **Dropped rows:** Rows with no valid email and no valid phone are removed before matching, so their opt-out flags, tags, and so on don't affect anyone. Such a row has nothing to send to anyway.
- **Opt-outs:** If any row in a group has opted out, the person is marked `"Y"`. Filter on `opted_out == "N"` and a non-empty `email` before sending.
- **"ext" vs. "x":** The regex `ext|x|#` finds the leftmost marker, so `555-123-4567 ext 12` is cut at the "e".
- **ZIP+4:** Any `digits-4digits` form takes the part before the hyphen, so `2139-4307` becomes `02139`. Anything that isn't exactly 5 digits afterward becomes `""`.
- **Unparseable `updated` values:** The spec says these shouldn't occur, but they're treated as the earliest possible datetime instead of raising an error. Check the export for such values, since a bad one could print as `0001-01-01`.
- **Tie-breaking:** Preference is manual source first, then the latest `updated`, then the earlier input row, applied independently for each field.
