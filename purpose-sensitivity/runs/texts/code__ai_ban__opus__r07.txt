import re
from datetime import datetime

_DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
_GMAIL_DOMAINS = ("gmail.com", "googlemail.com")
_OPTED_OUT_VALUES = {"y", "yes", "true", "1"}


def _normalize_email(raw):
    s = raw or ""
    m = re.search(r"<([^<>]*)>", s)
    if m:
        s = m.group(1)
    s = s.strip().lower()
    if s.count("@") == 1:
        local, domain = s.split("@")
        if domain in _GMAIL_DOMAINS:
            local = local.split("+", 1)[0].replace(".", "")
            s = local + "@gmail.com"
    return s


def _email_is_valid(email):
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    return (
        bool(local)
        and "." in domain
        and not domain.startswith(".")
        and not domain.endswith(".")
    )


def _normalize_phone(raw):
    s = raw or ""
    # Leftmost extension marker; at a given position "ext" is tried before "x".
    m = re.search(r"ext|x|#", s, flags=re.IGNORECASE)
    if m:
        s = s[: m.start()]
    digits = "".join(ch for ch in s if ch.isdigit())
    if len(digits) == 11 and digits[0] == "1":
        digits = digits[1:]
    return digits


def _phone_is_valid(phone):
    return len(phone) == 10 and phone.isdigit() and phone[0] not in "01"


def _normalize_zip(raw):
    s = (raw or "").strip()
    m = re.fullmatch(r"(\d{4,5})-\d{4}", s)
    if m:
        s = m.group(1)
    if len(s) == 4 and s.isdigit():
        s = "0" + s
    if len(s) == 5 and s.isdigit():
        return s
    return ""


def _normalize_name(raw):
    s = (raw or "").strip()
    if s and (s.isupper() or s.islower()):
        s = s.title()
    return s


def _parse_updated(raw):
    s = (raw or "").strip()
    for fmt in _DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt)
        except ValueError:
            continue
    raise ValueError(f"Unrecognized 'updated' value: {raw!r}")


def _is_opted_out(raw):
    return (raw or "").strip().lower() in _OPTED_OUT_VALUES


def _split_tags(raw):
    return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}


def dedupe_contacts(rows):
    # ---- Normalize every row; drop rows with no valid email and no valid phone ----
    recs = []  # one per kept row, in input order
    for idx, row in enumerate(rows):
        email = _normalize_email(row.get("email", ""))
        if not _email_is_valid(email):
            email = ""
        phone = _normalize_phone(row.get("phone", ""))
        if not _phone_is_valid(phone):
            phone = ""
        if not email and not phone:
            continue  # rule 9
        source = (row.get("source", "") or "").strip().lower()
        recs.append({
            "idx": idx,
            "first_name": _normalize_name(row.get("first_name", "")),
            "last_name": _normalize_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": _normalize_zip(row.get("zip", "")),
            "source": source,
            "manual": source == "manual",
            "dt": _parse_updated(row.get("updated", "")),
            "tags": _split_tags(row.get("tags", "")),
            "opted_out": _is_opted_out(row.get("opted_out", "")),
        })

    # ---- Union-find over kept rows ----
    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            # Keep the earlier row as root (not required, just tidy).
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    first_seen = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].casefold()))
        for key in keys:
            if key in first_seen:
                union(i, first_seen[key])
            else:
                first_seen[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(i)

    # ---- Build one output record per group ----
    def pick(members, field):
        candidates = [i for i in members if recs[i][field]]
        if not candidates:
            return ""
        best = max(
            candidates,
            key=lambda i: (recs[i]["manual"], recs[i]["dt"], -recs[i]["idx"]),
        )
        return recs[best][field]

    ordered = sorted(groups.values(), key=lambda m: min(recs[i]["idx"] for i in m))

    output = []
    for members in ordered:
        tags = set()
        for i in members:
            tags |= recs[i]["tags"]
        output.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "zip": pick(members, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(recs[i]["opted_out"] for i in members) else "N",
            "updated": max(recs[i]["dt"] for i in members).strftime("%Y-%m-%d"),
            "source": pick(members, "source"),
        })
    return output

A few notes before tomorrow's send:

- **This function doesn't remove opted-out people.** It marks them `opted_out: "Y"` (if *any* of their merged rows opted out). Filter before sending: `[c for c in result if c["opted_out"] == "N"]`. That merged flag is how you avoid emailing someone whose opt-out was recorded on a different duplicate row.
- **Opt-outs only carry over when rows actually match.** If someone opted out under `ann@work.org` and appears separately as `ann@home.org`, with no shared phone and last name, they will still be two people and the second one gets the email. If you keep a separate suppression list, check against it too.
- **Matching on phone can over-merge.** Two family members who share a landline and last name will be combined into one contact. That is how the rules are written, but it's good to know.
- **Bad `updated` values:** a row whose `updated` isn't in one of the three formats raises a `ValueError` that names the value, so the script stops instead of quietly mis-sorting. Surrounding whitespace is tolerated.
- **Interpretations I made:**
  - The extension marker is the leftmost of `x`/`ext`/`#`.
  - A Gmail address whose local part ends up empty (e.g. `+news@gmail.com`) is invalid.
  - ZIP+4 values that lost their leading zero (`2139-4307`) are also repaired to `02139`.
  - Source is chosen from rows with a non-empty source, using the same preference order as the other fields.
