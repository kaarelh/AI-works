import re
from collections import defaultdict
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}

    # ---------- normalizers ----------

    def norm_email(raw):
        s = raw or ""
        m = re.search(r"<([^>]*)>", s)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            s = m.group(1)
        s = s.strip().lower()
        if s.count("@") == 1:
            local, domain = s.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                s = local + "@gmail.com"
        # validity (rule 2); invalid -> "" so it never matches or gets output
        if s.count("@") != 1:
            return ""
        local, domain = s.split("@")
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return s

    def norm_phone(raw):
        s = re.split(r"ext|x|#", raw or "", maxsplit=1, flags=re.IGNORECASE)[0]
        digits = re.sub(r"\D", "", s)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""                                # invalid -> ""

    def norm_zip(raw):
        s = (raw or "").strip()
        if "-" in s:                             # ZIP+4 -> keep the part before the hyphen
            s = s.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", s):         # Excel ate the leading zero
            s = "0" + s
        return s if re.fullmatch(r"[0-9]{5}", s) else ""

    def norm_name(raw):
        s = (raw or "").strip()
        if s.isupper() or s.islower():
            return s.title()
        return s

    def parse_updated(raw):
        s = (raw or "").strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        return datetime.min                      # unparseable -> treated as oldest possible

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw or "") if t.strip()}

    # ---------- normalize rows, dropping those with no valid email or phone ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        if not email and not phone:              # rule 9
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(row.get("zip", "")),
            "tags": norm_tags(row.get("tags", "")),
            "opted_out": (row.get("opted_out", "") or "").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", "")),
            "source": (row.get("source", "") or "").strip().lower(),
        })

    # ---------- group transitively (union-find) ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)    # root is always the earliest record

    key_owner = {}
    for i, r in enumerate(recs):
        keys = []
        if r["email"]:
            keys.append(("email", r["email"]))
        if r["phone"] and r["last_name"]:
            keys.append(("phone", r["phone"], r["last_name"].lower()))
        for k in keys:
            if k in key_owner:
                union(i, key_owner[k])
            else:
                key_owner[k] = i

    groups = defaultdict(list)
    for i in range(len(recs)):
        groups[find(i)].append(recs[i])

    # ---------- build one output row per person ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["source"] == "manual", r["updated"], -r["idx"]))
        return best[field]

    out = []
    for root in sorted(groups):                  # root index == earliest input row in group
        g = groups[root]
        tags = set().union(*(r["tags"] for r in g))
        out.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "zip": pick(g, "zip"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "source": pick(g, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": max(r["updated"] for r in g).strftime("%Y-%m-%d"),
        })
    return out

A few notes on how I resolved details the rules leave open:

- **Invalid email/phone become `""`** right away, so they can't match anyone and can't be selected for output (consistent with rules 9–11).
- **Phone extension cutoff** splits on the first `x`, `ext`, or `#` (case-insensitive); only digits are kept afterward, so stray letters before the marker don't matter.
- **Unparseable `updated`** values are treated as the oldest possible date rather than crashing the run. Per the spec this shouldn't happen, but I'd rather the export finish tonight than die on one bad cell.
- **Gmail normalization** is only applied when the address has exactly one `@`; anything else is simply invalid.
- Rows missing a key entirely are treated as having `""` for that field.
