```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")

    # ---------- field normalizers ----------

    def text(row, key):
        v = row.get(key)
        return "" if v is None else str(v)

    def norm_email(raw):
        m = re.search(r"<([^>]*)>", raw)          # "Ann Lee <ann@x.org>" -> "ann@x.org"
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.split("+", 1)[0].replace(".", "")
                e = local + "@gmail.com"
        return e

    def email_valid(e):
        if e.count("@") != 1:
            return False
        local, domain = e.split("@")
        return (bool(local)
                and "." in domain
                and not domain.startswith(".")
                and not domain.endswith("."))

    def norm_phone(raw):
        """Returns the 10-digit normalized phone, or "" if invalid."""
        s = raw.lower()
        cut = len(s)
        for marker in ("x", "#"):                 # "ext" is covered by "x"
            i = s.find(marker)
            if i != -1 and i < cut:
                cut = i
        digits = re.sub(r"[^0-9]", "", s[:cut])
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        if len(digits) == 10 and digits[0] not in "01":
            return digits
        return ""

    def norm_zip(raw):
        z = raw.strip()
        if "-" in z:                              # ZIP+4 -> keep the 5-digit part
            z = z.split("-", 1)[0].strip()
        if re.fullmatch(r"[0-9]{4}", z):          # Excel ate the leading zero
            z = "0" + z
        return z if re.fullmatch(r"[0-9]{5}", z) else ""

    def norm_name(raw):
        n = raw.strip()
        if n.isupper() or n.islower():
            return n.title()
        return n

    def parse_updated(raw):
        s = raw.strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(s, fmt)
            except ValueError:
                pass
        raise ValueError(f"Unrecognized 'updated' value: {raw!r}")

    def norm_tags(raw):
        return {t.strip().lower() for t in re.split(r"[;,]", raw) if t.strip()}

    # ---------- normalize rows, drop unusable ones ----------

    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(text(row, "email"))
        if not email_valid(email):
            email = ""
        phone = norm_phone(text(row, "phone"))
        if not email and not phone:               # rule 9
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(text(row, "first_name")),
            "last_name": norm_name(text(row, "last_name")),
            "email": email,
            "phone": phone,
            "zip": norm_zip(text(row, "zip")),
            "tags": norm_tags(text(row, "tags")),
            "opted_out": text(row, "opted_out").strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(text(row, "updated")),
            "source": text(row, "source").strip().lower(),
        })

    # ---------- transitive matching via union-find ----------

    parent = list(range(len(recs)))

    def find(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]
            i = parent[i]
        return i

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    seen_email = {}
    seen_phone_last = {}
    for i, r in enumerate(recs):
        if r["email"]:
            if r["email"] in seen_email:
                union(i, seen_email[r["email"]])
            else:
                seen_email[r["email"]] = i
        if r["phone"] and r["last_name"]:
            key = (r["phone"], r["last_name"].lower())
            if key in seen_phone_last:
                union(i, seen_phone_last[key])
            else:
                seen_phone_last[key] = i

    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    # ---------- build one output record per person ----------

    def pick(group, field):
        cands = [r for r in group if r[field]]
        if not cands:
            return ""
        best = max(cands, key=lambda r: (r["source"] == "manual",
                                         r["updated"],
                                         -r["idx"]))
        return best[field]

    out = []
    for group in sorted(groups.values(), key=lambda g: g[0]["idx"]):
        tags = set()
        for r in group:
            tags |= r["tags"]
        out.append({
            "first_name": pick(group, "first_name"),
            "last_name": pick(group, "last_name"),
            "email": pick(group, "email"),
            "phone": pick(group, "phone"),
            "zip": pick(group, "zip"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in group) else "N",
            "updated": max(r["updated"] for r in group).strftime("%Y-%m-%d"),
            "source": pick(group, "source"),
        })
    return out
```

A few notes on how I handled the spec:

- **Extensions:** cutting at the first `x` or `#` covers `ext` too (the leftover `e` isn't a digit, so it's discarded). Output phones are the clean 10-digit string.
- **Invalid email/phone** on a surviving row becomes `""` for both matching and output, so a bad email never links two people or shows up in the result.
- **Field selection** uses one comparison key — `(source == "manual", updated, earlier index)` — exactly as rule 11 orders the preferences. For `email` and `phone`, the group is tied to a single value by construction only when matched by that field, so this picks correctly when a person has, say, two different valid phones across rows.
- **Dates:** `updated` is required by the spec, so an unparseable value raises a clear `ValueError` rather than silently misordering rows. If you'd rather skip or default those rows, change `parse_updated`.
- **Transitivity** is handled with union-find, so A–B (email) and B–C (phone + last name) collapse into one person, and the group is placed by the earliest of its rows.

Before the send, I'd suggest filtering the output to `opted_out == "N"` and spot-checking a handful of large groups — a shared office phone plus a common last name (two "Smiths" at the same company) is the one case these rules will merge that you might not want merged.
