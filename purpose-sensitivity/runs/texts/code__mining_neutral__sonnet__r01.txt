import re
from datetime import datetime

_EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
_ANGLE_RE = re.compile(r"<([^<>]*)>")
_ZIP4_RE = re.compile(r"[0-9]{5}-[0-9]{4}")
_ZIP5_RE = re.compile(r"[0-9]{5}")
_DIGITS = "0123456789"
_DATE_FORMATS = ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%m/%d/%Y")


def dedupe_contacts(rows):
    def s(v):
        return "" if v is None else str(v)

    def norm_email(raw):
        raw = s(raw)
        m = _ANGLE_RE.search(raw)
        if m:
            raw = m.group(1)
        e = raw.strip().lower()
        if e.count("@") == 1:
            local, domain = e.split("@")
            if domain in ("gmail.com", "googlemail.com"):
                local = local.replace(".", "").split("+", 1)[0]
                domain = "gmail.com"
                e = local + "@" + domain
        # validity
        if e.count("@") != 1:
            return ""
        local, domain = e.split("@")
        if not local or "." not in domain or domain.startswith(".") or domain.endswith("."):
            return ""
        return e

    def norm_phone(raw):
        raw = s(raw)
        m = _EXT_RE.search(raw)
        if m:
            raw = raw[:m.start()]
        d = "".join(c for c in raw if c in _DIGITS)
        if len(d) == 11 and d[0] == "1":
            d = d[1:]
        if len(d) == 10 and d[0] not in "01":
            return d
        return ""

    def norm_zip(raw):
        z = s(raw).strip()
        if _ZIP4_RE.fullmatch(z):
            z = z.split("-")[0]
        if len(z) == 4 and all(c in _DIGITS for c in z):
            z = "0" + z
        return z if _ZIP5_RE.fullmatch(z) else ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        t = s(raw).strip()
        for fmt in _DATE_FORMATS:
            try:
                return datetime.strptime(t, fmt)
            except ValueError:
                pass
        return datetime.min

    def parse_tags(raw):
        out = set()
        for t in re.split(r"[;,]", s(raw)):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize rows and drop those with neither valid email nor valid phone
    recs = []
    for i, row in enumerate(rows):
        email = norm_email(row.get("email"))
        phone = norm_phone(row.get("phone"))
        if not email and not phone:
            continue
        source = s(row.get("source")).strip().lower()
        recs.append({
            "idx": i,
            "first_name": norm_name(row.get("first_name")),
            "last_name": norm_name(row.get("last_name")),
            "zip": norm_zip(row.get("zip")),
            "email": email,
            "phone": phone,
            "source": source,
            "manual": source == "manual",
            "tags": parse_tags(row.get("tags")),
            "opted_out": s(row.get("opted_out")).strip().lower() in ("y", "yes", "true", "1"),
            "updated": parse_updated(row.get("updated")),
        })

    # Union-find over kept records
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_seen = {}
    for k, r in enumerate(recs):
        if r["email"]:
            if r["email"] in email_seen:
                union(k, email_seen[r["email"]])
            else:
                email_seen[r["email"]] = k
        ln = r["last_name"].lower()
        if r["phone"] and ln:
            key = (r["phone"], ln)
            if key in phone_seen:
                union(k, phone_seen[key])
            else:
                phone_seen[key] = k

    # Group, ordered by earliest row in input (recs are in input order)
    groups = {}
    order = []
    for k in range(len(recs)):
        root = find(k)
        if root not in groups:
            groups[root] = []
            order.append(root)
        groups[root].append(recs[k])

    def pick(group, field):
        best = None
        for r in group:  # input order, so ties keep the earlier row
            if not r[field]:
                continue
            if best is None:
                best = r
            elif r["manual"] and not best["manual"]:
                best = r
            elif r["manual"] == best["manual"] and r["updated"] > best["updated"]:
                best = r
        return best[field] if best else ""

    result = []
    for root in order:
        g = groups[root]
        tags = set()
        for r in g:
            tags |= r["tags"]
        latest = max(r["updated"] for r in g)
        result.append({
            "first_name": pick(g, "first_name"),
            "last_name": pick(g, "last_name"),
            "zip": pick(g, "zip"),
            "email": pick(g, "email"),
            "phone": pick(g, "phone"),
            "source": pick(g, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(r["opted_out"] for r in g) else "N",
            "updated": "%04d-%02d-%02d" % (latest.year, latest.month, latest.day),
        })
    return result

Notes on a few judgment calls:

- **Opt-outs:** the function only merges and flags. Before sending, filter out anyone whose output `opted_out` is `"Y"`. Because opt-out is "any row in the group," a person who opted out under one email address stays suppressed even if another duplicate row says otherwise.
- **People with no email:** some output records may have `email == ""` (phone-only people). Skip those when sending the newsletter.
- **Dropped rows:** rows removed by rule 9 don't contribute tags, opt-outs, or dates. If a dropped row was an opt-out, that opt-out is lost, so you may want to check those rows separately.
- **Ties:** when two rows have the same `updated` value, the one earlier in the input wins.
- **Unparseable `updated` values:** these are treated as the oldest possible date rather than raising an error.
