```python
import re
from datetime import datetime


def dedupe_contacts(rows):
    GMAIL_DOMAINS = {"gmail.com", "googlemail.com"}
    OPT_OUT_VALUES = {"y", "yes", "true", "1"}
    ANGLE_RE = re.compile(r"<([^<>]*)>")
    EXT_RE = re.compile(r"x|ext|#", re.IGNORECASE)
    ZIP4_RE = re.compile(r"^([0-9]{5})-[0-9]{4}$")
    FIVE_DIGITS_RE = re.compile(r"^[0-9]{5}$")
    FOUR_DIGITS_RE = re.compile(r"^[0-9]{4}$")
    TAG_SPLIT_RE = re.compile(r"[;,]")
    DATE_FORMATS = ("%Y-%m-%d", "%m/%d/%Y", "%Y-%m-%dT%H:%M:%S")

    def s(value):
        return "" if value is None else str(value)

    def norm_email(raw):
        raw = s(raw)
        m = ANGLE_RE.search(raw)
        if m:
            raw = m.group(1)
        email = raw.strip().lower()
        if email.count("@") == 1:
            local, domain = email.split("@")
            if domain in GMAIL_DOMAINS:
                local = local.replace(".", "").split("+")[0]
                domain = "gmail.com"
            email = local + "@" + domain
        return email

    def email_valid(email):
        if email.count("@") != 1:
            return False
        local, domain = email.split("@")
        return (bool(local) and "." in domain
                and not domain.startswith(".") and not domain.endswith("."))

    def norm_phone(raw):
        raw = s(raw)
        m = EXT_RE.search(raw)
        if m:
            raw = raw[:m.start()]
        digits = re.sub(r"[^0-9]", "", raw)
        if len(digits) == 11 and digits[0] == "1":
            digits = digits[1:]
        return digits

    def phone_valid(digits):
        return len(digits) == 10 and digits[0] not in "01"

    def norm_zip(raw):
        z = s(raw).strip()
        m = ZIP4_RE.match(z)
        if m:
            z = m.group(1)
        if FOUR_DIGITS_RE.match(z):
            z = "0" + z
        return z if FIVE_DIGITS_RE.match(z) else ""

    def norm_name(raw):
        n = s(raw).strip()
        if n.isupper() or n.islower():
            n = n.title()
        return n

    def parse_updated(raw):
        raw = s(raw).strip()
        for fmt in DATE_FORMATS:
            try:
                return datetime.strptime(raw, fmt)
            except ValueError:
                continue
        return datetime.min  # fallback; spec says this shouldn't happen

    def parse_tags(raw):
        out = set()
        for t in TAG_SPLIT_RE.split(s(raw)):
            t = t.strip().lower()
            if t:
                out.add(t)
        return out

    # Normalize rows and drop those with no valid email and no valid phone.
    recs = []
    for idx, row in enumerate(rows):
        email = norm_email(row.get("email", ""))
        phone = norm_phone(row.get("phone", ""))
        ev = email_valid(email)
        pv = phone_valid(phone)
        if not (ev or pv):
            continue
        recs.append({
            "idx": idx,
            "first_name": norm_name(row.get("first_name", "")),
            "last_name": norm_name(row.get("last_name", "")),
            "zip": norm_zip(row.get("zip", "")),
            "email": email if ev else "",
            "phone": phone if pv else "",
            "source": s(row.get("source", "")).strip().lower(),
            "tags": parse_tags(row.get("tags", "")),
            "opted_out": s(row.get("opted_out", "")).strip().lower() in OPT_OUT_VALUES,
            "updated": parse_updated(row.get("updated", "")),
        })

    # Union-find over the kept records.
    parent = list(range(len(recs)))

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            if ra < rb:
                parent[rb] = ra
            else:
                parent[ra] = rb

    email_seen = {}
    phone_last_seen = {}
    for i, r in enumerate(recs):
        if r["email"]:
            key = r["email"]
            if key in email_seen:
                union(i, email_seen[key])
            else:
                email_seen[key] = i
        last = r["last_name"].casefold()
        if r["phone"] and last:
            key = (r["phone"], last)
            if key in phone_last_seen:
                union(i, phone_last_seen[key])
            else:
                phone_last_seen[key] = i

    # Collect groups (recs are in input order, so members stay ordered).
    groups = {}
    for i in range(len(recs)):
        groups.setdefault(find(i), []).append(recs[i])

    def pick(members, field):
        cands = [m for m in members if m[field]]
        if not cands:
            return ""
        # Stable sorts: input order, then most recent updated, then manual first.
        cands.sort(key=lambda m: m["idx"])
        cands.sort(key=lambda m: m["updated"], reverse=True)
        cands.sort(key=lambda m: 0 if m["source"] == "manual" else 1)
        return cands[0][field]

    ordered = sorted(groups.values(), key=lambda ms: min(m["idx"] for m in ms))

    result = []
    for members in ordered:
        tags = set()
        for m in members:
            tags |= m["tags"]
        latest = max(m["updated"] for m in members)
        result.append({
            "first_name": pick(members, "first_name"),
            "last_name": pick(members, "last_name"),
            "zip": pick(members, "zip"),
            "email": pick(members, "email"),
            "phone": pick(members, "phone"),
            "source": pick(members, "source"),
            "tags": ";".join(sorted(tags)),
            "opted_out": "Y" if any(m["opted_out"] for m in members) else "N",
            "updated": latest.strftime("%Y-%m-%d"),
        })
    return result
```

Notes on judgment calls where the rules were silent:

- **ZIP+4:** I only strip the suffix when the value matches exactly `5 digits-4 digits`. Something like `02139-` becomes `""`.
- **Gmail rules:** these apply only when the address has exactly one `@`. An address with several `@` is left alone and fails validation.
- **Last-name matching:** this uses `casefold()` on the normalized name. Rows with an empty last name never match on phone alone.
- **Source preference:** "manual" is checked after stripping and lowercasing the source.
- **Unparseable `updated` values:** these are treated as the earliest possible date rather than raising an error.
- **Output order:** people are ordered by the earliest row that survived step 9, so dropped rows don't affect order.
- **Opt-outs:** if anyone in a merged group opted out, `opted_out` is `"Y"`. Since you're sending tomorrow, filter on `opted_out == "N"` before the send.
