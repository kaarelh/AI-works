\section{Proofs for the relation to operator induction}\label{app:soi}

This appendix proves the results of \cref{sec:soi} and states those that the main text mentions in one sentence. Results of our earlier report \citep{claude2026axiominduction} and of the companion paper \citep{claudeSFI2026fundamentals} are cited by their printed numbers and re-proved only where the setting changes. Throughout, $c,c',\dots$ are machine constants. For a computable map $h$ on indices, universality of $V$ gives a string $r_h$ with $V(r_hs)=h(V(s))$ for all $s\in\dom V$; so $p\mapsto r_hp$ is injective, maps $\dom V$ into $\dom V$ and adds $\lvert r_h\rvert$ to the length. ``Tape by tape'' means: for every tape $\omega$.

\subsection{Decompositions and the four maps}\label{sec:app:soi:maps}

\emph{Measurability.} $\{\omega:f_\omega\supseteq D\}$ is open. ``$f_\omega$ is inconsistent'' is open: an inconsistency uses finitely many labels, each certified under a prefix of $\omega$, and these prefixes are comparable. ``$f_\omega$ is closed'' is a countable intersection of Borel conditions. For theory machines $\{\omega:B\cup A_\omega\vdash\psi\}$ is open and consistency is closed (\cref{lem:setting:measurable}). All events below are Borel.

\begin{lemma}[Mixture decompositions]\status{proved}\src{sfi Lemma 1.2}\label{lem:soi:mixture}
For every finite $D\ne\emptyset$,
\begin{equation}\label{eq:soi:mixF}
\MFcons(D)=\sum_{p\in\dom V}2^{-\lvert p\rvert}\nu^{\mathrm{cons}}_{F_{V(p)}}(D),
\end{equation}
and the same holds for $\nu^{\mathrm{cl}}$ (with $\MFcl$) and for $\nu$ (with $\MF$). For $D=\emptyset$ add $1-\Omega_V$ on the right for $\nu^{\mathrm{cons}}$ and $\nu$, and nothing for $\nu^{\mathrm{cl}}$. For every finite $D$, the empty one included,
\begin{equation}\label{eq:soi:mixA}
\MA(D)=\sum_{p\in\dom V}2^{-\lvert p\rvert}\nu_{G_{V(p)}}(D)+(1-\Omega_V)\,\ind[B\vdash\psi_D].
\end{equation}
\end{lemma}

\begin{proof}
$\dom V$ is prefix-free, so every tape has at most one prefix $p\in\dom V$. On a tape $p\omega'$, $f^{F_U}_{p\omega'}=f^{F_{V(p)}}_{\omega'}$ and $A^{G_U}_{p\omega'}=A^{G_{V(p)}}_{\omega'}$, with $\omega'$ uniform given $p$. On the tapes with no prefix in $\dom V$ (measure $1-\Omega_V$), $f_\omega$ is nowhere defined, so it extends $D$ only if $D=\emptyset$, and then it is consistent, $B$ being consistent; it is never closed, because $B$ proves every tautology and the empty labeller labels none. On these tapes $A_\omega=\emptyset$, so the theory is $B$, which is consistent and proves $\psi_D$ iff $B\vdash\psi_D$.
\end{proof}

The weight of an index $i$ is $\mm(i)=\sum_{V(p)=i}2^{-\lvert p\rvert}$, so \eqref{eq:soi:mixF} reads $\MFcons=\sum_i\mm(i)\nu^{\mathrm{cons}}_{F_i}$ on nonempty $D$.

\begin{lemma}[The four maps]\status{proved; computed}\src{sfi Lemma 1.3}\label{lem:soi:maps}
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item \emph{Proof search.} For a theory machine $G$ let $F_G$ answer a query $\varphi$ as follows: in stages $s=1,2,\dots$ read the tape up to bit $s$, run $s$ steps of the enumeration of $G$ restricted to pairs $(w,\chi)$ with $w$ a prefix of the bits read, and test all derivations of size at most $s$ from $B$ and the sentences emitted so far; halt with $1$ at the first derivation of $\varphi$ and with $0$ at the first derivation of $\neg\varphi$ (within a stage $\varphi$ is tested first). Then $F_G$ is a function machine, and on every tape $\omega$ with $B\cup A^G_\omega$ consistent, $f^{F_G}_\omega$ is consistent and closed and decides every sentence exactly as $B\cup A^G_\omega$ does.
\item \emph{Emission.} For a function machine $F$ let $G_F:=\{(w,\varphi):(\varphi,w,1)\in F\}\cup\{(w,\neg\varphi):(\varphi,w,0)\in F\}$. Then $A^{G_F}_\omega=\Gamma_{f^F_\omega}$ for every $\omega$. So if $f^F_\omega$ is consistent and extends $D$, then $G_F$ is compatible with $D$ on $\omega$; if $f^F_\omega$ is closed and consistent, $B\cup A^{G_F}_\omega$ decides every sentence exactly as $f^F_\omega$ does.
\item \emph{Craig's map with the tape.} For a function machine $F$, read as a machine that reads the tape one-way, let $C_F$ answer a query $\chi$ as follows. Find the at most two decompositions $\chi=\psi^{\wedge m}$ (the right-nested $m$-fold conjunction: $m=1$, or $\psi$ the left conjunct of $\chi$ and $m$ fixed by $\lvert\chi\rvert$). For each, test both readings: (1) run $F$ on $\psi$ for exactly $m-1$ steps, reading the tape as $F$ does, and accept if $F$ halts at step $m-1$ with label $1$; (2) if $\psi=\neg\varphi'$, run $F$ on $\varphi'$ for exactly $m-1$ steps and accept if it halts at step $m-1$ with label $0$. Answer $1$ if some test accepts and $0$ otherwise. Then $C_F$ is a random decider, and $\Cn(A^{\mathrm{dec}}_{C_F}(\omega))=\Cn(\Gamma_{f^F_\omega})$ for every $\omega$.
\item \emph{Closure.} For a partial computable labeller $f$ let $\bar f$ search for proofs of $\varphi$ and of $\neg\varphi$ from $B\cup\Gamma_f$, dovetailing the enumeration of $\Gamma_f$. If $f$ is consistent, $\bar f$ extends $f$, is consistent and closed, and $\Cn(\Gamma_{\bar f})=\Cn(\Gamma_f)$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) The run on a query is a deterministic function of the bits read, and a halting run that has read $w$ runs identically on every tape extending $w$; so the triples $(\varphi,w,y)$ of halting runs form a consistent c.e.\ set. Let $B\cup A^G_\omega$ be consistent. If it proves $\varphi^b$, a derivation uses finitely many axioms, each emitted under a prefix of $\omega$; at a late enough stage all of them have been enumerated and the derivation has size at most $s$, so the run halts, and it cannot first find a derivation of $\varphi^{1-b}$, since none exists. If the theory decides neither, no derivation is found and the label is undefined. So $f^{F_G}_\omega(\varphi)=b$ iff $B\cup A^G_\omega\vdash\varphi^b$. Then $\Gamma_{f}\subseteq\Cn(A^G_\omega)$ is consistent with $B$, and $f$ is closed: $B\cup\Gamma_f\vdash\varphi^b$ implies $B\cup A^G_\omega\vdash\varphi^b$, which implies $f(\varphi)=b$; the converse holds as $\varphi^b\in\Gamma_f$.
(ii) $G_F$ is c.e.\ and emits $\varphi^y$ exactly under the prefixes that certify $f(\varphi)=y$, so $A^{G_F}_\omega=\Gamma_{f^F_\omega}$. If $f$ is closed, $B\cup\Gamma_f\vdash\varphi^b$ iff $f(\varphi)=b$.
(iii) A decomposition $\chi=\psi^{\wedge m}$ with $m\ge2$ has $\psi$ the left conjunct and $m$ fixed by $\lvert\chi\rvert$, since $\lvert\psi^{\wedge m}\rvert=m\lvert\psi\rvert+3(m-1)$ in primitive syntax (Lemma~E.12(b) of the earlier report); the only other candidate is $m=1$. Every run is bounded by $m-1$ simulated steps, so $C_F$ halts on every query and every tape: it is a random decider. Its answer depends only on the tape bits read by the simulated runs, so its triples are consistent. Its accepted set on $\omega$ is $\{\varphi^{\wedge(k+1)}:F\text{ halts on }\varphi\text{ with }1\text{ in exactly }k\text{ steps}\}\cup\{(\neg\varphi)^{\wedge(k+1)}:F\text{ halts on }\varphi\text{ with }0\text{ in exactly }k\text{ steps}\}$: reading (1) gives the first set (also when $\varphi$ is itself a negation), reading (2) the second. Each member is equivalent to a member of $\Gamma_{f^F_\omega}$, and each member of $\Gamma_{f^F_\omega}$ has one, so $\Cn$ is preserved. (Reading (1) alone would lose the members $(\neg\varphi')^{\wedge(k+1)}$ that come from $f(\neg\varphi')=1$, and $\Cn$ would change.)
(iv) As (i), with the r.e.\ set $\Gamma_f$ in place of $A^G_\omega$; $f(\varphi)=b$ puts $\varphi^b\in\Gamma_f$, so $\bar f(\varphi)=b$.
\end{proof}

Only inclusions of events are claimed. The inclusion given by proof search can be strict: on a tape on which the theory is inconsistent, the labeller may already have committed to a consistent set of labels before the inconsistency appeared. A computation on a finite propositional analogue (code in the accompanying repository) finds many such tapes.

\subsection{The equivalences}\label{sec:app:soi:equiv}

\begin{proof}[Proof of \cref{thm:soi:coinfree}]
We use coin-free forms of the maps of \cref{lem:soi:maps}: deterministic machines ignore the tape.
\emph{Enumerators to labellers} (proof search): $p\mapsto r_{\rm ps}p$; if the enumerator $p$ is compatible with $D$, the proof-search labeller is consistent and closed, extends $D$, and decides each $\varphi$ as the theory does.
\emph{Labellers to enumerators} (emission): $p\mapsto r_{\rm em}p$, the enumerator of $\Gamma_{\{V(p)\}}$; if the labeller is consistent and extends $D$, the theory is consistent and proves $D^\pm$; if it is closed, it decides as the labeller does.
\emph{Labellers to deciders} (Craig's map without tape): the decider of Craig's set of the labeller, which halts on every input even when the labeller is partial, since every run is bounded by the exponent of the candidate conjunction.
\emph{Deciders to enumerators}: enumerate the decided set; the theory is the same.
\emph{Consistent to closed}: the closure map.
All maps are injective on programs and add a constant to the length. Summing $2^{-\lvert p\rvert}$ over compatible programs gives (a) and (c), where (c) uses proof search and emission restricted to closed labellers, which preserve the decision of every sentence; taking shortest programs gives (b). Finiteness in (b): if $D$ is consistent, the lookup table of $D$ is a consistent labeller, its closure a closed one, and the decider of the finite set $D^\pm$ is compatible; if not, no consistent theory proves $D^\pm$ and no consistent labeller extends $D$. For the losses, each of the two terms of $-\log W(D)+\log W(\emptyset)$ changes by at most $c$. For deciders this is Theorem~6.3 of the earlier report, stated there for programs that read their input; the change of program format costs a constant.
\end{proof}

With weights $2^{-\lambda\lvert p\rvert}$, $\lambda\ge1$, as in the companion paper on steeper priors \citep{claudeSFI2026steeper}, the same maps give the equivalence within $2^{\pm\lambda c}$.

\begin{remark}[The schema route]\status{(a) proved; (b) proved, the schema argument fails in one sort; (c) proved, given MRDP; (d) proved, given partial truth predicates}\src{sfi Prop 2.2}\label{rem:soi:schema}
A different route to \cref{thm:soi:coinfree} turns a labeller $f$ into the axiom schema $A_f:=\{\mathrm{Acc}_f(\gn\varphi)\to\varphi\}\cup\{\mathrm{Rej}_f(\gn\varphi)\to\neg\varphi\}$, with $\Sigma_1$ formulas $\mathrm{Acc}_f$, $\mathrm{Rej}_f$, written without $<$, that express that $f$ accepts or rejects.
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item \emph{Two sorts.} Let $B=B_A\cup B_L$, with arithmetic axioms $\Q\subseteq B_A\subseteq\Th(\mathbb N)$ on one sort, axioms $B_L$ on the other and no mixed axioms. If $B_L\cup\Gamma_f$ is consistent, so is $B\cup A_f$, and the sentences of the non-arithmetic sort's language that $B\cup A_f$ proves are exactly those that $B_L\cup\Gamma_f$ proves (Proposition~6.4 of the earlier report; it uses the $\Sigma_1$-completeness of $\Q$).
\item \emph{One sort over $\PA$.} Assuming $\Con(\PA)$, the labeller $f$ that accepts $\neg\Con(\PA)$ and, on $\Con(\PA)$, searches for a $\PA$-proof of $\bot$ has $\Gamma_f=\{\neg\Con(\PA)\}$, which is consistent with $\PA$, while $\PA\cup A_f$ is inconsistent (Proposition~6.5 there). So the argument through the schema fails in one sort; the conclusion of \cref{thm:soi:coinfree} does not.
\item \emph{Bounded witnesses.} In one sort, for every decidable $B\supseteq\Q$, true or not, let $A^{\mathrm{bd}}_f$ use the forms $\mathrm{Acc}^{\le k}_f$, $\mathrm{Rej}^{\le k}_f$ whose witnesses are bounded by a binary numeral $k$. Then $\Cn(A^{\mathrm{bd}}_f)=\Cn(\Gamma_f)$ for every labeller $f$, and membership in $A^{\mathrm{bd}}_f$ is decidable in polynomial time without running $f$ (Proposition~E.17 there, given the MRDP theorem): each axiom is $\Q$-equivalent to a member of $\Gamma_f$ or is a $\Q$-theorem. In substance this is Craig's idea, the bound playing the role of the padding.
\item \emph{Templates.} No non-ground template of that report's template class has all its instances in the acceptance part of $A_f$; but for every $\Sigma_n$-sound labeller $f$ one ground reflection sentence of length at most $a\lvert f\rvert+b_n$ reproduces $f$ on $\Sigma_n$ sentences over $\PA$ (Propositions~6.6 and~6.7 there, given partial truth predicates).
\end{enumerate}
The one-tape equivalence avoids the schema altogether: a theory machine is an enumerator that reads a tape, so it may emit $\Gamma_{f_\omega}$ itself (emission); neither decidability, nor arithmetic, nor a schema is needed. The schema, like Craig's padding, matters only for the decider format.
\end{remark}

\begin{proof}[Proof of \cref{thm:soi:onetape}]
(b) is \cref{lem:soi:maps}(i)--(iii). (a), $\MA$ against $\MFcons$, for $D\ne\emptyset$: by \eqref{eq:soi:mixA}, (b) and \eqref{eq:soi:mixF}, with $p\mapsto r_{\rm ps}p$ (an index of $F_{G_{V(p)}}$) and $p\mapsto r_{\rm em}p$ (an index of $G_{F_{V(p)}}$),
\begin{gather*}
\sum_p2^{-\lvert p\rvert}\nu_{G_{V(p)}}(D)\le\sum_p2^{-\lvert p\rvert}\nu^{\mathrm{cons}}_{F_{V(r_{\rm ps}p)}}(D)\le2^{\lvert r_{\rm ps}\rvert}\MFcons(D),\\
\MFcons(D)\le2^{\lvert r_{\rm em}\rvert}\sum_p2^{-\lvert p\rvert}\nu_{G_{V(p)}}(D)\le2^{\lvert r_{\rm em}\rvert}\MA(D).
\end{gather*}
The remaining term $(1-\Omega_V)\ind[B\vdash\psi_D]$ of \eqref{eq:soi:mixA} is at most $\ind[B\vdash\psi_D]$; if $B\vdash\psi_D$, the deterministic proof-search labeller of $B$ is closed and consistent and extends $D$ on every tape, so $\MFcons(D)\ge2^{-c_B}$. Hence $\MA\le(2^{\lvert r_{\rm ps}\rvert}+2^{c_B})\MFcons$. For $D=\emptyset$ both $\MA(\emptyset)$ and $\MFcons(\emptyset)$ lie in $[1-\Omega_V,1]$ (the tapes without a program give the consistent theory $B$ and the empty labeller), so their ratio is at most $1/(1-\Omega_V)$.
$\MA$ against $\MAdec$: a random decider is a theory machine (emit $\chi$ under the prefix that certifies the answer $1$), so $\MAdec\le2^c\MA$; and Craig's map $C_{F_{V(p)}}$ gives $\MAdec\ge2^{-c}\MFcons\ge2^{-2c}\MA$ on $D\ne\emptyset$; for $D=\emptyset$ the decider of $\emptyset$ gives $\MAdec(\emptyset)\ge2^{-c}$.
\end{proof}

\begin{corollary}[Closed labellers and the trichotomy]\status{proved; computed}\src{sfi Cor 3.2}\label{cor:soi:closed}
For every finite $D$ and $\varphi\notin\dom D$, the three outcome weights $\MA(D+(\varphi,1))$, $\MA(D+(\varphi,0))$, $\MAthree(D+(\varphi,\iota))$ (the theory proves $\varphi$, proves $\neg\varphi$, decides neither) agree within $2^{\pm c}$ with the measures of the tapes on which $f_\omega$ is closed, consistent, extends $D$ and has $f_\omega(\varphi)=1$, $0$, $\bot$ respectively; and $2^{-c}\MA\le\MFcl\le\MFcons$.
\end{corollary}

\begin{proof}
Proof search maps tapes with a consistent theory to tapes on which $F_G$ is closed and decides as the theory does; emission maps tapes with a closed consistent labeller to tapes on which the theory decides as the labeller does. The program-less tapes (theory $B$) are covered by the closed deterministic labeller ``proof search from $B$'', of weight $2^{-c_B}$, which decides as $B$ does. Sum as in the proof of \cref{thm:soi:onetape}. The outcome weights of $\MA$ are those of the corollary: $\MA(D+(\varphi,1))$ is the measure of the tapes whose theory is consistent and proves $\psi_D$ and $\varphi$, and likewise for the other two.
\end{proof}

\paragraph{Penalised versions.} One may charge a hypothesis for the time it takes to check its axioms (a decider) or to generate them (an enumerator), with deduction still free (\cref{def:soi:penalised}). The results on these versions rest on an assumption about the universal machine $U$ whose programs are the hypotheses, taken from Appendix~E.6 of the earlier report; it is standard for multitape machines, and we do not prove it, or its form with a tape, for a particular $U$.

\begin{assumption}[(S): efficient self-simulation]\status{assumption}\src{sfi \S7 (the earlier report's (S))}\label{ass:soi:S}
Penalised hypotheses are programs of a fixed universal prefix machine $U$, read to their end before the input. There are constants $\alpha,a\ge1$ such that the fixed wrappers used in the proofs read, copy and measure a program $q$ within $\alpha(\lvert q\rvert+1)$ steps; parse, compare or copy strings of total length $L$ within $\alpha(L+1)^2$ steps; and run the \emph{dovetailing schedule} of $q$ (at stage $s$, start $q$ on the $s$-th sentence and advance every unfinished run by one simulated step) while keeping a \emph{work counter} $T$ (the simulated steps, plus the lengths of the sentences started, plus $\lvert q\rvert$), within $\alpha(T+1)^a$ steps up to work $T$, writing a string of length $L$ at cost at most $\alpha L$. Put $a^*:=\max(a,3)$.
\end{assumption}

\begin{assumption}[(S$_{\rm tape}$): self-simulation with a tape]\status{assumption}\src{sfi \S3.3, Assumption (S$_{\rm tape}$)}\label{ass:soi:Stape}
There are constants $\alpha,a\ge1$ with the properties of \cref{ass:soi:S} when $q$ also reads a one-way random tape: all runs of the dovetailing schedule read the same tape $\omega$ one-way from its start, the work counter also counts tape reads, and up to work $T$ the schedule costs at most $\alpha(T+1)^a$ steps and reads at most $T$ tape bits.
\end{assumption}

We use the format of Appendix~E.6 of the earlier report: hypotheses are programs of $U$, read to their end before the input, now with access to a one-way read-only random tape.

\begin{definition}[Penalised one-tape \SAIabbr]\status{definition}\src{sfi Def 3.3}\label{def:soi:penalised}
A \emph{random decider program} $q$ halts on every pair $(\chi,\omega)$ with yes or no; $A^q_\omega$ is its yes-set on $\omega$, and $\mu^{\mathrm{dec}}_q(D):=\Unif\{\omega:B\cup A^q_\omega\text{ consistent and}\vdash\psi_D\}$. A \emph{random enumerator program} $q$ has no input and writes sentences $a_1(\omega),a_2(\omega),\dots$, completing $a_i$ at step $\mathrm{st}_i(\omega)$ ($\mathrm{st}_0:=0$). For time-constructible $t$:
\begin{itemize}[leftmargin=*]
\item $\MAchk{t}(D):=\sum_q2^{-\lvert q\rvert}\mu^{\mathrm{dec}}_q(D)$, the sum over the random decider programs $q$ whose running time on every query $\chi$ and every tape is at most a constant multiple of $t(\lvert\chi\rvert)$, that is, $\sup_{\chi,\omega}\mathrm{time}_q(\chi,\omega)/t(\lvert\chi\rvert)<\infty$;
\item $\MAgen{t}(D):=\sum_q2^{-\lvert q\rvert}\nu^{\mathrm{gen}}_q(D)$, with $\nu^{\mathrm{gen}}_q(D):=\Unif\{\omega:B\cup\{a_i(\omega)\}_i$ consistent and $\vdash\psi_D\}$, the sum over the random enumerator programs that meet the \emph{delay form} $\sup_{\omega,i}(\mathrm{st}_i-\mathrm{st}_{i-1})/t(\lvert a_i\rvert)<\infty$; in the \emph{strict form} the condition is on $\mathrm{st}_i/t(\lvert a_i\rvert)$, in the \emph{cumulative form} on $\mathrm{st}_i/t(\lvert a_1\rvert+\dots+\lvert a_i\rvert)$.
\end{itemize}
The coin-free $\WAchk{t}$, $\WAgen{t}$ and the Levin-type $\WAKt{t}$ are those of Definition~E.10 of the earlier report, which also assumes $t$ nondecreasing and superadditive with $t(m)\ge m$.
\end{definition}

\begin{remark}[On \cref{ass:soi:Stape}]\status{proof sketch}\src{sfi \S3.3, justification of (S$_{\rm tape}$)}
\emph{Proof sketch.} We do not write this out for a concrete $U$; like \cref{ass:soi:S}, the assumption is used, not proved. A tape-reading program can be simulated by keeping the bits read so far on a work tape and one pointer per run of the schedule; a simulated read copies a stored bit or reads the next tape bit, at cost $O(T)$ up to work $T$. So the exponent $a$ of \cref{ass:soi:S} should grow by at most $1$.
\end{remark}

In this format the unpenalised one-tape laws $M_A^U:=\sum_q2^{-\lvert q\rvert}\nu_{G_q}$, over all tape-reading programs read as theory machines, and the corresponding $\MFcons$ agree with $\MA$ and $\MFcons$ within constant factors, since each universal mixture contains the other's universal machine behind a constant prefix.

\begin{theorem}[Axiom-side penalties change nothing]\status{(a) proved, under (S$_{\rm tape}$); (b) proved, under (S)}\src{sfi Thm 3.4 (a); Thm 7.1 (b)}\label{thm:soi:penalty}
Let $t(m)\ge m^{a^*}$ be time-constructible. There is $c$ such that, for every finite $D$ and in the strict, delay and cumulative forms:
(a) under \cref{ass:soi:Stape}, $2^{-c}\MFcons(D)\le\MAchk{t}(D)\le2^c\MA(D)$ and $2^{-c}\MFcons(D)\le\MAgen{t}(D)\le2^c\MA(D)$, the empty $D$ included; so $\MAchk{t}\asymp\MAgen{t}\asymp\MA\asymp\MFcons$;
(b) under \cref{ass:soi:S}, $2^{-c}\Wcons(D)\le\WAchk{t}(D)\le\WAdec(D)\le2^c\Wcons(D)$ and $2^{-c}\Wcons(D)\le\WAgen{t}(D)\le2^c\Wcons(D)$.
\end{theorem}

\begin{proof}
(a) \emph{Lower bounds.} Let $p$ be a tape-reading program for a function machine $F$. Let $q_F$ say yes on $\chi$ if $F$ labels $\chi$ with $1$, or if $\chi=\neg\varphi$ and $F$ labels $\varphi$ with $0$ (dovetailing the two runs on the same tape). On every tape its yes-set is $\Gamma_{f^F_\omega}$, and $\lvert q_F\rvert\le\lvert p\rvert+c$. The padded random decider $d$, a fixed wrapper followed by $q_F$, answers $\chi$ as in Lemma~E.12(b) of the earlier report: it finds the at most two candidates $\chi=\psi^{\wedge m}$, rejects if $m-1<\lvert q_F\rvert$, and otherwise runs the dovetailing schedule of $q_F$ on the tape, without writing, until the work counter exceeds $m-1$, accepting iff $q_F$ is first detected saying yes on $\psi$ exactly at counter $m-1$. Under \cref{ass:soi:Stape} this takes at most $\alpha(\lvert q_F\rvert+1)+3\alpha(\lvert\chi\rvert+1)^{a'}$ steps, $a':=\max(a,2)$, on every tape, uniformly in $\omega$ (the argument of that lemma, verbatim with the tape); as $(\lvert\chi\rvert+1)^{a'}\le2^{a'}\lvert\chi\rvert^{a^*}\le2^{a'}t(\lvert\chi\rvert)$, the membership bound holds. The answer depends only on the tape bits read by the schedule, so $d$ is a random decider. On each tape its yes-set is $\mathrm{Pad}_\omega:=\{\psi^{\wedge(T+1)}:q_F\text{ is first detected saying yes on }\psi\text{ at counter }T\text{ on }\omega\}$; distinct detections have distinct counters, so each $\psi\in\Gamma_{f^F_\omega}$ has exactly one padded copy and $\Cn(\mathrm{Pad}_\omega)=\Cn(\Gamma_{f^F_\omega})$ (Lemma~E.12(a) there, per tape). So on every tape on which $f^F_\omega$ is consistent and extends $D$, $d$ is compatible with $D$. The padded enumerator, a fixed wrapper followed by $q_F$, writes $\psi^{\wedge(T+1)}$ at each detection; Lemma~E.12(c) there gives, on every tape, delay at most $2\alpha\lvert a_i\rvert^a$, strict completion at most $5\alpha\lvert a_i\rvert^{a^*}$ and cumulative completion at most $2\alpha(\lvert a_1\rvert+\dots+\lvert a_i\rvert)^a$, uniformly in $\omega$. Both maps are injective and add a constant to the length. Summing over $p$ gives the lower bounds for $D\ne\emptyset$. For $D=\emptyset$, the decider that rejects every sentence runs in linear time with theory $B$, which is consistent, and the enumerator that writes nothing meets every delay condition vacuously; so $\MAchk{t}(\emptyset),\MAgen{t}(\emptyset)\ge2^{-c}\ge2^{-c}\MFcons(\emptyset)$.
\emph{Upper bounds.} A random decider program $q$ is read as the theory machine that emits $\chi$ under the tape prefix read by a run of $q$ on $\chi$ that answers yes; a random enumerator program as the theory machine that emits the $i$-th written sentence under the prefix read when it is completed. Each reading is a fixed computable map on programs, realised by a constant-size wrapper, which is injective and keeps the theory on every tape. So $\MAchk{t},\MAgen{t}\le2^cM_A^U\le2^{c'}\MA$ for every $D$. With \cref{thm:soi:onetape} this gives the chain of equivalences.
(b) This is Theorem~6.14(a) of the earlier report: the coin-free version of the construction above, with the padded sets of Lemma~E.12 there. The inequality $\WAchk{t}\le\WAdec$ holds because the penalised deciders are deciders with the same weights, and $\WAdec\le2^c\Wcons$ by \cref{thm:soi:coinfree}.
\end{proof}

The equivalence also holds class by class (coin-free, \cref{prop:soi:disguised}).

\begin{proposition}[Class posteriors, one tape]\status{proved, under (S$_{\rm tape}$)}\src{sfi Prop 3.5}\label{prop:soi:classes}
Let $t(m)\ge m^{a^*}$ be time-constructible and $X$ be $\MAchk{t}$ or $\MAgen{t}$ (any form). For a class $\mathcal C$ of deductively closed theories let $X(\mathcal C;D)$ be the measure, under $X$'s mixture, of the tapes on which the hypothesis is compatible with $D$ and the deductive closure of its axioms is in $\mathcal C$. Under \cref{ass:soi:Stape} there is $c$ with $2^{-c}\UA(\mathcal C;D)\le X(\mathcal C;D)\le2^c\UA(\mathcal C;D)$ for all $\mathcal C$ and $D$. So the posterior odds of any two classes of theories (genuine axiom systems and disguised labellers alike) under $X$ and under $\MA$ differ by a factor at most $2^{2c}$.
\end{proposition}

\begin{proof}
Lower bound: for a theory machine $G$ with program $p$, let $q_G$ say yes on $\chi$ iff $G$ emits $\chi$ on the tape, and pad as in the proof of \cref{thm:soi:penalty}. On every tape the padded hypothesis has the same deductive closure as $B\cup A^G_\omega$, so it lies in $\mathcal C$ and is compatible with $D$ exactly when $G$'s theory is. The program-less tapes of $\MA$ (theory $B$) are matched by the decider of $\emptyset$. Upper bound: a penalised hypothesis, read as a theory machine, has the same theory. For the odds, divide by $X(\text{all};D)$ and $\UA(\text{all};D)=\MA(D)$.
\end{proof}

\begin{proposition}[The chain for the consistent versions]\status{proved, given the companion paper's Theorem 3.21(a) and the construction of its Proposition 4.6(a)}\src{sfi Prop 4.1}\label{prop:soi:chain}
For every consistent $D\ne\emptyset$,
\[
2^{-\KFcons(D)}\le\Wcons(D)\le2^c\MFcons(D)\le2^{K(\dom D)+c}\,2^{-\KFcons(D)},
\]
and $\log(\MFcons(D)/\Wcons(D))\le\Kpad(L^{\mathrm{cons}}_{1/2}(D))+c$, where $L^{\mathrm{cons}}_{1/2}(D)$ is the least $L$ with $\Unif\{\omega:\text{every label of }D\text{ is certified within }\omega_{1:L}\text{ and }f_\omega\text{ is consistent}\}\ge\MFcons(D)/2$. By \cref{thm:soi:coinfree,thm:soi:onetape} the same holds for $\KA$, $\WA$, $\MA$ with a larger $c$.
\end{proposition}

\begin{proof}
A shortest consistent program extending $D$ is a term of $\Wcons$. A consistent deterministic labeller, read as a function machine that ignores the tape, is consistent on every tape; with \eqref{eq:soi:mixF} and the wrapper ``treat $\{V(p)\}$ as a machine'', $\Wcons\le2^c\MFcons$. Next, $\MFcons\le\MF\le2^{-K(D\mid\dom D)+c}$ by Theorem~3.21(a) of the companion paper, which holds for the input set $\Sent$; the lookup table of $D$ is consistent, so $\KFcons(D)\le K(D)+c\le K(\dom D)+K(D\mid\dom D)+c$. For the $\Kpad$ bound, take $L'\ge L$ with $K(L')=\Kpad(L)$, and for $w\in\{0,1\}^{L'}$ the stored-prefix labeller $\varphi\mapsto y$ iff some $(\varphi,w',y)\in F$ has $w'\sqsubseteq w$. It is a sub-function of $f^F_\omega$ for every $\omega\sqsupseteq w$, so it is consistent whenever some extension of $w$ is a consistent tape. The count in the proof of Proposition~4.6(a) of the companion paper, restricted to such $w$, gives, with $r$ the constant-size wrapper, $\Wcons(D)\ge2^{-\lvert r\rvert-K(F)-\Kpad(L)}\Unif\{\omega:D\text{ certified within }\omega_{1:L},\ f^F_\omega\text{ consistent}\}$; take $F=F_U$ and $L=L^{\mathrm{cons}}_{1/2}(D)$.
\end{proof}

\subsection{Positive literals}\label{sec:app:soi:literals}

\begin{proof}[Proof of \cref{thm:soi:transfer}]
By \cref{thm:soi:coinfree,thm:soi:onetape} it suffices to compare consistent \SOIabbr on sentences with plain \SOIabbr on $\mathcal I$. \emph{Strings to sentences:} for a partial computable $g$ on $\mathcal I$ let $g^P(R(t_x)):=g(x)$, undefined off the literals, and likewise for a function machine on $\mathcal I$ (decode $x$ from $t_x$). $\Gamma_{g^P}$ is a set of literals $\pm R(t_x)$, at most one per $x$, and it is consistent with $B$: in a model of $B$ in which the $t_x$ are pairwise distinct, the symbol $R$, which $B$ does not mention, can be interpreted as $\{t_x:g(x)=1\}$. So $g^P$, and the machine version on every tape, is consistent, extends $D$ iff $g$ extends $D^{\rm str}$, and costs a constant. \emph{Sentences to strings:} $f^{\rm str}(x):=f(R(t_x))$, at a constant cost; $f\supseteq D$ implies $f^{\rm str}\supseteq D^{\rm str}$, and consistency is not needed. Sum over the injective maps, and take shortest programs. For the three outcome weights of $R(t_y)$: a consistent set of literals of this kind proves exactly its members (for $R(t_y)^b$ not in the set, reinterpret $R$ at $t_y$), so the theory $\Cn(\Gamma_{g^P})$ decides $R(t_y)$ iff $g(y)$ is defined, with the same value; and the proof-search labeller of a theory, restricted to literals, labels $y$ exactly as the theory decides $R(t_y)$.
\end{proof}

\begin{corollary}[What transfers]\status{proved, by \cref{thm:soi:transfer} and the cited results of the companion paper}\src{sfi Cor 5.2, \S5.5}\label{cor:soi:transfer}
In the setting of \cref{thm:soi:transfer}, every result of the companion paper about binary labels that holds up to additive constants in $K$, $-\log W$ and $-\log\MF$ holds for \SAIabbr on literal data, with the same quantities. In particular:
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item \emph{no coding theorem for axiom systems}: for every $d\ge2$ there are distinct $x_1,x_2\in\{0,1\}^d$ such that $D:=\{(R(t_{x_1}),0),(R(t_{x_2}),1)\}$ has $\KA(D)\ge\log d-2-c$, $-\log\WA(D)\le2+K(\lceil\log(d+1)\rceil)+c$ and $-\log\MA(D)\le2+c$; and over two-point literal data $\log d-c\le\max_D(\KA(D)-\LA(D))\le\log d+O(\log\log d)$ (Theorem~4.9 and Corollary~4.10 of the companion paper); \cref{thm:basic:nocoding} proves the first three bounds directly;
\item \emph{random theories beat deterministic ones}: $\sup_D\MA(D)/\WA(D)=\infty$ over two-point literal data, and $\WA(D)\ge2^{-\lvert D\rvert-\Kpad(2^{d+1})-c}$ for every literal data set over words of length at most $d$ (Propositions~4.5(a) and~4.6 there);
\item \emph{shared randomness is not a mixture of private coins}, read as a comparison of $\MA$ on literal data $D$ with the companion paper's product mixture $\Mprod$ on $D^{\rm str}$: $\MA/\Mprod$ is unbounded (Proposition~4.7 there); no product law internal to \SAIabbr is defined here;
\item \emph{Solomonoff induction}: with unary numerals $t_i:=S^i0$ and $B\in\{\emptyset,\Q\}$, $\MA(\{(R(S^i0),x_{i+1}):i<n\})$ is within $2^{\pm c}$ of the universal semimeasure of $x\in\{0,1\}^n$ (Proposition~3.24 there).
\end{enumerate}
\end{corollary}

\begin{proof}
\Cref{thm:soi:transfer} changes $K$, $-\log W$ and $-\log\MF$ by at most a constant, so every statement that holds up to additive constants in these quantities transfers. (a) is Theorem~4.9 and Corollary~4.10 of the companion paper applied to $D^{\rm str}$; (b) is its Propositions~4.5(a) and~4.6; (c) is its Proposition~4.7, read through $\MA(D)\asymp\MF(D^{\rm str})$, so that it compares $\MA$ with $\Mprod$ on $D^{\rm str}$; (d) is its Proposition~3.24 with $t_i:=S^i0$, which are pairwise distinct in $\mathbb N$, a model of $B\in\{\emptyset,\Q\}$ (with $R$ interpreted arbitrarily). The statements that do not transfer are of the following kinds: labels from a set larger than $\{0,1\}$ (a fresh unary predicate carries one bit); cumulative statements in the normalised convention (each normalised conditional transfers only within $2^{\pm2c}$, so $n$ steps cost $2cn$; single-step quantities do transfer up to a constant factor); private randomness per input or per query, and repeated inputs with fresh noise (no mixture over theory machines with a private coin per atom is defined here, and conflicting labels on one sentence are inconsistent data); history-aware and weak-semimeasure variants, which have no analogue here; computability statements, which are not ``up to constants''; and other objects (time-bounded mixtures, circuits), which we compare directly (\cref{thm:soi:onetapetime,thm:soi:circuit}).
\end{proof}

\begin{proposition}[The price of the coin on literal data]\status{proved, given the companion paper's Proposition 4.6(b) and Theorem 3.21(a)}\src{sfi Prop 5.3}\label{prop:soi:price}
In the setting of \cref{thm:soi:transfer}, for every literal data set $D$ over words of length at most $d$,
\[
\log\frac{\MA(D)}{\WA(D)}\le\delta_D+\Kpad(2^{d+1})+c\le\delta_D+K(\lceil\log(d+1)\rceil)+c',\qquad\delta_D:=\lvert D\rvert-K(D^{\rm str}\mid\dom D^{\rm str}).
\]
\end{proposition}

\begin{proof}
$\WA(D)\ge2^{-c}W(D^{\rm str})\ge2^{-\lvert D\rvert-\Kpad(2^{d+1})-c'}$ by \cref{thm:soi:transfer} and Proposition~4.6(b) of the companion paper, and $\MA(D)\le2^c\MF(D^{\rm str})\le2^{-K(D^{\rm str}\mid\dom D^{\rm str})+c''}$ by its Theorem~3.21(a).
\end{proof}

So on literal data whose labels are algorithmically random given the inputs ($\delta_D=O(1)$), the coin-free version loses at most $\Kpad(2^{d+1})+O(1)$ bits against the one-tape version, however many labels are predicted.

\begin{remark}[An encoding effect]\status{proved; computed}\src{sfi Rem 5.4}\label{rem:soi:encoding}
Let $\chi_i$ be $R(t_{w_i})$ or $\neg R(t_{w_i})$ by independent fair coins, for distinct words $w_i$, with data $(\chi_i,1)$, $i\le n$. Plain coin-free \SOIabbr on sentences, $\Wall$, has bounded loss, through the labeller ``always $1$'', while \SAIabbr and consistent \SOIabbr, coin-free or one-tape, lose at least $n$ bits in expectation (Proposition~E.1 of the earlier report). This is an encoding effect: the map $(\neg R(t_w),1)\mapsto(R(t_w),0)$ turns these data into fair-coin labels on positive literals and leaves $\MA$ and $\WA$ unchanged, since they depend on $D$ only through $\Cn(D^\pm)$ (\cref{prop:basic:structure}); there \SAIabbr is \SOIabbr on strings, which loses at least $n-c$ bits in expectation on fair coins. The labeller ``always $1$'' is inconsistent; consistency is what ties $\neg R(t_w)$ to $R(t_w)$.
\end{remark}

\begin{proof}
The lower bound for \SAIabbr and consistent \SOIabbr is Proposition~E.1 of the earlier report: on every tape the theory (or the consistent labeller) cannot accept both $R(t_w)$ and $\neg R(t_w)$, so the two predictive probabilities sum to at most $1$, and convexity of $-\log$ gives at least one bit per step in expectation; the argument applies tape by tape to $\MA$ and $\MFcons$. The re-encoding $(\neg R(t_w),1)\mapsto(R(t_w),0)$ is a bijection of data sets that keeps $D^\pm$, so it keeps $\MA$ and $\WA$ (\cref{prop:basic:structure}). On positive literals \SAIabbr is \SOIabbr on strings (\cref{thm:soi:transfer}), and on fair-coin labels the latter loses at least $n-c$ bits in expectation, by Theorem~3.21(a) of the companion paper and Gibbs' inequality.
\end{proof}

\subsection{What breaks}\label{sec:app:soi:breaks}

\begin{proof}[Proof of \cref{prop:soi:closure}]
(a) A consistent theory proving $\psi_D$ proves $\chi$; and $\MA$ depends on $D$ only through $\Cn(D^\pm)$ (\cref{prop:basic:structure}). (b) A labeller $f\supseteq D$ with $f(\chi)=0$ has $D^\pm\cup\{\neg\chi\}\subseteq\Gamma_f$, which is inconsistent with $B$. By \cref{thm:soi:coinfree}, $\Wcons(D+(\chi,1))\ge2^{-c}\WA(D+(\chi,1))=2^{-c}\WA(D)\ge2^{-2c}\Wcons(D)$; for one tape use \cref{thm:soi:onetape}. (c) Let $q_\chi$ be a shortest $V$-program for $\chi$. The prefix machine that reads $q_\chi$, then $s\in\dom V$, and outputs an index of ``$\{V(s)\}$, made undefined at $V(q_\chi)$'' is simulated by $V$ with a prefix $r$. The map $s\mapsto rq_\chi s$ is injective; removing one label preserves consistency, and the restricted labeller still extends $D$ because $\chi\notin\dom D$. So the consistent labellers extending $D$ and undefined at $\chi$ have weight at least $2^{-\lvert r\rvert-K(\chi)}\Wcons(D)$. For one tape, apply the same map to function machines and use \eqref{eq:soi:mixF}; on the program-less tapes the empty labeller is already undefined at $\chi$. (d) is \cref{thm:soi:coinfree}(c) and \cref{cor:soi:closed}.
\end{proof}

\begin{remark}[The three outcomes are a measure for \SAIabbr]\status{proved; computed}\src{sfi Rem 6.2}\label{rem:soi:outcomes}
For $\varphi\notin\dom D$, $\MA(D+(\varphi,1))+\MA(D+(\varphi,0))+\MAthree(D+(\varphi,\iota))=\MA(D)$ (\cref{prop:basic:structure}), and likewise for $\WA$: each consistent theory proving $\psi_D$ falls in exactly one of the three classes. For \SOIabbr the corresponding third mass is the defect, which collects abstention and divergence alike.
\end{remark}

\paragraph{Productivity of the independent sentences.} $\mathrm{Ref}_B$ and $\mathrm{Thm}_B$ are the sets of $B$-refutable and $B$-provable sentences. We use three textbook facts about $\Q$ (standard; see \citealp{boolos2007computability}): (R) every decidable relation on numbers is strongly represented in $\Q$, by a formula $\theta(\bar x)$ with $\Q\vdash\theta(\bar{\num n})$ when the relation holds of $\bar n$ and $\Q\vdash\neg\theta(\bar{\num n})$ when it does not; (O) for every $n$, $\Q\vdash\forall z\,(z\le\num n\to z=\num0\vee\dots\vee z=\num n)$ and $\Q\vdash\forall y\,(y\le\num n\vee\num n\le y)$, with $\le$ the usual abbreviation; (Diag) for every formula $\vartheta(x)$ there is a sentence $\rho$, computable from $\vartheta$, with $\Q\vdash\rho\leftrightarrow\vartheta(\gn\rho)$.

\begin{lemma}[Productivity of the independent sentences]\status{known (classical: effective inseparability); proved here, given textbook facts}\src{sfi Lemma 6.3$'$}\label{lem:soi:productive}
Let $\mathcal L\supseteq\Lar$, and let $B$ be decidable and consistent with $\Q\subseteq\Cn(B)$. From an index of an r.e.\ set $S\subseteq\IndB$ one computes an $\Lar$-sentence $\rho\in\IndB\setminus S$. In particular $\IndB$ is productive, hence not r.e., and $\mathrm{Ref}_B$ is r.e.\ and not decidable.
\end{lemma}

The lemma is classical: it is a form of the effective inseparability of the provable and the refutable sentences of consistent r.e.\ extensions of $\Q$ (cf.\ \citealp{smullyan1961theory}). We include the proof, which is Rosser's argument relative to $S$.

\begin{proof}
Let $\mathsf{Neg}(y,x)$ be the decidable relation ``$y$ codes a $B$-derivation of the negation of the sentence coded by $x$, or $x$ is enumerated into $S$ within $y$ steps'', and $\mathsf{Prf}(y,x)$ the decidable relation ``$y$ codes a $B$-derivation of the sentence coded by $x$'' ($B$ is decidable). Let $\theta_{\mathrm{neg}},\theta_{\mathrm{prf}}$ strongly represent them (R), and let $\rho$ satisfy $\Q\vdash\rho\leftrightarrow\exists y\,(\theta_{\mathrm{neg}}(y,\gn\rho)\wedge\forall z\le y\ \neg\theta_{\mathrm{prf}}(z,\gn\rho))$ (Diag); $\rho$ is computed uniformly from the index of $S$.
\emph{$B\nvdash\rho$.} Suppose $k$ codes a $B$-derivation of $\rho$. Then $\rho\notin\IndB\supseteq S$, and $B\nvdash\neg\rho$ by consistency, so $\mathsf{Neg}(n,\gn\rho)$ fails for every $n$ and $\Q\vdash\neg\theta_{\mathrm{neg}}(\num n,\gn\rho)$ for every $n$ (R); also $\Q\vdash\theta_{\mathrm{prf}}(\num k,\gn\rho)$. Argue in $\Q$: given $y$ with $\theta_{\mathrm{neg}}(y,\gn\rho)\wedge\forall z\le y\,\neg\theta_{\mathrm{prf}}(z,\gn\rho)$, by (O) either $\num k\le y$, which contradicts $\theta_{\mathrm{prf}}(\num k,\gn\rho)$, or $y\le\num k$, so that $y$ is one of $\num0,\dots,\num k$, which contradicts $\neg\theta_{\mathrm{neg}}(\num n,\gn\rho)$. So $\Q\vdash\neg\rho$, hence $B\vdash\neg\rho$, and $B$ is inconsistent, a contradiction.
\emph{$B\nvdash\neg\rho$ and $\rho\notin S$.} Otherwise $\mathsf{Neg}(n,\gn\rho)$ holds for some $n$, while $\mathsf{Prf}(m,\gn\rho)$ fails for every $m$ by the previous step. By (R) and (O), $\Q\vdash\theta_{\mathrm{neg}}(\num n,\gn\rho)\wedge\forall z\le\num n\,\neg\theta_{\mathrm{prf}}(z,\gn\rho)$, so $\Q\vdash\rho$ and $B\vdash\rho$, against the previous step.
So $\rho\in\IndB\setminus S$. A productive set is not r.e.\ (take $S$ to be the set itself). $\mathrm{Ref}_B$ is r.e.\ since $B$ is decidable; if it were decidable, so would be $\mathrm{Thm}_B=\{\varphi:\neg\varphi\in\mathrm{Ref}_B\}$, hence $\IndB$, which is not even r.e.
\end{proof}

\begin{proof}[Proof of \cref{prop:soi:independent}]
(a) Coin-free: map a consistent three-valued labeller $f$ to the enumerator of $\Gamma_f$ (its $1/0$ part); the theory $\Cn(\Gamma_f)$ proves the $1/0$ data and, by the consistency of $f$, decides none of the sentences $f$ labels $\iota$, which include those of $D$. One tape: emission, tape by tape, on the tapes on which $f^F_\omega$ is consistent and extends $D$, and \eqref{eq:soi:mixF}.
(b) Coin-free: let $\mathcal F_n$ be the finite (not computable) set of the $p\in\dom V$ with $\lvert p\rvert\le n$ for which $\{V(p)\}$ is a consistent three-valued labeller, and $I_n$ the union of their $\iota$-sets. It is a finite union of r.e.\ sets, hence r.e., and $I_n\subseteq\IndB$, since $f(\sigma)=\iota$ for a consistent $f$ implies $B\cup\Gamma_f\nvdash\sigma,\neg\sigma$. Let $\sigma_n\in\IndB\setminus I_n$ (\cref{lem:soi:productive}). Every consistent labeller extending $\{(\sigma_n,\iota)\}$ labels $\sigma_n$ with $\iota$, so its program is not in $\mathcal F_n$ and is longer than $n$; hence $\Wthreecons(\{(\sigma_n,\iota)\})\le\Omtail{n+1}$. The enumerator of $\emptyset$ (theory $B$) is compatible with $\{(\sigma_n,\iota)\}$, so $\WAthree(\{(\sigma_n,\iota)\})\ge2^{-c}$.
One tape: fix $n$ and $\varepsilon:=2^{-n}$. For $p\in\dom V$ with $\lvert p\rvert\le n$ let $F_p:=F_{V(p)}$ and $\mathrm{Cons}_p$ the (Borel) set of tapes on which $f^{F_p}_\omega$ is consistent. If $\Unif(\mathrm{Cons}_p)=0$, $p$ contributes nothing. Otherwise, by the Lebesgue density theorem (\cref{lem:logic:density}) almost every $\omega\in\mathrm{Cons}_p$ has a prefix $w$ with $\Unif(\mathrm{Cons}_p\cap[w])>(1-\varepsilon)2^{-\lvert w\rvert}$. Let $\mathcal W_p$ be the set of minimal such prefixes, a prefix-free set with $\Unif(\mathrm{Cons}_p\setminus\bigcup_{w\in\mathcal W_p}[w])=0$, and choose a finite $\mathcal W'_p\subseteq\mathcal W_p$ with $\sum_{w\in\mathcal W_p\setminus\mathcal W'_p}2^{-\lvert w\rvert}\le\varepsilon$. For $w\in\mathcal W'_p$ put
\[
J_w:=\{\sigma:\Unif([w]\cap\{\omega:f^{F_p}_\omega(\sigma)=\iota\})>\varepsilon2^{-\lvert w\rvert}\}.
\]
$J_w$ is r.e., uniformly in $(p,w)$ and the rational $\varepsilon$, because the measure of the open set $[w]\cap\{f(\sigma)=\iota\}$ is lower semicomputable. And $J_w\subseteq\IndB$: if $\sigma\in J_w$, the set $[w]\cap\{f(\sigma)=\iota\}$ has measure larger than $\varepsilon2^{-\lvert w\rvert}>\Unif([w]\setminus\mathrm{Cons}_p)$, so it meets $\mathrm{Cons}_p$; on such a tape $f_\omega$ is consistent and labels $\sigma$ with $\iota$, so $B\cup\Gamma_{f_\omega}$, and hence $B$, decides neither $\sigma$ nor $\neg\sigma$. Let $I'_n:=\bigcup_{\lvert p\rvert\le n}\bigcup_{w\in\mathcal W'_p}J_w$, a finite union of r.e.\ subsets of $\IndB$, and $\sigma'_n\in\IndB\setminus I'_n$ (\cref{lem:soi:productive}). For $\lvert p\rvert\le n$, since $\sigma'_n\notin J_w$,
\begin{align*}
\nu^{3,\mathrm{cons}}_{F_p}(\{(\sigma'_n,\iota)\})&\le\sum_{w\in\mathcal W'_p}\Unif([w]\cap\{f(\sigma'_n)=\iota\})+\sum_{w\in\mathcal W_p\setminus\mathcal W'_p}2^{-\lvert w\rvert}+\Unif\Big(\mathrm{Cons}_p\setminus\bigcup_{w\in\mathcal W_p}[w]\Big)\\
&\le\sum_{w\in\mathcal W'_p}\varepsilon2^{-\lvert w\rvert}+\varepsilon\le2\varepsilon.
\end{align*}
By \eqref{eq:soi:mixF}, $\MFthreecons(\{(\sigma'_n,\iota)\})\le\sum_{\lvert p\rvert\le n}2^{-\lvert p\rvert}2\varepsilon+\Omtail{n+1}\le2^{1-n}+\Omtail{n+1}$. The deterministic theory machine that emits nothing (theory $B$) is compatible with $\{(\sigma'_n,\iota)\}$ on every tape, so $\MAthree(\{(\sigma'_n,\iota)\})\ge2^{-c}$.
(c) For $\sigma\in\IndB$ the labeller ``$\iota$ at $\sigma$, undefined elsewhere'' is consistent and has length $K(\sigma)+c$; with $\WAthree\le1$ this gives the coin-free bound, and as a function machine that ignores the tape it gives the one-tape bound.
\end{proof}

The coin-free part of (b) is the special case of machines that ignore the tape: then $J_w$ is the labeller's $\iota$-set and no density argument is needed. One tape, the obstacle is that the set of sentences a random machine labels $\iota$ on consistent tapes need not be r.e., since it depends on the positivity of the measure of a $\Pi^0_1$ class; the density point restores an r.e.\ subset of $\IndB$ that carries almost all of the consistent $\iota$-mass. A computation on finite analogues (code in the accompanying repository), with machines consistent on most tapes and noise that labels $B$-decided sentences $\iota$, confirms the density step and the summed bound in every case tested.

\begin{proposition}[Large gaps have no r.e.\ witnesses]\status{proved}\src{sfi Prop 6.4}\label{prop:soi:noRE}
In the setting of \cref{prop:soi:independent}: (a) for every r.e.\ $S\subseteq\IndB$ and every $\sigma\in S$, $\Wthreecons(\{(\sigma,\iota)\})\ge2^{-K(S)-c}$ and $\MFthreecons(\{(\sigma,\iota)\})\ge2^{-K(S)-c}$, with $K(S)$ the complexity of an index of $S$; so no computable sequence $\sigma_n\in\IndB$ has $\Wthreecons(\{(\sigma_n,\iota)\})\to0$. (b) If $n(\sigma)$ is the length of a shortest program of a consistent three-valued labeller that labels $\sigma$ with $\iota$, then $2^{-n(\sigma)}\le\Wthreecons(\{(\sigma,\iota)\})\le\Omtail{n(\sigma)}$.
\end{proposition}

\begin{proof}
(a) The labeller ``$\iota$ on every member of $S$ (enumerate $S$), undefined elsewhere'' has $\Gamma=\emptyset$ and labels only $B$-independent sentences, so it is consistent; it has length $K(S)+c$. As a function machine that ignores the tape it gives the one-tape bound. (b) Every program counted has length at least $n(\sigma)$, and one has length $n(\sigma)$.
\end{proof}

\begin{remark}[Shared versus private randomness]\status{proved}\src{sfi Rem 6.6}\label{rem:soi:product}
If a product law (private coins per input, Proposition~3.9(a) of the companion paper) gives positive probability to labelling $\varphi$ true and to labelling $\psi$ true, with $B\cup\{\varphi,\psi\}$ inconsistent, the sampled labeller is inconsistent with probability at least the product of the two. The consistent part of a product law is not a product law: for a $B$-independent $\varphi$, on the two inputs $\varphi$, $\neg\varphi$ with fair independent labels, the consistent outcomes are $(1,0)$ and $(0,1)$, each of mass $\frac14$, while a product would need $P(1,1)P(0,0)=P(1,0)P(0,1)\ne0$. (If $B$ decides $\varphi$, one of these two outcomes is inconsistent too.) The shared tape is what makes a random labeller a random \emph{theory}; private randomness is harmless on logically independent inputs such as positive literals (\cref{thm:soi:transfer}). Product laws are machine laws, so $\MFcons$ dominates their consistent parts.
\end{remark}

\begin{proposition}[Computability of the versions]\status{(a), (b), (c), (c$'$) proved; $\emptyset'$-computability of the closed and decider versions not claimed}\src{sfi Prop 6.7}\label{prop:soi:computability}
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item Suppose the set of finite data sets $D$ with $B\cup D^\pm$ consistent is not r.e.\ (for instance if $\mathcal L\supseteq\Lar$ and $B$ is decidable and consistent with $\Q\subseteq\Cn(B)$, since then $\{\varphi:\{(\varphi,1)\}\text{ is consistent}\}$ is the complement of $\mathrm{Ref}_B$, which is r.e.\ and not decidable by \cref{lem:soi:productive}; or if $B=\emptyset$ and $\mathcal L$ has a binary relation symbol, by Church's theorem, \citealp{church1936note}). Then none of $\WA$, $\WAdec$, $\Wcons$, $\Wcl$, $\MA$, $\MAdec$, $\MFcons$, $\MFcl$ is lower semicomputable.
\item In every setting, $\MA(\emptyset)$ and $\MFcons(\emptyset)$ are not left-c.e.; so $\MA$ and $\MFcons$ are not lower semicomputable.
\item $\WA$, $\Wcons$, $\MA$ and $\MFcons$ are computable relative to $\emptyset'$, uniformly in $D$.
\item[(c$'$)] $\WAdec$, $\MAdec$, $\Wcl$ and $\MFcl$ are computable relative to $\emptyset''$, uniformly in $D$.
\end{enumerate}
\end{proposition}

Whether the four versions of (c$'$) are computable relative to $\emptyset'$ is not settled here. Being closed and being a (random) decider are $\Pi_2$ conditions, and for closedness the condition is $\Pi_2$-complete: with $\chi_0,\chi_1,\dots$ an injective enumeration of the $B$-decided sentences, the labeller that labels $\chi_n$ with $B$'s decision once program $e$ halts on $n$ is consistent, and it is closed iff $e$ is total.

\begin{proof}
(a) For a function $\nu$ that is lower semicomputable uniformly in $D$, the support $\{D:\nu(D)>0\}$ is r.e. Each object listed has support exactly the consistent data sets: positive weight needs a consistent hypothesis that proves or extends $D$; conversely, for a consistent $D$ the lookup table of $D$ (as a labeller, a finite theory, a decider, or a machine that ignores the tape) has positive weight in $\WA$, $\WAdec$, $\Wcons$, $\MA$, $\MAdec$ and $\MFcons$, and for the closed versions $\Wcl$, $\MFcl$ its closure ``proof search from $B\cup D^\pm$'', which is closed, consistent and extends $D$, has positive weight (the lookup table itself is never closed).
(b) For $\MA$ this is \cref{thm:basic:inc}. For $\MFcons$: for a left-c.e.\ real $\zeta$ with approximations $\zeta_s\uparrow\zeta$, let $F_\zeta$ label the sentence $\exists x\,x\ne x$ with $1$ under a prefix $w$ as soon as some stage $s$ has $0.w+2^{-\lvert w\rvert}\le\zeta_s$, and be undefined elsewhere; its labelling is inconsistent with probability $\Unif\{\omega:0.\omega<\zeta\}=\zeta$. With $p_\zeta$ a $V$-program for $F_\zeta$, the probability that $f_\omega$ is inconsistent is $2^{-\lvert p_\zeta\rvert}\zeta+\kappa$ with $\kappa$ left-c.e. If $\MFcons(\emptyset)$, which is one minus that probability, were left-c.e., the probability would be computable and $\zeta$ right-c.e.; take $\zeta=\Omega_V$.
(c) $\WA$ and $\Wcons$ are sums $\sum_p2^{-\lvert p\rvert}\ind[\cdot]$ of $\Delta_2$ indicators (``proves $\psi_D$'' and ``extends $D$'' are $\Sigma_1$, consistency is $\Pi_1$), and the tails $\Omtail{\ell}$ are $\emptyset'$-computable; so $\emptyset'$ computes each value to any precision. For one tape, $\MA(D)=\Unif(O_{\psi_D})-\Unif(O_\bot)$ (\cref{prop:basic:form}), and $\MFcons(D)=\Unif(O)-\Unif(O\setminus C)$ with $O:=\{\omega:f_\omega\supseteq D\}$ effectively open and $C:=\{\omega:f_\omega\text{ consistent}\}$ effectively closed: differences of left-c.e.\ reals, uniformly in $D$.
(c$'$) Coin-free: ``$V(p)$ is a total $\{0,1\}$-valued program'' and ``$\{V(p)\}$ is closed'' ($\forall\varphi,b:B\cup\Gamma\vdash\varphi^b\to\{V(p)\}(\varphi)=b$) are $\Pi_2$, so $\emptyset''$ decides each indicator and computes the sums. One tape, deciders: ``$F_{V(p)}$ halts on every query and every tape'' is $\Pi_2$, because for each query, halting on all tapes is, by compactness of Cantor space, equivalent to the $\Sigma_1$ condition that some $n$ and $s$ make every tape prefix of length $n$ halt within $s$ steps. For a random decider $\mu^{\mathrm{dec}}_F(D)$ is a difference of left-c.e.\ reals uniformly in an index, as in (c); so $\emptyset''$ computes $\MAdec(D)$. One tape, closed: ``$f_\omega$ is closed'' is the countable intersection over $(\varphi,b)$ of the sets $\{\omega:B\cup\Gamma_{f_\omega}\nvdash\varphi^b\}\cup\{\omega:f_\omega(\varphi)=b\}$, each the union of an effectively closed and an effectively open set, hence an effective $G_\delta$. So the event in $\MFcl(D)$ is an effective $\Pi^0_2$ class $\bigcap_kU_k$ with $U_k$ uniformly effectively open and decreasing; its measure $\inf_k\Unif(U_k)$ is an infimum of uniformly left-c.e., hence uniformly $\emptyset'$-computable, reals, so it is right-c.e.\ relative to $\emptyset'$ and computable from $\emptyset''$.
\end{proof}

\begin{question}\status{open}\src{sfi \S6.4, open question after Prop 6.7}\label{q:soi:lsc}
Can $\WA$ be lower semicomputable when consistency of finite sets of sentences with $B$ is decidable (for instance for a propositional language, or a monadic language without function symbols and $B=\emptyset$)?
\end{question}

There the support argument of \cref{prop:soi:computability}(a) says nothing, and the argument of (b) does not carry over directly to the coin-free version, because the weight of the inconsistent programs is not separated from $\Omega_V$ by a computable amount.

\paragraph{Pathwise and machine-wise consistency.} $\MFcons$ conditions on the event that the \emph{sampled} labeller is consistent. The machine-wise alternative is
\[
\MFmw(D):=\sum_{p\in\dom V}2^{-\lvert p\rvert}\,\ind[F_{V(p)}\text{ is consistent on every tape}]\,\nu_{F_{V(p)}}(D),
\]
and $M_A^{\mathrm{mw}}$ is defined likewise over the theory machines whose theory is consistent on every tape.

\begin{proposition}[Pathwise against machine-wise consistency]\status{(a)--(c) proved; computed}\src{sfi Prop 6.8 (a)--(c)}\label{prop:soi:pathwise}
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item $\Wcons\le2^c\MFmw\le2^c\MFcons$ on nonempty $D$, and $\MFmw\asymp M_A^{\mathrm{mw}}$.
\item For every consistent $D\ne\emptyset$, $\MFcons(D)\le2^{\min\{\Kpad(L^{\mathrm{cons}}_{1/2}(D)),\,K(\dom D)\}+c}\,\MFmw(D)$, with $L^{\mathrm{cons}}_{1/2}$ as in \cref{prop:soi:chain}.
\item If consistency of finite sets with $B$ is decidable (an algorithm decides, for finite $\Lambda\subseteq\Sent$, whether $B\cup\Lambda$ is consistent; for example if $\Cn(B)$ is decidable, for propositional languages, and for monadic languages without function symbols with $B=\emptyset$, by the decidability of monadic first-order logic, \citealp{lowenheim1915moglichkeiten,behmann1922beitrage}), then $\MFcons\asymp\MFmw$ and $\MA\asymp M_A^{\mathrm{mw}}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(a) A deterministic consistent labeller is consistent on every tape, and a machine consistent on every tape has $\nu=\nu^{\mathrm{cons}}$. Proof search and emission preserve ``consistent on every tape'', since on each tape proof search of a consistent theory is consistent and emission of a consistent labelling is consistent. (b) \Cref{prop:soi:chain} with (a). (c) \emph{The global-schedule filter.} For a function machine $F$, read as a machine that reads the tape one-way, let $F'$ answer a query $\varphi$ by running the \emph{global schedule} of $F$: stage $s$ reads tape bit $s$, starts $F$ on the $s$-th sentence and advances every started run by one step; certification events (a run halting with a label) are ordered by stage, ties by sentence index. When $\varphi$'s event occurs, $F'$ checks whether the set of labels certified up to and including it is consistent with $B$ (decidable by hypothesis); if so it outputs $\varphi$'s label, otherwise it diverges. The answer is a deterministic function of the bits read, so $F'$ is a function machine, and $K(F')\le K(F)+c$. On a tape $\omega$ let $l_1,l_2,\dots$ be the labelled sentences in event order. $F'$ outputs the label of $l_i$ iff $\{l_1,\dots,l_i\}$ is consistent; consistency of these prefixes is monotone, so the outputs form an initial segment all of whose finite subsets are consistent, hence a consistent set, by compactness. So $F'$ is consistent on every tape, and $F'=F$ on every tape on which $F$ is consistent. Hence $\nu_{F'}(D)\ge\nu^{\mathrm{cons}}_F(D)$ for every $D$, and summing with \eqref{eq:soi:mixF} gives $\MFmw\ge2^{-c}\MFcons$. The theory-machine version filters emissions in the same way.
\end{proof}

With two unary function symbols, consistency of finite sets is not decidable, so in (c) ``monadic'' must exclude function symbols.

\begin{question}\status{open}\src{sfi Prop 6.8(d)}\label{q:soi:pathwise}
Is $\MFcons/\MFmw$ bounded in general, for instance for $B=\Q$, or for $B=\emptyset$ and a language with a binary relation symbol?
\end{question}

A machine consistent on every tape must avoid inconsistency, which is only semi-decidable, so its support is an r.e.\ family of consistent data sets; a pathwise-conditioned machine can exploit the $\Pi^0_1$ event that nothing inconsistent ever happens. The lottery does not separate the two: the lottery machine is consistent on every tape (\cref{thm:logic:lottery}).

\subsection{Time bounds with free deduction}\label{sec:app:soi:time}

As in Appendix~E.6 of the earlier report, the penalised coin-free hypotheses are programs of the machine $U$ of \cref{ass:soi:S}, so $\mathrm{time}_p(x)\ge\lvert p\rvert$, and $a':=\max(a,2)$. Clocked strings, their weights $w(f)$ and $c_\bot$ are as in \cref{sec:soi:time}, with $E$ the Elias-$\delta$ code of $\lvert f\rvert+1$ followed by $f$; $c_1$ is the length of the fixed wrapper that runs a clocked string and diverges when it abstains. Simulations between $U_0$, $U$ and multitape machines have polynomial overhead (standard; used in the same form as there).

\begin{proposition}[A Levin penalty costs a logarithm]\status{proved, under (S)}\src{sfi Prop 7.2}\label{prop:soi:kt}
Let $t(m)\ge m^{a^*}$ be time-constructible, $m_0$ the least length of a sentence, $\WAKt{t}$ the mixture over all deciders $p$ with weight $2^{-\lvert p\rvert}/c_p$, $c_p:=\sup_\chi\mathrm{time}_p(\chi)/t(\lvert\chi\rvert)$ (Definition~E.10 of the earlier report), and $W^1(D):=\sum_f2^{-\lvert f\rvert}/(\lvert f\rvert+1)$ over the programs $f$ of consistent labellers extending $D$. Under \cref{ass:soi:S}, $\WAKt{t}(\emptyset)\le t(m_0)$, and there is $c$ with $2^{-c}W^1(D)\le\WAKt{t}(D)\le2^ct(m_0)W^1(D)$ for every $D$; so a labeller $f$ pays $\log(\lvert f\rvert+1)+O(1)$ bits beyond its unpenalised weight.
\end{proposition}

\begin{proof}
This is Proposition~E.13 of the earlier report. Since $\mathrm{time}_p\ge\lvert p\rvert$, $c_p\ge\lvert p\rvert/t(m_0)$, so $p$ has weight at most $t(m_0)2^{-\lvert p\rvert}/\lvert p\rvert$, and the total is at most $t(m_0)$ by Kraft's inequality. For the lower bound, the padded decider of a consistent labeller $f$ (as in the proof of \cref{thm:soi:penalty}) has $c_d\le\beta(\lvert f\rvert+1)$ for a constant $\beta$, by Lemma~E.12(b) there and $(\lvert\chi\rvert+1)^{a'}\le2^{a'}t(\lvert\chi\rvert)$, so it has weight at least $2^{-\lvert f\rvert-2c}/(\beta(\lvert f\rvert+1))$; the map is injective. For the upper bound, map a compatible decider $p$ to its proof-search labeller $f_p$, with $\lvert p\rvert\le\lvert f_p\rvert\le\lvert p\rvert+c$; the weight of $p$ is at most $t(m_0)(c+2)2^c\,2^{-\lvert f_p\rvert}/(\lvert f_p\rvert+1)$, and $p\mapsto f_p$ is injective.
\end{proof}

\begin{theorem}[Domination lemma]\status{proved}\src{sfi Thm 7.3}\label{thm:soi:domination}
Let $X$ be a coin-free mixture and $v_X$ a weight on consistent labellers such that $W_X(D)\ge2^{-c_X}\sum_gv_X(g)$, the sum over the consistent labellers $g$ extending $D$, and $W_X(\emptyset)\le\Omega_X$. Let $Y$ be a mixture with $W_Y(D)=\sum_hv_Y(h,D)$, the sum over the hypotheses $h$ compatible with $D$, with an injective map $h\mapsto g_h$ to consistent labellers that compute the same partial map as $h$, with $v_Y(h,D)\le k_Y\,v_X(g_h)$, and with $W_Y(\emptyset)\ge\kappa_Y>0$. Then $\ell_X(D)\le\ell_Y(D)+c_X+\log(k_Y\Omega_X/\kappa_Y)$ for every $D$.
\end{theorem}

\begin{proof}
A labeller $g_h$ computes the same partial map as $h$, so it extends $D$ iff $h$ is compatible with $D$. Hence $W_X(D)\ge2^{-c_X}\sum_hv_X(g_h)\ge2^{-c_X}k_Y^{-1}W_Y(D)$, and $\ell_X(D)=-\log W_X(D)+\log W_X(\emptyset)\le-\log W_Y(D)+c_X+\log(k_Y\Omega_X)=\ell_Y(D)-\log W_Y(\emptyset)+c_X+\log(k_Y\Omega_X)$.
\end{proof}

The weaker hypothesis $\Gamma_{g_h}=\Gamma_h$ does not suffice: a labeller that rejects $\psi$ and one that accepts $\neg\psi$ and abstains on $\psi$ have the same $\Gamma$ but differ on the datum $(\psi,0)$.

\begin{corollary}[A one-sided equivalence]\status{proved; for the penalised versions under (S)}\src{sfi Cor 7.3$'$; Thm 7.8(a)}\label{cor:soi:onesided}
For every time-constructible $\tau$ there is $c_\tau$ such that, on every data set $D$, the losses of $\WA$ and $\WAdec$ exceed that of $W^{\mathrm{cons},\tau}$ by at most $c_\tau$, and the loss of $\MA$ exceeds that of $\MF^{\mathrm{cons},\tau}$ by at most $c_\tau$. Under \cref{ass:soi:S} the same holds for $\WAchk{t}$ and $\WAgen{t}$, $t(m)\ge m^{a^*}$ time-constructible.
\end{corollary}

\begin{proof}
Coin-free (Theorem~6.14(b) of the earlier report): apply \cref{thm:soi:domination} with $X\in\{\WA,\WAdec,\WAchk{t},\WAgen{t}\}$ and $Y=W^{\mathrm{cons},\tau}$. By \cref{thm:soi:coinfree} and, for the penalised versions under \cref{ass:soi:S}, \cref{thm:soi:penalty}(b), $W_X\ge2^{-c}\Wcons$, and $W_X(\emptyset)\le1$ by Kraft's inequality. For a clocked string $h$ let $g_h$ be the wrapper that runs $h^\tau$ and diverges when it abstains, followed by $E(h)$; it computes the same partial map as $h^\tau$, $\lvert g_h\rvert=\lvert E(h)\rvert+c_1$, and $h\mapsto g_h$ is injective. $W^{\mathrm{cons},\tau}(\emptyset)\ge2^{-c_\bot}$, the weight of the always-abstaining string ($B$ is consistent). So $c_\tau:=c+c_1+c_\bot$ works. One tape: the clocked function machine of a string $f$ is a function machine whose $V$-program is a fixed wrapper followed by $E(f)$, injectively, so \eqref{eq:soi:mixF} gives $\MFcons\ge2^{-c_1}\MF^{\mathrm{cons},\tau}$ on $D\ne\emptyset$; \cref{thm:soi:onetape} gives $\MA\ge2^{-c}\MFcons$; and $\MA(\emptyset)\le1$, $\MF^{\mathrm{cons},\tau}(\emptyset)\ge2^{-c_\bot}$.
\end{proof}

\begin{proof}[Proof of \cref{thm:soi:diagonal}]
This is Theorem~6.15 of the earlier report; we recall the argument. By the MRDP theorem \citep{matiyasevich1970enumerable,davis1973hilbert} there are polynomials $E,E'$ with natural-number coefficients such that program $e$ halts on $x$ with output $y$ iff $\mathbb N\models\exists\bar u\,(E(e,x,y,\bar u)=E'(e,x,y,\bar u))$; $\mathrm{Out}(e,x,y)$ is this formula (Lemma~E.14(a) there), and $\mathrm{DET}:=\forall e\,\forall x\,\forall y\,\forall y'\,\forall\bar u\,\forall\bar u'\,(E(e,x,y,\bar u)=E'(e,x,y,\bar u)\wedge E(e,x,y',\bar u')=E'(e,x,y',\bar u')\to y=y')$, a true $\Pi_1$ sentence. Let $H_0:=\Q\cup\{\mathrm{DET}\}$.
\emph{Construction.} Let $\mathcal G(e',n)$ be the program that, for $j=1,\dots,n$, forms $\sigma_j:=\mathrm{Out}(\num{e'},\num j,S0)$, asks $P$ for approximations $(a_0,a_1)$ within $2^{-j-3}$ of its pair on the history $((\sigma_1,\beta_1),\dots,(\sigma_{j-1},\beta_{j-1}))$ and the sentence $\sigma_j$, and sets $\beta_j:=0$ if $a_0\le a_1$ and $\beta_j:=1$ otherwise; it outputs $\beta_n$. $\mathcal G$ is total, as $P$ is. By Kleene's recursion theorem (standard) there is $e$ with $e(n)=\mathcal G(e,n)$ for all $n$. The values $\beta_j$ computed inside $\mathcal G(e,n)$ do not depend on $n$, so $e(j)=\beta_j\in\{0,1\}$ and $\sigma_j=s_j$; and $\mathbb N\models s_n$ iff $e(n)=1$, so the label $b_n$ is $\beta_n$, and $P$ is queried exactly on the true histories $D_{j-1}$.
\emph{$H_0$ decides each $s_n$.} If $b_n=1$, $s_n$ is a true existential equation, which $\Q$ proves (Lemma~E.14(c) there). If $b_n=0$, $\mathrm{Out}(\num e,\num n,0)$ is one, so $\Q$ proves it, and with $s_n$ and $\mathrm{DET}$ it gives $S0=0$, which $\Q$ refutes; so $H_0\vdash\neg s_n$.
\emph{The loss of $P$.} With $q_b:=q_b(D_{j-1},s_j)$ and $\varepsilon:=2^{-j-3}$, the tie rule gives $q_{b_j}\le a_{b_j}+\varepsilon\le a_{1-b_j}+\varepsilon\le q_{1-b_j}+2\varepsilon$, and with $q_0+q_1\le1$ this gives $q_{b_j}\le\frac12+2^{-j-3}$. So $-\log q_{b_j}\ge1-\log(1+2^{-j-2})$, and $\sum_j\log(1+2^{-j-2})\le\sum_j2^{-j-2}/\ln2=1/(4\ln2)<0.361$.
\emph{Axiom induction.} $H_0$ is true in $\mathbb N$ and $B$ in an expansion of $\mathbb N$, so $B\cup H_0$ is consistent, and $H_0$ is compatible with every $D_n$. It is finite, so it has a decider running in linear time and an enumerator writing finitely many axioms; let $\omega_0>0$ be the least of their weights in $\WA$, $\WAchk{t}$ and $\WAgen{t}$. Then $W(D_n)\ge\omega_0$, $W(\emptyset)\le1$, and the loss is at most $\log(1/\omega_0)$, whatever $P$ is.
\emph{Clocked \SOIabbr.} The last claim concerns the sequence built against one predictor. Put $P^\ast(D):=(W^{\mathrm{all},\tau}(D)+2^{-\lvert D\rvert})/2$ and $q_b(D,\varphi):=P^\ast(D+(\varphi,b))/P^\ast(D)$. Then $q_0+q_1\le1$, since $W^{\mathrm{all},\tau}$ is a semimeasure and the uniform part halves at each step; $P^\ast$ is computable, since clocked compatibility is decidable and the weights have computable tails; $P^\ast(D)\ge2^{-\lvert D\rvert-1}>0$; and $W^{\mathrm{all},\tau}(\emptyset)=1$, so $P^\ast(\emptyset)=1$ and $\ell_{P^\ast}(D_n)=-\log P^\ast(D_n)$. On the sequence built against $P^\ast$, $W^{\mathrm{cons},\tau}(D_n)\le W^{\mathrm{all},\tau}(D_n)\le2P^\ast(D_n)$, so $\ell_{W^{\mathrm{all},\tau}}(D_n)\ge n-1.37$ and $\ell_{W^{\mathrm{cons},\tau}}(D_n)\ge\ell_{P^\ast}(D_n)-1+\log W^{\mathrm{cons},\tau}(\emptyset)\ge n-1.37-c_\bot$.
\end{proof}

\begin{proposition}[Incomparability with time-bounded predictors that may be inconsistent]\status{proved; the strengthening to every time-bounded predictor refuted}\src{sfi Prop 7.5}\label{prop:soi:incomparable}
On the data of \cref{rem:soi:encoding}, $W^{\mathrm{all},\tau}$ has bounded loss for every $\tau$ that covers the constant running time of ``always $1$'', while every consistent inducer (\SAIabbr and consistent \SOIabbr, coin-free or one-tape) loses at least $n$ bits in expectation; on a computable sequence (\cref{thm:soi:diagonal}, applied to $P^\ast$), $W^{\mathrm{all},\tau}$ loses at least $n-1.37$ while \SAIabbr loses $O(1)$. So $W^{\mathrm{all},\tau}$ and \SAIabbr are incomparable.
\end{proposition}

\begin{proof}
Once $\tau$ covers its running time, the clocked string ``always $1$'' is compatible with every prefix of the data of \cref{rem:soi:encoding}, so the loss of $W^{\mathrm{all},\tau}$ is at most $-\log w(\text{always }1)$. The other two claims are \cref{rem:soi:encoding} and \cref{thm:soi:diagonal}, the latter for the sequence built against $P^\ast$.
\end{proof}

\begin{proof}[Proof of \cref{prop:soi:freededuction}]
This is Proposition~E.15(a), (b) of the earlier report. (a) $\mathrm{Out}(\num e,\num n,S0)$ is true iff $e$ outputs $1$ on $n$; if $e$ halts on $n$ with another output, $\neg\mathrm{Out}(\num e,\num n,S0)$ is true, as a program has at most one output. So every label of $u$ is true in $\mathbb N$. (b) In \cref{thm:soi:diagonal} $e$ is total with values in $\{0,1\}$, so $u$ labels every $s_n$ correctly and extends every $D_n$; hence $\Wcons(D_n)\ge2^{-\lvert u\rvert-c}$, where $c$ pays for turning $u$ into a $V$-program for a labeller index, and with $\Wcons(\emptyset)\le1$ the loss is at most $\lvert u\rvert+c$. The same holds for $\Wall\ge\Wcons$.
\end{proof}

\begin{remark}[A time-hierarchy effect]\status{proof sketch}\src{sfi Prop 7.6 (c)}\label{rem:soi:freededuction}
\emph{Proof sketch.} We do not write out the version of the diagonal theorem that gives the compared predictors the same reuse of state, nor the dependence on the machine model. The claim is that under a time bound the sequence that defeats a time-bounded predictor is itself computable within a slightly larger bound. For the predictors $S_p$ of Proposition~E.15(c) of the earlier report, which spend time $p(n)\log n$ per step and reuse their state, the labels that defeat $S_p$ are produced by a predictor with per-step time $O(p(n)\log n)$ and state reuse that predicts its own labels with certainty; so $S_{p'}$, for a slightly larger $p'$ (for example $np(n)$), has bounded loss on the sequence that defeats $S_p$.
\end{remark}

\begin{proposition}[Disguised labellers keep their weight]\status{proved, under (S)}\src{sfi Prop 7.7}\label{prop:soi:disguised}
Let $t(m)\ge m^{a^*}$ be time-constructible and $X$ be $\WAchk{t}$ or $\WAgen{t}$ (any form). For a class $\mathcal C$ of deductively closed theories let $X(\mathcal C;D)$ and $\WAdec(\mathcal C;D)$ be the weights of the hypotheses compatible with $D$ whose theory is in $\mathcal C$. Under \cref{ass:soi:S} there is $c$ with $2^{-c}\WAdec(\mathcal C;D)\le X(\mathcal C;D)\le2^c\WAdec(\mathcal C;D)$ for all $\mathcal C$ and $D$. So the posterior odds of any two classes of positive weight, genuine and disguised alike, move by at most $2^{2c}$, and a slow consistent labeller $f$ keeps prior weight at least $2^{-\lvert f\rvert-c}$, as a padded Craig set.
\end{proposition}

\begin{proof}
This is Proposition~E.16 of the earlier report: the padded sets of Lemma~E.12 there keep the theory and meet the penalty, which gives the lower bound; for the upper bound the penalised deciders are deciders with the same weights, and an enumerator $q$ maps to the padded decider of ``yes iff $q$ writes the input'', which keeps the theory. Divide by the total weights.
\end{proof}

\begin{theorem}[One tape and the companion paper's time-bounded mixtures]\status{proved, given \cref{thm:soi:onetape,thm:soi:domination,thm:soi:diagonal}, Proposition E.1 of the earlier report and the companion paper's Proposition 9.4; (b)--(d) under the hypotheses of \cref{thm:soi:diagonal}; (d) for the diagonal sequences only}\src{sfi Thm 7.8}\label{thm:soi:onetapetime}
Let $\MF^{\mathrm{cons},\tau}:=\sum_fw(f)\nu^{\mathrm{cons}}_{F^\tau_f}$, where $F^\tau_f$ runs $f$ on $U_0$ with a tape-reading instruction for at most $\tau(\lvert\varphi\rvert)$ steps and abstains otherwise, and let $\MF^{\mathrm{all},\tau}$ drop the consistency restriction. Parts (b)--(d) assume that $\mathcal L\supseteq\Lar$ and that $B$ is decidable and true in an expansion of $\mathbb N$.
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item $\ell_{\MA}(D)\le\ell_{\MF^{\mathrm{cons},\tau}}(D)+c_\tau$ for every $D$.
\item On a computable sequence of arithmetic sentences, $\ell_{\MF^{\mathrm{all},\tau}}\ge n-1.37$ and $\ell_{\MF^{\mathrm{cons},\tau}}\ge n-1.37-c_\bot$, while $\ell_{\MA}\le\log(1/\omega_0)$.
\item The length-activated and round-activated mixtures of the companion paper (Definition~9.1 and Proposition~9.4 there), with computable weights whose tails are computable, are computable predictors, so each loses at least $n-0.361$ on a computable sequence on which \SAIabbr loses $O(1)$. Conversely, on the coin literals of \cref{rem:soi:encoding}, all labelled true, the round-activated mixture has bounded loss, while \SAIabbr and consistent \SOIabbr lose at least $n$ bits in expectation. So the companion paper's time-bounded \SOIabbr and \SAIabbr are incomparable.
\item Unpenalised $\Wcons$ and $\MFcons$ have loss at most $\lvert u\rvert+c$ on the diagonal sequences of (b) and (c).
\end{enumerate}
\end{theorem}

\begin{proof}
(a) is the one-tape part of the proof of \cref{cor:soi:onesided}. (b) Each query of $F^\tau_f$ reads at most $\tau(\lvert\varphi\rvert)$ tape bits, so $\nu_{F^\tau_f}(D)$ is a finite union of cylinders, computable exactly; with computable tails, $\MF^{\mathrm{all},\tau}$ is computable, a semimeasure with value $1$ at $\emptyset$, and $P(D):=(\MF^{\mathrm{all},\tau}(D)+2^{-\lvert D\rvert})/2$ is a computable predictor; run \cref{thm:soi:diagonal} against $P$, as in its last step. The theory machine that emits $H_0$ is compatible with every $D_n$ on every tape. (c) \Cref{thm:soi:diagonal} needs only computability of the predictor. The converse is Proposition~9.4 of the companion paper with the constant comparator ``label true with probability $1$'', which is in its class $\mathcal C_t$, and Proposition~E.1 of the earlier report, applied tape by tape (\cref{rem:soi:encoding}). (d) \Cref{prop:soi:freededuction}, with \eqref{eq:soi:mixF} for $\MFcons$. On the coin literals of (c), by contrast, every consistent inducer, $\Wcons$ and $\MFcons$ included, loses at least $n$ bits in expectation.
\end{proof}

\subsection{Charged deduction}\label{sec:app:soi:charged}

We use the setting of Definition~E.18 of the earlier report: $\mathcal L$ is finite, $B$ is decidable in polynomial time, derivations are in the Hilbert calculus $\mathsf K$ of that report, with sizes counting symbols, and a hypothesis is a program $p$ that decides an axiom set $A_p$ in time $O(t)$. Each datum $\varphi$ must have a derivation of $\varphi^b$ from $B\cup A_p$ whose size is at most $s(\lvert\varphi\rvert)$ ($W_A[t,s]$); in which every axiom instance, logical ones included, has at most $h(\lvert\varphi\rvert)$ symbols (reading (a1), $W_A^{a1}[t,h]$), or every nonlogical instance does (reading (a2), $W_A^{a2}[t,h]$); or whose \emph{material}, the total size of the distinct instances used, is at most $g(\lvert\varphi\rvert)$ (reading (b), $W_A^{b}[t,g]$), or whose nonlogical material is (reading (b2), $W_A^{b2}[t,g]$). \emph{Certificate} \SOIabbr, $\Wcert[t',s']$, weighs the consistent labellers $f_{\mathcal V}$ defined by verifiers $\mathcal V$ running in time $O(t')$: $f_{\mathcal V}(\varphi)$ is $1$ (resp.\ $0$) if some certificate of length at most $s'(\lvert\varphi\rvert)$ makes $\mathcal V$ accept (resp.\ reject). All charged results assume the renaming condition of that report; without it the index of a single parameter, which costs one symbol, could carry a certificate of any length (\cref{rem:soi:F1}).

\begin{assumption}[(F1): renaming closure]\status{assumption}\src{sfi \S8 (the earlier report's (F1))}\label{ass:soi:F1}
For every hypothesis $p$ of a charged reading, $B\cup A_p$ is closed under permutations of the parameters.
\end{assumption}

\begin{theorem}[Charged deduction is certificate \SOIabbr]\status{proved, under (F1); the correspondence of proof systems with NP known}\src{sfi Thm 8.1}\label{thm:soi:sandwich}
Under \cref{ass:soi:F1}, with time-constructible nondecreasing budgets:
(a) in a normal derivation of $\varphi$ every line sits at its own node of an axiom instance used or of $\varphi$, up to abstracting parameters; so its size is at most $(M+\lvert\varphi\rvert)(M+\lvert\varphi\rvert+1)/2$, with $M$ its material, and the factor $\frac12$ is attained;
(b) with $G(m):=(g+m+1)(g+m+2)/2$, $t^+(m):=m^{k_0}(t(m)+1)$ and $t'^\#(m):=m^{k_0}+t'(m)$, for constants $k_0$, $\beta$ and $c$: $W_A[t,g]\le W_A^b[t,g]\le W_A[t,G]$ hypothesis by hypothesis; $W_A^b[t,g]\le2^c\,\Wcert[t^+,\beta G\log(G+2)]$; and $\Wcert[t',s']\le2^c$ times each of $W_A^b[t'^\#,2s'+m+31]$, $W_A^{a1}[t'^\#,s'+m+16]$ and $W_A[t'^\#,2s'+2m+31]$.
\end{theorem}

Its classical core is the correspondence of propositional proof systems with $\cplx{NP}$ \citep{cook1979relative}: a derivation is a certificate; conversely, a certificate $\zeta$ becomes the axiom $\theta_\zeta\to\varphi^b$, where $\theta_\zeta$ is a tautology of size $\lvert\zeta\rvert+14$ that encodes $\zeta$.

\begin{proof}
This is Theorem~6.16 of the earlier report, with Lemma~E.19 and Proposition~E.21 there, whose budgets are time-constructible and nondecreasing. Part (a) is the subformula lemma (Lemma~E.19(b), (d) there): in a normal derivation every line sits at its own formula node of an instance used or of the conclusion, up to abstracting the parameters of a chain of Gen steps, and a family with $k$ vacuous Gen steps attains the factor $\frac12$. Part (b): a derivation of size at most $g$ has material at most $g$, and by (a) material at most $g$ gives size at most $G$; this is the first chain, hypothesis by hypothesis. Under \cref{ass:soi:F1} the parameters of a derivation of size at most $G$ can be renamed to $p_0,\dots,p_{G-1}$, so the derivation is a certificate of at most $\beta G\log(G+2)$ bits that a verifier checks in time $t^+$ (membership in $A_p$ in time $t$ per line); this is the second inequality. For the third, let $Z$ be a closed tautology of size $4$ and $C_\zeta$ a sentence of size $\lvert\zeta\rvert+4$ that encodes the certificate $\zeta$, and put $\theta_\zeta:=Z\to(C_\zeta\to Z)$, a tautology of size $\lvert\zeta\rvert+14$. The axiom set $\{\theta_\zeta\to\varphi^b:\zeta\text{ a certificate on which }\mathcal V\text{ accepts }(b=1)\text{ or rejects }(b=0)\ \varphi\}$ is decidable in time $t'^\#$, satisfies \cref{ass:soi:F1}, and has the same consequences as $\Gamma_{f_{\mathcal V}}$. The derivation $\theta_\zeta$, $\theta_\zeta\to\varphi^b$, $\varphi^b$ has material $2\lvert\zeta\rvert+29+\lvert\varphi^b\rvert$, largest instance $\lvert\zeta\rvert+15+\lvert\varphi^b\rvert$ and size $2\lvert\zeta\rvert+29+2\lvert\varphi^b\rvert$, with $\lvert\varphi^b\rvert\le\lvert\varphi\rvert+1$; these give the three stated budgets. All maps are injective and add a constant to the length.
\end{proof}

\begin{corollary}[Fit classes]\status{proved, under (F1); conditional parts as stated; hierarchy theorems and APSPACE = EXPTIME known}\src{sfi Cor 8.2}\label{cor:soi:fit}
Under \cref{ass:soi:F1}, on literal sequences $D^X$, at polynomial budgets whose degrees are charged in the prior, with standard simulation overheads:
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item reading (b) fits exactly the $X\in\cplx{NP}\cap\cplx{coNP}$; it beats polynomial-time consistent \SOIabbr, $W^{\mathrm{cons},\mathrm{poly}}$, on some sequence if $\cplx{NP}\cap\cplx{coNP}\ne\cplx P$, and fits the same sequences if $\cplx P=\cplx{NP}$; with degrees uncharged, $W^{\mathrm{cons},\mathrm{poly}}$ fails to dominate it, unconditionally; and, under \cref{ass:soi:S}, it beats $W^{\mathrm{cons},\tau}$ at a budget a hierarchy gap above $\tau$;
\item reading (a1) fits exactly the $X\in\cplx{EXPTIME}$; at any computable budgets some decidable $X$ defeats it; and every predictor computable in time polynomial in the number of data loses at least $n-0.361$ on a sequence on which (a1) at a linear budget loses $O(1)$;
\item readings (a2) and (b2) fit every decidable $X$, by a finite set of universally closed Horn sentences.
\end{enumerate}
\end{corollary}

\begin{proof}
This is Corollary~6.17(a)--(c) of the earlier report, with its proofs there. (ii) uses $\cplx{APSPACE}=\cplx{EXPTIME}$ \citep{chandra1981alternation} through Proposition~E.23 there, and the comparisons use the time hierarchy theorems (standard; cf.\ \citealp{hartmanis1965computational}, and \citealp{seiferas1978separating} for nondeterministic time).
\end{proof}

\begin{proposition}[Soft charges]\status{proved}\src{sfi Prop 8.3}\label{prop:soi:soft}
Let a hypothesis pay a factor $2^{-\kappa\ell}$ per datum, $\ell$ the bit length of a shortest derivation of the datum (soft \SAIabbr), and let a verifier pay likewise for its shortest certificate (soft certificate \SOIabbr). The two correspond only up to a constant factor in the rate $\kappa$; no pair of rates gives two-sided constant regret; and every soft loss diverges on every infinite sequence, so there are no soft fit classes.
\end{proposition}

\begin{proof}
Proposition~E.26 of the earlier report, which gives the explicit rates.
\end{proof}

\begin{remark}[The role of the renaming condition]\status{proved (in the earlier report)}\src{sfi Rem 8.4}\label{rem:soi:F1}
Without \cref{ass:soi:F1} the index of one parameter, which costs one symbol, can carry a certificate of any length (Proposition~E.27 of the earlier report). The coarse statements (Theorem~6.9 there, and the inclusion of the fit class of reading (b) in $\cplx{NP}\cap\cplx{coNP}$) survive in the reading that counts symbols, by truncating parameter indices (Lemma~E.28 there); the fine bounds ($2^{O(h\log h)}$ for (a1), certificate lengths) are not claimed without it.
\end{remark}

\begin{remark}[One tape]\status{proof sketch}\src{sfi Rem 8.6}\label{rem:soi:onetapesandwich}
\emph{Proof sketch.} We do not write out the bookkeeping of a verifier that reads a tape, which needs \cref{ass:soi:Stape}-style accounting. The maps of \cref{thm:soi:sandwich}(b) act tape by tape: a random decider with time $O(t)$ on every tape gives a verifier that reads the tape to answer membership queries, and certificate axioms $\theta_\zeta\to\varphi^b$ can be emitted from a random verifier. So a one-tape form of the sandwich should hold with the same budgets.
\end{remark}

\subsection{Presburger arithmetic}\label{sec:app:soi:presburger}

\begin{remark}[Lengths]\status{proved, given a proviso on Fischer and Rabin's measure of length}\src{sfi Rem 9.0}\label{rem:soi:lengths}
(i) The length $\lvert\varphi\rvert$ of a sentence is its symbol count: every occurrence of a connective, quantifier, nonlogical symbol, parenthesis or variable counts $1$, whatever the variable's index. This is the size measure of the calculus $\mathsf K$ of the earlier report, in which a variable or parameter counts one symbol. Machines read $\varphi$ as a bit string of length $b(\varphi)\le\kappa_0\lvert\varphi\rvert\log(\lvert\varphi\rvert+2)$, since a sentence with $m$ symbols has at most $m$ distinct variables, which can be renamed $x_1,\dots,x_m$ and written in binary. Clocks $\tau$, derivation budgets $s,g,h$ and the length $m$ in the lower bounds are functions of $\lvert\varphi\rvert$.
(ii) If a clock were a function of the bit length $b$, which can exceed the symbol count by a factor $\log m$, a clock $2^{2^{c'b}}$ could exceed every $2^{2^{cm}}$, and the contradictions below would fail. For clocks in bit length the results hold with $\tau(b)\le2^{2^{c'b/\log(b+2)}}$ in place of $\tau(m)\le2^{2^{c'm}}$, with $c'<c/\kappa_0$ (and likewise for the singly exponential clocks), since $b/\log(b+2)\le\kappa_0\lvert\varphi\rvert$.
(iii) \emph{Proviso:} Fischer and Rabin's length counts every symbol occurrence, variables included, at least once, and their nonlogical symbols are among $0,1,+,=,<$. Under the proviso every sentence of their length $n$ has an equivalent $\mathcal L_{\rm Pr}$-sentence with at most $\kappa n$ symbols, computable in polynomial time (rewrite $x<y$ as $\exists z\,(x+z+1=y)$ with $z$ fresh, and copy every other symbol). \emph{Padding:} every $\mathcal L_{\rm Pr}$-sentence $\psi$ has an equivalent sentence with exactly $m$ symbols for every $m\ge\lvert\psi\rvert+c_0$, computable in polynomial time, because there are two equivalence-preserving paddings with coprime increments (in the size measure of $\mathsf K$, $\psi\wedge0=0$ adds $4$ and $\psi\wedge\neg\,0=1$ adds $5$; every standard syntax has such a pair). \emph{Consequence:} let $Q$ be a decision procedure for $\mathrm{PrA}$ in our syntax, deterministic or nondeterministic. For $j=0,\dots,\kappa-1$ let $P_j$ decide their sentences of length $n$ by translating, padding to exactly $\kappa n+c_0+j$ symbols and running $Q$. Each $P_j$ is a decision procedure in their syntax, with the same reading of nondeterministic time as $Q$; so by their theorem, for all large $n$, $Q$ takes more than $2^{2^{c_{\rm FR}n}}-\mathrm{poly}(n)$ steps on some sentence with exactly $\kappa n+c_0+j$ symbols. As $j$ runs through all residues, for all large $m$, $Q$ takes more than $2^{2^{cm}}$ steps on some sentence with exactly $m$ symbols, for every $c<c_{\rm FR}/\kappa$; $c$ denotes such a constant. The exact-length form is what the ``for all large $m$'' arguments below use.
\end{remark}

We use two known results. ($\mathrm F_{\rm P}$) Presburger's axioms for successor and addition, with the induction schema for $\mathcal L_{\rm Pr}$-formulas, form a decidable, consistent and complete axiomatisation of $\mathrm{PrA}$ \citep{presburger1929vollstandigkeit}, with membership decidable in polynomial time. ($\mathrm F_{\rm FR}$) The bound of \citet{fischer1974super}, in the form of \cref{rem:soi:lengths}(iii). Its statement does not say how the time of a nondeterministic procedure is measured. We distinguish two readings: (max) the time is the maximum over the computation paths, all of which halt, every output being correct and some path giving an output; (short) the time on a sentence is the length of the shortest computation that outputs the correct answer. (short) implies (max). Where only (max) is needed nothing is assumed; the one place that uses (short) says so.

\begin{proposition}[Presburger arithmetic separates \SAIabbr from clocked \SOIabbr]\status{proved, given Presburger's theorem, the Fischer--Rabin bound and, for (c), the Lebesgue density theorem}\src{sfi Props 9.1, 9.2}\label{prop:soi:presburger}
(a) $\WA$, $\WAdec$ and $\MA$, and the penalised $\WAchk{t}$, $\WAgen{t}$ and $\MAchk{t}$ for a large enough polynomial $t$, have bounded loss on $D^{\rm Pres}$.
(b) If $\tau$ is time-constructible and $\tau(m)\le2^{2^{c'm}}$ for large $m$, with $c'<c$, the losses of $W^{\mathrm{cons},\tau}$ and $W^{\mathrm{all},\tau}$ on $D^{\rm Pres}_n$ tend to infinity.
(c) If $\tau(m)\le2^{c'm}$ for large $m$, with $c'<c$, the losses of $\MF^{\mathrm{cons},\tau}$ and $\MF^{\mathrm{all},\tau}$ on $D^{\rm Pres}_n$ tend to infinity.
\end{proposition}

\begin{proof}
(a) The decider of Presburger's axioms is compatible with every prefix of $D^{\rm Pres}$ (the axiomatisation is complete and consistent), runs in polynomial time and is a deterministic theory machine; with an enumerator of the same axioms it meets the axiom-side penalties for a large enough polynomial $t$. The losses are at most $-\log$ of its weight.
(b) In a clocked mixture compatibility is inherited by prefixes, so the compatible strings form a decreasing family and bounded loss gives one $\tau$-clocked $f$ that labels every sentence correctly (Lemma~E.11 of the earlier report). By the standard simulation overhead it yields a multitape decision procedure for $\mathrm{PrA}$ that runs in time $q(C_f\tau(m)+b)$, with $q$ a fixed polynomial and $b\le\kappa_0m\log(m+2)$ the input's bit length. When $c'<c$ this is at most $2^{2^{cm}}$ for large $m$, which contradicts $\mathrm F_{\rm FR}$ in the deterministic case.
(c) Write $M:=\sum_fw(f)\nu_f$ for either one-tape mixture, each $\nu_f(D^{\rm Pres}_n)$ being non-increasing in $n$. Bounded loss gives $\inf_nM(D^{\rm Pres}_n)>0$; by dominated convergence some $f$ has $\lim_n\nu_f(D^{\rm Pres}_n)>0$, which by continuity from above is $\Unif(\mathcal G_f)$ for the set $\mathcal G_f$ of tapes on which $F^\tau_f$ labels every sentence correctly (and consistently, for the consistent version). By the Lebesgue density theorem (\cref{lem:logic:density}) there is $w_0$ with $\Unif(\mathcal G_f\cap[w_0])>\frac34\,2^{-\lvert w_0\rvert}$, so for every sentence the tapes in $[w_0]$ on which $F^\tau_f$ answers correctly have conditional measure greater than $\frac34$. A query of length $m$ reads at most $\tau(m)$ tape bits, so the procedure ``for every $v$ of length $\max(0,\tau(m)-\lvert w_0\rvert)$ run $F^\tau_f$ with tape $w_0v$ and output the majority answer'' ($w_0$ hard-coded) decides $\mathrm{PrA}$ deterministically within $2^{\tau(m)}q(C_f\tau(m)+b)\le2^{2^{cm}}$ steps for large $m$: a contradiction. The majority vote over the tape completions at a density point is the technique of \citet{deleeuw1956computability}; the argument applies to every mixture of machines whose tape use per query is bounded by $\tau$.
\end{proof}

A computation with illustrative values of the constants (code in the accompanying repository; $c$ is not known explicitly) gives the lengths from which these contradictions apply: for $c=1$ and $c'=0.9$, from about $11$--$26$ symbols in (b) and $6$--$9$ in (c). The separations are asymptotic: below these lengths a clocked hypothesis may label every short sentence correctly.

\begin{assumption}[(B$^*$): a $\Sigma_2$-time lower bound for Presburger arithmetic]\status{assumption}\src{sfi Prop 9.3, assumption (B*)}\label{ass:soi:Bstar}
There is $c_B>0$ such that, for all large $m$, no alternating machine with at most two alternations (a $\Sigma_2$ machine) decides $\mathrm{PrA}$ within $2^{2^{c_Bm}}$ steps on all sentences with $m$ symbols.
\end{assumption}

\begin{assumption}[(SGL): time-bounded $\cplx{BPP}\subseteq\Sigma_2$]\status{assumption (standard; time bound derived)}\src{sfi Prop 9.3, assumption (SGL)}\label{ass:soi:SGL}
For time-constructible $T(m)\ge m$ and large $T$, $\cplx{BPTIME}(T)\subseteq\Sigma_2\text{-}\cplx{TIME}(T^3)$.
\end{assumption}

We do not prove either assumption. $\mathrm{PrA}$ is complete for alternating doubly exponential time with linearly many alternations (cf.\ \citealp{berman1980complexity}); (B$^*$) would follow from the hardness part if the reductions were of linear size, and we do not assume that they are. If they are only of polynomial size, a bound $2^{2^{c_Bm^\epsilon}}$ follows for some $\epsilon\in(0,1]$, and the clock condition below must be weakened to $\tau(m)\le2^{2^{c'm^\epsilon}}$. (SGL) is the time-bounded form of $\cplx{BPP}\subseteq\Sigma_2^p$ \citep{sipser1983complexity,lautemann1983bpp}: amplifying by $O(\log T)$ repetitions to $R=O(T\log T)$ random bits and error below $1/(2R)$, a $\Sigma_2$ machine guesses $R$ shifts of $R$ bits, universally chooses $R$ bits, and runs the amplified machine $R$ times, in time $O(T^2\log^2T)$.

\begin{proposition}[One tape, doubly exponential clocks]\status{conditional: proved, given (B*) and (SGL)}\src{sfi Prop 9.3}\label{prop:soi:presburgerCond}
Under \cref{ass:soi:Bstar,ass:soi:SGL}, if $\tau(m)\le2^{2^{c'm}}$ for large $m$ with $c'<c_B$, then the losses of $\MF^{\mathrm{cons},\tau}$ and $\MF^{\mathrm{all},\tau}$ on $D^{\rm Pres}_n$ tend to infinity.
\end{proposition}

\begin{proof}
As in the proof of \cref{prop:soi:presburger}(c), obtain $w_0$ such that on $[w_0]$ the conditional probability of a correct answer exceeds $\frac34$ for every sentence. The algorithm ``draw $v$ of length $\tau(m)$ at random, run $F^\tau_f$ with tape $w_0v$, and output its answer'' errs with probability below $\frac14$ on every sentence and runs in time $T(m):=q(C_f\tau(m)+b)$, so $\mathrm{PrA}\in\cplx{BPTIME}(T)\subseteq\Sigma_2\text{-}\cplx{TIME}(T^3)$ by \cref{ass:soi:SGL}. Since $T(m)^3\le2^{2^{c_Bm}}$ for large $m$ when $c'<c_B$, this contradicts \cref{ass:soi:Bstar}.
\end{proof}

A nondeterministic lower bound alone does not finish the doubly exponential case by this route: separating $\cplx{NTIME}(2^{2^n})$ from $\Sigma_2$-time below it is not known in general.

\begin{proposition}[Charged deduction on Presburger data]\status{proved, given the Fischer--Rabin bound, Presburger's theorem, Lemma E.28, Proposition E.23(a) and the proof of Theorem 6.9 of the earlier report}\src{sfi Props 9.4, 9.4$'$}\label{prop:soi:presburgerCharged}
Let $p$ be a hypothesis of a charged reading over $\mathcal L_{\rm Pr}$ with $B=\emptyset$, deciding $A_p$ in polynomial time and satisfying \cref{ass:soi:F1}, and compatible with every prefix of $D^{\rm Pres}$. Let $c''<c$.
(a) Size reading, budget $s$: $s(m)>2^{2^{c''m}}$ for infinitely many $m$, and for all large $m$ if $s$ is time-constructible.
(b) Material reading (b), budget $g$: the same with $g$ in place of $s$.
(c) Reading (a1), with $h$ nondecreasing and time-constructible: $h(m)>2^{c''m}$ for all large $m$.
\end{proposition}

A mixture of such hypotheses has bounded loss on $D^{\rm Pres}$ only if one of them is compatible with every prefix.

\begin{proof}
Since compatibility is inherited by prefixes, it suffices to treat one hypothesis $p$, deciding $A_p$ in time $O(m^{k_p})$ and compatible with every prefix of $D^{\rm Pres}$. Then $B\cup A_p=A_p$ is consistent and proves every true sentence, so it proves exactly $\mathrm{PrA}$.
(a) \emph{Infinitely often.} Fix $c''<c$ and suppose $s(m)\le\tau''(m):=2^{2^{c''m}}$ for all $m\ge m_1$; $\tau''$ is time-constructible. The nondeterministic procedure $N$: on a sentence $\varphi$ with $m<m_1$ symbols, look $\varphi$ up in a table; otherwise compute $\tau''(m)$, guess a derivation of $\varphi$ or of $\neg\varphi$ of size at most $\tau''(m)$ whose parameter numbers have at most $K+1+\lceil\log(\tau''(m)+1)\rceil$ bits, $K=O(\tau''(m)^{k_p})$ (if a derivation of size at most $\tau''(m)$ exists, such a one exists, by Lemma~E.28 of the earlier report), check it as in the proof of Theorem~6.9 there (each line a logical axiom, by matching, a member of $A_p$, in time $O(\tau''^{k_p})$, or a step of modus ponens or generalisation), and output the label it proves; a path whose guess is not a valid derivation outputs nothing. Every output is correct, every sentence has a path with an output, and every path halts within $\tau''(m)^{c_k}$ steps, $c_k$ depending only on $k_p$ and the checking overhead of $\mathsf K$. Since $\tau''(m)^{c_k}=2^{c_k2^{c''m}}\le2^{2^{cm}}$ once $m\ge\log(c_k)/(c-c'')$, $N$ contradicts $\mathrm F_{\rm FR}$ (nondeterministic case, reading (max)). So $s(m)>2^{2^{c''m}}$ for infinitely many $m$.
\emph{For all large $m$, $s$ time-constructible.} Suppose $s(m)\le\tau''(m)$ for infinitely many $m$, the \emph{good} lengths. The procedure $N''$: on a sentence with $m$ symbols, compute $s(m)$ with a clock of $\tau''(m)$ steps; if the computation finishes with $s(m)\le\tau''(m)$, run $N$'s guess-and-check with the budget $s(m)$; otherwise run a deterministic decision procedure for $\mathrm{PrA}$, which exists by $\mathrm F_{\rm P}$. Every output is correct and every path halts, and at each good length every path halts within $O(\tau''(m))+\tau''(m)^{c_k}\le2^{2^{cm}}$ steps beyond the thresholds. By $\mathrm F_{\rm FR}$ (reading (max)), for all large $m$ some sentence with $m$ symbols takes $N''$ more than $2^{2^{cm}}$ steps; a good length beyond the thresholds gives a contradiction. (For an arbitrary $s$ the same conclusion follows if $\mathrm F_{\rm FR}$ holds in reading (short): let a path whose guess is not a valid derivation run the deterministic procedure; at each good length every sentence then has a correct output within $\tau''(m)^{c_k}+O(1)$ steps.)
(b) By the subformula lemma (\cref{thm:soi:sandwich}(a)), material at most $g$ gives a derivation of size at most $(g+m)(g+m+1)/2$. If $g(m)\le2^{2^{c''m}}$ for all large $m$, this is at most $2^{2^{c'''m}}$ for large $m$, for any $c'''\in(c'',c)$, against (a) with $c'''$. The form for all large $m$ transfers in the same way.
(c) Proposition~E.23(a) of the earlier report gives the labeller $g_p$: with $H(\varphi):=\max(h(\lvert\varphi\rvert),\lvert\varphi\rvert+1)$ and $\mathrm{Der}_H$ the formulas with a derivation from $B\cup A_p$ all of whose lines have size at most $H$, it accepts $\varphi$ if $\varphi\in\mathrm{Der}_{H(\varphi)}$, rejects it if $\neg\varphi\in\mathrm{Der}_{H(\varphi)}$ and $\varphi\notin\mathrm{Der}_{H(\varphi)}$, and abstains otherwise. It runs in time $2^{O(H\log H)}(1+t(H))^{O(1)}$, by closing the at most $2^{O(H\log H)}$ canonical formulas of size at most $H$ under the rules, using \cref{ass:soi:F1}; and it labels correctly every datum of every $D$ with which $p$ is (a1)-compatible, since a normal (a1)-derivation of $\varphi^b$ has every line of size at most $\max(h(\lvert\varphi\rvert),\lvert\varphi^b\rvert)\le H(\varphi)$ (the subformula lemma) and $B\cup A_p$ is consistent. Nothing in this argument uses the form of $\varphi$. On $D^{\rm Pres}$, $g_p$ therefore decides $\mathrm{PrA}$ deterministically (it never abstains, since every sentence is a datum) in time $T(m)\le2^{C_aH\log H}$ with $H=\max(h(m),m+1)$ and $C_a$ depending on $k_p$ (computing $h(m)$ costs $O(h(m))$). By $\mathrm F_{\rm FR}$, $T(m)>2^{2^{cm}}$ for all large $m$, so $C_aH\log H>2^{cm}$, hence $H>2^{cm}/(C_acm)>2^{c''m}$ for all large $m$; as $m+1<2^{c''m}$ for large $m$, $H=h(m)$.
\end{proof}

\begin{question}\status{open}\src{sfi Prop 9.4$'$ and the remark after it; \S9 open rows}\label{q:soi:presburger}
(i) Under reading (a1), does every hypothesis compatible with every prefix of $D^{\rm Pres}$ need doubly exponential instance budgets $h$? (ii) Does some hypothesis of reading (a1) in $\mathcal L_{\rm Pr}$ fit $D^{\rm Pres}$ at exponential budgets? (iii) What do readings (a2) and (b2) need on $D^{\rm Pres}$, in either direction?
\end{question}

(i) would follow from a deterministic lower bound $\mathrm{PrA}\notin\cplx{DTIME}(2^{2^{2^{c_3m}}})$ for some $c_3>0$, by the argument of \cref{prop:soi:presburgerCharged}(c), which would then give $h(m)>2^{2^{c'_3m}}$ infinitely often for every $c'_3<c_3$; no such bound is known to us. For (ii), Proposition~E.23(b) of the earlier report simulates alternating space-bounded machines, but it needs the literal language with two extra binary relation symbols, which $\mathcal L_{\rm Pr}$ with $B=\emptyset$ lacks. For (iii), logical instances are free under (a2) and (b2), so neither the guess-and-check route nor the enumeration route gives a bound, and \cref{cor:soi:fit}(iii) concerns literal data in the literal language, not $D^{\rm Pres}$.

\paragraph{Each charged reading matches one version of \SOIabbr.}\label{rem:soi:presburgerSummary}\status{assessment (summary of proved results; the (a1) correspondence is a reading for sentence data)}\src{sfi Rem 9.5} Once deduction is charged, the budget \SAIabbr needs on $D^{\rm Pres}$ is that of the matching version of \SOIabbr: under the size and material readings \SAIabbr is certificate \SOIabbr (\cref{thm:soi:sandwich}), and the nondeterministic Fischer--Rabin bound forces doubly exponential budgets on both; under (a1) \SAIabbr behaves like deterministic \SOIabbr with an exponential clock (Proposition~E.23 and Corollary~6.17(b) of the earlier report; the converse direction is proved there for literal data only, so for sentence data we offer this as a reading), and the deterministic bound forces exponential instance sizes, which the clock turns into doubly exponential time. So on a natural theory, as on the diagonal sequences, the separation from clocked \SOIabbr is the free deduction, for every reading in which the derivation is charged and the conclusion is proved; for (a2) and (b2) the question is open.

\subsection{The circuit form of the sandwich}\label{sec:app:soi:circuit}

The sandwich of \cref{thm:soi:sandwich} has a routine circuit form, which connects finite axiom sets with the circuit Bayes theorem of the companion paper. Fix $d\ge1$; the data are the literals $\varphi_x:=R(t_x)$, $x\in\{0,1\}^d$, labelled by a partial $f:\{0,1\}^d\rightharpoonup\{0,1\}$.

A \emph{verifier circuit} $\mathcal V$ of size $\lvert\mathcal V\rvert$ is a list of $\lvert\mathcal V\rvert$ two-input gates (any of the $16$ Boolean functions) over the inputs $x_1,\dots,x_d$, certificate bits $z_1,\dots,z_k$ (without loss of generality $k\le2\lvert\mathcal V\rvert$), the constants and earlier gates; its last two gates are the outputs $\mathsf a$ (accept) and $\mathsf r$ (reject). It is \emph{consistent} if no $x$ has certificates $\zeta,\zeta'$ with $\mathsf a(x,\zeta)=1$ and $\mathsf r(x,\zeta')=1$, and it defines $f_{\mathcal V}(x):=1$ if some $\zeta$ has $\mathsf a(x,\zeta)=1$, $0$ if some $\zeta$ has $\mathsf r(x,\zeta)=1$, and undefined otherwise. $\mathrm{vs}(f)$ is the least size of a consistent verifier circuit with $f_{\mathcal V}\supseteq f$; it is at most the circuit size of $f$ in the sense of Definition~9.7 of the companion paper plus a constant. The \emph{fixed language} is $\mathcal L_{\rm circ}:=\{\mathtt e,\mathtt s_0,\mathtt s_1,\mathrm{Dig},\mathrm{Gate},R\}$, with $\mathrm{Dig}$ binary and $\mathrm{Gate}$ ternary. Index terms $\mathbf n_i$ and gate terms $\mathbf g_j$ are the word terms $t_{\beta(i)}$, $t_{\beta(j)}$, with $\beta(i)$ the binary expansion of $i+1$ without its leading $1$; so $\lvert\mathbf n_i\rvert=O(\log i)$, the terms of each family are pairwise distinct, and the language does not grow with the circuit. $\mathrm{ax}_s(f)$ is the least total size of a finite set $T$ of sentences of $\mathcal L_{\rm circ}$ (or of any fixed finite language containing $\mathtt e,\mathtt s_0,\mathtt s_1,R$) with $T\cup\Gamma_f$ consistent such that every datum $\varphi_x^{f(x)}$, $x\in\dom f$, has a $\mathsf K$-derivation from $T$ of size at most $s$.

\emph{The translation $T_{\mathcal V}$} (all sentences universally closed; $\leftrightarrow$ abbreviates $\neg((A\to B)\to\neg(B\to A))$). Digits: $\forall y\,\mathrm{Dig}(\mathbf n_0,\mathtt s_1y)$, $\forall y\,\neg\mathrm{Dig}(\mathbf n_0,\mathtt s_0y)$, and $\forall y\,(\mathrm{Dig}(\mathbf n_{i+1},\mathtt s_ay)\leftrightarrow\mathrm{Dig}(\mathbf n_i,y))$ for $i<\max(d,k)-1$ and $a\in\{0,1\}$; so $\mathrm{Dig}(\mathbf n_i,t_w)$ holds iff the $(i+1)$-st symbol of $w$ is $1$. Gates: a gate $j$ that reads $x_i$ (resp.\ $z_i$) has the axiom $\forall x\,\forall z\,(\mathrm{Gate}(\mathbf g_j,x,z)\leftrightarrow\mathrm{Dig}(\mathbf n_{i-1},x))$ (resp.\ with $z$ in place of $x$), and an internal gate $j$ with operation $\mathrm{op}$ on gates $a,b$ has $\forall x\,\forall z\,(\mathrm{Gate}(\mathbf g_j,x,z)\leftrightarrow\mathrm{op}(\mathrm{Gate}(\mathbf g_a,x,z),\mathrm{Gate}(\mathbf g_b,x,z)))$. Outputs: $\forall x\,\forall z\,(\mathrm{Gate}(\mathbf g_{\mathsf a},x,z)\to R(x))$ and $\forall x\,\forall z\,(\mathrm{Gate}(\mathbf g_{\mathsf r},x,z)\to\neg R(x))$.

\begin{theorem}[The circuit sandwich]\status{(i) proved; computed; (ii) proved, given the circuit simulation of Turing machines}\src{sfi Thm 10.1}\label{thm:soi:circuit}
(i) For every consistent verifier circuit $\mathcal V$ with $k$ certificate bits, $T_{\mathcal V}$ has total size at most $C_1(\lvert\mathcal V\rvert+d+k)\log(\lvert\mathcal V\rvert+d+k+2)$, $T_{\mathcal V}\cup\Gamma_{f_{\mathcal V}}$ is consistent, and every $x\in\dom f_{\mathcal V}$ has a derivation of $\varphi_x^{f_{\mathcal V}(x)}$ from $T_{\mathcal V}$ of size at most $s_{\mathcal V}:=C_2\big(\lvert\mathcal V\rvert(\log(\lvert\mathcal V\rvert+2)+d+k)+(d+k)^3\big)$, with absolute constants $C_1,C_2$. Hence, using $k\le2\lvert\mathcal V\rvert$, $\mathrm{ax}_s(f)\le C'_1(\mathrm{vs}(f)+d)\log(\mathrm{vs}(f)+d+2)$ for $s=C'_2\big(\mathrm{vs}(f)(\log\mathrm{vs}(f)+d)+(d+\mathrm{vs}(f))^3\big)$.
(ii) Conversely, $\mathrm{vs}(f)\le P(\mathrm{ax}_s(f),s,d)$ for a fixed polynomial $P$ that depends only on the language and $\mathsf K$.
\end{theorem}

This is the routine circuit form of the sandwich, not a separate result of independent interest; we include it because it carries the circuit Bayes theorem of the companion paper over to verifier circuits (\cref{cor:soi:circuitBayes}).

\begin{proof}
(i) \emph{Size.} There are $O(d+k)$ digit axioms of size $O(\log(d+k))$, one axiom of size $O(\log(\lvert\mathcal V\rvert+d+k))$ per gate (two or three gate terms and at most two index terms), and two output axioms.
\emph{Consistency.} Take the free term algebra on $\mathtt e,\mathtt s_0,\mathtt s_1$, with equality the identity; every element is a word term. Read off the bits of an element $y$ (the symbols of the word it spells, $0$ beyond its length). Let $\mathrm{Dig}(t,y)$ hold iff $t=\mathbf n_i$ for some $i<\max(d,k)$ and bit $i$ of $y$ is $1$; let $\mathrm{Gate}(t,x,z)$ hold iff $t=\mathbf g_j$ and gate $j$ of $\mathcal V$ outputs $1$ on the first $d$ bits of $x$ and the first $k$ bits of $z$; and let $R(x)$ hold iff some $\zeta$ has $\mathsf a(\mathrm{bits}_d(x),\zeta)=1$. The digit and gate axioms hold by construction. The first output axiom holds by the definition of $R$; the second holds because $\mathcal V$ is consistent: $\mathsf r(\mathrm{bits}_d(x),z)=1$ excludes $\mathsf a(\mathrm{bits}_d(x),\zeta)=1$ for every $\zeta$. On word terms of length $d$, $R(t_x)$ holds iff $f_{\mathcal V}(x)=1$, and $f_{\mathcal V}(x)=0$ forces $\neg R(t_x)$; so the structure satisfies $T_{\mathcal V}\cup\Gamma_{f_{\mathcal V}}$.
\emph{Derivations.} For $x\in\dom f_{\mathcal V}$ take a certificate $\zeta$ for its label. Instantiate every gate axiom with $x:=t_x$, $z:=t_\zeta$ (two instances of the quantifier axiom and two modus ponens steps per axiom, each line of size $O(\log\lvert\mathcal V\rvert+d+k)$). Derive each needed digit, $\mathrm{Dig}(\mathbf n_i,t_w)$ or its negation, from an instance of the base axiom and $i$ instances of the successor axioms (each of size $O(d+k)$), substituting at each step a fixed propositional derivation for biconditional elimination: $O((d+k)^2)$ per digit, $O((d+k)^3)$ in all. Then, gate by gate in order, derive $\mathrm{Gate}(\mathbf g_j,t_x,t_\zeta)$ or its negation from the ground biconditional and the literals of its inputs, by a substitution instance of a fixed propositional derivation (the propositional fragment of $\mathsf K$ is complete, and substitution instances of derivations are derivations of size linear in the substituted formulas, as in the proof of Theorem~6.16 of the earlier report). Finish with the output instance and modus ponens. The total is $O(\lvert\mathcal V\rvert(\log\lvert\mathcal V\rvert+d+k)+(d+k)^3)$.
(ii) A certificate is the code of a derivation of size at most $s$. Since $T$ is a set of sentences, \cref{ass:soi:F1} holds, and the parameters can be renamed to $p_0,\dots,p_{s-1}$, so the code has $O(s\log s)$ bits (Proposition~E.21(b) of the earlier report). The checker, with $T$ hard-wired, verifies each line (logical axioms by matching, quantifier axioms by first-order matching, members of $T$ by comparison, modus ponens, generalisation) and the last line against $\varphi_x$ or $\neg\varphi_x$, in time polynomial in $s+\lvert T\rvert+d$; it accepts if the code is a derivation of $\varphi_x$ and rejects if it is one of $\neg\varphi_x$. Simulating it by circuits (standard; size $O(T\log T)$ for time $T$, \citealp{pippenger1979relations}) gives a verifier circuit of size $P(\lvert T\rvert,s,d)$. It is consistent because $T$ is consistent, and $f_{\mathcal V}\supseteq f$ because every datum has a derivation within $s$ and $T$ proves no wrong label, $T\cup\Gamma_f$ being consistent.
\end{proof}

A computation (code in the accompanying repository) checks the translation on random consistent verifier circuits with $d\in\{2,3\}$: every axiom of $T_{\mathcal V}$ is true in the intended structure on a finite fragment, forward propagation through the ground instances derives the right label for every $x\in\dom f_{\mathcal V}$, every corruption of one gate that changes $f_{\mathcal V}$ is detected, and $\lvert T_{\mathcal V}\rvert$ stays between $7.9$ and $9.0$ times $(\lvert\mathcal V\rvert+d)\log(\lvert\mathcal V\rvert+d)$ for sizes from $33$ to $1099$ gates.

\begin{corollary}[Bayes over verifier circuits]\status{proved, by the proof of the companion paper's Theorem 9.8 with the stated changes; computed}\src{sfi Cor 10.2}\label{cor:soi:circuitBayes}
Let $W_{\rm ver}$ be the mixture over consistent verifier circuits with prior $w_{\rm ver}(\mathcal V):=2^{-\lvert\mathcal V\rvert}\,(2\lvert\mathcal V\rvert+1)^{-1}\,\Lambda_{\lvert\mathcal V\rvert,k}^{-1}$, where $k\le2\lvert\mathcal V\rvert$ is uniform and $\Lambda_{j,k}:=\prod_{i=1}^{j}16(d+k+1+i)^2$ is the number of gate lists of size $j$. For a nonempty $J\subseteq\{0,1\}^d$, an ordering of $J$ and the labels of $f$, let $L:=-\log W_{\rm ver}(f|_J)$, $\ell_\emptyset:=-\log W_{\rm ver}(\emptyset)$, and $\mathrm{Err}:=\sum_i\epsilon_i$, with $\epsilon_i$ the posterior mass at step $i$ that is wrong \emph{or undefined} at $x_i$. Then:
(a) some consistent verifier circuit of size at most $\lceil L\rceil$ agrees with $f$ on $J$, and $L\le\mathrm{vs}_J+\log(2\mathrm{vs}_J+1)+\log\Lambda_{\mathrm{vs}_J,2\mathrm{vs}_J}=O(\mathrm{vs}_J\log(\mathrm{vs}_J+d))$, with $\mathrm{vs}_J:=\mathrm{vs}(f|_J)$;
(b) $\mathrm{Err}\le(\ln2)(L-\ell_\emptyset)$ and $L-\ell_\emptyset\le2\,\mathrm{Err}\,(\mathrm{gad}(\lceil L\rceil)+3)$, with $\mathrm{gad}(j):=\mathrm{gg}\,(5+2\log(d+3j+\mathrm{gg}+1))+\log(\mathrm{gg}+1)$ and $\mathrm{gg}:=2d+2$.
\end{corollary}

So Bayes over nondeterministic verifier circuits makes few posterior mistakes on $f$ exactly when $f$ has a small consistent verifier circuit, and, by \cref{thm:soi:circuit}, exactly when $f$ has a small finite axiomatisation in a fixed language whose data have polynomially short derivations, up to polynomial changes.

\begin{proof}
(a) The prior has an exponential tail in the size (the total weight of sizes above $S$ is at most $2^{-S}$), and the verifier with no active output disagrees with $f$ at every size, so the argument of Theorem~9.8(a) of the companion paper applies; the bound on $L$ is the weight of one smallest consistent verifier circuit for $f|_J$. (b) The first inequality is $\epsilon_i\le-\ln(1-\epsilon_i)$ with telescoping; now $W_{\rm ver}(\emptyset)<1$, hence $\ell_\emptyset$. For the second we use an exception gadget for verifiers: given $\mathcal V$, a point $a$ and a label $b$, append a circuit for $[x=a]$ (at most $2d-1$ gates), its negation, and new outputs $\mathsf a':=\mathsf a\vee[x=a]$, $\mathsf r':=\mathsf r\wedge\neg[x=a]$ if $b=1$ (symmetrically if $b=0$): at most $\mathrm{gg}=2d+2$ gates. The new circuit $\mathcal V'$ is consistent (at $a$ only the $b$-output can fire, and elsewhere it is $\mathcal V$), is defined at $a$ with label $b$, agrees with $\mathcal V$ off $a$, and $\mathcal V\mapsto\mathcal V'$ is injective, with $w_{\rm ver}(\mathcal V')/w_{\rm ver}(\mathcal V)\ge2^{-\mathrm{gad}(\lvert\mathcal V\rvert)}$. It maps the hypotheses of size at most $S$ that are wrong or undefined at $x_i$ injectively to hypotheses that are right at $x_i$ and still compatible with $D_{i-1}$, whose inputs differ from $x_i$. The rest of the proof of Theorem~9.8(b) of the companion paper is unchanged.
\end{proof}

\begin{remark}[The axiom-set prior itself]\status{proof sketch}\src{sfi Rem 10.3}\label{rem:soi:axiomsetprior}
\emph{Proof sketch.} We do not write out the converse direction for a prior over finite axiom sets; the obstacles are listed here. For a mixture over finite axiom sets with a derivation budget, the upper bounds $L=O(a\log a)$, with $a$ the size of a smallest compatible axiom set, and $\mathrm{Err}\le(\ln2)L$ hold directly (\cref{thm:soi:circuit}(i) and telescoping). The converse needs an exception gadget for \emph{every} hypothesis $T$: replace $R$ by a fresh unary predicate $R'$ in $T$ and add $\forall x\,(x\ne t_a\to(R(x)\leftrightarrow R'(x)))$ and $R(t_a)^b$. This works if the language supplies fresh predicate symbols (so it is not a fixed finite language, and the symbol's index costs $O(\log)$ bits), if $B$ contains the free-algebra axioms needed to derive $t_x\ne t_a$ (of size $O(d^2)$), and if the budget is charged in the prior, so that the gadget's extra derivation size $O(d^2)$ moves a hypothesis to a nearby budget at a cost of $O(\log)$ bits.
\end{remark}
