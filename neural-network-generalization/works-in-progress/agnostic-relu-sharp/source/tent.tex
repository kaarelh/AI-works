\section{A bounded steep teacher}
\begin{lemma}[A four-group tent]
\label{lem:tent}
Let $m\ge4$, $B\ge1$, $r=\lfloor m/4\rfloor$, and suppose
$\delta=2/(r^HB^L)\le1/2$. A width-at-most-$m$, depth-$L$ ReLU network with weights bounded by $B$ realizes an odd function $g(t)$ of its first coordinate such that
\[
 |g(t)|\le1\quad(t\in\R),\qquad
 g(0)=g(1)=g(-1)=0,\qquad g(\delta)=1,\quad g(-\delta)=-1.
\]
\end{lemma}
\begin{proof}
Use four groups of $r$ neurons. First-layer features are
$\relu(B(t+2\delta))$, $\relu(B(t+\delta))$,
$\relu(B(t-\delta))$, and $\relu(B(t-2\delta))$.
The biases have magnitude at most $2B\delta\le B$. Each internal within-group edge is $B$, and all other edges vanish. These nonnegative channels are multiplied by $(rB)^{H-1}$ through the internal layers. The readout coefficients per group are $-B/2,B,-B,B/2$.

The output is $g(t)=S\psi_\delta(t)$, where $S=r^HB^L/2=1/\delta$ and
\[
 \psi_\delta(t)=-\relu(t+2\delta)+2\relu(t+\delta)
                 -2\relu(t-\delta)+\relu(t-2\delta).
\]
This function equals $t$ on $[-\delta,\delta]$, is odd, vanishes outside $[-2\delta,2\delta]$, and has absolute value at most $\delta$. The assertions follow. Unused neurons are zero.
\end{proof}
