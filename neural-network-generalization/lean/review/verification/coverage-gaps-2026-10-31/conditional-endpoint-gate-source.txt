import ArbitraryPosteriorCeiling
import FixedHardFamily
import Lean.Util.CollectAxioms
import Lean.Elab.Command

open Lean Elab Command in
run_cmd do
  let targets : Array Name := #[
    `ConditionalNonpolynomial.arbitraryPosteriorCeiling,
    `ConditionalNonpolynomial.posterior_circuit_ceiling_all_beta,
    `ConditionalNonpolynomial.fixed_hard_family]
  let env ← getEnv
  for name in targets do
    unless (env.find? name).isSome do
      throwError "Missing theorem {name}"
    let axioms ← collectAxioms name
    for ax in axioms do
      unless #[`propext, `Classical.choice, `Quot.sound].contains ax do
        throwError "Unexpected axiom {ax} in {name}"
    logInfo m!"{name}: standard axioms only"
  logInfo m!"Checked {targets.size} conditional-package endpoints."
